    99→function formatTime(seconds: number) {
   100→  const h = Math.floor(seconds / 3600);
   101→  const m = Math.floor((seconds % 3600) / 60);
   102→  const s = seconds % 60;
   103→  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
   104→}
   105→
   106→function formatDuration(minutes: number | undefined) {
   107→  if (!minutes) return "0h 0m";
   108→  const hours = Math.floor(minutes / 60);
   109→  const mins = minutes % 60;
   110→  return `${hours}h ${mins}m`;
   111→}
   112→
   113→function formatDate(ts: number) {
   114→  return new Date(ts).toLocaleDateString("en-US", {
   115→    month: "short",
   116→    day: "numeric",
   117→    year: "numeric",
   118→  });
   119→}
