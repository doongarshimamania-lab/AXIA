   168→// ─── Status badge colors ─────────────────────────────────────────────────
   169→function statusColor(status: string): "default" | "secondary" | "destructive" | "outline" {
   170→  if (["won", "signed", "paid", "completed"].includes(status)) return "default";
   171→  if (["lost", "declined", "overdue", "expired", "rejected"].includes(status)) return "destructive";
   172→  return "secondary";
   173→}

   175→function statusIcon(type: string, status: string) {
   176→  if (type === "deal") {
   177→    if (status === "won") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   178→    if (status === "lost") return <AlertCircle className="h-4 w-4 text-red-500" />;
   179→    return <TrendingUp className="h-4 w-4 text-blue-500" />;
   180→  }
   181→  if (type === "proposal") {
   182→    if (status === "signed") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   183→    if (status === "sent" || status === "viewed") return <Send className="h-4 w-4 text-blue-500" />;
   184→    return <FileText className="h-4 w-4 text-muted-foreground" />;
   185→  }
   186→  // invoice
   187→  if (status === "paid") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   188→  if (status === "overdue") return <AlertCircle className="h-4 w-4 text-red-500" />;
   189→  return <DollarSign className="h-4 w-4 text-amber-500" />;
   190→}

   192→// ─── Format currency ─────────────────────────────────────────────────────
   193→function fmtCurrency(n: number): string {
   194→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
   195→}
   196→
   197→function fmtRelative(ts: number): string {
   198→  const diff = Date.now() - ts;
   199→  const mins = Math.floor(diff / 60000);
   200→  if (mins < 1) return "just now";
   201→  if (mins < 60) return `${mins}m ago`;
   202→  const hrs = Math.floor(mins / 60);
   203→  if (hrs < 24) return `${hrs}h ago`;
   204→  const days = Math.floor(hrs / 24);
   205→  if (days < 30) return `${days}d ago`;
   206→  return new Date(ts).toLocaleDateString();
   207→}
   208→
   209→function fmtCompactCurrency(n: number): string {
   210→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
   211→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
   212→  return fmtCurrency(n);
   213→}

   216→function TrendIndicator({ value, label }: { value: number; label: string }) {
   217→  if (value > 0) return (
   218→    <span className="flex items-center gap-0.5 text-emerald-600 dark:text-emerald-400 text-xs">
   219→      <ArrowUpRight className="h-3 w-3" /> +{value}% {label}
   220→    </span>
   221→  );
   222→  if (value < 0) return (
   223→    <span className="flex items-center gap-0.5 text-red-600 dark:text-red-400 text-xs">
   224→      <ArrowDownRight className="h-3 w-3" /> {value}% {label}
   225→    </span>
   226→  );
   227→  return (
   228→    <span className="flex items-center gap-0.5 text-muted-foreground text-xs">
   229→      <Minus className="h-3 w-3" /> No change
   230→    </span>
   231→  );
   232→}
