   118→export default function TeamManagement() {
   119→  const { activeWorkspaceId, isTeamMode, isOwner, canManageTeam, activeWorkspace } = useWorkspaceContext();
   120→
   121→  // ─── Convex Queries ────────────────────────────────────────────────────────
   122→  const hasRealWorkspaceId = activeWorkspaceId && !activeWorkspaceId.startsWith("ws_");
   123→
   124→  const convexMembers = useQuery(
   125→    hasRealWorkspaceId ? api.workspaces.members.getMembers : "skip",
   126→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   127→  ) as any[] | undefined;
   128→
   129→  const convexStats = useQuery(
   130→    hasRealWorkspaceId ? api.workspaces.crud.getWorkspaceStats : "skip",
   131→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   132→  ) as any | undefined;
   133→
   134→  const convexInvitations = useQuery(
   135→    hasRealWorkspaceId ? api.workspaces.invitations.getInvitations : "skip",
   136→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any, status: "pending" as const } : "skip"
   137→  ) as any[] | undefined;
   138→
   139→  // ─── Teams queries ─────────────────────────────────────────────────────────
   140→  const convexTeams = useQuery(
   141→    hasRealWorkspaceId ? api.teams.crud.getTeams : "skip",
   142→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   143→  ) as any[] | undefined;
   144→
   145→  // ─── Mutations ─────────────────────────────────────────────────────────────
   146→  const inviteMemberMutation = useMutation(api.workspaces.invitations.createInvitation);
   147→  const removeMemberMutation = useMutation(api.workspaces.members.removeMember);
   148→  const updateRoleMutation = useMutation(api.workspaces.members.updateMemberRole);
   149→  const cancelInvitationMutation = useMutation(api.workspaces.invitations.cancelInvitation);
   150→
   151→  // Team mutations
   152→  const createTeamMutation = useMutation(api.teams.crud.createTeam);
   153→  const updateTeamMutation = useMutation(api.teams.crud.updateTeam);
   154→  const deleteTeamMutation = useMutation(api.teams.crud.deleteTeam);
   155→  const addTeamMemberMutation = useMutation(api.teams.crud.addTeamMember);
   156→  const removeTeamMemberMutation = useMutation(api.teams.crud.removeTeamMember);
   157→
   158→  // ─── State ─────────────────────────────────────────────────────────────────
   159→  const [activeTab, setActiveTab] = useState<string>("members");
   160→
   161→  // Invite dialog
   162→  const [showInviteDialog, setShowInviteDialog] = useState(false);
   163→  const [inviteEmail, setInviteEmail] = useState("");
   164→  const [inviteRole, setInviteRole] = useState<"manager" | "member">("member");
   165→  const [inviteMessage, setInviteMessage] = useState("");
   166→  const [isInviting, setIsInviting] = useState(false);
   167→
   168→  // Remove/change role
   169→  const [memberToRemove, setMemberToRemove] = useState<any>(null);
   170→  const [memberToChangeRole, setMemberToChangeRole] = useState<any>(null);
   171→  const [newRole, setNewRole] = useState<"manager" | "member">("member");
   172→
   173→  // Team CRUD state
   174→  const [showCreateTeamDialog, setShowCreateTeamDialog] = useState(false);
   175→  const [teamName, setTeamName] = useState("");
   176→  const [teamColor, setTeamColor] = useState(TEAM_COLORS[0]);
   177→  const [teamDescription, setTeamDescription] = useState("");
   178→  const [teamIsCrossTeam, setTeamIsCrossTeam] = useState(false);
   179→  const [isCreatingTeam, setIsCreatingTeam] = useState(false);
   180→
   181→  // Edit team
   182→  const [editingTeam, setEditingTeam] = useState<any>(null);
   183→  const [editTeamName, setEditTeamName] = useState("");
   184→  const [editTeamColor, setEditTeamColor] = useState("");
   185→  const [editTeamDescription, setEditTeamDescription] = useState("");
   186→  const [isUpdatingTeam, setIsUpdatingTeam] = useState(false);
   187→
   188→  // Delete team
   189→  const [teamToDelete, setTeamToDelete] = useState<any>(null);
   190→  const [isDeletingTeam, setIsDeletingTeam] = useState(false);
   191→
   192→  // Add member to team
   193→  const [teamToAddMember, setTeamToAddMember] = useState<any>(null);
   194→  const [addMemberUserId, setAddMemberUserId] = useState("");
   195→  const [addMemberRole, setAddMemberRole] = useState<"lead" | "member">("member");
   196→  const [isAddingTeamMember, setIsAddingTeamMember] = useState(false);
   197→
   198→  // Remove member from team
   199→  const [teamMemberToRemove, setTeamMemberToRemove] = useState<any>(null);
   200→  const [teamMemberRemoveTeamId, setTeamMemberRemoveTeamId] = useState<string>("");
   201→
   202→  // ─── Map Convex data to UI shape ───────────────────────────────────────────
   203→  const members = useMemo(() => {
   204→    if (convexMembers && convexMembers.length > 0) {
   205→      return convexMembers.map((m: any) => ({
   206→        _id: m._id,
   207→        userId: m.userId,
   208→        name: m.userName ?? m.userEmail ?? "Unknown",
   209→        displayName: m.userName ?? m.userEmail ?? "Unknown",
   210→        email: m.userEmail ?? "",
   211→        image: m.userImage ?? "",
   212→        role: m.role,
   213→        status: m.status,
   214→        joinedAt: m.joinedAt,
   215→        lastActiveAt: m.lastActiveAt ?? null,
   216→        projectsAssigned: 0,
   217→        hoursThisWeek: 0,
   218→      }));
   219→    }
   220→    return [];
   221→  }, [convexMembers]);
   222→
   223→  const teams = useMemo(() => {
   224→    if (convexTeams && convexTeams.length > 0) {
   225→      return convexTeams;
   226→    }
   227→    return [];
   228→  }, [convexTeams]);
   229→
   230→  const stats = useMemo(() => {
   231→    if (convexStats) {
   232→      return {
   233→        memberCount: convexStats.memberCount ?? 0,
   234→        clientCount: convexStats.clientCount ?? 0,
   235→        activeProjectCount: convexStats.activeProjectCount ?? 0,
   236→        pendingInvoiceCount: 0,
   237→        totalRevenue: 0,
   238→        totalHoursThisWeek: 0,
   239→        protectionScore: 0,
   240→      };
   241→    }
   242→    return null;
   243→  }, [convexStats]);
   244→
   245→  const invitations = useMemo(() => {
   246→    if (convexInvitations && convexInvitations.length > 0) {
   247→      return convexInvitations.map((inv: any) => ({
   248→        _id: inv._id,
   249→        name: inv.email,
   250→        displayName: inv.email,
   251→        email: inv.email,
   252→        role: inv.role,
   253→        invitedAt: inv.createdAt,
   254→        invitedBy: inv.inviterName,
   255→      }));
   256→    }
   257→    return [];
   258→  }, [convexInvitations]);
   259→
   260→  const { isDisconnected } = useConvexConnectionState();
   261→  const isLoading = hasRealWorkspaceId && (convexMembers === undefined);
   262→  const isDemoMode = !hasRealWorkspaceId;
   263→  const loadingTimedOut = useQueryTimeout(isLoading, 3000);
   264→  const showLoading = isLoading && !loadingTimedOut && !isDisconnected;
   265→
   266→  // ─── Handlers ──────────────────────────────────────────────────────────────
   267→  const handleInvite = async () => {
   268→    if (!inviteEmail.trim() || !activeWorkspaceId) return;
   269→    setIsInviting(true);
   270→    try {
   271→      if (hasRealWorkspaceId) {
   272→        await inviteMemberMutation({
   273→          workspaceId: activeWorkspaceId as any,
   274→          email: inviteEmail.trim(),
   275→          role: inviteRole,
   276→        });
   277→      }
   278→      toast.success(`Invitation sent to ${inviteEmail}`);
   279→      setInviteEmail("");
   280→      setInviteMessage("");
   281→      setShowInviteDialog(false);
   282→    } catch (e: any) {
   283→      toast.error(e.message || "Failed to send invitation");
   284→    } finally {
   285→      setIsInviting(false);
   286→    }
   287→  };
   288→
   289→  const handleRemoveMember = async () => {
   290→    if (!memberToRemove) return;
   291→    try {
   292→      await removeMemberMutation({ memberId: memberToRemove._id });
   293→      toast.success(`${memberToRemove.name} removed from workspace`);
   294→      setMemberToRemove(null);
   295→    } catch (e: any) {
   296→      toast.error(e.message || "Failed to remove member");
   297→    }
   298→  };
   299→
   300→  const handleChangeRole = async () => {
   301→    if (!memberToChangeRole) return;
   302→    try {
   303→      await updateRoleMutation({ memberId: memberToChangeRole._id, role: newRole });
   304→      toast.success(`Role updated to ${newRole}`);
   305→      setMemberToChangeRole(null);
   306→    } catch (e: any) {
   307→      toast.error(e.message || "Failed to update role");
   308→    }
   309→  };
   310→
   311→  const handleCancelInvitation = async (invitation: any) => {
   312→    try {
   313→      await cancelInvitationMutation({ invitationId: invitation._id });
   314→      toast.success(`Invitation to ${invitation.email || invitation.name} cancelled`);
   315→    } catch (e: any) {
   316→      toast.error(e.message || "Failed to cancel invitation");
   317→    }
   318→  };
   319→
   320→  // ─── Team CRUD Handlers ────────────────────────────────────────────────────
   321→  const handleCreateTeam = async () => {
   322→    if (!teamName.trim() || !activeWorkspaceId) return;
   323→    setIsCreatingTeam(true);
   324→    try {
   325→      if (createTeamMutation && hasRealWorkspaceId) {
   326→        await createTeamMutation({
   327→          workspaceId: activeWorkspaceId as any,
   328→          name: teamName.trim(),
   329→          color: teamColor,
   330→          description: teamDescription.trim() || undefined,
   331→          isCrossTeam: teamIsCrossTeam || undefined,
   332→        });
   333→      }
   334→      toast.success(`Team "${teamName}" created!`);
   335→      setShowCreateTeamDialog(false);
   336→      setTeamName("");
   337→      setTeamColor(TEAM_COLORS[0]);
   338→      setTeamDescription("");
   339→      setTeamIsCrossTeam(false);
   340→    } catch (e: any) {
   341→      toast.error(e.message || "Failed to create team");
   342→    } finally {
   343→      setIsCreatingTeam(false);
   344→    }
   345→  };
   346→
   347→  const handleUpdateTeam = async () => {
   348→    if (!editingTeam) return;
   349→    setIsUpdatingTeam(true);
   350→    try {
   351→      if (updateTeamMutation) {
   352→        await updateTeamMutation({
   353→          teamId: editingTeam._id,
   354→          name: editTeamName.trim() || undefined,
   355→          color: editTeamColor || undefined,
   356→          description: editTeamDescription.trim() || undefined,
   357→        });
   358→      }
   359→      toast.success("Team updated!");
   360→      setEditingTeam(null);
   361→    } catch (e: any) {
   362→      toast.error(e.message || "Failed to update team");
   363→    } finally {
   364→      setIsUpdatingTeam(false);
   365→    }
   366→  };
   367→
   368→  const handleDeleteTeam = async () => {
   369→    if (!teamToDelete) return;
   370→    setIsDeletingTeam(true);
   371→    try {
   372→      if (deleteTeamMutation) {
   373→        await deleteTeamMutation({ teamId: teamToDelete._id });
   374→      }
   375→      toast.success(`Team "${teamToDelete.name}" deleted`);
   376→      setTeamToDelete(null);
   377→    } catch (e: any) {
   378→      toast.error(e.message || "Failed to delete team");
   379→    } finally {
   380→      setIsDeletingTeam(false);
   381→    }
   382→  };
   383→
   384→  const handleAddTeamMember = async () => {
   385→    if (!teamToAddMember || !addMemberUserId) return;
   386→    setIsAddingTeamMember(true);
   387→    try {
   388→      if (addTeamMemberMutation) {
   389→        await addTeamMemberMutation({
   390→          teamId: teamToAddMember._id,
   391→          userId: addMemberUserId as any,
   392→          role: addMemberRole,
   393→        });
   394→      }
   395→      toast.success("Member added to team!");
   396→      setTeamToAddMember(null);
   397→      setAddMemberUserId("");
   398→      setAddMemberRole("member");
   399→    } catch (e: any) {
   400→      toast.error(e.message || "Failed to add member to team");
   401→    } finally {
   402→      setIsAddingTeamMember(false);
   403→    }
   404→  };
   405→
   406→  const handleRemoveTeamMember = async () => {
   407→    if (!teamMemberToRemove || !teamMemberRemoveTeamId) return;
   408→    try {
   409→      if (removeTeamMemberMutation) {
   410→        await removeTeamMemberMutation({
   411→          teamId: teamMemberRemoveTeamId as any,
   412→          userId: teamMemberToRemove.userId as any,
   413→        });
   414→      }
   415→      toast.success("Member removed from team");
   416→      setTeamMemberToRemove(null);
   417→      setTeamMemberRemoveTeamId("");
   418→    } catch (e: any) {
   419→      toast.error(e.message || "Failed to remove member from team");
   420→    }
   421→  };
   422→
   423→  // ─── Not in team mode: Show upgrade prompt ────────────────
   424→  if (!isTeamMode) {
   425→    return <SoloModePrompt />;
   426→  }
   427→
   428→  const activeMembers = members.filter((m: any) => m.status === "active");
   429→  const ownerCount = activeMembers.filter((m: any) => m.role === "owner").length;
   430→  const managerCount = activeMembers.filter((m: any) => m.role === "manager").length;
   431→  const memberCount = activeMembers.filter((m: any) => m.role === "member").length;
   432→
   433→  const roleIcon = (role: string) => {
   434→    switch (role) {
   435→      case "owner": return <Crown className="w-3.5 h-3.5 text-amber-500" />;
   436→      case "manager": return <Shield className="w-3.5 h-3.5 text-blue-500" />;
   437→      case "lead": return <Shield className="w-3.5 h-3.5 text-emerald-500" />;
   438→      case "member": return <Eye className="w-3.5 h-3.5 text-gray-500" />;
   439→      default: return null;
   440→    }
   441→  };
   442→
   443→  const roleBadge = (role: string) => {
   444→    const variants: Record<string, any> = {
   445→      owner: { className: "bg-amber-500/10 text-amber-600 border-amber-500/20", label: "Owner" },
   446→      manager: { className: "bg-blue-500/10 text-blue-600 border-blue-500/20", label: "Manager" },
   447→      lead: { className: "bg-emerald-500/10 text-emerald-600 border-emerald-500/20", label: "Lead" },
   448→      member: { className: "bg-gray-500/10 text-gray-600 border-gray-500/20", label: "Member" },
   449→    };
   450→    const v = variants[role] || variants.member;
   451→    return (
   452→      <Badge variant="outline" className={`gap-1 ${v.className}`}>
   453→        {roleIcon(role)}
   454→        {v.label}
   455→      </Badge>
   456→    );
   457→  };
   458→
   459→  const formatLastActive = (timestamp: number | null) => {
   460→    if (!timestamp) return "Never";
   461→    const diffMs = Date.now() - timestamp;
   462→    const diffMins = Math.floor(diffMs / 60000);
   463→    if (diffMins < 5) return "Just now";
   464→    if (diffMins < 60) return `${diffMins}m ago`;
   465→    const diffHours = Math.floor(diffMins / 60);
   466→    if (diffHours < 24) return `${diffHours}h ago`;
   467→    const diffDays = Math.floor(diffHours / 24);
   468→    return `${diffDays}d ago`;
   469→  };
   470→
   471→  const formatJoined = (timestamp: number | null) => {
   472→    if (!timestamp) return "";
   473→    const diffMs = Date.now() - timestamp;
   474→    const diffDays = Math.floor(diffMs / (24 * 60 * 60 * 1000));
   475→    if (diffDays < 1) return "Joined today";
   476→    if (diffDays < 30) return `Joined ${diffDays}d ago`;
   477→    const diffMonths = Math.floor(diffDays / 30);
   478→    return `Joined ${diffMonths}mo ago`;
   479→  };
   480→
   481→  return (
   482→    <PageLayout>
   483→      <div className="w-full space-y-6">
   484→        {/* Header */}
   485→        <div className="flex items-center justify-between">
   486→          <div>
   487→            <h1 className="text-3xl font-bold text-foreground mb-2 flex items-center gap-3">
   488→              <Building2 className="w-8 h-8 text-purple-500" />
   489→              {activeWorkspace?.name || "Team Workspace"}
   490→            </h1>
   491→            <p className="text-muted-foreground">
   492→              Manage your team members, teams, and invitations
   493→            </p>
   494→          </div>
   495→          {canManageTeam && (
   496→            <div className="flex gap-2">
   497→              <Button className="gap-2" onClick={() => setShowInviteDialog(true)}>
   498→                <UserPlus className="w-4 h-4" />
   499→                Invite Member
   500→              </Button>
   501→            </div>
   502→          )}
   503→        </div>
   504→
   505→        {/* Demo Mode Banner */}
   506→        {isDemoMode && <DemoModeBanner />}
   507→
   508→        {/* Loading / Error State */}
   509→        {showLoading ? (
   510→          <StatsSkeleton />
   511→        ) : (
   512→          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
   513→            <Card className="p-5">
   514→              <div className="flex items-center gap-3">
   515→                <div className="p-2.5 bg-purple-500/10 rounded-lg"><Users className="w-5 h-5 text-purple-500" /></div>
   516→                <div>
   517→                  <p className="text-sm text-muted-foreground">Members</p>
   518→                  <p className="text-2xl font-bold">{stats?.memberCount ?? activeMembers.length}</p>
   519→                </div>
   520→              </div>
   521→            </Card>
   522→            <Card className="p-5">
   523→              <div className="flex items-center gap-3">
   524→                <div className="p-2.5 bg-emerald-500/10 rounded-lg"><Shield className="w-5 h-5 text-emerald-500" /></div>
   525→                <div>
   526→                  <p className="text-sm text-muted-foreground">Teams</p>
   527→                  <p className="text-2xl font-bold">{teams.length}</p>
   528→                </div>
   529→              </div>
   530→            </Card>
   531→            <Card className="p-5">
   532→              <div className="flex items-center gap-3">
   533→                <div className="p-2.5 bg-blue-500/10 rounded-lg"><Building2 className="w-5 h-5 text-blue-500" /></div>
