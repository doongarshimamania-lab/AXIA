    55→function fmtCompactCurrency(n: number): string {
    56→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
    57→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
    58→  return fmtCurrency(n);

    60→// ─── Animated Number Counter ──────────────────────────────────────────────
    61→function AnimatedNumber({ value, duration = 1200, prefix = "", suffix = "", decimals = 0 }: {
    62→  value: number; duration?: number; prefix?: string; suffix?: string; decimals?: number;
    63→}) {
    64→  const [display, setDisplay] = useState(0);
    65→  const prevValue = useRef(0);
    66→  const rafRef = useRef<number>();
    67→
    68→  useEffect(() => {
    69→    const start = prevValue.current;
    70→    const end = value;
    71→    const startTime = performance.now();
    72→
    73→    const animate = (now: number) => {
    74→      const elapsed = now - startTime;
    75→      const progress = Math.min(elapsed / duration, 1);
    76→      // Ease out cubic
    77→      const eased = 1 - Math.pow(1 - progress, 3);
    78→      const current = start + (end - start) * eased;
    79→      setDisplay(current);
    80→      if (progress < 1) {
    81→        rafRef.current = requestAnimationFrame(animate);
    82→      } else {
    83→        prevValue.current = end;
    84→      }
    85→    };
    86→
    87→    rafRef.current = requestAnimationFrame(animate);
    88→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
    89→  }, [value, duration]);
    90→
    91→  const formatted = decimals > 0
    92→    ? display.toFixed(decimals)
    93→    : Math.round(display).toLocaleString();
    94→
    95→  return <span>{prefix}{formatted}{suffix}</span>;
    96→}

    98→// ─── Mini Sparkline Chart (SVG) ──────────────────────────────────────────
    99→function Sparkline({ data, color = "#8B5CF6", height = 32, width = 80 }: {
   100→  data: number[]; color?: string; height?: number; width?: number;
   101→}) {
   102→  if (!data || data.length < 2) return null;
   103→
   104→  const min = Math.min(...data);
   105→  const max = Math.max(...data);
   106→  const range = max - min || 1;
   107→  const padding = 2;
   108→
   109→  const points = data.map((v, i) => {
   110→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   111→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   112→    return `${x},${y}`;
   113→  }).join(" ");
   114→
   115→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   116→
   117→  return (
   118→    <svg width={width} height={height} className="overflow-visible">
   119→      <defs>
   120→        <linearGradient id={`spark-grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
   121→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   122→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   123→        </linearGradient>
   124→      </defs>
   125→      <polygon points={areaPoints} fill={`url(#spark-grad-${color.replace("#", "")})`} />
   126→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   127→      {/* Last point dot */}
   128→      {data.length > 0 && (
   129→        <circle
   130→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   131→          cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)}
   132→          r="2.5"
   133→          fill={color}
   134→          className="animate-pulse"
   135→        />
   136→      )}
   137→    </svg>
   138→  );
   139→}

   141→// ─── Circular Progress Ring ──────────────────────────────────────────────
   142→function ProgressRing({ value, size = 56, strokeWidth = 4, color = "#8B5CF6", label }: {
   143→  value: number; size?: number; strokeWidth?: number; color?: string; label?: string;
   144→}) {
   145→  const radius = (size - strokeWidth) / 2;
   146→  const circumference = 2 * Math.PI * radius;
   147→  const offset = circumference - (value / 100) * circumference;
   148→
   149→  return (
   150→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   151→      <svg width={size} height={size} className="-rotate-90">
   152→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   153→        <motion.circle
   154→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
   155→          strokeLinecap="round"
   156→          strokeDasharray={circumference}
   157→          initial={{ strokeDashoffset: circumference }}
   158→          animate={{ strokeDashoffset: offset }}
   159→          transition={{ duration: 1.2, ease: "easeOut" }}
   160→        />
   161→      </svg>
   162→      <span className="absolute text-[10px] font-bold" style={{ color }}>{value}%</span>
   163→    </div>
   164→  );
   165→}
