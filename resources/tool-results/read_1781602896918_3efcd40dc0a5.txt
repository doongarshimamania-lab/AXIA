   100→   100→function timeAgo(ts: number): string {
   101→   101→  const diff = Date.now() - ts;
   102→   102→  const minutes = Math.floor(diff / 60000);
   103→   103→  const hours = Math.floor(minutes / 60);
   104→   104→  const days = Math.floor(hours / 24);
   105→   105→  if (days > 0) return `${days}d ago`;
   106→   106→  if (hours > 0) return `${hours}h ago`;
   107→   107→  return `${minutes}m ago`;
   108→   108→}

   158→   158→// ─── Scope Creep Detection ──────────────────────────────────────────────────

   160→   160→function calculateScopeHealth(scope: ScopeDefinition, changeOrders: ScopeChangeOrder[]) {
   161→   161→  const scopeChanges = changeOrders.filter(co => co.scopeId === scope._id);
   162→   162→  const pendingChanges = scopeChanges.filter(co => co.status === "pending" || co.status === "auto_generated");
   163→   163→  const revisionUsage = scope.revisionLimit > 0 ? (scope.revisionCount / scope.revisionLimit) * 100 : 0;
   164→   164→  const totalHoursAdded = scopeChanges.reduce((sum, co) => sum + co.impact.hoursAdded, 0);
   165→   165→  const totalCostImpact = scopeChanges.reduce((sum, co) => sum + co.impact.costImpact, 0);
   166→   166→  const hoursOverrun = scope.totalEstimatedHours ? (totalHoursAdded / scope.totalEstimatedHours) * 100 : 0;
   167→   167→
   168→   168→  let riskLevel: "low" | "medium" | "high" = "low";
   169→   169→  if (revisionUsage >= 100 || hoursOverrun > 30 || pendingChanges.length >= 3) riskLevel = "high";
   170→   170→  else if (revisionUsage >= 66 || hoursOverrun > 15 || pendingChanges.length >= 2) riskLevel = "medium";
   171→   171→
   172→   172→  const completedDeliverables = scope.deliverables.filter(d => d.status === "completed").length;
   173→   173→  const progressPercent = scope.deliverables.length > 0 ? (completedDeliverables / scope.deliverables.length) * 100 : 0;
   174→   174→
   175→   175→  return {
   176→   176→    riskLevel,
   177→   177→    revisionUsage,
   178→   178→    totalHoursAdded,
   179→   179→    totalCostImpact,
   180→   180→    hoursOverrun,
   181→   181→    pendingChanges: pendingChanges.length,
   182→   182→    completedDeliverables,
   183→   183→    totalDeliverables: scope.deliverables.length,
   184→   184→    progressPercent,
   185→   185→  };
   186→   186→}
