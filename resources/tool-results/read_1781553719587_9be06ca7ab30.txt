   138→export default function Proposals() {
   139→  const navigate = useNavigate();
   140→
   141→  // ── Workspace Context ──
   142→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();

   144→
   145→  // ── Permissions ──

   149→  const [searchQuery, setSearchQuery] = useState("");
   150→  const [deleteDialogId, setDeleteDialogId] = useState<string | null>(null);
   151→  const [isSeeding, setIsSeeding] = useState(false);
   152→  const [showShareDialog, setShowShareDialog] = useState(false);
   153→  const [shareProposalId, setShareProposalId] = useState<string | null>(null);
   154→  const [sharingRecord, setSharingRecord] = useState<{id: string, type: string, sharing: any[]} | null>(null);
   155→
   156→  // ── Convex mutations for sharing ──
   157→  const shareRecordMutation = useMutation(api.permissions.shareRecord.shareRecord);
   158→  const unshareRecordMutation = useMutation(api.permissions.shareRecord.unshareRecord);
   159→
   160→  // Convex queries
   161→  const convexProposals = useQuery(api.proposals.crud.getProposals, workspaceId ? (activeFilter === "all" ? { workspaceId } : { workspaceId, status: activeFilter }) : "skip") as Proposal[] | undefined;
   162→  const convexStats = useQuery(api.proposals.crud.getProposalStats, workspaceId ? { workspaceId } : "skip") as {
   163→    total: number;
   164→    sent: number;
   165→    signed: number;
   166→    draft: number;
   167→    signatureRate: number;
   168→    totalValue: number;
   169→  } | undefined;
   170→
   171→  // Convex mutations
   172→  const sendProposal = useMutation(api.proposals.crud.sendProposal);

   176→

   180→  const { isDisconnected } = useConvexConnectionState();
   181→  const isLoading = convexProposals === undefined || convexStats === undefined;
   182→  const timedOut = useQueryTimeout(isLoading, 3000);
   183→  const showLoading = isLoading && !timedOut && !isDisconnected;
   184→
   185→  // Use Convex data only
   186→  const proposals = useMemo(() => {

   190→  const stats = useMemo(() => {

   192→  }, [convexStats]);
   193→

   201→      counts[p.status] = (counts[p.status] || 0) + 1;

   213→        p.title.toLowerCase().includes(q) ||
   214→        (p.clientName && p.clientName.toLowerCase().includes(q))

   223→    const d = new Date(timestamp);
   224→    return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
   225→  };
   226→
   227→  const handleSeed = async () => {

   233→          toast.success("Mock proposals seeded!", {
   234→            description: `${(result as any).count} proposals created.`,

   237→          toast.info("Proposals already exist", {
   238→            description: `${(result as any).count} proposals found.`,
   239→          });
   240→        }
   241→      }
   242→    } catch (err: any) {
   243→      toast.error("Failed to seed proposals", { description: err.message });

   245→      setIsSeeding(false);
   246→    }
   247→  };

   251→      await sendProposal({ proposalId: proposalId as any });

   253→        description: "Follow-ups have been scheduled automatically.",
   254→      });
   255→    } catch (err: any) {
   256→      toast.error("Failed to send proposal", { description: err.message });
   257→    }
   258→  };
   259→
   260→  const handleDuplicate = async (proposalId: string) => {
   261→    try {
   262→      const newId = await duplicateProposal({ proposalId: proposalId as any });
   263→      toast.success("Proposal duplicated!", {
   264→        description: "A draft copy has been created.",
   265→      });
   266→    } catch (err: any) {
   267→      toast.error("Failed to duplicate proposal", { description: err.message });
   268→    }
   269→  };
   270→
   271→  const handleDelete = async (proposalId: string) => {
   272→    try {
   273→      await deleteProposal({ proposalId: proposalId as any });
   274→      setDeleteDialogId(null);
   275→      toast.success("Proposal deleted");
   276→    } catch (err: any) {
   277→      toast.error("Failed to delete proposal", { description: err.message });
   278→    }
   279→  };
   280→
   281→  return (
   282→    <motion.div
   283→      className="flex-1 min-h-screen bg-background text-foreground transition-colors"
   284→      initial={{ opacity: 0 }}
   285→      animate={{ opacity: 1 }}
   286→      transition={{ duration: 0.5 }}
   287→    >
   288→      <PageLayout wide>
   289→        {/* Header */}

   292→            <h1 className="text-2xl md:text-[32px] font-bold text-foreground tracking-tight mb-1" style={{ fontFamily: "Space Grotesk, sans-serif" }}>
   293→              Proposals
   294→            </h1>
   295→            <p className="text-[16px] text-muted-foreground">
   296→              Create, send, and track proposals with automated follow-ups
   297→            </p>
   298→          </div>
   299→          <div className="flex items-center gap-2">
   300→            <Button
   301→              variant="outline"
   302→              size="sm"
   303→              className="gap-2 text-muted-foreground"
   304→              onClick={handleSeed}
   305→              disabled={isSeeding}
   306→            >
   307→              {isSeeding ? (
   308→                <Loader2 className="h-4 w-4 animate-spin" />
   309→              ) : (
   310→                <Sparkles className="h-4 w-4" />
   311→              )}
   312→              Seed Data
   313→            </Button>
   314→            <Button
   315→              className="bg-[#8B5CF6] hover:bg-[#7C3AED] text-white gap-2"
   316→              onClick={() => navigate("/proposals/new")}
   317→            >
   318→              <Plus className="h-4 w-4" />
   319→              Create Proposal
   320→            </Button>
   321→          </div>
   322→        </div>
   323→
   324→        {/* Loading State */}
   325→        {showLoading ? (
   326→          <div className="space-y-6">
   327→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   328→              {[1, 2, 3, 4, 5, 6].map((i) => (
   329→                <Card key={i}>
   330→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-1.5 px-4 pt-4">
   331→                    <Skeleton className="h-4 w-16" />
   332→                    <Skeleton className="h-4 w-4" />
   333→                  </CardHeader>
   334→                  <CardContent className="px-4 pb-4">
   335→                    <Skeleton className="h-6 w-12 mb-1" />
   336→                  </CardContent>
   337→                </Card>
   338→              ))}
   339→            </div>
   340→            <Skeleton className="h-10 rounded-lg" />
   341→            <div className="space-y-3">
   342→              {[1, 2, 3].map((i) => (
   343→                <Skeleton key={i} className="h-[100px] rounded-xl" />
   344→              ))}
   345→            </div>
   346→          </div>
   347→        ) : (
   348→          <>
   349→        {/* Stats Bar */}
   350→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3 mb-6">
   351→          {[
   352→            {
   353→              title: "Total",
   354→              value: stats?.total ?? 0,
   355→              icon: FileText,
   356→              color: "text-[#8B5CF6]",
   357→              bgColor: "bg-[#8B5CF6]/10",
   358→              format: (v: number) => String(v),
   359→            },
   360→            {
   361→              title: "Sent",
   362→              value: stats?.sent ?? 0,
   363→              icon: Send,

   366→              format: (v: number) => String(v),
   367→            },
   368→            {

   370→              value: stats?.signed ?? 0,

   378→              value: stats?.draft ?? 0,

   390→              format: (v: number) => `${v}%`,
   391→            },
   392→            {
   393→              title: "Total Value",

   400→          ].map((stat, idx) => (
   401→            <motion.div
   402→              key={stat.title}
   403→              initial={{ opacity: 0, y: 20 }}

   407→              <Card>

   413→                    <stat.icon className={`h-3.5 w-3.5 ${stat.color}`} />

   415→                </CardHeader>

   457→                      ? "bg-[#8B5CF6]/20 text-[#8B5CF6]"

   459→                  }`}
   460→                >
   461→                  {filterCounts[tab.key] ?? 0}
   462→                </span>
   463→                <div
   464→                  className={`absolute bottom-0 left-0 right-0 h-[2px] ${
   465→                    isActive ? "bg-[#8B5CF6]" : "bg-transparent"
   466→                  }`}
   467→                />
   468→              </button>

   470→          })}
   471→        </div>
   472→
   473→        {/* Proposal Cards Grid */}
   474→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   475→          <AnimatePresence mode="popLayout">

   478→                initial={{ opacity: 0 }}
   479→                animate={{ opacity: 1 }}
   480→                exit={{ opacity: 0 }}
   481→                className="col-span-full"
   482→              >
   483→                <Card>
   484→                  <CardContent className="py-16">
   485→                    <div className="text-center text-muted-foreground">
   486→                      <FileText className="h-12 w-12 mx-auto mb-4 opacity-40" />
   487→                      <p className="text-[16px] font-medium mb-1">No proposals found</p>
   488→                      <p className="text-[14px] mb-4">
   489→                        {searchQuery
   490→                          ? "Try adjusting your search terms"
   491→                          : "Create your first proposal or seed sample data"}
   492→                      </p>
   493→                      {!searchQuery && (
   494→                        <div className="flex items-center justify-center gap-3">
   495→                          <Button
   496→                            variant="outline"
   497→                            className="gap-2"
   498→                            onClick={handleSeed}
   499→                            disabled={isSeeding}
   500→                          >
   501→                            <Sparkles className="h-4 w-4" />
   502→                            Seed Sample Data
   503→                          </Button>
   504→                          <Button
   505→                            className="bg-[#8B5CF6] hover:bg-[#7C3AED] text-white gap-2"
   506→                            onClick={() => navigate("/proposals/new")}
   507→                          >
   508→                            <Plus className="h-4 w-4" />
   509→                            Create Proposal
   510→                          </Button>
   511→                        </div>
   512→                      )}
   513→                    </div>
   514→                  </CardContent>
   515→                </Card>
   516→              </motion.div>
   517→            ) : (
   518→              filteredProposals.map((proposal, idx) => (
   519→                <ProposalCard
   520→                  key={proposal._id}
   521→                  proposal={proposal}
   522→                  idx={idx}
   523→                  isMock={false}
   524→                  formatCurrency={formatCurrency}
   525→                  formatDate={formatDate}
   526→                  onSend={handleSend}
   527→                  onDuplicate={handleDuplicate}
   528→                  onDelete={canDeleteRecords ? setDeleteDialogId : () => {}}
   529→                  onView={() => navigate(`/proposals/new?edit=${proposal._id}`)}
   530→                  onShare={canShareRecords ? (id: string) => {
   531→                    const p = proposals.find((p: any) => p._id === id);
   532→                    if (canShareRecords) {
   533→                      setSharingRecord({
   534→                        id,
   535→                        type: "proposal",
   536→                        sharing: (p as any)?.sharing || [],
   537→                      });
   538→                      setShareProposalId(id);
   539→                      setShowShareDialog(true);
   540→                    }
   541→                  } : () => {}}
   542→                  canDelete={canDeleteRecords}
   543→                  canShare={canShareRecords}
   544→                />
   545→              ))
   546→            )}
   547→          </AnimatePresence>
   548→        </div>
   549→        </>
   550→        )}
   551→      </PageLayout>
   552→
   553→      {/* Delete Confirmation Dialog */}
   554→      <Dialog open={!!deleteDialogId} onOpenChange={() => setDeleteDialogId(null)}>
   555→        <DialogContent>
   556→          <DialogHeader>
   557→            <DialogTitle>Delete Proposal</DialogTitle>
   558→            <DialogDescription>
   559→              Are you sure you want to delete this proposal? This action cannot be undone. All associated follow-ups will also be removed.
   560→            </DialogDescription>
   561→          </DialogHeader>
   562→          <DialogFooter className="gap-2">
   563→            <Button variant="outline" onClick={() => setDeleteDialogId(null)}>
   564→              Cancel
   565→            </Button>
   566→            <Button
   567→              variant="destructive"
   568→              onClick={() => deleteDialogId && handleDelete(deleteDialogId)}
   569→            >
   570→              Delete
   571→            </Button>
   572→          </DialogFooter>
   573→        </DialogContent>
   574→      </Dialog>
   575→
   576→      {/* Share Dialog */}
   577→      <ShareDialog
   578→        open={showShareDialog}
   579→        onOpenChange={setShowShareDialog}
   580→        recordId={sharingRecord?.id || shareProposalId || ""}
   581→        recordType={sharingRecord?.type || "proposal"}
   582→        currentSharing={sharingRecord?.sharing || []}
   583→        onShare={async (args) => {
   584→          try {
   585→            if (shareRecordMutation) {
   586→              await shareRecordMutation({
   587→                recordId: sharingRecord?.id || shareProposalId,
   588→                recordType: sharingRecord?.type || "proposal",
   589→                ...args,
   590→              });
   591→            }
   592→            toast.success("Record shared successfully");
   593→          } catch (err: any) {
   594→            toast.error(err?.message || "Failed to share record");
   595→          }
   596→        }}
   597→        onUnshare={async (args) => {
   598→          try {
   599→            if (unshareRecordMutation) {
   600→              await unshareRecordMutation({
   601→                recordId: sharingRecord?.id || shareProposalId,
   602→                recordType: sharingRecord?.type || "proposal",
   603→                ...args,
   604→              });
   605→            }
   606→            toast.success("Access removed");
   607→          } catch (err: any) {
   608→            toast.error(err?.message || "Failed to remove access");
   609→          }
   610→        }}
   611→      />
   612→    </motion.div>
   613→  );
   614→}
