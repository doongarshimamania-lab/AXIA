   426→   426→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });

   429→   429→    }
   430→   430→    setIsExporting(false);
   431→   431→  };

   436→   436→      return;
   437→   437→    }
   438→   438→    setSelectedFormat(format);
   439→   439→  };
   440→   440→
   441→   441→  const previewItems = useMemo(() => {
   442→   442→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
