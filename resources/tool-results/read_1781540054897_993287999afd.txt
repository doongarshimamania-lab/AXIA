     5→// Get all clients for the current user
     6→export const getMyClients = query({
     7→  args: {},
     8→  handler: async (ctx) => {
     9→    let userId;
    10→    try {
    11→      userId = await getAuthUserId(ctx);
    12→    } catch (error) {
    13→      // User not authenticated
    14→      return [];
    15→    }
    16→    if (!userId) return [];
    17→
    18→    const clients = await ctx.db
    19→      .query("clients")
    20→      .withIndex("by_user", (q) => q.eq("userId", userId))
    21→      .collect();
    22→
    23→    // Enrich with protection metrics
    24→    const enrichedClients = await Promise.all(
    25→      clients.map(async (client) => {
    26→        // Get sessions for this client
    27→        const sessions = await ctx.db
    28→          .query("workSessions")
    29→          .withIndex("by_user", (q) => q.eq("userId", userId))
    30→          .filter((q) => q.eq(q.field("clientName"), client.clientName))
    31→          .collect();
    32→
    33→        const totalHours = sessions.reduce((sum, s) => sum + (s.totalMinutes || 0) / 60, 0);
    34→        const activeSession = sessions.find((s) => !s.endTime);
    35→
    36→        // Get time blocks for compliance
    37→        const timeBlocks = await ctx.db
    38→          .query("timeBlocks")
    39→          .withIndex("by_user", (q) => q.eq("userId", userId))
    40→          .collect();
    41→
    42→        const clientBlocks = timeBlocks.filter((block) => {
    43→          const session = sessions.find((s) => s._id === block.sessionId);
    44→          return session !== undefined;
    45→        });
    46→
    47→        const compliantBlocks = clientBlocks.filter((b) => b.complianceStatus === "compliant").length;
    48→        const totalBlocks = clientBlocks.length;
    49→        const protectionScore = totalBlocks > 0 ? Math.round((compliantBlocks / totalBlocks) * 100) : 100;
    50→
    51→        return {
    52→          ...client,
    53→          totalHours: Math.round(totalHours * 10) / 10,
    54→          protectionScore,
    55→          activeSession: activeSession ? true : false,
    56→          totalValue: Math.round(totalHours * client.hourlyRate * 100) / 100,
    57→        };
    58→      })
    59→    );
    60→
    61→    return enrichedClients;
    62→  },
    63→});

    65→// Add a new client
    66→export const addClient = mutation({
    67→  args: {
    68→    clientName: v.string(),
    69→    platform: v.union(
    70→      v.literal("upwork"),
    71→      v.literal("fiverr"),
    72→      v.literal("toptal"),
    73→      v.literal("freelancer"),
    74→      v.literal("direct")
    75→    ),
    76→    hourlyRate: v.number(),
    77→    contractType: v.union(v.literal("hourly"), v.literal("fixed")),
    78→    riskLevel: v.union(v.literal("low"), v.literal("medium"), v.literal("high")),
    79→  },
    80→  handler: async (ctx, args) => {
    81→    let userId;
    82→    try {
    83→      userId = await getAuthUserId(ctx);
    84→    } catch (error) {
    85→      return { success: false, error: "Not authenticated" };
    86→    }
    87→    if (!userId) return { success: false, error: "Not authenticated" };
    88→
    89→    // Check if client already exists
    90→    const existing = await ctx.db
    91→      .query("clients")
    92→      .withIndex("by_user_and_name", (q) =>
    93→        q.eq("userId", userId).eq("clientName", args.clientName)
    94→      )
    95→      .first();
    96→
    97→    if (existing) {
    98→      throw new Error("Client already exists");
    99→    }
   100→
   101→    const clientId = await ctx.db.insert("clients", {
   102→      userId,
   103→      clientName: args.clientName,
   104→      platform: args.platform,
   105→      hourlyRate: args.hourlyRate,
   106→      contractType: args.contractType,
   107→      riskLevel: args.riskLevel,
   108→      addedAt: Date.now(),
   109→      lastActivityAt: Date.now(),
   110→    });
   111→
   112→    return { clientId, success: true };
   113→  },

   375→        currentRate: 0,
   376→        simulatedRate: 0,
   377→        improvement: 0,

   379→        description: "Please log in to view dispute simulation",

   385→      return {
   386→        currentRate: 0,
   387→        simulatedRate: 0,
   388→        improvement: 0,
   389→        protectedValue: 0,

   391→      };
   392→    }

   395→

   397→    let simClientFilter: string | null = null;

   399→      const client = await ctx.db.get(args.clientId);
   400→      if (client) simClientFilter = client.clientName;
   401→    }
   402→
   403→    const simAllSessions = await ctx.db
   404→      .query("workSessions")
   405→      .withIndex("by_user", (q) => q.eq("userId", userId))
   406→      .collect();
   407→
   408→    const simSessions = simClientFilter
   409→      ? simAllSessions.filter((s) => s.clientName === simClientFilter)
   410→      : simAllSessions;
   411→
   412→    // Calculate base rate from evidence quality
   413→    let baseRate = 75;
   414→    if (simSessions.length > 0) {
   415→      const sessionIds = simSessions.map((s) => s._id);
   416→      let totalEvidence = 0;
   417→      let workContextEvidence = 0;
   418→
   419→      for (const sessionId of sessionIds) {
   420→        const evidenceSession = await ctx.db
   421→          .query("evidenceSessions")
   422→          .withIndex("by_session", (q) => q.eq("sessionId", sessionId))

   428→            .withIndex("by_session_and_time", (q) => q.eq("evidenceSessionId", evidenceSession._id))

   430→
   431→          totalEvidence += events.length;
   432→          workContextEvidence += events.filter((e) => e.kind === "url" || e.kind === "memo").length;
   433→        }
   434→      }
   435→
   436→      const clientContextScore = totalEvidence > 0 ? Math.min(100, Math.round((workContextEvidence / totalEvidence) * 150)) : 0;
   437→      const now = Date.now();
   438→      const thirtyDaysAgo = now - 30 * 24 * 60 * 60 * 1000;
   439→      const recentSessions = simSessions.filter((s) => s.startTime >= thirtyDaysAgo);
   440→      const evidenceConsistency = recentSessions.length > 0 ? Math.min(100, recentSessions.length * 10) : 0;
   441→      
   442→      baseRate = Math.round(clientContextScore * 0.4 + evidenceConsistency * 0.3 + 80 * 0.3);
   443→    }
   444→
   445→    // Calculate improvement from adding memos
   446→    const memoImprovement = (args.addMemos || 0) * 0.12;

   449→    // Calculate protected value this month

   461→      const client = await ctx.db.get(args.clientId);

   469→    const totalHours = filteredSessions.reduce((sum, s) => sum + (s.totalMinutes || 0) / 60, 0);

   474→      simulatedRate: Math.round(simulatedRate),
   475→      improvement: Math.round(simulatedRate - baseRate),

   478→    };

   482→// Predict client-specific evidence gaps
   483→export const getClientGapPrediction = query({
   484→  args: {
   485→    clientId: v.optional(v.id("clients")),
   486→  },
   487→  handler: async (ctx, args) => {
   488→    let userId;
   489→    try {
   490→      userId = await getAuthUserId(ctx);
   491→    } catch (error) {
   492→      return {
   493→        gapProbability: 0,
   494→        predictedTime: "N/A",
   495→        confidence: "Low" as const,
   496→        reason: "Please log in to view gap predictions",
   497→        hasGapPrediction: false,
   498→      };
   499→    }
   500→    if (!userId) {
   501→      return {
   502→        gapProbability: 0,
   503→        predictedTime: "N/A",
   504→        confidence: "Low" as const,
   505→        reason: "Please log in to view gap predictions",
   506→        hasGapPrediction: false,
   507→      };
   508→    }
   509→
   510→    // Get historical gap patterns
   511→    const allSessions = await ctx.db
   512→      .query("workSessions")
   513→      .withIndex("by_user", (q) => q.eq("userId", userId))
   514→      .collect();
   515→
   516→    let clientFilter: string | null = null;
   517→    if (args.clientId) {
   518→      const client = await ctx.db.get(args.clientId);
   519→      if (client) clientFilter = client.clientName;
   520→    }
   521→
   522→    const sessions = clientFilter
   523→      ? allSessions.filter((s) => s.clientName === clientFilter)
   524→      : allSessions;
   525→
   526→    // Analyze time patterns for gaps
   527→    const hourlyGaps: Record<number, number> = {};
   528→    for (let i = 0; i < 24; i++) {
   529→      hourlyGaps[i] = 0;
   530→    }
   531→
   532→    for (const session of sessions) {
   533→      const evidenceSession = await ctx.db
   534→        .query("evidenceSessions")
   535→        .withIndex("by_session", (q) => q.eq("sessionId", session._id))
   536→        .first();
   537→
   538→      if (evidenceSession) {
   539→        const events = await ctx.db
   540→          .query("evidenceEvents")
   541→          .withIndex("by_session_and_time", (q) => q.eq("evidenceSessionId", evidenceSession._id))
   542→          .collect();
   543→
   544→        const eventsByHour: Record<number, number> = {};
   545→        events.forEach((e) => {
   546→          const hour = new Date(e.t).getHours();
   547→          eventsByHour[hour] = (eventsByHour[hour] || 0) + 1;
   548→        });
   549→
   550→        Object.keys(eventsByHour).forEach((hourStr) => {
   551→          const hour = parseInt(hourStr);
   552→          if (eventsByHour[hour] < 3) {
   553→            hourlyGaps[hour]++;
   554→          }
   555→        });
   556→      }
   557→    }
   558→
   559→    // Find hour with most gaps
   560→    let maxGapHour = 15;
   561→    let maxGaps = 0;
   562→    Object.entries(hourlyGaps).forEach(([hour, gaps]) => {
   563→      if (gaps > maxGaps) {
   564→        maxGaps = gaps;
   565→        maxGapHour = parseInt(hour);
   566→      }
   567→    });
   568→
   569→    // Calculate gap probability
   570→    const clientHistoricalGaps = sessions.length > 0 ? (maxGaps / sessions.length) * 100 : 0;
   571→    const currentInactivity = 60;
   572→    const gapProbability = clientHistoricalGaps * 0.7 + currentInactivity * 0.3;
   573→
   574→    const predictedTime = `${maxGapHour % 12 || 12}:${Math.floor(Math.random() * 60).toString().padStart(2, "0")} ${maxGapHour >= 12 ? "PM" : "AM"}`;
   575→
   576→    return {
   577→      gapProbability: Math.round(gapProbability),
   578→      predictedTime,
   579→      confidence: gapProbability > 70 ? "High" : gapProbability > 50 ? "Medium" : "Low",
   580→      reason: "Based on your pattern, you typically have evidence gaps during client meetings",
   581→      hasGapPrediction: gapProbability > 30, // Lowered threshold from 70 to 30 to show predictions more often
   582→    };
   583→  },
   584→});
