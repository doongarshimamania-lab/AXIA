   128→   128→function formatDate(ts: number | undefined) {
   129→   129→  if (!ts) return "—";
   130→   130→  return new Date(ts).toLocaleDateString("en-US", {
   131→   131→    year: "numeric",
   132→   132→    month: "short",
   133→   133→    day: "numeric",
   134→   134→  });
   135→   135→}
   136→   136→
   137→   137→function formatCurrency(amount: number, currency = "USD") {
   138→   138→  return new Intl.NumberFormat("en-US", {
   139→   139→    style: "currency",
   140→   140→    currency,
   141→   141→    minimumFractionDigits: 0,
   142→   142→    maximumFractionDigits: 2,
   143→   143→  }).format(amount);
   144→   144→}
   145→   145→
   146→   146→function relativeTime(ts: number | undefined) {
   147→   147→  if (!ts) return "Never";
   148→   148→  const diff = Date.now() - ts;
   149→   149→  const mins = Math.floor(diff / 60000);
   150→   150→  if (mins < 1) return "Just now";
   151→   151→  if (mins < 60) return `${mins}m ago`;
   152→   152→  const hours = Math.floor(mins / 60);
   153→   153→  if (hours < 24) return `${hours}h ago`;
   154→   154→  const days = Math.floor(hours / 24);
   155→   155→  if (days < 30) return `${days}d ago`;
   156→   156→  return formatDate(ts);
   157→   157→}

   159→   159→function getInitials(name: string) {
   160→   160→  return name
   161→   161→    .split(" ")
   162→   162→    .map((w) => w[0])
   163→   163→    .join("")
   164→   164→    .toUpperCase()
   165→   165→    .slice(0, 2);
   166→   166→}

   168→   168→function getAvatarColor(name: string) {
   169→   169→  const colors = [
   170→   170→    "bg-violet-500",
   171→   171→    "bg-blue-500",
   172→   172→    "bg-emerald-500",
   173→   173→    "bg-amber-500",
   174→   174→    "bg-rose-500",
   175→   175→    "bg-cyan-500",
   176→   176→    "bg-pink-500",
   177→   177→    "bg-teal-500",
   178→   178→  ];
   179→   179→  let hash = 0;
   180→   180→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   181→   181→  return colors[Math.abs(hash) % colors.length];
   182→   182→}
