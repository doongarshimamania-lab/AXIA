   507→                  } catch {
   508→                    toast.error("Failed to sign out");
   509→                  }
   510→                }} className="w-full text-left">
   511→                  <NavItem icon={LogOut} label="Sign Out" isExpanded={true} />
   512→                </button>
   513→              </motion.div>
   514→            )}
   515→          </AnimatePresence>
   516→          {/* Collapsed theme toggle icon */}
   517→          {!isExpanded && (
   518→            <div className="flex flex-col items-center gap-4 py-2">
   519→              <button
   520→                onClick={(e) => {
   521→                  e.stopPropagation();
   522→                  toggleTheme();
   523→                }}
   524→                title={theme === "dark" ? "Light Mode" : "Dark Mode"}
   525→                className="p-2 hover:bg-sidebar-accent rounded-lg transition-colors"
   526→              >
   527→                {theme === "dark" ? (
   528→                  <Sun className="w-5 h-5 text-sidebar-foreground/60 hover:text-sidebar-foreground" />
   529→                ) : (
   530→                  <Moon className="w-5 h-5 text-sidebar-foreground/60 hover:text-sidebar-foreground" />
   531→                )}
   532→              </button>
   533→            </div>
   534→          )}
   535→        </div>
   536→      </motion.div>
   537→    </>
   538→  );
   539→}
