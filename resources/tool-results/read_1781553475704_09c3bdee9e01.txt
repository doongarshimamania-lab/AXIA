   116→
   117→function getTierLevel(tier: string): number {
   118→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
   119→  return levels[tier] ?? 0;
   120→}
   121→
   122→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   123→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   124→}

   218→export default function EvidenceLibrary() {
   219→  const { tier: subscriptionTier } = useSubscriptionTier();
   220→  const { isAuthenticated } = useConvexAuth();
   221→  const { isDisconnected } = useConvexConnectionState();
   222→
   223→  // ── Workspace Context ──
   224→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   225→  const workspaceId = isConvexConnected ? (activeWorkspaceId as Id<"workspaces">) : undefined;
   226→
   227→  const [viewMode, setViewMode] = useState<ViewType>("date");
   228→
   229→  // ── Library date range (±30 days for monitoring) ──
   230→  const libraryDateRange = useMemo(() => {
   231→    const now = new Date();
   232→    const start = new Date(now);
   233→    start.setDate(start.getDate() - 30);
   234→    const end = new Date(now);
   235→    end.setDate(end.getDate() + 30);
   236→    return { start: start.getTime(), end: end.getTime() };
   237→  }, []);
   238→
   239→  const startOfDay = useMemo(() => new Date().setHours(0, 0, 0, 0), []);
   240→
   241→  // ── Export date range (1 year for comprehensive export) ──
   242→  const [dateFrom, setDateFrom] = useState(() => {
   243→    const d = new Date();
   244→    d.setMonth(d.getMonth() - 3);
   245→    return d.toISOString().split("T")[0];
   246→  });
   247→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   248→
   249→  // ── Convex Queries ──
   250→
   251→  // Library data with ±30 day window for monitoring
   252→  const evidenceData = useQuery(
   253→    api.evidence.library.getEvidenceLibraryData,
   254→    {
   255→      view: viewMode,
   256→      startDate: libraryDateRange.start,
   257→      endDate: libraryDateRange.end,
   258→    }
   259→  );
   260→
   261→  // Export data with full year window for comprehensive exports
   262→  const exportData = useQuery(
   263→    api.evidence.library.getEvidenceLibraryData,
   264→    {
   265→      view: "date" as const,
   266→      startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   267→      endDate: Date.now(),
   268→    }
   269→  ) as any | undefined;
   270→
   271→  const timelineData = useQuery(
   272→    api.evidence.library.getEvidenceTimeline,
   273→    { date: startOfDay }
   274→  );
   275→
   276→  const clients = useQuery(api.clients.crud.getClients, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   277→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   278→
   279→  const isLoading = evidenceData === undefined || timelineData === undefined;
   280→  const exportLoading = exportData === undefined || clients === undefined;
   281→  const timedOut = useQueryTimeout(isLoading, 5000);
   282→  const showLoading = isLoading && !timedOut && !isDisconnected;
   283→
   284→  // ── Safe fallbacks ──
   285→  const safeEvidenceData = evidenceData ?? {
   286→    totalCount: 0,
   287→    disputeSuccessRate: 0,
   288→    contentQualityScore: 0,
   289→    gapPrediction: {
   290→      status: "no_gaps" as const,
   291→      message: "No Gaps Predicted",
   292→      description: "Start collecting evidence to build your protection.",
   293→      nextGapTime: null,
   294→      missingTypes: [],
   295→    },
   296→    evidenceItems: [],
   297→    healthScore: null,
   298→    disputeData: null,
   299→    contentData: null,
   300→  };
   301→
   302→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   303→
   304→  // ── Export State ──
   305→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   306→  const [selectedProject, setSelectedProject] = useState("all");
   307→  const [selectedClient, setSelectedClient] = useState("all");
   308→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   309→  const [isExporting, setIsExporting] = useState(false);
   310→  const [previewOpen, setPreviewOpen] = useState(false);
   311→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   312→
   313→  // ── Derived Export Data ──
   314→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   315→    if (!exportData) return [];
   316→    const items = exportData.evidenceItems ?? [];
   317→    const typeCounts: Record<string, number> = {};
   318→    const typeLabels: Record<string, string> = {
   319→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   320→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   321→    };
   322→    const typeIcons: Record<string, React.ReactNode> = {
   323→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,
   324→      url: <FileJson className="h-4 w-4" />, keyboard: <FileText className="h-4 w-4" />,
   325→      mouse: <FileText className="h-4 w-4" />, platform_status: <Shield className="h-4 w-4" />,
   326→    };
   327→    for (const item of items) {
   328→      const kind = item.type ?? "unknown";
   329→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   330→    }
   331→    return Object.entries(typeCounts).map(([id, count]) => ({
   332→      id,
   333→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   334→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   335→      count,
   336→    }));
   337→  }, [exportData]);
   338→
   339→  const projectOptions = useMemo(() => {
   340→    const all = [{ id: "all", label: "All Projects" }];
   341→    if (scopeDefinitions) {
   342→      for (const scope of scopeDefinitions) {
   343→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   344→      }
   345→    }
   346→    return all;
   347→  }, [scopeDefinitions]);
   348→
   349→  const clientOptions = useMemo(() => {
   350→    const all = [{ id: "all", label: "All Clients" }];
   351→    if (clients) {
   352→      for (const client of clients) {
   353→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   354→      }
   355→    }
   356→    return all;
   357→  }, [clients]);
   358→
   359→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;
   360→  const totalItems = exportData?.totalCount ?? safeEvidenceData.totalCount ?? 0;
   361→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;
   362→
   363→  // Auto-select all evidence types when they load
   364→  useEffect(() => {
   365→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   366→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   367→    }
   368→  }, [evidenceTypes]);
   369→
   370→  const totalEvidenceItems = useMemo(
   371→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   372→    [evidenceTypes, selectedEvidenceTypes]
   373→  );
   374→
   375→  const isFormatLocked = (format: ExportFormat): boolean => {
   376→    return !hasTierAccess(subscriptionTier, FORMAT_CONFIG[format].tierRequired);
   377→  };
   378→
   379→  const toggleEvidenceType = (id: string) => {
   380→    setSelectedEvidenceTypes((prev) =>
   381→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   382→    );
   383→  };
   384→
   385→  const handleExport = async () => {
   386→    if (selectedEvidenceTypes.length === 0) {
   387→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   388→      return;
   389→    }
   390→    if (isFormatLocked(selectedFormat)) {
   391→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   392→      return;
   393→    }
   394→    setIsExporting(true);
   395→    try {
   396→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   397→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   398→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   399→
   400→      exportEvidence(selectedFormat, {
   401→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   402→        dateRange: `${dateFrom} to ${dateTo}`,
   403→        project: selectedProjectLabel,
   404→        client: selectedClientLabel,
   405→        complianceScore,
   406→        totalItems: totalEvidenceItems,
   407→      });
   408→
   409→      const newExport: RecentExport = {
   410→        id: `exp_${Date.now()}`,
   411→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   412→        format: selectedFormat,
   413→        status: "completed",
   414→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   415→        date: new Date().toISOString().split("T")[0],
   416→        itemCount: totalEvidenceItems,
   417→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   418→        project: selectedProjectLabel,
   419→      };
   420→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   421→
   422→      toast.success("Export complete", {
   423→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   424→      });
   425→
   426→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   427→    } catch {
   428→      toast.error("Export failed", { description: "An error occurred while generating the export file." });
   429→    }
   430→    setIsExporting(false);
   431→  };
   432→
   433→  const handleFormatSelect = (format: ExportFormat) => {
   434→    if (isFormatLocked(format)) {
   435→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.` });
   436→      return;
   437→    }
   438→    setSelectedFormat(format);
   439→  };
   440→
   441→  const previewItems = useMemo(() => {
   442→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   443→  }, [evidenceTypes, selectedEvidenceTypes]);
   444→
   445→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   446→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   447→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   448→  const hasTierAccessFn = (requiredTier: string) => getTierLevel(subscriptionTier) >= getTierLevel(requiredTier);
   449→
   450→  // ── Loading / Timeout State ──
   451→  const [showTimeout, setShowTimeout] = useState(false);
   452→  useEffect(() => {
   453→    if (isLoading) {
   454→      const timer = setTimeout(() => setShowTimeout(true), 5000);
   455→      return () => clearTimeout(timer);
   456→    }
   457→    setShowTimeout(false);
   458→  }, [isLoading]);
   459→
   460→  return (
   461→    <div className="w-full min-h-screen bg-background">
   462→      <PageLayout spaced>
   463→        {/* ── Page Header ── */}
   464→        <div>
   465→          <h1 className="text-2xl md:text-[32px] font-bold text-foreground tracking-tight mb-2">
   466→            Evidence Library
   467→          </h1>
   468→          <p className="text-[16px] text-muted-foreground">
   469→            Monitor capture health, anticipate gaps early, and export airtight proof on demand.
   470→          </p>
   471→        </div>
   472→
   473→        {/* Demo Mode Banner */}
   474→        {!isAuthenticated && <DemoModeBanner />}
   475→
   476→        {/* ════════════════════════════════════════════════════════════════════════
   477→            SECTION 1: EVIDENCE MONITORING (was Evidence Library)
   478→            - Health dashboard, timeline, quality scorecard, item browser
   479→        ════════════════════════════════════════════════════════════════════════ */}
   480→        <section id="monitoring">
   481→          <div className="flex items-center gap-3 mb-4">
   482→            <div className="p-2 rounded-lg bg-primary/10">
   483→              <DatabaseIcon className="w-5 h-5 text-primary" />
   484→            </div>
   485→            <div>
   486→              <h2 className="text-xl font-semibold text-foreground">Evidence Monitoring</h2>
   487→              <p className="text-sm text-muted-foreground">Real-time health, quality analysis, and gap detection</p>
   488→            </div>
   489→          </div>
   490→
   491→          {showLoading ? (
   492→            <div className="text-center py-12">
   493→              <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary mx-auto"></div>
   494→              <p className="mt-4 text-muted-foreground">Loading evidence data...</p>
   495→              {showTimeout && (
   496→                <div className="mt-4 p-4 bg-yellow-50 dark:bg-yellow-900/20 border border-yellow-200 dark:border-yellow-800 rounded-lg">
   497→                  <p className="text-sm text-yellow-800 dark:text-yellow-200">
   498→                    Taking longer than expected. Check your connection or try refreshing the page.
   499→                  </p>
   500→                </div>
   501→              )}
   502→            </div>
   503→          ) : (
   504→            <div className="space-y-6">
   505→              <WorkContentAnalysis
   506→                qualityScore={safeEvidenceData.contentQualityScore}
   507→                hasAccess={hasTierAccessFn("pro")}
   508→                contentData={safeEvidenceData.contentData}
   509→              />
   510→
   511→              <EvidenceTimeline timeline={safeTimelineData.timeline} />
   512→
   513→              <EvidenceQualityScorecard
   514→                hasAccess={hasTierAccessFn("pro")}
   515→                healthScore={safeEvidenceData.healthScore}
   516→              />
   517→
   518→              <TeamValidation hasAccess={hasTierAccessFn("expert")} />
   519→
   520→              <EvidenceItemsList
   521→                evidenceItems={safeEvidenceData.evidenceItems}
   522→                viewMode={viewMode}
   523→                setViewMode={setViewMode}
   524→              />
   525→            </div>
   526→          )}
   527→        </section>
   528→
