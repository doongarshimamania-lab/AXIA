    98→const formatDate = (timestamp: number) => {
    99→  const date = new Date(timestamp);
   100→  return date.toLocaleDateString("en-US", {
   101→    month: "short",
   102→    day: "numeric",
   103→    year: "numeric",
   104→  });
   105→};
   106→
   107→const daysUntil = (timestamp: number) => {
   108→  const now = new Date();
   109→  now.setHours(0, 0, 0, 0);
   110→  const target = new Date(timestamp);
   111→  target.setHours(0, 0, 0, 0);
   112→  const diff = Math.ceil((target.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   113→  return diff;
   114→};

   680→                                      variant="ghost"
   681→                                      size="icon"

   695→                                    variant="ghost"
   696→                                    size="icon"

   704→                                    variant="ghost"
   705→                                    size="icon"
   706→                                    className="h-8 w-8 text-destructive hover:text-destructive"
   707→                                    onClick={() => openDelete(goal)}
   708→                                    title="Delete goal"
   709→                                  >
   710→                                    <Trash2 className="h-4 w-4" />

   713→                              </div>
   714→
   715→                              {/* Description */}
   716→                              {goal.description && (
   717→                                <p className="text-sm text-muted-foreground line-clamp-2">
   718→                                  {goal.description}
   719→                                </p>

   721→
   722→                              {/* Progress bar */}
   723→                              <div className="space-y-1.5">
   724→                                <div className="flex items-center justify-between text-sm">

   734→                              {totalMilestones > 0 && (

   738→                                  </p>
   739→                                  <div className="flex flex-wrap gap-2">

   744→                                        onClick={() => handleToggleMilestone(goal._id, ms.id, ms.completed)}

   763→                              {/* Bottom meta row */}

   800→                            </div>
   801→                          </div>
   802→                        </CardContent>
   803→                      </Card>
   804→                    </motion.div>

   806→                })}
   807→              </div>
   808→            )}
   809→          </>

   816→        onOpenChange={(open) => {

   821→          }
   822→        }}
   823→      >
   824→        <DialogContent className="sm:max-w-lg">

   829→          <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
