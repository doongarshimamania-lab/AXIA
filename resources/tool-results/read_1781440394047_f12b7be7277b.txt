   117→   117→// --- Helpers ---
   118→   118→function formatDate(ts: number): string {
   119→   119→  return new Date(ts).toLocaleDateString("en-US", {
   120→   120→    year: "numeric",
   121→   121→    month: "short",
   122→   122→    day: "numeric",
   123→   123→  });
   124→   124→}
   125→   125→
   126→   126→function timeAgo(ts: number | null): string {
   127→   127→  if (!ts) return "Never";
   128→   128→  const diff = Date.now() - ts;
   129→   129→  const mins = Math.floor(diff / 60000);
   130→   130→  if (mins < 1) return "Just now";
   131→   131→  if (mins < 60) return `${mins}m ago`;
   132→   132→  const hrs = Math.floor(mins / 60);
   133→   133→  if (hrs < 24) return `${hrs}h ago`;
   134→   134→  const days = Math.floor(hrs / 24);
   135→   135→  return `${days}d ago`;
   136→   136→}

   138→   138→function generateMockKey(prefix: string = "tl_prod_"): string {
   139→   139→  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
   140→   140→  let result = prefix + "sk_";
   141→   141→  for (let i = 0; i < 20; i++) {
   142→   142→    result += chars.charAt(Math.floor(Math.random() * chars.length));
   143→   143→  }
   144→   144→  return result;
   145→   145→}

   147→   147→function maskKey(key: string): string {
   148→   148→  const prefix = key.substring(0, key.indexOf("sk_") + 3);
   149→   149→  const suffix = key.slice(-4);
   150→   150→  return `${prefix}${"•".repeat(16)}${suffix}`;
   151→   151→}
