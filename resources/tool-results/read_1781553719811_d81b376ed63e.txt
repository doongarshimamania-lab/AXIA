    24→function formatTime(seconds: number) {
    25→  const h = Math.floor(seconds / 3600);
    26→  const m = Math.floor((seconds % 3600) / 60);
    27→  const s = seconds % 60;
    28→  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    29→}
    30→
    31→function formatDuration(minutes: number | undefined) {
    32→  if (!minutes) return "0h 0m";
    33→  const hours = Math.floor(minutes / 60);
    34→  const mins = minutes % 60;
    35→  return `${hours}h ${mins}m`;
    36→}
    37→
    38→function formatDate(ts: number) {
    39→  return new Date(ts).toLocaleDateString("en-US", {
    40→    month: "short",
    41→    day: "numeric",
    42→    year: "numeric",
    43→  });
    44→}
