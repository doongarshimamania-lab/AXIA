   117→function getTierLevel(tier: string): number {
   118→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
   119→  return levels[tier] ?? 0;
   120→}
   121→
   122→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   123→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   124→}

   218→export default function EvidenceLibrary() {
   219→  const { tier: subscriptionTier } = useSubscriptionTier();
   220→  const { isAuthenticated } = useConvexAuth();
   221→  const { isDisconnected } = useConvexConnectionState();
   222→
   223→  // ── Workspace Context ──
   224→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   225→  const workspaceId = isConvexConnected ? (activeWorkspaceId as Id<"workspaces">) : undefined;
   226→
   227→  const [viewMode, setViewMode] = useState<ViewType>("date");
   228→
   229→  // ── Library date range (±30 days for monitoring) ──
   230→  const libraryDateRange = useMemo(() => {
   231→    const now = new Date();
   232→    const start = new Date(now);
   233→    start.setDate(start.getDate() - 30);
   234→    const end = new Date(now);
   235→    end.setDate(end.getDate() + 30);
   236→    return { start: start.getTime(), end: end.getTime() };
   237→  }, []);
   238→
   239→  const startOfDay = useMemo(() => new Date().setHours(0, 0, 0, 0), []);
   240→
   241→  // ── Export date range (1 year for comprehensive export) ──
   242→  const [dateFrom, setDateFrom] = useState(() => {
   243→    const d = new Date();
   244→    d.setMonth(d.getMonth() - 3);
   245→    return d.toISOString().split("T")[0];
   246→  });
   247→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   248→
   249→  // ── Convex Queries ──
   250→  // Safely access evidence API — it may not exist in local Convex deployment
   251→  const evidenceApi = (api as any).evidence?.library;
   252→  const hasEvidenceApi = !!(evidenceApi?.getEvidenceLibraryData);
   253→  const hasTimelineApi = !!(evidenceApi?.getEvidenceTimeline);
   254→
   255→  // Library data with ±30 day window for monitoring
   256→  const evidenceData = useQuery(
   257→    hasEvidenceApi ? evidenceApi.getEvidenceLibraryData : "skip",
   258→    hasEvidenceApi ? {
   259→      view: viewMode,
   260→      startDate: libraryDateRange.start,
   261→      endDate: libraryDateRange.end,
   262→    } : "skip"
   263→  );
   264→
   265→  // Export data with full year window for comprehensive exports
   266→  const exportData = useQuery(
   267→    hasEvidenceApi ? evidenceApi.getEvidenceLibraryData : "skip",
   268→    hasEvidenceApi ? {
   269→      view: "date" as const,
   270→      startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   271→      endDate: Date.now(),
   272→    } : "skip"
   273→  ) as any | undefined;
   274→
   275→  const timelineData = useQuery(
   276→    hasTimelineApi ? evidenceApi.getEvidenceTimeline : "skip",
   277→    hasTimelineApi ? { date: startOfDay } : "skip"
   278→  );
   279→
   280→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   281→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   282→
   283→  const isLoading = evidenceData === undefined || timelineData === undefined;
   284→  const exportLoading = exportData === undefined || clients === undefined;
   285→  const timedOut = useQueryTimeout(isLoading, 5000);
   286→  const showLoading = isLoading && !timedOut && !isDisconnected;
   287→
   288→  // ── Safe fallbacks ──
   289→  const safeEvidenceData = evidenceData ?? {
   290→    totalCount: 0,
   291→    disputeSuccessRate: 0,
   292→    contentQualityScore: 0,
   293→    gapPrediction: {
   294→      status: "no_gaps" as const,
   295→      message: "No Gaps Predicted",
   296→      description: "Start collecting evidence to build your protection.",
   297→      nextGapTime: null,
   298→      missingTypes: [],
   299→    },
   300→    evidenceItems: [],
   301→    healthScore: null,
   302→    disputeData: null,
   303→    contentData: null,
   304→  };
   305→
   306→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   307→
   308→  // ── Export State ──
   309→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   310→  const [selectedProject, setSelectedProject] = useState("all");
   311→  const [selectedClient, setSelectedClient] = useState("all");
   312→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   313→  const [isExporting, setIsExporting] = useState(false);
   314→  const [previewOpen, setPreviewOpen] = useState(false);
   315→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   316→
   317→  // ── Derived Export Data ──
   318→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   319→    if (!exportData) return [];
   320→    const items = exportData.evidenceItems ?? [];
   321→    const typeCounts: Record<string, number> = {};
   322→    const typeLabels: Record<string, string> = {
   323→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   324→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   325→    };
   326→    const typeIcons: Record<string, React.ReactNode> = {
   327→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,
   328→      url: <FileJson className="h-4 w-4" />, keyboard: <FileText className="h-4 w-4" />,
   329→      mouse: <FileText className="h-4 w-4" />, platform_status: <Shield className="h-4 w-4" />,
   330→    };
   331→    for (const item of items) {
   332→      const kind = item.type ?? "unknown";
   333→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   334→    }
   335→    return Object.entries(typeCounts).map(([id, count]) => ({
   336→      id,
   337→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   338→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   339→      count,
   340→    }));
   341→  }, [exportData]);
   342→
   343→  const projectOptions = useMemo(() => {
   344→    const all = [{ id: "all", label: "All Projects" }];
   345→    if (scopeDefinitions) {
   346→      for (const scope of scopeDefinitions) {
   347→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   348→      }
   349→    }
   350→    return all;
   351→  }, [scopeDefinitions]);
   352→
   353→  const clientOptions = useMemo(() => {
   354→    const all = [{ id: "all", label: "All Clients" }];
   355→    if (clients) {
   356→      for (const client of clients) {
   357→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   358→      }
   359→    }
   360→    return all;
   361→  }, [clients]);
   362→
   363→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;
   364→  const totalItems = exportData?.totalCount ?? safeEvidenceData.totalCount ?? 0;
   365→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;
   366→
   367→  // Auto-select all evidence types when they load
   368→  useEffect(() => {
   369→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   370→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   371→    }
   372→  }, [evidenceTypes]);
   373→
   374→  const totalEvidenceItems = useMemo(
   375→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   376→    [evidenceTypes, selectedEvidenceTypes]
   377→  );
   378→
   379→  const isFormatLocked = (format: ExportFormat): boolean => {
   380→    return !hasTierAccess(subscriptionTier, FORMAT_CONFIG[format].tierRequired);
   381→  };
   382→
   383→  const toggleEvidenceType = (id: string) => {
   384→    setSelectedEvidenceTypes((prev) =>
   385→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   386→    );
   387→  };
   388→
   389→  const handleExport = async () => {
   390→    if (selectedEvidenceTypes.length === 0) {
   391→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   392→      return;
   393→    }
   394→    if (isFormatLocked(selectedFormat)) {
   395→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   396→      return;
   397→    }
   398→    setIsExporting(true);
   399→    try {
   400→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   401→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   402→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   403→
   404→      exportEvidence(selectedFormat, {
   405→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   406→        dateRange: `${dateFrom} to ${dateTo}`,
   407→        project: selectedProjectLabel,
   408→        client: selectedClientLabel,
   409→        complianceScore,
   410→        totalItems: totalEvidenceItems,
   411→      });
   412→
   413→      const newExport: RecentExport = {
   414→        id: `exp_${Date.now()}`,
   415→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   416→        format: selectedFormat,
   417→        status: "completed",
   418→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   419→        date: new Date().toISOString().split("T")[0],
   420→        itemCount: totalEvidenceItems,
   421→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   422→        project: selectedProjectLabel,
   423→      };
   424→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   425→
   426→      toast.success("Export complete", {
   427→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   428→      });
   429→
   430→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   431→    } catch {
   432→      toast.error("Export failed", { description: "An error occurred while generating the export file." });
   433→    }
   434→    setIsExporting(false);
   435→  };
   436→
   437→  const handleFormatSelect = (format: ExportFormat) => {
   438→    if (isFormatLocked(format)) {
   439→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.` });
   440→      return;
   441→    }
   442→    setSelectedFormat(format);
   443→  };
   444→
   445→  const previewItems = useMemo(() => {
   446→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   447→  }, [evidenceTypes, selectedEvidenceTypes]);
   448→
   449→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   450→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   451→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   452→  const hasTierAccessFn = (requiredTier: string) => getTierLevel(subscriptionTier) >= getTierLevel(requiredTier);
   453→
   454→  // ── Loading / Timeout State ──
   455→  const [showTimeout, setShowTimeout] = useState(false);
   456→  useEffect(() => {
   457→    if (isLoading) {
   458→      const timer = setTimeout(() => setShowTimeout(true), 5000);
   459→      return () => clearTimeout(timer);
   460→    }
   461→    setShowTimeout(false);
   462→  }, [isLoading]);
   463→
   464→  return (
   465→    <div className="w-full min-h-screen bg-background">
   466→      <PageLayout spaced>
   467→        {/* ── Page Header ── */}
   468→        <div>
   469→          <h1 className="text-2xl md:text-[32px] font-bold text-foreground tracking-tight mb-2">
   470→            Evidence Library
   471→          </h1>
   472→          <p className="text-[16px] text-muted-foreground">
   473→            Monitor capture health, anticipate gaps early, and export airtight proof on demand.
   474→          </p>
   475→        </div>
   476→
   477→        {/* Demo Mode Banner */}
   478→        {!isAuthenticated && <DemoModeBanner />}
   479→
   480→        {/* ════════════════════════════════════════════════════════════════════════
   481→            SECTION 1: EVIDENCE MONITORING (was Evidence Library)
   482→            - Health dashboard, timeline, quality scorecard, item browser
   483→        ════════════════════════════════════════════════════════════════════════ */}
   484→        <section id="monitoring">
   485→          <div className="flex items-center gap-3 mb-4">
   486→            <div className="p-2 rounded-lg bg-primary/10">
   487→              <DatabaseIcon className="w-5 h-5 text-primary" />
   488→            </div>
   489→            <div>
   490→              <h2 className="text-xl font-semibold text-foreground">Evidence Monitoring</h2>
   491→              <p className="text-sm text-muted-foreground">Real-time health, quality analysis, and gap detection</p>
   492→            </div>
   493→          </div>
   494→
   495→          {showLoading ? (
   496→            <div className="text-center py-12">
   497→              <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary mx-auto"></div>
   498→              <p className="mt-4 text-muted-foreground">Loading evidence data...</p>
   499→              {showTimeout && (
   500→                <div className="mt-4 p-4 bg-yellow-50 dark:bg-yellow-900/20 border border-yellow-200 dark:border-yellow-800 rounded-lg">
   501→                  <p className="text-sm text-yellow-800 dark:text-yellow-200">
   502→                    Taking longer than expected. Check your connection or try refreshing the page.
   503→                  </p>
   504→                </div>
   505→              )}
   506→            </div>
   507→          ) : (
   508→            <div className="space-y-6">
   509→              <WorkContentAnalysis
   510→                qualityScore={safeEvidenceData.contentQualityScore}
   511→                hasAccess={hasTierAccessFn("pro")}
   512→                contentData={safeEvidenceData.contentData}
   513→              />
   514→
   515→              <EvidenceTimeline timeline={safeTimelineData.timeline} />
   516→
   517→              <EvidenceQualityScorecard
   518→                hasAccess={hasTierAccessFn("pro")}
   519→                healthScore={safeEvidenceData.healthScore}
   520→              />
   521→
   522→              <TeamValidation hasAccess={hasTierAccessFn("expert")} />
