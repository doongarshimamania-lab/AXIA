    10→export const getProposals = query({
    11→  args: {
    12→    workspaceId: v.optional(v.id("workspaces")),
    13→    status: v.optional(v.string()),
    14→  },
    15→  handler: async (ctx, { workspaceId, status }) => {
    16→    const userId = await getAuthUserId(ctx);
    17→    if (!userId) return [];
    18→
    19→    if (workspaceId) {
    20→      const visibility = await getUserVisibility(ctx, workspaceId);
    21→      if (!visibility) return [];
    22→
    23→      const allProposals = await ctx.db
    24→        .query("proposals")
    25→        .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
    26→        .collect();
    27→
    28→      let visible = allProposals.filter((p) => isRecordVisible(p, visibility));
    29→      if (status) visible = visible.filter((p) => p.status === status);
    30→      return visible;
    31→    }
    32→
    33→    // Backward compat: no workspaceId
    34→    if (status) {
    35→      return await ctx.db
    36→        .query("proposals")
    37→        .withIndex("by_user_and_status", (q) => q.eq("userId", userId).eq("status", status as any))
    38→        .order("desc")
    39→        .collect();
    40→    }
    41→    return await ctx.db
    42→      .query("proposals")
    43→      .withIndex("by_user", (q) => q.eq("userId", userId))
    44→      .order("desc")
    45→      .collect();
    46→  },
    47→});

    78→export const getTemplates = query({
    79→  args: { workspaceId: v.optional(v.id("workspaces")) },
    80→  handler: async (ctx, { workspaceId }) => {
    81→    const userId = await getAuthUserId(ctx);
    82→
    83→    const systemTemplates = await ctx.db
    84→      .query("proposalTemplates")
    85→      .withIndex("by_system", (q) => q.eq("isSystem", true))
    86→      .collect();
    87→
    88→    let userTemplates: any[] = [];
    89→    if (userId) {
    90→      if (workspaceId) {
    91→        userTemplates = await ctx.db
    92→          .query("proposalTemplates")
    93→          .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
    94→          .collect();
    95→        // Also get user's own templates
    96→        const personalTemplates = await ctx.db
    97→          .query("proposalTemplates")
    98→          .withIndex("by_user", (q) => q.eq("userId", userId))
    99→          .collect();
   100→        // Merge and deduplicate
   101→        const ids = new Set(userTemplates.map((t: any) => t._id));
   102→        for (const t of personalTemplates) {
   103→          if (!ids.has(t._id)) userTemplates.push(t);
   104→        }
   105→      } else {
   106→        userTemplates = await ctx.db
   107→          .query("proposalTemplates")
   108→          .withIndex("by_user", (q) => q.eq("userId", userId))
   109→          .collect();
   110→      }
   111→    }
   112→
   113→    return [...systemTemplates, ...userTemplates];
   114→  },

   117→export const getProposalStats = query({
   118→  args: { workspaceId: v.optional(v.id("workspaces")) },
   119→  handler: async (ctx, { workspaceId }) => {
   120→    const userId = await getAuthUserId(ctx);
   121→    if (!userId) return { total: 0, sent: 0, signed: 0, draft: 0, signatureRate: 0, totalValue: 0 };
   122→
   123→    let proposals;
   124→    if (workspaceId) {
   125→      const visibility = await getUserVisibility(ctx, workspaceId);
   126→      if (!visibility) return { total: 0, sent: 0, signed: 0, draft: 0, signatureRate: 0, totalValue: 0 };
   127→
   128→      const allProposals = await ctx.db
   129→        .query("proposals")
   130→        .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
   131→        .collect();
   132→      proposals = allProposals.filter((p) => isRecordVisible(p, visibility));
   133→    } else {
   134→      proposals = await ctx.db
   135→        .query("proposals")
   136→        .withIndex("by_user", (q) => q.eq("userId", userId))
   137→        .collect();
   138→    }
   139→
   140→    const sent = proposals.filter(p => p.status === "sent" || p.status === "viewed").length;
   141→    const signed = proposals.filter(p => p.status === "signed").length;
   142→    const draft = proposals.filter(p => p.status === "draft").length;
   143→    const totalValue = proposals.filter(p => p.status === "signed").reduce((sum, p) => sum + p.totalValue, 0);
   144→
   145→    return {
   146→      total: proposals.length,
   147→      sent,
   148→      signed,
   149→      draft,
   150→      signatureRate: sent + signed > 0 ? Math.round((signed / (sent + signed)) * 100) : 0,
   151→      totalValue,
   152→    };
   153→  },
   154→});

   168→function generateToken(): string {
   169→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
   170→  let result = "";
   171→  for (let i = 0; i < 32; i++) {
   172→    result += chars.charAt(Math.floor(Math.random() * chars.length));
   173→  }
   174→  return result;
   175→}

   325→export const signProposal = mutation({
   326→  args: { publicToken: v.string(), signatureData: v.string() },
   327→  handler: async (ctx, { publicToken, signatureData }) => {
   328→    // This is a public action via shared link — client signs without being a platform user
   329→    const proposal = await ctx.db
   330→      .query("proposals")
   331→      .withIndex("by_public_token", (q) => q.eq("publicToken", publicToken))
   332→      .first();
   333→
   334→    if (!proposal) throw new Error("Proposal not found");
   335→    if (proposal.status !== "sent" && proposal.status !== "viewed") {
   336→      throw new Error("Proposal is not in a signable state");
   337→    }
   338→    if (proposal.status === "signed") {
   339→      throw new Error("Proposal has already been signed");
   340→    }
   341→
   342→    await ctx.db.patch(proposal._id, {
   343→      status: "signed",
   344→      signedAt: Date.now(),
   345→      signatureData,
   346→      updatedAt: Date.now(),
   347→    });
   348→
   349→    // Cancel remaining follow-ups
   350→    const followUps = await ctx.db
   351→      .query("proposalFollowUps")
   352→      .withIndex("by_proposal", (q) => q.eq("proposalId", proposal._id))
   353→      .collect();
   354→
   355→    for (const fu of followUps) {
   356→      if (fu.status === "scheduled") {
   357→        await ctx.db.patch(fu._id, { status: "cancelled" });
   358→      }
   359→    }
   360→  },
   361→});

   545→function getFollowUpSubject(dayNumber, title) {
   546→  if (dayNumber <= 3) return `Following up: ${title}`;
   547→  if (dayNumber <= 10) return `Checking in: ${title}`;
   548→  return `Final reminder: ${title}`;
   549→}
   550→
   551→function getFollowUpBody(dayNumber, title, clientName, sentAt) {
   552→  const name = clientName || "there";
   553→  const sentDate = new Date(sentAt).toLocaleDateString();
   554→  if (dayNumber <= 3) {
   555→    return `Hi ${name},\n\nI wanted to follow up on the proposal "${title}" I sent on ${sentDate}.\n\nI'd love to hear your thoughts and answer any questions you might have.\n\nBest regards`;
   556→  }
   557→  if (dayNumber <= 10) {
   558→    return `Hi ${name},\n\nI'm checking in on the proposal "${title}" I sent on ${sentDate}.\n\nPlease let me know if you need any additional information or would like to discuss any adjustments.\n\nBest regards`;
   559→  }
   560→  return `Hi ${name},\n\nThis is a final reminder regarding the proposal "${title}" I sent on ${sentDate}.\n\nThis proposal will expire soon. Please let me know if you have any questions or if you'd like to proceed.\n\nBest regards`;
   561→}

   563→export const startFollowUps = mutation({
   564→  args: {
   565→    proposalId: v.id("proposals"),
   566→    intervals: v.optional(v.array(v.number())),
   567→  },
   568→  handler: async (ctx, { proposalId, intervals }) => {
   569→    const userId = await getAuthUserId(ctx);
   570→    if (!userId) throw new Error("Not authenticated");
   571→
   572→    const proposal = await ctx.db.get(proposalId);
   573→    if (!proposal) throw new Error("Proposal not found");
   574→
   575→    // Check workspace access or direct ownership
   576→    if (proposal.workspaceId) {
   577→      const access = await getRecordAccess(ctx, proposal, userId);
   578→      if (!access) throw new Error("Not authorized");
   579→    } else if (proposal.userId !== userId) {
   580→      throw new Error("Not authorized");
   581→    }
   582→
   583→    // Only allow for proposals with status "sent" or "viewed"
   584→    if (proposal.status !== "sent" && proposal.status !== "viewed") {
   585→      throw new Error("Follow-ups can only be started for sent or viewed proposals");
   586→    }
   587→
   588→    // Cancel any existing scheduled follow-ups for this proposal
   589→    const existingFollowUps = await ctx.db
   590→      .query("proposalFollowUps")
   591→      .withIndex("by_proposal", (q) => q.eq("proposalId", proposalId))
   592→      .collect();
   593→
   594→    for (const fu of existingFollowUps) {
   595→      if (fu.status === "scheduled") {
   596→        await ctx.db.patch(fu._id, { status: "cancelled" });
   597→      }
   598→    }
   599→
   600→    // Create new follow-ups based on provided intervals or defaults
   601→    const dayIntervals = intervals || [3, 7, 14];
   602→    const sentAt = proposal.sentAt || Date.now();
   603→
   604→    for (const dayNumber of dayIntervals) {
   605→      await ctx.db.insert("proposalFollowUps", {
   606→        userId: proposal.userId,
   607→        workspaceId: proposal.workspaceId ?? undefined,
   608→        proposalId,
   609→        dayNumber,
   610→        subject: getFollowUpSubject(dayNumber, proposal.title),
   611→        body: getFollowUpBody(dayNumber, proposal.title, proposal.clientName, sentAt),
   612→        channel: "email",
   613→        status: "scheduled",
   614→        scheduledAt: sentAt + dayNumber * 24 * 60 * 60 * 1000,
   615→        createdAt: Date.now(),
   616→      });
   617→    }
   618→
   619→    return { success: true, scheduledCount: dayIntervals.length };
   620→  },
   621→});
