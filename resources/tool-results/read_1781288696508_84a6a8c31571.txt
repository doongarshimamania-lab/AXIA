   170→function statusColor(status: string): "default" | "secondary" | "destructive" | "outline" {
   171→  if (["won", "signed", "paid", "completed"].includes(status)) return "default";
   172→  if (["lost", "declined", "overdue", "expired", "rejected"].includes(status)) return "destructive";
   173→  return "secondary";
   174→}

   176→function statusIcon(type: string, status: string) {
   177→  if (type === "deal") {
   178→    if (status === "won") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   179→    if (status === "lost") return <AlertCircle className="h-4 w-4 text-red-500" />;
   180→    return <TrendingUp className="h-4 w-4 text-blue-500" />;
   181→  }
   182→  if (type === "proposal") {
   183→    if (status === "signed") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   184→    if (status === "sent" || status === "viewed") return <Send className="h-4 w-4 text-blue-500" />;
   185→    return <FileText className="h-4 w-4 text-muted-foreground" />;
   186→  }
   187→  // invoice
   188→  if (status === "paid") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   189→  if (status === "overdue") return <AlertCircle className="h-4 w-4 text-red-500" />;
   190→  return <DollarSign className="h-4 w-4 text-amber-500" />;
   191→}

   193→// ─── Format currency ─────────────────────────────────────────────────────
   194→function fmtCurrency(n: number): string {
   195→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
   196→}
   197→
   198→function fmtRelative(ts: number): string {
   199→  const diff = Date.now() - ts;
   200→  const mins = Math.floor(diff / 60000);
   201→  if (mins < 1) return "just now";
   202→  if (mins < 60) return `${mins}m ago`;
   203→  const hrs = Math.floor(mins / 60);
   204→  if (hrs < 24) return `${hrs}h ago`;
   205→  const days = Math.floor(hrs / 24);
   206→  if (days < 30) return `${days}d ago`;
   207→  return new Date(ts).toLocaleDateString();
   208→}
   209→
   210→function fmtCompactCurrency(n: number): string {
   211→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
   212→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
   213→  return fmtCurrency(n);
   214→}

   216→// ─── Metric trend indicator ──────────────────────────────────────────────
   217→function TrendIndicator({ value, label }: { value: number; label: string }) {
   218→  if (value > 0) return (
   219→    <span className="flex items-center gap-0.5 text-emerald-600 dark:text-emerald-400 text-xs">
   220→      <ArrowUpRight className="h-3 w-3" /> +{value}% {label}
   221→    </span>
   222→  );
   223→  if (value < 0) return (
   224→    <span className="flex items-center gap-0.5 text-red-600 dark:text-red-400 text-xs">
   225→      <ArrowDownRight className="h-3 w-3" /> {value}% {label}
   226→    </span>
   227→  );
   228→  return (
   229→    <span className="flex items-center gap-0.5 text-muted-foreground text-xs">
   230→      <Minus className="h-3 w-3" /> No change
   231→    </span>
   232→  );
   233→}
