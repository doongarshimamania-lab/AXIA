   133→   133→function formatCurrency(amount: number, currency: string = "USD"): string {
   134→   134→  try {
   135→   135→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   136→   136→  } catch {
   137→   137→    return `$${amount.toFixed(2)}`;
   138→   138→  }
   139→   139→}
   140→   140→
   141→   141→function formatDate(ts: number): string {
   142→   142→  return new Date(ts).toLocaleDateString("en-US", {
   143→   143→    month: "short",
   144→   144→    day: "numeric",
   145→   145→    year: "numeric",
   146→   146→  });
   147→   147→}
   148→   148→
   149→   149→function daysOverdue(dueDate: number): number {
   150→   150→  const now = Date.now();
   151→   151→  return Math.max(0, Math.floor((now - dueDate) / (1000 * 60 * 60 * 24)));
   152→   152→}
