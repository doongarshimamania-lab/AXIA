    40→// 1. CREATE
    41→// ─────────────────────────────────────────────
    42→export const create = mutation({
    43→  args: {
    44→    workspaceId: v.optional(v.id("workspaces")),
    45→    teamId: v.optional(v.id("teams")),
    46→    clientId: v.id("clients"),
    47→    dealId: v.optional(v.id("deals")),
    48→    title: v.string(),
    49→    content: v.array(contentSectionValidator),
    50→    templateId: v.optional(v.id("proposalTemplates")),
    51→    totalValue: v.optional(v.number()),
    52→    currency: v.optional(v.string()),
    53→    validUntil: v.optional(v.number()),
    54→    customFields: v.optional(v.any()),
    55→  },
    56→  handler: async (ctx, args) => {
    57→    const userId = await getAuthUserId(ctx);
    58→    if (!userId) throw new ConvexError("Not authenticated");
    59→
    60→    // If workspaceId provided, verify membership
    61→    if (args.workspaceId) {
    62→      await requireWorkspaceAccess(ctx, args.workspaceId, "member");
    63→    }
    64→
    65→    // Verify client access
    66→    const client = await ctx.db.get(args.clientId);
    67→    if (!client) throw new ConvexError("Client not found");
    68→    if (client.workspaceId) {
    69→      const access = await getRecordAccess(ctx, client, userId);
    70→      if (!access) throw new ConvexError("Not authorized: client belongs to another workspace");
    71→    } else if (client.userId !== userId) {
    72→      throw new ConvexError("Not authorized: client belongs to another user");
    73→    }
    74→
    75→    // Verify deal access if provided
    76→    if (args.dealId) {
    77→      const deal = await ctx.db.get(args.dealId);
    78→      if (!deal) throw new ConvexError("Deal not found");
    79→      if (deal.workspaceId) {
    80→        const access = await getRecordAccess(ctx, deal, userId);
    81→        if (!access) throw new ConvexError("Not authorized: deal belongs to another workspace");
    82→      } else if (deal.userId !== userId) {
    83→        throw new ConvexError("Not authorized: deal belongs to another user");
    84→      }
    85→    }
    86→
    87→    // Verify template exists if provided
    88→    if (args.templateId) {
    89→      const template = await ctx.db.get(args.templateId);
    90→      if (!template) throw new ConvexError("Template not found");
    91→    }
    92→
    93→    // Validate title is not empty
    94→    if (args.title.trim().length === 0) {
    95→      throw new ConvexError("Title cannot be empty");
    96→    }
    97→
    98→    // Validate totalValue is non-negative if provided
    99→    if (args.totalValue !== undefined && args.totalValue < 0) {
   100→      throw new ConvexError("Total value cannot be negative");
   101→    }
   102→
   103→    // Generate unique publicToken using crypto
   104→    const publicToken = crypto.randomUUID();
   105→
   106→    const now = Date.now();
   107→    const { workspaceId, teamId, customFields, ...rest } = args;
   108→
   109→    const proposalId = await ctx.db.insert("proposals", {
   110→      userId,
   111→      workspaceId: workspaceId ?? undefined,
   112→      createdBy: userId,
   113→      teamId: teamId ?? undefined,
   114→      customFields: customFields ?? undefined,
   115→      clientId: rest.clientId,
   116→      dealId: rest.dealId,
   117→      title: rest.title.trim(),
   118→      content: rest.content,
   119→      templateId: rest.templateId,
   120→      status: "draft",
   121→      totalValue: rest.totalValue,
   122→      currency: rest.currency ?? "USD",
   123→      validUntil: rest.validUntil,
   124→      sentAt: undefined,
   125→      viewedAt: undefined,
   126→      viewedCount: 0,
   127→      signedAt: undefined,
   128→      signedIp: undefined,
   129→      signedName: undefined,
   130→      declinedAt: undefined,
   131→      declinedReason: undefined,
   132→      publicToken,
   133→      version: 1,
   134→      createdAt: now,
   135→      updatedAt: now,
   136→    });
   137→
   138→    // Increment template usage count if a template was used
   139→    if (args.templateId) {
   140→      const template = await ctx.db.get(args.templateId);
   141→      if (template) {
   142→        await ctx.db.patch(args.templateId, {
   143→          usageCount: (template.usageCount ?? 0) + 1,
   144→        });
   145→      }
   146→    }
   147→
   148→    return proposalId;
   149→  },
   150→});

   192→    // Validate totalValue if provided
   193→    if (args.totalValue !== undefined && args.totalValue < 0) {
   194→      throw new ConvexError("Total value cannot be negative");

   197→    // Verify client access if changing client
   198→    if (args.clientId !== undefined) {
   199→      const client = await ctx.db.get(args.clientId);
   200→      if (!client) throw new ConvexError("Client not found");
   201→      if (client.workspaceId) {

   205→        throw new ConvexError("Not authorized: client belongs to another user");
   206→      }
   207→    }
   208→
   209→    // Verify deal access if changing deal
   210→    if (args.dealId !== undefined) {
   211→      const deal = await ctx.db.get(args.dealId);
   212→      if (!deal) throw new ConvexError("Deal not found");
   213→      if (deal.workspaceId) {
   214→        const access = await getRecordAccess(ctx, deal, userId);
   215→        if (!access) throw new ConvexError("Not authorized: deal belongs to another workspace");

   217→        throw new ConvexError("Not authorized: deal belongs to another user");
   218→      }
   219→    }
   220→
   221→    // Build partial update object
   222→    const updates: Record<string, unknown> = { updatedAt: Date.now() };
   223→
   224→    if (args.clientId !== undefined) updates.clientId = args.clientId;
   225→    if (args.dealId !== undefined) updates.dealId = args.dealId;
   226→    if (args.title !== undefined) updates.title = args.title.trim();

   228→    if (args.templateId !== undefined) updates.templateId = args.templateId;
   229→    if (args.totalValue !== undefined) updates.totalValue = args.totalValue;
   230→    if (args.currency !== undefined) updates.currency = args.currency;
   231→    if (args.validUntil !== undefined) updates.validUntil = args.validUntil;
   232→    if (args.teamId !== undefined) updates.teamId = args.teamId;
   233→
   234→    await ctx.db.patch(args.proposalId, updates);
   235→
   236→    return args.proposalId;
   237→  },

   319→  args: {
   320→    publicToken: v.string(),

   324→  handler: async (ctx, args) => {

   327→      .withIndex("by_public_token", (q) => q.eq("publicToken", args.publicToken))
   328→      .first();
   329→
   330→    if (!proposal) throw new ConvexError("Invalid or expired proposal link");
   331→
   332→    // Only sent or viewed proposals can be signed

   334→      throw new ConvexError("Proposal cannot be signed in its current state");
   335→    }
   336→
   337→    // Check if proposal has expired
   338→    if (proposal.validUntil && proposal.validUntil < Date.now()) {
   339→      throw new ConvexError("This proposal has expired and can no longer be signed");

   341→
   342→    if (args.signedName.trim().length === 0) {
   343→      throw new ConvexError("Signed name is required");
   344→    }
   345→
   346→    const now = Date.now();
   347→
   348→    await ctx.db.patch(proposal._id, {
   349→      status: "signed",
   350→      signedAt: now,
   351→      signedIp: args.signedIp,
   352→      signedName: args.signedName.trim(),
   353→      updatedAt: now,
   354→    });
   355→
   356→    // Cancel all pending follow-ups for this proposal
   357→    await cancelFollowUpsInternal(ctx, proposal._id);
   358→
   359→    // Auto-create scope definition from proposal content
   360→    await convertToScopeInternal(ctx, proposal.userId, proposal._id, proposal.clientId, proposal.content, proposal.title);
   361→
   362→    return { success: true };
   363→  },

   403→// ─────────────────────────────────────────────
   404→// 7. EXPIRE (batch — mark expired proposals)
   405→// ─────────────────────────────────────────────
   406→export const expire = mutation({
   407→  args: {},
   408→  handler: async (ctx) => {
   409→    const now = Date.now();
   410→
   411→    // Query all sent proposals and filter for expired ones
   412→    const sentProposals = await ctx.db
   413→      .query("proposals")
   414→      .filter((q) => q.eq(q.field("status"), "sent"))
   415→      .collect();
   416→
   417→    let expiredCount = 0;
   418→
   419→    for (const proposal of sentProposals) {
   420→      if (proposal.validUntil && proposal.validUntil < now) {
   421→        await ctx.db.patch(proposal._id, {
   422→          status: "expired",
   423→          updatedAt: now,
   424→        });
   425→
   426→        // Cancel pending follow-ups for expired proposals
   427→        await cancelFollowUpsInternal(ctx, proposal._id);
   428→
   429→        expiredCount++;
   430→      }
   431→    }
   432→
   433→    return { expiredCount };
   434→  },
   435→});

   463→// ─────────────────────────────────────────────
   464→// 9. LIST (workspace-aware, with optional status filter)
   465→// ─────────────────────────────────────────────
   466→export const list = query({
   467→  args: {
   468→    workspaceId: v.optional(v.id("workspaces")),
   469→    status: v.optional(proposalStatusValidator),
   470→  },
   471→  handler: async (ctx, args) => {
   472→    const userId = await getAuthUserId(ctx);
   473→    if (!userId) throw new ConvexError("Not authenticated");
   474→
   475→    // If workspaceId provided, filter by workspace with team-aware visibility
   476→    if (args.workspaceId) {
   477→      const visibility = await getUserVisibility(ctx, args.workspaceId);
   478→      if (!visibility) throw new ConvexError("Not a member of this workspace");
   479→
   480→      const allProposals = await ctx.db
   481→        .query("proposals")
   482→        .withIndex("by_workspace", (q) => q.eq("workspaceId", args.workspaceId!))
   483→        .collect();
   484→
   485→      let visible = allProposals.filter((p) => isRecordVisible(p, visibility));
   486→
   487→      if (args.status) {
   488→        visible = visible.filter((p) => p.status === args.status);
   489→      }
   490→
   491→      return visible.sort((a, b) => b.updatedAt - a.updatedAt);
   492→    }
   493→
   494→    // Backward compat: no workspaceId
   495→    if (args.status) {
   496→      const proposals = await ctx.db
   497→        .query("proposals")
   498→        .withIndex("by_user_and_status", (q) =>
   499→          q.eq("userId", userId).eq("status", args.status!)
   500→        )
   501→        .order("desc")
   502→        .collect();
   503→      return proposals;
   504→    }
   505→
   506→    // No status filter — return all proposals for this user
   507→    const proposals = await ctx.db
   508→      .query("proposals")
   509→      .withIndex("by_user", (q) => q.eq("userId", userId))
   510→      .order("desc")
   511→      .collect();
   512→
   513→    // Sort by updatedAt descending (index order is by creation)
   514→    return proposals.sort((a, b) => b.updatedAt - a.updatedAt);
   515→  },
   516→});

   601→      const visibility = await getUserVisibility(ctx, args.workspaceId);
   602→      if (!visibility) return {
   603→        counts: { draft: 0, sent: 0, viewed: 0, signed: 0, expired: 0, declined: 0, total: 0 },
   604→        totalSignedValue: 0,
   605→        closeRate: 0,
   606→        viewToSignRate: 0,
   607→        avgDaysToSign: 0,
   608→      };
   609→
   610→      const workspaceProposals = await ctx.db
   611→        .query("proposals")
   612→        .withIndex("by_workspace", (q) => q.eq("workspaceId", args.workspaceId))
   613→        .collect();
   614→      allProposals = workspaceProposals.filter((p) => isRecordVisible(p, visibility));
   615→    } else {
   616→      allProposals = await ctx.db
   617→        .query("proposals")
   618→        .withIndex("by_user", (q) => q.eq("userId", userId))
   619→        .collect();
   620→    }
   621→
   622→    const draft = allProposals.filter((p) => p.status === "draft").length;
   623→    const sent = allProposals.filter((p) => p.status === "sent").length;
   624→    const viewed = allProposals.filter((p) => p.status === "viewed").length;
   625→    const signed = allProposals.filter((p) => p.status === "signed").length;
   626→    const expired = allProposals.filter((p) => p.status === "expired").length;
   627→    const declined = allProposals.filter((p) => p.status === "declined").length;
   628→
   629→    // Total value of signed proposals
   630→    const totalSignedValue = allProposals
   631→      .filter((p) => p.status === "signed")
   632→      .reduce((sum, p) => sum + (p.totalValue ?? 0), 0);
   633→
   634→    // Average close rate: signed / (signed + declined + expired) for proposals that have reached a terminal state
   635→    const terminalStates = signed + declined + expired;
   636→    const closeRate = terminalStates > 0 ? signed / terminalStates : 0;
   637→
   638→    // Average view-to-sign conversion
   639→    const viewedOrBeyond = allProposals.filter(
   640→      (p) => p.status === "viewed" || p.status === "signed" || p.status === "declined"
   641→    );
   642→    const viewToSignRate = viewedOrBeyond.length > 0
   643→      ? signed / viewedOrBeyond.length
   644→      : 0;
   645→
   646→    // Average days to sign (from sent to signed)
   647→    const signedProposals = allProposals.filter((p) => p.status === "signed" && p.sentAt && p.signedAt);
   648→    const avgDaysToSign = signedProposals.length > 0
   649→      ? signedProposals.reduce((sum, p) => {
   650→          const days = (p.signedAt! - p.sentAt!) / (1000 * 60 * 60 * 24);
   651→          return sum + days;
   652→        }, 0) / signedProposals.length
   653→      : 0;
   654→
   655→    return {
   656→      counts: { draft, sent, viewed, signed, expired, declined, total: allProposals.length },
   657→      totalSignedValue,
   658→      closeRate: Math.round(closeRate * 100) / 100,
   659→      viewToSignRate: Math.round(viewToSignRate * 100) / 100,
   660→      avgDaysToSign: Math.round(avgDaysToSign * 10) / 10,
   661→    };
   662→  },
   663→});

   811→// 16. LIST TEMPLATES (user's own + system, workspace-aware)
   812→// ─────────────────────────────────────────────
   813→export const listTemplates = query({
   814→  args: {
   815→    industry: v.optional(v.string()),
   816→    workspaceId: v.optional(v.id("workspaces")),
   817→  },
   818→  handler: async (ctx, args) => {
   819→    const userId = await getAuthUserId(ctx);
   820→    if (!userId) throw new ConvexError("Not authenticated");
   821→
   822→    // Get user's own templates
   823→    const userTemplates = await ctx.db
   824→      .query("proposalTemplates")
   825→      .withIndex("by_user", (q) => q.eq("userId", userId))
   826→      .collect();
   827→
   828→    // Get workspace templates if provided
   829→    let workspaceTemplates: any[] = [];
   830→    if (args.workspaceId) {
   831→      workspaceTemplates = await ctx.db
   832→        .query("proposalTemplates")
   833→        .withIndex("by_workspace", (q) => q.eq("workspaceId", args.workspaceId))
   834→        .collect();
   835→    }
   836→
   837→    // Get system templates (userId is null)
   838→    let systemTemplates: any[] = [];
   839→
   840→    if (args.industry) {
   841→      systemTemplates = await ctx.db
   842→        .query("proposalTemplates")
   843→        .withIndex("by_industry", (q) => q.eq("industry", args.industry!))
   844→        .collect();
   845→      // Filter to only system templates (null userId)
   846→      systemTemplates = systemTemplates.filter((t) => t.userId === undefined);
   847→    } else {
   848→      const industries = ["web_design", "consulting", "marketing", "development"];
   849→      for (const ind of industries) {
   850→        const templates = await ctx.db
   851→          .query("proposalTemplates")
   852→          .withIndex("by_industry", (q) => q.eq("industry", ind))
   853→          .collect();
   854→        systemTemplates.push(...templates.filter((t) => t.userId === undefined));
   855→      }
   856→    }
   857→
   858→    // Combine and deduplicate
   859→    const allTemplateIds = new Set<string>();
   860→    const combined: any[] = [];
   861→
   862→    for (const t of [...userTemplates, ...workspaceTemplates, ...systemTemplates]) {
   863→      if (!allTemplateIds.has(t._id)) {
   864→        allTemplateIds.add(t._id);
   865→        combined.push(t);
   866→      }
   867→    }
   868→
   869→    // Sort: system templates first, then by usage count descending
   870→    combined.sort((a, b) => {
   871→      const aIsSystem = a.userId === undefined ? 0 : 1;
   872→      const bIsSystem = b.userId === undefined ? 0 : 1;
   873→      if (aIsSystem !== bIsSystem) return aIsSystem - bIsSystem;
   874→      return (b.usageCount ?? 0) - (a.usageCount ?? 0);
   875→    });
   876→
   877→    return combined;
   878→  },
   879→});

  1069→export const scheduleFollowUps = mutation({
  1070→  args: {
  1071→    proposalId: v.id("proposals"),
  1072→  },
  1073→  handler: async (ctx, args) => {
  1074→    const userId = await getAuthUserId(ctx);
  1075→    if (!userId) throw new ConvexError("Not authenticated");
  1076→
  1077→    const proposal = await ctx.db.get(args.proposalId);
  1078→    if (!proposal) throw new ConvexError("Proposal not found");
  1079→
  1080→    // Check workspace access or direct ownership
  1081→    if (proposal.workspaceId) {
  1082→      await requireRecordAccess(ctx, proposal, "collaborate");
  1083→    } else if (proposal.userId !== userId) {
  1084→      throw new ConvexError("Not authorized");
  1085→    }
  1086→
  1087→    if (proposal.status !== "sent") {
  1088→      throw new ConvexError("Follow-ups can only be scheduled for sent proposals");
  1089→    }
  1090→
  1091→    if (!proposal.sentAt) {
  1092→      throw new ConvexError("Proposal has not been sent yet");
  1093→    }
  1094→
  1095→    await scheduleFollowUpsInternal(
  1096→      ctx, args.proposalId, userId, proposal.title, proposal.totalValue, proposal.sentAt
  1097→    );
  1098→
  1099→    return { success: true };
  1100→  },

  1109→  args: {},
  1110→  handler: async (ctx) => {

  1112→
  1113→    // Find all scheduled follow-ups that are due

  1118→
  1119→    const dueFollowUps = allFollowUps.filter(
  1120→      (fu) => fu.scheduledFor <= now
  1121→    );
  1122→
  1123→    let processedCount = 0;
  1124→    for (const fu of dueFollowUps) {
  1125→      // Mark as sent (in a real system, this would also send an email)
  1126→      await ctx.db.patch(fu._id, {
  1127→        status: "sent",
  1128→        sentAt: now,
  1129→      });
  1130→      processedCount++;
  1131→    }
  1132→
  1133→    return { processedCount, totalDue: dueFollowUps.length };

  1166→        channel: "email",

  1170→        createdAt: now,
