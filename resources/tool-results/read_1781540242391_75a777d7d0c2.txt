   207→   207→    month: "short",

   213→   213→function daysUntilClose(timestamp: number | undefined): number | null {
   214→   214→  if (!timestamp) return null;
   215→   215→  const diff = timestamp - Date.now();
   216→   216→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   217→   217→}
