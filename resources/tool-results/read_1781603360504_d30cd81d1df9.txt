   218→function DayView({ data, onBlockClick }: { data: TimeBlock[]; onBlockClick: (b: TimeBlock) => void }) {
   219→  // 8 AM – 8 PM = 13 hours displayed in a fixed grid
   220→  const hours = Array.from({ length: 13 }, (_, i) => i + 8);

   236→              key={hour}

   243→        </div>
   244→
   245→        {/* Grid area */}
   246→        <div className="absolute left-14 top-0 bottom-0 right-0">
   247→          {/* Hour divider lines */}
   248→          {hours.map((hour) => (
   249→            <div
   250→              key={hour}
   251→              className="absolute left-0 right-0 border-t border-border"
   252→              style={{ top: `${((hour - 8) / 13) * 100}%` }}
   253→            />
   254→          ))}
   255→          {/* Half-hour dashed lines */}
   256→          {hours.map((hour) => (
   257→            <div
   258→              key={`half-${hour}`}
   259→              className="absolute left-0 right-0 border-t border-dashed border-border/40"
   260→              style={{ top: `${((hour - 8 + 0.5) / 13) * 100}%` }}
   261→            />
   262→          ))}
   263→
   264→          {/* Current time indicator */}
   265→          {(() => {
   266→            const now = new Date();
   267→            const nowHour = now.getHours() + now.getMinutes() / 60;
   268→            if (nowHour >= 8 && nowHour <= 21) {
   269→              const topPct = ((nowHour - 8) / 13) * 100;
   270→              return (
   271→                <div className="absolute left-0 right-0 z-10 flex items-center" style={{ top: `${topPct}%` }}>
   272→                  <div className="w-2.5 h-2.5 rounded-full bg-red-500 -ml-1" />
   273→                  <div className="flex-1 h-[2px] bg-red-500" />
   274→                </div>
   275→              );
   276→            }
   277→            return null;
   278→          })()}
   279→
   280→          {/* Time blocks */}
   281→          {data.map((block) => {
   282→            const colors = STATUS_COLORS[block.complianceStatus];
   283→            const startHour = new Date(block.startTime).getHours() + new Date(block.startTime).getMinutes() / 60;
   284→            const endHour = new Date(block.endTime).getHours() + new Date(block.endTime).getMinutes() / 60;
   285→            const topPct = Math.max(0, ((startHour - 8) / 13) * 100);
   286→            const heightPct = Math.max(3, ((endHour - startHour) / 13) * 100);
   287→
   288→            return (
   289→              <button
   290→                key={block._id}
   291→                onClick={() => onBlockClick(block)}
   292→                className={`absolute left-1 right-2 rounded-md border ${colors.border} ${colors.bg} ${colors.text} px-2 py-0.5 text-left transition-shadow hover:shadow-md focus:outline-none focus:ring-2 focus:ring-primary/50 overflow-hidden`}
   293→                style={{ top: `${topPct}%`, height: `${heightPct}%` }}
   294→              >
   295→                <div className="flex items-center gap-1">
   296→                  <div className={`w-1.5 h-1.5 rounded-full ${colors.dot} flex-shrink-0`} />
   297→                  <span className="text-[11px] font-semibold truncate">{block.activity}</span>
   298→                </div>
   299→                <div className="text-[9px] opacity-70 truncate mt-px">
   300→                  {formatHour(startHour)} – {formatHour(endHour)} &middot; {block.website}
   301→                </div>
   302→              </button>
   303→            );
   304→          })}
   305→        </div>
   306→      </div>
   307→    </div>
   308→  );

   313→function WeekView({ data, currentDate, onDayClick }: { data: TimeBlock[]; currentDate: Date; onDayClick: (d: Date) => void }) {
   314→  const weekStart = new Date(currentDate);
   315→  weekStart.setDate(currentDate.getDate() - currentDate.getDay());
   316→  const days = Array.from({ length: 7 }, (_, i) => {
   317→    const d = new Date(weekStart);
   318→    d.setDate(weekStart.getDate() + i);
   319→    return d;
   320→  });
   321→
   322→  const getDaySummary = (date: Date) => {
   323→    const dayBlocks = data.filter((b) => new Date(b.startTime).toDateString() === date.toDateString());
   324→    const compliant = dayBlocks.filter((b) => b.complianceStatus === "compliant").length;
   325→    const atRisk = dayBlocks.filter((b) => b.complianceStatus === "at_risk").length;
   326→    const rejected = dayBlocks.filter((b) => b.complianceStatus === "rejected").length;
   327→    const totalHours = dayBlocks.length;
   328→    const earnings = totalHours * 25;
   329→    return { compliant, atRisk, rejected, totalHours, earnings };
   330→  };
   331→
   332→  const today = new Date();
   333→
   334→  return (
   335→    <div className="h-full p-3 flex flex-col">
   336→      {/* Day headers */}
   337→      <div className="grid grid-cols-7 gap-1 mb-1.5 flex-shrink-0">
   338→        {days.map((day, i) => {
   339→          const isToday = day.toDateString() === today.toDateString();
   340→          return (
   341→            <div key={i} className="text-center py-1">
   342→              <div className="text-[10px] text-muted-foreground font-medium">{DAY_NAMES[day.getDay()]}</div>
   343→              <div className={`text-sm font-semibold mt-0.5 w-6 h-6 rounded-full flex items-center justify-center mx-auto ${isToday ? "bg-primary text-primary-foreground" : "text-foreground"}`}>
   344→                {day.getDate()}
   345→              </div>
   346→            </div>
   347→          );
   348→        })}
   349→      </div>
   350→
   351→      {/* Day cells */}
   352→      <div className="grid grid-cols-7 gap-1 flex-1 min-h-0">
   353→        {days.map((day, i) => {
   354→          const summary = getDaySummary(day);
   355→          const hasData = summary.totalHours > 0;
   356→          const isToday = day.toDateString() === today.toDateString();
   357→
   358→          return (
   359→            <button
   360→              key={i}
   361→              onClick={() => onDayClick(day)}
   362→              className={`rounded-lg border transition-all hover:shadow-md flex flex-col items-center justify-center p-1.5 min-h-0 ${
   363→                isToday ? "border-primary/50 bg-primary/5" : "border-border bg-card hover:bg-muted/50"
   364→              }`}
   365→            >
   366→              {hasData ? (
   367→                <>
   368→                  <div className={`text-sm font-bold ${summary.rejected > 0 ? "text-red-600" : summary.atRisk > 0 ? "text-amber-600" : "text-emerald-600"}`}>
   369→                    +${summary.earnings}
   370→                  </div>
   371→                  <div className="text-[9px] text-muted-foreground mt-0.5">{summary.totalHours}h</div>
   372→                  <div className="flex gap-0.5 mt-1">
   373→                    {summary.compliant > 0 && <div className="w-1.5 h-1.5 rounded-full bg-blue-500" />}
   374→                    {summary.atRisk > 0 && <div className="w-1.5 h-1.5 rounded-full bg-amber-500" />}
   375→                    {summary.rejected > 0 && <div className="w-1.5 h-1.5 rounded-full bg-red-500" />}
   376→                  </div>
   377→                </>
   378→              ) : (
   379→                <div className="text-xs text-muted-foreground">$0</div>
   380→              )}
   381→            </button>
   382→          );
   383→        })}
   384→      </div>
   385→    </div>
   386→  );
   387→}

   391→function MonthView({ data, currentDate, onDayClick }: { data: TimeBlock[]; currentDate: Date; onDayClick: (d: Date) => void }) {
   392→  const year = currentDate.getFullYear();
   393→  const month = currentDate.getMonth();
   394→  const firstDay = new Date(year, month, 1).getDay();
   395→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   396→  const today = new Date();
   397→
   398→  const cells: (Date | null)[] = [];
   399→  for (let i = 0; i < firstDay; i++) cells.push(null);
   400→  for (let d = 1; d <= daysInMonth; d++) cells.push(new Date(year, month, d));
   401→  while (cells.length % 7 !== 0) cells.push(null);
   402→
   403→  const getDaySummary = (date: Date | null) => {
   404→    if (!date) return { totalHours: 0, earnings: 0, rejected: 0, atRisk: 0 };
   405→    const dayBlocks = data.filter((b) => new Date(b.startTime).toDateString() === date.toDateString());
   406→    return {
   407→      totalHours: dayBlocks.length,
   408→      earnings: dayBlocks.length * 25,
   409→      rejected: dayBlocks.filter((b) => b.complianceStatus === "rejected").length,
   410→      atRisk: dayBlocks.filter((b) => b.complianceStatus === "at_risk").length,
   411→    };
   412→  };
   413→
   414→  return (
   415→    <div className="h-full p-3 flex flex-col">
   416→      {/* Day-of-week headers */}
   417→      <div className="grid grid-cols-7 gap-0.5 mb-1 flex-shrink-0">
   418→        {DAY_NAMES.map((name) => (
   419→          <div key={name} className="text-center text-[10px] font-medium text-muted-foreground py-0.5">{name}</div>
   420→        ))}
   421→      </div>
   422→
   423→      {/* Calendar grid */}
   424→      <div className="grid grid-cols-7 gap-0.5 flex-1 min-h-0 auto-rows-fr">
   425→        {cells.map((day, i) => {
   426→          if (!day) return <div key={i} />;
   427→          const summary = getDaySummary(day);
   428→          const isToday = day.toDateString() === today.toDateString();
   429→          const hasData = summary.totalHours > 0;
   430→
   431→          return (
   432→            <button
   433→              key={i}
   434→              onClick={() => onDayClick(day)}
   435→              className={`rounded border text-left px-1 py-0.5 transition-all hover:shadow-sm min-h-0 flex flex-col justify-start ${
   436→                isToday ? "border-primary/50 bg-primary/5" : "border-border/50 hover:bg-muted/50"
   437→              }`}
   438→            >
   439→              <div className={`text-[10px] font-medium leading-none ${isToday ? "text-primary" : "text-foreground"}`}>
   440→                {day.getDate()}
   441→              </div>
   442→              {hasData && (
   443→                <div className={`text-[9px] font-semibold mt-px ${summary.rejected > 0 ? "text-red-600" : summary.atRisk > 0 ? "text-amber-600" : "text-emerald-600"}`}>
   444→                  ${summary.earnings}
   445→                </div>
   446→              )}
   447→            </button>
   448→          );
   449→        })}
   450→      </div>
   451→    </div>
   452→  );
   453→}

   457→function YearView({ data, currentDate, onMonthClick }: { data: TimeBlock[]; currentDate: Date; onMonthClick: (d: Date) => void }) {
   458→  const year = currentDate.getFullYear();
   459→  const today = new Date();
   460→
   461→  const getMonthSummary = (monthIdx: number) => {
   462→    const monthBlocks = data.filter((b) => {
   463→      const bd = new Date(b.startTime);
   464→      return bd.getFullYear() === year && bd.getMonth() === monthIdx;
   465→    });
   466→    const compliant = monthBlocks.filter((b) => b.complianceStatus === "compliant").length;
   467→    const atRisk = monthBlocks.filter((b) => b.complianceStatus === "at_risk").length;
   468→    const rejected = monthBlocks.filter((b) => b.complianceStatus === "rejected").length;
   469→    const totalHours = monthBlocks.length;
   470→    const earnings = totalHours * 25;
   471→    const compliancePct = monthBlocks.length > 0 ? Math.round((compliant / monthBlocks.length) * 100) : 0;
   472→    return { compliant, atRisk, rejected, totalHours, earnings, compliancePct };
   473→  };
   474→

   614→function formatHourShort(hour: number): string {
   615→  if (hour === 12) return "12 PM";
   616→  if (hour > 12) return `${hour - 12} PM`;
   617→  return `${hour} AM`;
   618→}
   619→
   620→function formatHour(h: number): string {
   621→  const hour = Math.floor(h);
   622→  const min = Math.round((h - hour) * 60);
   623→  const suffix = hour >= 12 ? "PM" : "AM";
   624→  const display = hour > 12 ? hour - 12 : hour === 0 ? 12 : hour;
   625→  return min > 0 ? `${display}:${min.toString().padStart(2, "0")} ${suffix}` : `${display} ${suffix}`;
   626→}

   628→function formatPeriodLabel(date: Date, view: TimeView): string {
   629→  switch (view) {
   630→    case "day":
   631→      return date.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" });
   632→    case "week": {
   633→      const ws = new Date(date);
   634→      ws.setDate(date.getDate() - date.getDay());
   635→      const we = new Date(ws);
   636→      we.setDate(ws.getDate() + 6);
   637→      return `${ws.toLocaleDateString("en-US", { month: "short", day: "numeric" })} – ${we.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })}`;
   638→    }
   639→    case "month":
   640→      return date.toLocaleDateString("en-US", { month: "long", year: "numeric" });
   641→    case "year":
   642→      return String(date.getFullYear());
   643→  }
   644→}

   646→function calculateComplianceScore(data: TimeBlock[]): number {
   647→  if (!data.length) return 100;
   648→  return Math.round((data.filter((b) => b.complianceStatus === "compliant").length / data.length) * 100);
   649→}

   651→function calculateAtRiskHours(data: TimeBlock[]): number {
   652→  return Math.round((data.filter((b) => b.complianceStatus !== "compliant").length * 5) / 60 * 10) / 10;
   653→}
