   117→// --- Helpers ---
   118→function formatDate(ts: number): string {
   119→  return new Date(ts).toLocaleDateString("en-US", {
   120→    year: "numeric",
   121→    month: "short",
   122→    day: "numeric",
   123→  });
   124→}
   125→
   126→function timeAgo(ts: number | null): string {
   127→  if (!ts) return "Never";
   128→  const diff = Date.now() - ts;
   129→  const mins = Math.floor(diff / 60000);
   130→  if (mins < 1) return "Just now";
   131→  if (mins < 60) return `${mins}m ago`;
   132→  const hrs = Math.floor(mins / 60);
   133→  if (hrs < 24) return `${hrs}h ago`;
   134→  const days = Math.floor(hrs / 24);
   135→  return `${days}d ago`;
   136→}

   138→function generateMockKey(prefix: string = "tl_prod_"): string {
   139→  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
   140→  let result = prefix + "sk_";
   141→  for (let i = 0; i < 20; i++) {
   142→    result += chars.charAt(Math.floor(Math.random() * chars.length));
   143→  }
   144→  return result;
   145→}
   146→
   147→function maskKey(key: string): string {
   148→  const prefix = key.substring(0, key.indexOf("sk_") + 3);
   149→  const suffix = key.slice(-4);
   150→  return `${prefix}${"•".repeat(16)}${suffix}`;
   151→}
