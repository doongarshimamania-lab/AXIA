   100→function timeAgo(ts: number): string {
   101→  const diff = Date.now() - ts;
   102→  const minutes = Math.floor(diff / 60000);
   103→  const hours = Math.floor(minutes / 60);
   104→  const days = Math.floor(hours / 24);
   105→  if (days > 0) return `${days}d ago`;
   106→  if (hours > 0) return `${hours}h ago`;
   107→  return `${minutes}m ago`;
   108→}

   160→function calculateScopeHealth(scope: ScopeDefinition, changeOrders: ScopeChangeOrder[]) {
   161→  const scopeChanges = changeOrders.filter(co => co.scopeId === scope._id);
   162→  const pendingChanges = scopeChanges.filter(co => co.status === "pending" || co.status === "auto_generated");
   163→  const revisionUsage = scope.revisionLimit > 0 ? (scope.revisionCount / scope.revisionLimit) * 100 : 0;
   164→  const totalHoursAdded = scopeChanges.reduce((sum, co) => sum + co.impact.hoursAdded, 0);
   165→  const totalCostImpact = scopeChanges.reduce((sum, co) => sum + co.impact.costImpact, 0);
   166→  const hoursOverrun = scope.totalEstimatedHours ? (totalHoursAdded / scope.totalEstimatedHours) * 100 : 0;
   167→
   168→  let riskLevel: "low" | "medium" | "high" = "low";
   169→  if (revisionUsage >= 100 || hoursOverrun > 30 || pendingChanges.length >= 3) riskLevel = "high";
   170→  else if (revisionUsage >= 66 || hoursOverrun > 15 || pendingChanges.length >= 2) riskLevel = "medium";
   171→
   172→  const completedDeliverables = scope.deliverables.filter(d => d.status === "completed").length;
   173→  const progressPercent = scope.deliverables.length > 0 ? (completedDeliverables / scope.deliverables.length) * 100 : 0;
   174→
   175→  return {
   176→    riskLevel,
   177→    revisionUsage,
   178→    totalHoursAdded,
   179→    totalCostImpact,
   180→    hoursOverrun,
   181→    pendingChanges: pendingChanges.length,
   182→    completedDeliverables,
   183→    totalDeliverables: scope.deliverables.length,
   184→    progressPercent,
   185→  };
   186→}

   436→// ─── ScopeCard with its own change orders query ─────────────────────────────
   437→
   438→function ScopeCardContainer({
   439→  scope,
   440→  isSelected,
   441→  onSelect,
   442→  onDelete,
   443→  deleting,
   444→  onMetricsReady,
   445→}: {
   446→  scope: ScopeDefinition;
   447→  isSelected: boolean;
   448→  onSelect: () => void;
   449→  onDelete: () => void;
   450→  deleting: boolean;
   451→  onMetricsReady: (scopeId: string, metrics: { pendingChanges: number; costAtRisk: number; hoursAtRisk: number }) => void;
   452→}) {
   453→  const changeOrders = useQuery(api.scope.crud.getChangeOrders, { scopeId: scope._id as any });
   454→
   455→  const prevMetricsRef = useRef<string>("");
   456→
   457→  useEffect(() => {
   458→    if (changeOrders && Array.isArray(changeOrders)) {
   459→      const pending = (changeOrders as ScopeChangeOrder[]).filter(co => co.status === "pending" || co.status === "auto_generated");
   460→      const costAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.costImpact, 0);
   461→      const hoursAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.hoursAdded, 0);
   462→      const key = `${pending.length}-${costAtRisk}-${hoursAtRisk}`;
   463→      if (key !== prevMetricsRef.current) {
   464→        prevMetricsRef.current = key;
   465→        onMetricsReady(scope._id, { pendingChanges: pending.length, costAtRisk, hoursAtRisk });
   466→      }
   467→    }
   468→  }, [changeOrders, scope._id, onMetricsReady]);
   469→
   470→  return (
   471→    <ScopeCard
   472→      scope={scope}
   473→      changeOrders={(changeOrders as ScopeChangeOrder[]) ?? []}
   474→      isSelected={isSelected}
