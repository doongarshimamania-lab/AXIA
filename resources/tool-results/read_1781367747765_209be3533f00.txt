     1→/**
     2→ * Template Parser — Rules-based document structure extraction
     3→ *
     4→ * Parses PDF, DOCX, and TXT files into ProposalSection or InvoiceSection
     5→ * objects using purely heuristic approaches (NO AI).
     6→ *
     7→ * Supports:
     8→ *  - DOCX: mammoth → HTML → structure extraction
     9→ *  - PDF:  pdfjs-dist → text items with font/position → heading/table detection
    10→ *  - TXT:  simple line-by-line parsing
    11→ *
    12→ * Accuracy notes:
    13→ *  - Well-structured DOCX: ~85-95% (HTML preserves most structure)
    14→ *  - Well-structured PDF (text-based): ~70-85% (position/font heuristics)
    15→ *  - Scanned PDF (image-based): 0% (no OCR — returns "no text" message)
    16→ *  - TXT: ~60-75% (no formatting metadata to work with)
    17→ *
    18→ *  Users can always reclassify and edit sections after import.
    19→ *  The parser gets them 80% there; the UI handles the rest.
    20→ */
    21→
    22→// ─── Types ────────────────────────────────────────────────────────────────────

    63→/** Detect if a string looks like a placeholder pattern */
    64→function detectPlaceholders(text: string): string[] {
    65→  const patterns = [
    66→    /\{\{([^}]+)\}\}/g,       // {{var}}
    67→    /\[([^\]]+)\]/g,          // [Variable]
    68→    /<([A-Z_]+)>/g,           // <COMPANY>
    69→    /_{3,}/g,                 // ___ (underscores)
    70→    /%([A-Z_]+)%/g,           // %VARIABLE%
    71→  ];
    72→  const found: string[] = [];
    73→  for (const pat of patterns) {
    74→    let m: RegExpExecArray | null;
    75→    while ((m = pat.exec(text)) !== null) {
    76→      found.push(m[1] || m[0]);
    77→    }
    78→  }
    79→  return found;
    80→}

   195→/** Check if lines form a table-like grid structure */
   196→function detectTableRows(lines: string[]): { isTable: boolean; rows: string[][] } {
   197→  if (lines.length < 2) return { isTable: false, rows: [] };
   198→
   199→  const rows: string[][] = [];
   200→  let consistentCols = 0;
   201→  let firstColCount = 0;
   202→
   203→  for (const line of lines) {
   204→    // Split by tab OR by 2+ consecutive spaces (common in PDF text extraction)
   205→    const cols = line.split(/\t|  {2,}/).map(c => c.trim()).filter(Boolean);
   206→    if (cols.length >= 2) {
   207→      rows.push(cols);
   208→      if (firstColCount === 0) firstColCount = cols.length;
   209→      if (cols.length === firstColCount) consistentCols++;
   210→    }
   211→  }
   212→
   213→  const isTable = rows.length >= 2 && consistentCols >= 2;
   214→  return { isTable, rows };
   215→}

   217→/** Detect multi-column layout from PDF text items (e.g., side-by-side address blocks) */
   218→function detectMultiColumnLayout(
   219→  items: { x: number; y: number; text: string; fontSize: number; fontName: string }[]
   220→): { left: string; right: string } | null {
   221→  if (items.length < 4) return null;
   222→
   223→  // Find the median X position
   224→  const xPositions = items.map(i => i.x).sort((a, b) => a - b);
   225→  const medianX = xPositions[Math.floor(xPositions.length / 2)];
   226→
   227→  // If there's a clear gap (items clustered on left and right of median)
   228→  const leftItems = items.filter(i => i.x < medianX * 0.7);
   229→  const rightItems = items.filter(i => i.x > medianX * 1.3);
   230→
   231→  if (leftItems.length >= 2 && rightItems.length >= 2) {
   232→    const leftText = leftItems.sort((a, b) => b.y - a.y).map(i => i.text).join("\n");
   233→    const rightText = rightItems.sort((a, b) => b.y - a.y).map(i => i.text).join("\n");
   234→    return { left: leftText, right: rightText };
   235→  }
   236→
   237→  return null;
   238→}

   240→/** Parse a quantity × rate = amount pattern from text */
   241→function parseQtyRateAmount(text: string): { description: string; quantity: number; rate: number; amount: number }[] {
   242→  const lines = text.split("\n");
   243→  const items: { description: string; quantity: number; rate: number; amount: number }[] = [];
   244→
   245→  for (const line of lines) {
   246→    const trimmed = line.trim();
   247→    if (!trimmed) continue;
   248→
   249→    // Skip header-like rows
   250→    if (/^(item|description|qty|quantity|rate|price|amount|total|service|#)/i.test(trimmed)) continue;
   251→    // Skip subtotal/total rows
   252→    if (/^(subtotal|total|tax|vat|gst|discount)/i.test(trimmed)) continue;
   253→
   254→    // Try to detect tab/space-separated columns: description  qty  rate  amount
   255→    const cols = trimmed.split(/\t|  {2,}/).map(c => c.trim()).filter(Boolean);
   256→
   257→    if (cols.length >= 4) {
   258→      // Full format: description, qty, rate, amount
   259→      const description = cols[0];
   260→      const qtyMatch = cols[1].match(/([\d.]+)/);
   261→      const rateMatch = cols[2].match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   262→      const amountMatch = cols[3].match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   263→
   264→      if (amountMatch) {
   265→        items.push({
   266→          description,
   267→          quantity: qtyMatch ? parseFloat(qtyMatch[1]) : 1,
   268→          rate: rateMatch ? parseFloat(rateMatch[1].replace(/,/g, "")) : 0,
   269→          amount: parseFloat(amountMatch[1].replace(/,/g, "")),
   270→        });
   271→        continue;
   272→      }
   273→    }
   274→
   275→    if (cols.length >= 3) {
   276→      // 3-column format: description, rate/qty, amount
   277→      const description = cols[0];
   278→      const secondLast = cols[cols.length - 2];
   279→      const lastCol = cols[cols.length - 1];
   280→      const amountMatch = lastCol.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   281→      const rateMatch = secondLast.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   282→
   283→      if (amountMatch) {
   284→        const amount = parseFloat(amountMatch[1].replace(/,/g, ""));
   285→        const rate = rateMatch ? parseFloat(rateMatch[1].replace(/,/g, "")) : 0;
   286→        const quantity = rate > 0 ? Math.round((amount / rate) * 100) / 100 : 1;
   287→        items.push({ description, quantity, rate, amount });
   288→      }
   289→    }
   290→  }
   291→
   292→  return items;
   293→}

   297→export async function parseDOCX(arrayBuffer: ArrayBuffer): Promise<ProposalSection[]> {
   298→  const mammoth = await import("mammoth");
   299→  const result = await mammoth.convertToHtml({ arrayBuffer });
   300→  const html = result.value;
   301→
   302→  return parseHTML(html);
   303→}

   305→function parseHTML(html: string): ProposalSection[] {
   306→  const sections: ProposalSection[] = [];
   307→  const doc = new DOMParser().parseFromString(html, "text/html");
   308→  const body = doc.body;
   309→  if (!body) return sections;
   310→
   311→  const children = Array.from(body.children);
   312→
   313→  for (const el of children) {
   314→    const tagName = el.tagName.toLowerCase();
   315→    const text = el.textContent?.trim() || "";
   316→
   317→    if (!text) continue;
   318→
   319→    if (tagName.match(/^h[1-6]$/)) {
   320→      // Heading element — detect special heading types
   321→      const sectionType = classifyHeading(text);
   322→      sections.push({
   323→        id: generateId(),
   324→        type: sectionType,
   325→        content: text,
   326→      });
   327→    } else if (tagName === "table") {
   328→      // Table element — check what kind of table
   329→      const tableText = el.textContent || "";
   330→
   331→      if (hasNumericContent(tableText)) {
   332→        // Check for qty/rate/amount columns (invoice-style)
   333→        const headerRow = el.querySelector("tr");
   334→        const headerText = headerRow?.textContent?.toLowerCase() || "";
   335→        if (headerText.includes("qty") || headerText.includes("quantity") || headerText.includes("rate")) {
   336→          // Invoice-style line items
   337→          const items = parseHTMLTableToLineItems(el);
   338→          sections.push({
   339→            id: generateId(),
   340→            type: "pricing",
   341→            content: "Line Items",
   342→            metadata: { items, hasQuantity: true },
   343→          });
   344→        } else {
   345→          // Standard pricing table
   346→          const items = parseHTMLTableToPricingItems(el);
   347→          sections.push({
   348→            id: generateId(),
   349→            type: "pricing",
   350→            content: "Pricing",
   351→            metadata: { items },
   352→          });
   353→        }
   354→      } else if (looksLikeClientInfo(tableText)) {
   355→        sections.push({
   356→          id: generateId(),
   357→          type: "client_info",
   358→          content: tableText.trim(),
   359→        });
   360→      } else if (looksLikeSenderInfo(tableText)) {
   361→        sections.push({
   362→          id: generateId(),
   363→          type: "sender_info",
   364→          content: tableText.trim(),
   365→        });
   366→      } else {
   367→        sections.push({
   368→          id: generateId(),
   369→          type: "text",
   370→          content: tableText.trim(),
   371→        });
   372→      }
   373→    } else if (tagName === "ul" || tagName === "ol") {
   374→      // List items — format as text with bullets
   375→      const items = Array.from(el.querySelectorAll("li")).map(
   376→        li => `• ${li.textContent?.trim() || ""}`
   377→      );
   378→      const listText = items.join("\n");
   379→
   380→      // Check if this list is part of scope of work
   381→      if (looksLikeScopeOfWork(listText)) {
   382→        sections.push({
   383→          id: generateId(),
   384→          type: "scope_of_work",
   385→          content: listText,
   386→        });
   387→      } else if (looksLikeMilestone(listText)) {
   388→        sections.push({
   389→          id: generateId(),
   390→          type: "milestone",
   391→          content: listText,
   392→          metadata: { milestones: parseMilestoneText(listText) },
   393→        });
   394→      } else {
   395→        sections.push({
   396→          id: generateId(),
   397→          type: "text",
   398→          content: listText,
   399→        });
   400→      }
   401→    } else if (tagName === "p") {
   402→      // Paragraph — classify by content
   403→      classifyParagraph(sections, text);
   404→    } else {
   405→      // Other elements (div, etc.) — try to classify
   406→      classifyParagraph(sections, text);
   407→    }
   408→  }
   409→
   410→  return sections.length > 0 ? sections : [
   411→    { id: generateId(), type: "text", content: html.replace(/<[^>]*>/g, "").trim() },
   412→  ];
   413→}

   415→/** Classify a heading into a specific section type */
   416→function classifyHeading(text: string): ProposalSectionType {
   417→  const lower = text.toLowerCase().trim();
   418→
   419→  if (/^(executive\s*summary|overview|introduction|about\s*this\s*proposal)/i.test(lower)) return "summary";
   420→  if (/^(scope\s*of\s*work|deliverables|what\s*we\s*deliver|our\s*services)/i.test(lower)) return "scope_of_work";
   421→  if (/^(pricing|investment|cost|budget|fees|rates)/i.test(lower)) return "pricing";
   422→  if (/^(terms|terms\s*and\s*conditions|conditions|legal)/i.test(lower)) return "terms";
   423→  if (/^(timeline|milestones|schedule|phases|project\s*plan)/i.test(lower)) return "milestone";
   424→  if (/^(about\s*us|our\s*team|team|who\s*we\s*are|company)/i.test(lower)) return "sender_info";
   425→  if (/^(client|customer|bill\s*to|for)/i.test(lower)) return "client_info";
   426→
   427→  return "heading";
   428→}

   430→/** Classify a paragraph of text into the right section type */
   431→function classifyParagraph(sections: ProposalSection[], text: string): void {
   432→  if (!text.trim()) return;
   433→
   434→  if (looksLikeTerms(text)) {
   435→    sections.push({ id: generateId(), type: "terms", content: text.trim() });
   436→  } else if (looksLikeClientInfo(text)) {
   437→    sections.push({ id: generateId(), type: "client_info", content: text.trim() });
   438→  } else if (looksLikeSenderInfo(text)) {
   439→    sections.push({ id: generateId(), type: "sender_info", content: text.trim() });
   440→  } else if (looksLikeSummary(text)) {
   441→    sections.push({ id: generateId(), type: "summary", content: text.trim() });
   442→  } else if (looksLikeScopeOfWork(text)) {
   443→    sections.push({ id: generateId(), type: "scope_of_work", content: text.trim() });
   444→  } else if (looksLikeMilestone(text)) {
   445→    sections.push({
   446→      id: generateId(),
   447→      type: "milestone",
   448→      content: text.trim(),
   449→      metadata: { milestones: parseMilestoneText(text) },
   450→    });
   451→  } else if (hasNumericContent(text)) {
   452→    // Check if this looks like tabular pricing data
   453→    const lines = text.split("\n");
   454→    const { isTable, rows } = detectTableRows(lines);
   455→
   456→    if (isTable && rows.length > 0) {
   457→      const items = rows.map(row => ({
   458→        name: row[0] || "",
   459→        price: parsePrice(row[1] || row[row.length - 1] || "0"),
   460→      }));
   461→      sections.push({
   462→        id: generateId(),
   463→        type: "pricing",
   464→        content: "Pricing",
   465→        metadata: { items },
   466→      });
   467→    } else {
   468→      const placeholders = detectPlaceholders(text);
   469→      sections.push({
   470→        id: generateId(),
   471→        type: "text",
   472→        content: text.trim(),
   473→        metadata: placeholders.length > 0 ? { placeholders } : undefined,
   474→      });
   475→    }
   476→  } else {
   477→    const placeholders = detectPlaceholders(text);
   478→    sections.push({
   479→      id: generateId(),
   480→      type: "text",
   481→      content: text.trim(),
   482→      metadata: placeholders.length > 0 ? { placeholders } : undefined,
   483→    });
   484→  }
   485→}

   487→/** Parse milestone text into structured milestone objects */
   488→function parseMilestoneText(text: string): { name: string; weeks: number; description?: string }[] {
   489→  const milestones: { name: string; weeks: number; description?: string }[] = [];
   490→  const lines = text.split("\n");
   491→
   492→  for (const line of lines) {
   493→    const trimmed = line.trim().replace(/^[•\-–*]\s*/, "");
   494→    if (!trimmed) continue;
   495→
   496→    // Try to extract duration info
   497→    const weekMatch = trimmed.match(/(\d+)\s*[-–]\s*(\d+)\s*weeks?/i);
   498→    const weekMatch2 = trimmed.match(/(\d+)\s*weeks?/i);
   499→    const monthMatch = trimmed.match(/(\d+)\s*[-–]\s*(\d+)\s*months?/i);
   500→    const monthMatch2 = trimmed.match(/(\d+)\s*months?/i);
   501→
   502→    let weeks = 0;
   503→    if (weekMatch) {
   504→      weeks = parseInt(weekMatch[2]); // Use the upper bound
   505→    } else if (weekMatch2) {
   506→      weeks = parseInt(weekMatch2[1]);
   507→    } else if (monthMatch) {
   508→      weeks = parseInt(monthMatch[2]) * 4;
   509→    } else if (monthMatch2) {
   510→      weeks = parseInt(monthMatch2[1]) * 4;
   511→    }
   512→
   513→    // Extract name (remove the duration part)
   514→    let name = trimmed;
   515→    name = name.replace(/\s*[\(–\-—]\s*\d+\s*[-–]\s*\d+\s*(weeks?|months?)\s*\)?/gi, "");
   516→    name = name.replace(/\s*[\(–\-—]\s*\d+\s*(weeks?|months?)\s*\)?/gi, "");
   517→    name = name.replace(/:\s*\d+\s*[-–]\s*\d+\s*(weeks?|months?)/gi, "");
   518→    name = name.replace(/:\s*\d+\s*(weeks?|months?)/gi, "");
   519→    name = name.trim();
   520→
   521→    if (name) {
   522→      milestones.push({ name, weeks: weeks || 2 });
   523→    }
   524→  }
   525→
   526→  return milestones;
   527→}

   529→function parseHTMLTableToPricingItems(tableEl: Element): { name: string; price: number }[] {
   530→  const items: { name: string; price: number }[] = [];
   531→  const rows = Array.from(tableEl.querySelectorAll("tr"));
   532→
   533→  for (const row of rows) {
   534→    const cells = Array.from(row.querySelectorAll("td, th"));
   535→    if (cells.length < 2) continue;
   536→
   537→    const nameCell = cells[0].textContent?.trim() || "";
   538→    const priceText = cells[cells.length - 1].textContent?.trim() || "0";
   539→
   540→    // Skip header rows
   541→    if (/^(item|description|service|delivable|qty|quantity|rate|price|amount|total)/i.test(nameCell)) continue;
   542→    if (/^(subtotal|total|tax|vat|gst|discount)/i.test(nameCell)) continue;
   543→
   544→    // Parse price — try last column first, then second-to-last
   545→    const priceMatch = priceText.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   546→    const price = priceMatch ? parseFloat(priceMatch[1].replace(/,/g, "")) : 0;
   547→
   548→    if (nameCell) {
   549→      items.push({ name: nameCell, price });
   550→    }
   551→  }
   552→
   553→  if (items.length === 0) {
   554→    const tableText = tableEl.textContent?.trim() || "Service";
   555→    items.push({ name: tableText.substring(0, 100), price: 0 });
   556→  }
   557→
   558→  return items;

   561→/** Parse HTML table into line items with qty/rate/amount */
   562→function parseHTMLTableToLineItems(tableEl: Element): { description: string; quantity: number; rate: number; amount: number }[] {
   563→  const items: { description: string; quantity: number; rate: number; amount: number }[] = [];
   564→  const rows = Array.from(tableEl.querySelectorAll("tr"));
   565→
   566→  for (const row of rows) {
   567→    const cells = Array.from(row.querySelectorAll("td"));
   568→    if (cells.length < 3) continue;
   569→
   570→    const description = cells[0].textContent?.trim() || "";
   571→
   572→    // Skip header rows
   573→    if (/^(item|description|service|qty|quantity|rate|price|amount)/i.test(description)) continue;
   574→    if (/^(subtotal|total|tax|vat|gst|discount)/i.test(description)) continue;
   575→    if (!description) continue;
   576→
   577→    // Try to find qty, rate, amount in the remaining cells
   578→    let quantity = 1;
   579→    let rate = 0;
   580→    let amount = 0;
   581→
   582→    if (cells.length >= 4) {
   583→      // description, qty, rate, amount
   584→      const qtyMatch = cells[1].textContent?.match(/([\d.]+)/);
   585→      const rateMatch = cells[2].textContent?.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   586→      const amountMatch = cells[3].textContent?.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   587→      quantity = qtyMatch ? parseFloat(qtyMatch[1]) : 1;
   588→      rate = rateMatch ? parseFloat(rateMatch[1].replace(/,/g, "")) : 0;
   589→      amount = amountMatch ? parseFloat(amountMatch[1].replace(/,/g, "")) : 0;
   590→    } else if (cells.length === 3) {
   591→      // description, rate, amount
   592→      const rateMatch = cells[1].textContent?.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   593→      const amountMatch = cells[2].textContent?.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
   594→      rate = rateMatch ? parseFloat(rateMatch[1].replace(/,/g, "")) : 0;
   595→      amount = amountMatch ? parseFloat(amountMatch[1].replace(/,/g, "")) : 0;
   596→      quantity = rate > 0 ? Math.round((amount / rate) * 100) / 100 : 1;
   597→    }
   598→
   599→    items.push({ description, quantity, rate, amount });
   600→  }
   601→
   602→  if (items.length === 0) {
   603→    const tableText = tableEl.textContent?.trim() || "Service";
   604→    items.push({ description: tableText.substring(0, 100), quantity: 1, rate: 0, amount: 0 });
   605→  }
   606→
   607→  return items;
   608→}

   612→interface PDFTextItem {
   613→  text: string;
   614→  x: number;
   615→  y: number;

   617→  height: number;

   623→export async function parsePDF(arrayBuffer: ArrayBuffer): Promise<ProposalSection[]> {
   624→  const pdfjs = await import("pdfjs-dist");
   625→
   626→  // Set worker source
   627→  pdfjs.GlobalWorkerOptions.workerSrc =
   628→    "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/4.4.168/pdf.worker.min.mjs";
   629→
   630→  const pdf = await pdfjs.getDocument({ data: arrayBuffer }).promise;
   631→  const allItems: PDFTextItem[] = [];
   632→  let bodyFontSize = 12;
   633→
   634→  const fontSizes: number[] = [];
   635→
   636→  for (let pageIdx = 1; pageIdx <= pdf.numPages; pageIdx++) {
   637→    const page = await pdf.getPage(pageIdx);
   638→    const textContent = await page.getTextContent();
   639→
   640→    for (const item of textContent.items as any[]) {
   641→      if (!item.str || !item.str.trim()) continue;
   642→
   643→      const fontSize = item.height || item.transform?.[0] || 12;
   644→      fontSizes.push(fontSize);
   645→
   646→      allItems.push({
   647→        text: item.str.trim(),
   648→        x: item.transform?.[4] || 0,
   649→        y: item.transform?.[5] || 0,
   650→        width: item.width || 0,
   651→        height: item.height || fontSize,
   652→        fontSize,
   653→        fontName: item.fontName || "",
   654→        page: pageIdx,
   655→      });
   656→    }
   657→  }
   658→
   659→  if (allItems.length === 0) {
   660→    return [{ id: generateId(), type: "text", content: "(No text could be extracted from this PDF — it may be a scanned image. OCR is not supported.)" }];
   661→  }
   662→
   663→  // Determine body font size (most common)
   664→  if (fontSizes.length > 0) {
   665→    const sizeCounts = new Map<number, number>();
   666→    for (const s of fontSizes) {
   667→      sizeCounts.set(s, (sizeCounts.get(s) || 0) + 1);
   668→    }
   669→    let maxCount = 0;
   670→    for (const [size, count] of sizeCounts) {
   671→      if (count > maxCount) {
   672→        maxCount = count;
   673→        bodyFontSize = size;
   674→      }
   675→    }
   676→  }
   677→
   678→  // Group text items into lines by Y position (within tolerance)
   679→  const LINE_TOLERANCE = 3;
   680→  const lines: { y: number; items: PDFTextItem[]; page: number }[] = [];
   681→
   682→  for (const item of allItems) {
   683→    let foundLine = false;
   684→    for (const line of lines) {
   685→      if (Math.abs(line.y - item.y) <= LINE_TOLERANCE && line.page === item.page) {
   686→        line.items.push(item);
   687→        foundLine = true;
   688→        break;
   689→      }
   690→    }
   691→    if (!foundLine) {
   692→      lines.push({ y: item.y, items: [item], page: item.page });
   693→    }
   694→  }
   695→
   696→  // Sort lines: by page first, then by Y (top to bottom within each page)
   697→  lines.sort((a, b) => {
   698→    if (a.page !== b.page) return a.page - b.page;
   699→    return b.y - a.y;
   700→  });
   701→
   702→  // Sort items within each line by X (left to right)
   703→  for (const line of lines) {
   704→    line.items.sort((a, b) => a.x - b.x);
   705→  }
   706→
   707→  // Detect multi-column layouts (e.g., side-by-side address blocks)
   708→  // Group consecutive lines with similar Y-gap and check for column splits
   709→  const processedSections: ProposalSection[] = [];
   710→
   711→  // Convert lines to text
   712→  const lineTexts = lines.map(line => {
   713→    return line.items.map(i => i.text).join(" ");
   714→  });
   715→
   716→  // Detect headings by font size (significantly larger than body text)
   717→  const HEADING_THRESHOLD = bodyFontSize * 1.3;
   718→  const lineMeta = lines.map((line, idx) => {
   719→    const maxFontSize = Math.max(...line.items.map(i => i.fontSize));
   720→    const isBold = line.items.some(i =>
   721→      i.fontName.toLowerCase().includes("bold") ||
   722→      i.fontName.toLowerCase().includes("heavy") ||
   723→      i.fontName.toLowerCase().includes("black")
   724→    );
   725→    const isHeading = maxFontSize >= HEADING_THRESHOLD || (isBold && maxFontSize > bodyFontSize);
   726→    return { text: lineTexts[idx], isHeading, isBold, fontSize: maxFontSize, page: line.page, y: line.y };
   727→  });
   728→
   729→  // Check for multi-column layout on the first group of lines
   730→  // (common in invoices: company info on left, invoice meta on right)
   731→  const firstPageItems = allItems.filter(i => i.page === 1);
   732→  const multiColumn = detectMultiColumnLayout(firstPageItems);
   733→
   734→  // Group consecutive non-heading lines into paragraphs
   735→  const sections: ProposalSection[] = [];
   736→  let currentParagraph: string[] = [];
   737→
   738→  // If we detected multi-column layout on first page, create client/sender sections
   739→  if (multiColumn) {
   740→    if (looksLikeSenderInfo(multiColumn.left) || looksLikeSenderInfo(multiColumn.right)) {
   741→      const senderText = looksLikeSenderInfo(multiColumn.left) ? multiColumn.left : multiColumn.right;
   742→      const otherText = looksLikeSenderInfo(multiColumn.left) ? multiColumn.right : multiColumn.left;
   743→      sections.push({ id: generateId(), type: "sender_info", content: senderText });
   744→
   745→      if (looksLikeClientInfo(otherText)) {
   746→        sections.push({ id: generateId(), type: "client_info", content: otherText });
   747→      } else if (looksLikeInvoiceMeta(otherText)) {
   748→        sections.push({ id: generateId(), type: "text", content: otherText, metadata: { likelyInvoiceMeta: true } });
   749→      } else {
   750→        sections.push({ id: generateId(), type: "text", content: otherText });
   751→      }
   752→    }
   753→  }
   754→
   755→  for (const meta of lineMeta) {
   756→    if (meta.isHeading) {
   757→      // Flush current paragraph
   758→      if (currentParagraph.length > 0) {
   759→        const paraText = currentParagraph.join("\n");
   760→        addTextSection(sections, paraText);
   761→        currentParagraph = [];
   762→      }
   763→
   764→      // Classify the heading
   765→      const sectionType = classifyHeading(meta.text);
   766→      sections.push({
   767→        id: generateId(),
   768→        type: sectionType,
   769→        content: meta.text,
   770→      });
   771→    } else {
   772→      currentParagraph.push(meta.text);
   773→    }
   774→  }
   775→
   776→  // Flush remaining paragraph
   777→  if (currentParagraph.length > 0) {
   778→    const paraText = currentParagraph.join("\n");
   779→    addTextSection(sections, paraText);
   780→  }
   781→
   782→  return sections.length > 0 ? sections : [
   783→    { id: generateId(), type: "text", content: allItems.map(i => i.text).join(" ") },
   784→  ];
   785→}

   789→export function parseTXT(text: string): ProposalSection[] {
   790→  const sections: ProposalSection[] = [];
   791→  const lines = text.split(/\r?\n/);
   792→
   793→  let currentParagraph: string[] = [];
   794→
   795→  for (const rawLine of lines) {
   796→    const line = rawLine.trim();
   797→
   798→    if (!line) {
   799→      // Blank line — flush paragraph
   800→      if (currentParagraph.length > 0) {
   801→        const paraText = currentParagraph.join("\n");
   802→        addTextSection(sections, paraText);
   803→        currentParagraph = [];
   804→      }
   805→      continue;
   806→    }
   807→
   808→    // Detect headings: short lines, possibly ALL CAPS, or starting with #
   809→    const isMarkdownHeading = /^#{1,6}\s/.test(line);
   810→    const isAllCaps = line === line.toUpperCase() && line.length > 2 && /[A-Z]/.test(line);
   811→    const isShortLine = line.length < 60 && !line.endsWith(".") && !line.endsWith(",");
   812→
   813→    if (isMarkdownHeading || (isAllCaps && isShortLine)) {
   814→      // Flush current paragraph
   815→      if (currentParagraph.length > 0) {
   816→        const paraText = currentParagraph.join("\n");
   817→        addTextSection(sections, paraText);
   818→        currentParagraph = [];
   819→      }
   820→
   821→      const headingText = isMarkdownHeading ? line.replace(/^#+\s*/, "") : line;
   822→      const sectionType = classifyHeading(headingText);
   823→      sections.push({
   824→        id: generateId(),
   825→        type: sectionType,
   826→        content: headingText,
   827→      });
   828→    } else if (line.startsWith("---") || line.startsWith("===") || line.startsWith("***")) {
   829→      // Divider
   830→      if (currentParagraph.length > 0) {
   831→        const paraText = currentParagraph.join("\n");
   832→        addTextSection(sections, paraText);
   833→        currentParagraph = [];
   834→      }
   835→      sections.push({
   836→        id: generateId(),
   837→        type: "divider",
   838→        content: "",
   839→      });
   840→    } else {
   841→      currentParagraph.push(line);
   842→    }
   843→  }
   844→
   845→  // Flush remaining paragraph
   846→  if (currentParagraph.length > 0) {
   847→    const paraText = currentParagraph.join("\n");
   848→    addTextSection(sections, paraText);
   849→  }
   850→
   851→  return sections.length > 0 ? sections : [
   852→    { id: generateId(), type: "text", content: text },
   853→  ];
   854→}

   858→function addTextSection(sections: ProposalSection[], text: string): void {
   859→  if (!text.trim()) return;

   861→  // Try specialized classifications first
   862→  if (looksLikeTerms(text)) {
   863→    sections.push({ id: generateId(), type: "terms", content: text.trim() });
   864→  } else if (looksLikeClientInfo(text)) {
   865→    sections.push({ id: generateId(), type: "client_info", content: text.trim() });
   866→  } else if (looksLikeSenderInfo(text)) {
   867→    sections.push({ id: generateId(), type: "sender_info", content: text.trim() });
   868→  } else if (looksLikeSummary(text)) {
   869→    sections.push({ id: generateId(), type: "summary", content: text.trim() });
   870→  } else if (looksLikeScopeOfWork(text)) {
   871→    sections.push({ id: generateId(), type: "scope_of_work", content: text.trim() });
   872→  } else if (looksLikeMilestone(text)) {
   873→    sections.push({
   874→      id: generateId(),
   875→      type: "milestone",
   876→      content: text.trim(),
   877→      metadata: { milestones: parseMilestoneText(text) },
   878→    });
   879→  } else if (hasNumericContent(text)) {
   880→    // Check if this looks like tabular pricing data
   881→    const lines = text.split("\n");
   882→    const { isTable, rows } = detectTableRows(lines);
   883→
   884→    if (isTable && rows.length > 0) {
   885→      const items = rows.map(row => ({
   886→        name: row[0] || "",
   887→        price: parsePrice(row[1] || row[row.length - 1] || "0"),
   888→      }));
   889→      sections.push({
   890→        id: generateId(),
   891→        type: "pricing",
   892→        content: "Pricing",
   893→        metadata: { items },
   894→      });
   895→    } else {
   896→      const placeholders = detectPlaceholders(text);
   897→      sections.push({
   898→        id: generateId(),
   899→        type: "text",
   900→        content: text.trim(),
   901→        metadata: placeholders.length > 0 ? { placeholders } : undefined,
   902→      });
   903→    }
   904→  } else {
   905→    const placeholders = detectPlaceholders(text);
   906→    sections.push({
   907→      id: generateId(),
   908→      type: "text",
   909→      content: text.trim(),
   910→      metadata: placeholders.length > 0 ? { placeholders } : undefined,
   911→    });
   912→  }
   913→}

   943→/** Parse with detailed result including confidence and warnings */
   944→export async function parseUploadedTemplateWithMeta(file: File): Promise<ParseResult<ProposalSection>> {

   958→    } else if (ext === "pdf") {
   959→      confidence = typedSections.length >= 2 ? "high" : typedSections.length >= 1 ? "medium" : "low";
   960→    } else {
   961→      confidence = typedSections.length >= 2 ? "medium" : "low";
   962→    }
   963→
   964→    // Add warnings for common issues
   965→    if (textSections.length > typedSections.length * 2) {
   966→      warnings.push("Many sections were classified as generic text. You may want to reclassify them manually.");
   967→    }
   968→    if (ext === "pdf" && sections.length <= 2) {
   969→      warnings.push("Very few sections detected. The PDF may be scanned (image-based) — OCR is not supported.");
   970→    }
   971→    if (sections.some(s => s.content.includes("(No text could be extracted"))) {
   972→      warnings.push("No text could be extracted. This PDF appears to be a scanned image.");
   973→    }
   974→
   975→    return { sections, warnings, confidence, fileType: ext };
   976→  } catch (err: any) {
   977→    warnings.push(err.message || "Failed to parse file");
   978→    return {
   979→      sections: [{ id: generateId(), type: "text", content: `Error parsing file: ${err.message}` }],
   980→      warnings,
   981→      confidence: "low",
   982→      fileType: ext,
   983→    };
   984→  }
   985→}

  1022→/** Convert ProposalSection[] to InvoiceSection[] with invoice-specific reclassification */

  1044→        metadata: section.metadata,
  1045→      });
  1046→      continue;
  1047→    }
  1048→
  1049→    // Check for total/grand total
  1050→    if (section.type === "text" && looksLikeTotal(section.content)) {
  1051→      const amountMatch = section.content.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);

  1058→      continue;
  1059→    }
  1060→
  1061→    // Check for discount
  1062→    if (section.type === "text" && looksLikeDiscount(section.content)) {
  1063→      const amountMatch = section.content.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
  1064→      const rateMatch = section.content.match(/(\d+(?:\.\d+)?)\s*%/);
  1065→      invoiceSections.push({
  1066→        id: section.id,
  1067→        type: "discount",
  1068→        content: section.content,
  1069→        metadata: {
  1070→          ...(amountMatch ? { amount: parseFloat(amountMatch[1].replace(/,/g, "")) } : {}),
  1071→          ...(rateMatch ? { rate: parseFloat(rateMatch[1]) } : {}),
  1072→        },
  1073→      });
  1074→      continue;
  1075→    }
  1076→
  1077→    // Check for notes
  1078→    if (section.type === "text" && looksLikeNotes(section.content)) {
  1079→      invoiceSections.push({
  1080→        id: section.id,
  1081→        type: "notes",
  1082→        content: section.content,
  1083→        metadata: section.metadata,
  1084→      });
  1085→      continue;
  1086→    }
  1087→
  1088→    // Check for subtotal/tax
  1089→    if (section.type === "text") {
  1090→      const subTax = looksLikeSubtotalOrTax(section.content.split("\n")[0]);
  1091→      if (subTax === "subtotal") {
  1092→        const amountMatch = section.content.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
  1093→        invoiceSections.push({
  1094→          id: section.id,
  1095→          type: "subtotal",
  1096→          content: section.content,
  1097→          metadata: amountMatch ? { amount: parseFloat(amountMatch[1].replace(/,/g, "")) } : undefined,
  1098→        });
  1099→        continue;
  1100→      }
  1101→      if (subTax === "tax") {
  1102→        const amountMatch = section.content.match(/[$€£¥₹]?\s*([\d,]+(?:\.\d{2})?)/);
  1103→        const rateMatch = section.content.match(/(\d+(?:\.\d+)?)\s*%/);
  1104→        invoiceSections.push({
  1105→          id: section.id,
  1106→          type: "tax",
  1107→          content: section.content,
  1108→          metadata: {
  1109→            ...(amountMatch ? { amount: parseFloat(amountMatch[1].replace(/,/g, "")) } : {}),
  1110→            ...(rateMatch ? { rate: parseFloat(rateMatch[1]) } : {}),
  1111→          },
  1112→        });
  1113→        continue;
  1114→      }
  1115→    }
  1116→
  1117→    // Convert pricing to line_items
  1118→    if (section.type === "pricing") {
  1119→      const pricingItems = section.metadata?.items || [];
  1120→      const hasQuantity = section.metadata?.hasQuantity;
  1121→
  1122→      let lineItems: { description: string; quantity: number; rate: number; amount: number }[];
  1123→
  1124→      if (hasQuantity && pricingItems.length > 0) {
  1125→        // Items already have qty/rate/amount
  1126→        lineItems = pricingItems.map((item: any) => ({
  1127→          description: item.name || item.description,
  1128→          quantity: item.quantity || 1,
  1129→          rate: item.rate || item.price || 0,
  1130→          amount: item.amount || item.price || 0,
  1131→        }));
  1132→      } else {
  1133→        // Simple name+price items — convert to line items
  1134→        lineItems = pricingItems.map((item: { name: string; price: number }) => ({
  1135→          description: item.name,
  1136→          quantity: 1,
  1137→          rate: item.price,
  1138→          amount: item.price,
  1139→        }));
  1140→      }
  1141→
  1142→      invoiceSections.push({
  1143→        id: section.id,
  1144→        type: "line_items",
  1145→        content: section.content === "Pricing" ? "Line Items" : section.content,
  1146→        metadata: { items: lineItems.length > 0 ? lineItems : parseInvoiceLineItems(section.content) },
  1147→      });
  1148→      continue;
  1149→    }
  1150→
  1151→    // client_info and sender_info map directly
  1152→    if (section.type === "client_info" || section.type === "sender_info") {
  1153→      invoiceSections.push({
  1154→        id: section.id,
  1155→        type: section.type as any,
  1156→        content: section.content,
  1157→        metadata: section.metadata,
  1158→      });
  1159→      continue;
  1160→    }
  1161→
  1162→    // Direct mapping for types that are the same
  1163→    if (section.type === "heading" || section.type === "text" || section.type === "terms" || section.type === "divider") {
  1164→      invoiceSections.push({
  1165→        id: section.id,
  1166→        type: section.type as any,
  1167→        content: section.content,
  1168→        metadata: section.metadata,
  1169→      });
  1170→      continue;
  1171→    }
  1172→
  1173→    // scope_of_work, summary, milestone → text (not relevant for invoices)
  1174→    invoiceSections.push({
  1175→      id: section.id,
  1176→      type: "text",
  1177→      content: section.content,
  1178→      metadata: section.metadata,
  1179→    });
  1180→  }
  1181→
  1182→  return invoiceSections;
  1183→}
