   117→function getTierLevel(tier: string): number {
   118→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
   119→  return levels[tier] ?? 0;
   120→}
   121→
   122→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   123→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   124→}

   234→    const end = new Date(now);
   235→    end.setDate(end.getDate() + 30);

   240→

   242→  const [dateFrom, setDateFrom] = useState(() => {

   245→    return d.toISOString().split("T")[0];
   246→  });

   250→
   251→  // Build query args — only include workspaceId when it has a value so Convex
   252→  // doesn't reject the call with an invalid optional ID.
   253→  const libraryQueryArgs = useMemo(() => ({
   254→    view: viewMode,
   255→    startDate: libraryDateRange.start,
   256→    endDate: libraryDateRange.end,
   257→    ...(workspaceId ? { workspaceId } : {}),
   258→  }), [viewMode, libraryDateRange.start, libraryDateRange.end, workspaceId]);
   259→

   261→    view: "date" as const,
   262→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   263→    endDate: Date.now(),
   264→    ...(workspaceId ? { workspaceId } : {}),
   265→  }), [workspaceId]);
   266→
   267→  const timelineQueryArgs = useMemo(() => ({
   268→    date: startOfDay,
   269→    ...(workspaceId ? { workspaceId } : {}),
   270→  }), [startOfDay, workspaceId]);
   271→
   272→  // Library data with ±30 day window for monitoring
   273→  const evidenceData = useQuery(
   274→    api.evidence.library.getEvidenceLibraryData,
   275→    libraryQueryArgs
   276→  );
   277→
   278→  // Export data with full year window for comprehensive exports
   279→  const exportData = useQuery(
   280→    api.evidence.library.getEvidenceLibraryData,
   281→    exportQueryArgs
   282→  ) as any | undefined;
   283→
   284→  const timelineData = useQuery(
   285→    api.evidence.library.getEvidenceTimeline,
   286→    timelineQueryArgs
   287→  );
   288→
   289→  const clients = useQuery(api.clients.crud.getClients, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   290→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   291→
   292→  const isLoading = evidenceData === undefined && timelineData === undefined;
   293→  const exportLoading = exportData === undefined && clients === undefined;
   294→  const timedOut = useQueryTimeout(isLoading, 5000);
   295→  const showLoading = isLoading && !timedOut && !isDisconnected;
   296→
   297→  // ── Safe fallbacks ──
   298→  const safeEvidenceData = evidenceData ?? {
   299→    totalCount: 0,
   300→    disputeSuccessRate: 0,
   301→    contentQualityScore: 0,
   302→    gapPrediction: {
   303→      status: "no_gaps" as const,
   304→      message: "No Gaps Predicted",
   305→      description: "Start collecting evidence to build your protection.",
   306→      nextGapTime: null,
   307→      missingTypes: [],
   308→    },
   309→    evidenceItems: [],
   310→    healthScore: null,
   311→    disputeData: null,
   312→    contentData: null,
   313→  };
   314→
   315→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   316→
   317→  // ── Export State ──
   318→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   319→  const [selectedProject, setSelectedProject] = useState("all");
   320→  const [selectedClient, setSelectedClient] = useState("all");
   321→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   322→  const [isExporting, setIsExporting] = useState(false);
   323→  const [previewOpen, setPreviewOpen] = useState(false);
   324→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   325→
   326→  // ── Derived Export Data ──
   327→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   328→    if (!exportData) return [];
   329→    const items = exportData.evidenceItems ?? [];
   330→    const typeCounts: Record<string, number> = {};
   331→    const typeLabels: Record<string, string> = {
   332→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   333→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   334→    };
   335→    const typeIcons: Record<string, React.ReactNode> = {
   336→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,
   337→      url: <FileJson className="h-4 w-4" />, keyboard: <FileText className="h-4 w-4" />,
   338→      mouse: <FileText className="h-4 w-4" />, platform_status: <Shield className="h-4 w-4" />,
   339→    };
   340→    for (const item of items) {
   341→      const kind = item.type ?? "unknown";
   342→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   343→    }
   344→    return Object.entries(typeCounts).map(([id, count]) => ({
   345→      id,
   346→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   347→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   348→      count,
   349→    }));
   350→  }, [exportData]);
   351→
   352→  const projectOptions = useMemo(() => {
   353→    const all = [{ id: "all", label: "All Projects" }];
   354→    if (scopeDefinitions) {
   355→      for (const scope of scopeDefinitions) {
   356→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   357→      }
   358→    }
   359→    return all;
   360→  }, [scopeDefinitions]);
   361→
   362→  const clientOptions = useMemo(() => {
   363→    const all = [{ id: "all", label: "All Clients" }];
   364→    if (clients) {
   365→      for (const client of clients) {
   366→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   367→      }
   368→    }
   369→    return all;
   370→  }, [clients]);
   371→
   372→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;
   373→  const totalItems = exportData?.totalCount ?? safeEvidenceData.totalCount ?? 0;
   374→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;
   375→
   376→  // Auto-select all evidence types when they load
   377→  useEffect(() => {
   378→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   379→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   380→    }
   381→  }, [evidenceTypes]);
   382→
   383→  const totalEvidenceItems = useMemo(
   384→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   385→    [evidenceTypes, selectedEvidenceTypes]
   386→  );
   387→
   388→  const isFormatLocked = (format: ExportFormat): boolean => {
   389→    return !hasTierAccess(subscriptionTier, FORMAT_CONFIG[format].tierRequired);
   390→  };
   391→
   392→  const toggleEvidenceType = (id: string) => {
   393→    setSelectedEvidenceTypes((prev) =>
   394→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   395→    );
   396→  };
   397→
   398→  const handleExport = async () => {
   399→    if (selectedEvidenceTypes.length === 0) {
   400→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   401→      return;
   402→    }
   403→    if (isFormatLocked(selectedFormat)) {
   404→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   405→      return;
   406→    }
   407→    setIsExporting(true);
   408→    try {
   409→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   410→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   411→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   412→
   413→      exportEvidence(selectedFormat, {
   414→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   415→        dateRange: `${dateFrom} to ${dateTo}`,
   416→        project: selectedProjectLabel,
   417→        client: selectedClientLabel,
   418→        complianceScore,
   419→        totalItems: totalEvidenceItems,
   420→      });
   421→
   422→      const newExport: RecentExport = {
   423→        id: `exp_${Date.now()}`,
   424→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   425→        format: selectedFormat,
   426→        status: "completed",
   427→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   428→        date: new Date().toISOString().split("T")[0],
   429→        itemCount: totalEvidenceItems,
   430→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   431→        project: selectedProjectLabel,
   432→      };
   433→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   434→
   435→      toast.success("Export complete", {
   436→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   437→      });
   438→
   439→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   440→    } catch {
   441→      toast.error("Export failed", { description: "An error occurred while generating the export file." });
   442→    }
   443→    setIsExporting(false);
   444→  };
   445→
   446→  const handleFormatSelect = (format: ExportFormat) => {
   447→    if (isFormatLocked(format)) {
   448→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.` });
   449→      return;
   450→    }
   451→    setSelectedFormat(format);
   452→  };
   453→
   454→  const previewItems = useMemo(() => {
   455→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   456→  }, [evidenceTypes, selectedEvidenceTypes]);
   457→
   458→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   459→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   460→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   461→  const hasTierAccessFn = (requiredTier: string) => getTierLevel(subscriptionTier) >= getTierLevel(requiredTier);
   462→
   463→  // ── Loading / Timeout State ──
   464→  const [showTimeout, setShowTimeout] = useState(false);
   465→  useEffect(() => {
   466→    if (isLoading) {
   467→      const timer = setTimeout(() => setShowTimeout(true), 5000);
   468→      return () => clearTimeout(timer);
   469→    }
   470→    setShowTimeout(false);
   471→  }, [isLoading]);
   472→
   473→  return (
   474→    <div className="w-full min-h-screen bg-background">
   475→      <PageLayout spaced>
   476→        {/* ── Page Header ── */}
   477→        <div>
   478→          <h1 className="text-2xl md:text-[32px] font-bold text-foreground tracking-tight mb-2">
   479→            Evidence Library
   480→          </h1>
   481→          <p className="text-[16px] text-muted-foreground">
   482→            Monitor capture health, anticipate gaps early, and export airtight proof on demand.
   483→          </p>
   484→        </div>
   485→
   486→        {/* Demo Mode Banner */}
   487→        {!isAuthenticated && <DemoModeBanner />}
   488→
   489→        {/* ════════════════════════════════════════════════════════════════════════
   490→            SECTION 1: EVIDENCE MONITORING (was Evidence Library)
   491→            - Health dashboard, timeline, quality scorecard, item browser
   492→        ════════════════════════════════════════════════════════════════════════ */}
   493→        <section id="monitoring">
   494→          <div className="flex items-center gap-3 mb-4">
   495→            <div className="p-2 rounded-lg bg-primary/10">
   496→              <DatabaseIcon className="w-5 h-5 text-primary" />
   497→            </div>
   498→            <div>
   499→              <h2 className="text-xl font-semibold text-foreground">Evidence Monitoring</h2>
   500→              <p className="text-sm text-muted-foreground">Real-time health, quality analysis, and gap detection</p>
   501→            </div>
   502→          </div>
   503→
   504→          {showLoading ? (
   505→            <div className="text-center py-12">
   506→              <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary mx-auto"></div>
   507→              <p className="mt-4 text-muted-foreground">Loading evidence data...</p>
   508→              {showTimeout && (
   509→                <div className="mt-4 p-4 bg-yellow-50 dark:bg-yellow-900/20 border border-yellow-200 dark:border-yellow-800 rounded-lg">
   510→                  <p className="text-sm text-yellow-800 dark:text-yellow-200">
   511→                    Taking longer than expected. Check your connection or try refreshing the page.
   512→                  </p>
   513→                </div>
   514→              )}
   515→            </div>
   516→          ) : (
   517→            <div className="space-y-6">
   518→              <WorkContentAnalysis
   519→                qualityScore={safeEvidenceData.contentQualityScore}
   520→                hasAccess={hasTierAccessFn("pro")}
   521→                contentData={safeEvidenceData.contentData}
