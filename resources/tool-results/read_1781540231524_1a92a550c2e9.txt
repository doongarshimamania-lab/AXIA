   195→function formatCurrency(value: number): string {
   196→  return new Intl.NumberFormat("en-US", {
   197→    style: "currency",
   198→    currency: "USD",
   199→    minimumFractionDigits: 0,
   200→    maximumFractionDigits: 0,
   201→  }).format(value);
   202→}
   203→
   204→function formatDate(timestamp: number | undefined): string {
   205→  if (!timestamp) return "—";
   206→  return new Intl.DateTimeFormat("en-US", {
   207→    month: "short",
   208→    day: "numeric",
   209→    year: "numeric",
   210→  }).format(new Date(timestamp));
   211→}
   212→
   213→function daysUntilClose(timestamp: number | undefined): number | null {
   214→  if (!timestamp) return null;
   215→  const diff = timestamp - Date.now();
   216→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   217→}

   219→function getSourceInfo(source: string | undefined) {
   220→  return SOURCE_OPTIONS.find((s) => s.value === source) ?? SOURCE_OPTIONS[5];
