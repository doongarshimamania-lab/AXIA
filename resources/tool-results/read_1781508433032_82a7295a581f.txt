     5→function generateToken(): string {
     6→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
     7→  let result = "";
     8→  for (let i = 0; i < 32; i++) result += chars.charAt(Math.floor(Math.random() * chars.length));
     9→  return result;
    10→}
