   131→// ─── Component ───────────────────────────────────────────────────────────────
   132→
   133→export default function Tags() {
   134→  const { isAuthenticated, isLoading: authLoading } = useAuth();
   135→
   136→  // ─── Convex queries & mutations ──────────────────────────────────────────
   137→  const tagsData = useQuery(api.tags.crud.getTags, {});
   138→  const createTagMutation = useMutation(api.tags.crud.createTag);
   139→  const updateTagMutation = useMutation(api.tags.crud.updateTag);
   140→  const deleteTagMutation = useMutation(api.tags.crud.deleteTag);
   141→
   142→  // ─── Local state ────────────────────────────────────────────────────────
   143→  const [searchQuery, setSearchQuery] = useState("");
   144→  const [activeFilter, setActiveFilter] = useState<string | null>(null);
   145→  const [createOpen, setCreateOpen] = useState(false);
   146→  const [editOpen, setEditOpen] = useState(false);
   147→  const [editingTag, setEditingTag] = useState<any>(null);
   148→  const [deleteOpen, setDeleteOpen] = useState(false);
   149→  const [deletingTag, setDeletingTag] = useState<any>(null);
   150→
   151→  // Form state
   152→  const [formName, setFormName] = useState("");
   153→  const [formColor, setFormColor] = useState(PRESET_COLORS[0]);
   154→  const [formCategory, setFormCategory] = useState<string>("general");
   155→
   156→  // Mutation loading state
   157→  const [isCreating, setIsCreating] = useState(false);
   158→  const [isUpdating, setIsUpdating] = useState(false);
   159→  const [isDeleting, setIsDeleting] = useState(false);
   160→
   161→  // ─── Loading timeout pattern ────────────────────────────────────────────
   162→  const { isDisconnected } = useConvexConnectionState();
   163→  const timedOut = useQueryTimeout(!authLoading && tagsData === undefined, 3000);
   164→
   165→  const isLoading = !authLoading && tagsData === undefined && !timedOut && !isDisconnected;
   166→
   167→  // ─── Demo mode ──────────────────────────────────────────────────────────
   168→  const isDemoMode = !authLoading && !isAuthenticated;
   169→
   170→  // ─── Data resolution ────────────────────────────────────────────────────
   171→  const realTags = (tagsData ?? []).map((t: any) => ({
   172→    _id: t._id,
   173→    name: t.name,
   174→    color: t.color,
   175→    category: t.category ?? null,
   176→    usageCount: t.usageCount ?? 0,
   177→    createdAt: t.createdAt,
   178→  }));
   179→
   180→  const tags = isDemoMode ? MOCK_TAGS : realTags;
   181→
   182→  // ── Computed Stats ──
   183→  const totalTags = tags.length;
   184→  const mostUsedTag =
   185→    tags.length > 0
   186→      ? tags.reduce((prev: any, curr: any) => (curr.usageCount > prev.usageCount ? curr : prev))
   187→      : null;
   188→
   189→  // ── Filtering ──
   190→  const filteredTags = tags.filter((tag: any) => {
   191→    const matchesSearch =
   192→      tag.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   193→      (tag.category ?? "").toLowerCase().includes(searchQuery.toLowerCase());
   194→    const matchesFilter = activeFilter ? tag._id === activeFilter : true;
   195→    return matchesSearch && matchesFilter;
   196→  });
   197→
   198→  // ── Helpers ──
   199→  const resetForm = () => {
   200→    setFormName("");
   201→    setFormColor(PRESET_COLORS[0]);
   202→    setFormCategory("general");
   203→  };
   204→
   205→  const formatDate = (timestamp: number) => {
   206→    const date = new Date(timestamp);
   207→    return date.toLocaleDateString("en-US", {
   208→      month: "short",
   209→      day: "numeric",
   210→      year: "numeric",
   211→    });
   212→  };
   213→
   214→  // ── Create ──
   215→  const handleCreate = async () => {
   216→    if (!formName.trim()) {
   217→      toast.error("Tag name is required");
   218→      return;
   219→    }
   220→
   221→    if (isDemoMode) {
   222→      toast.success(`Tag "${formName.trim()}" created successfully! (Demo mode)`);
   223→      resetForm();
   224→      setCreateOpen(false);
   225→      return;
   226→    }
   227→
   228→    setIsCreating(true);
   229→    try {
   230→      await createTagMutation({
   231→        name: formName.trim(),
   232→        color: formColor,
   233→        category: formCategory,
   234→      });
   235→      resetForm();
   236→      setCreateOpen(false);
   237→      toast.success(`Tag "${formName.trim()}" created successfully`);
   238→    } catch (err: any) {
   239→      toast.error(err?.message || "Failed to create tag");
   240→    } finally {
   241→      setIsCreating(false);
   242→    }
   243→  };
   244→
   245→  // ── Edit ──
   246→  const openEdit = (tag: any) => {
   247→    setEditingTag(tag);
   248→    setFormName(tag.name);
   249→    setFormColor(tag.color);
   250→    setFormCategory(tag.category ?? "general");
   251→    setEditOpen(true);
   252→  };
   253→
   254→  const handleEdit = async () => {
   255→    if (!editingTag) return;
   256→    if (!formName.trim()) {
   257→      toast.error("Tag name is required");
   258→      return;
   259→    }
   260→
   261→    if (isDemoMode) {
   262→      toast.success(`Tag "${formName.trim()}" updated! (Demo mode)`);
   263→      resetForm();
   264→      setEditOpen(false);
   265→      setEditingTag(null);
   266→      return;
   267→    }
   268→
   269→    setIsUpdating(true);
   270→    try {
   271→      await updateTagMutation({
   272→        tagId: editingTag._id,
   273→        name: formName.trim(),
   274→        color: formColor,
   275→        category: formCategory,
   276→      });
   277→      resetForm();
   278→      setEditOpen(false);
   279→      setEditingTag(null);
   280→      toast.success(`Tag "${formName.trim()}" updated`);
   281→    } catch (err: any) {
   282→      toast.error(err?.message || "Failed to update tag");
   283→    } finally {
   284→      setIsUpdating(false);
   285→    }
   286→  };
   287→
   288→  // ── Delete ──
   289→  const openDelete = (tag: any) => {
   290→    setDeletingTag(tag);
   291→    setDeleteOpen(true);
   292→  };
   293→
   294→  const handleDelete = async () => {
   295→    if (!deletingTag) return;
   296→
   297→    if (isDemoMode) {
   298→      toast.success(`Tag "${deletingTag.name}" deleted! (Demo mode)`);
   299→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   300→      setDeleteOpen(false);
   301→      setDeletingTag(null);
   302→      return;
   303→    }
   304→
   305→    setIsDeleting(true);
   306→    try {
   307→      await deleteTagMutation({ tagId: deletingTag._id });
   308→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   309→      setDeleteOpen(false);
   310→      setDeletingTag(null);
   311→      toast.success(`Tag "${deletingTag.name}" deleted`);
   312→    } catch (err: any) {
   313→      toast.error(err?.message || "Failed to delete tag");
   314→    } finally {
   315→      setIsDeleting(false);
   316→    }
   317→  };
   318→
   319→  // ── Render ──
   320→  return (
   321→    <motion.div
   322→      initial={{ opacity: 0 }}
   323→      animate={{ opacity: 1 }}
   324→      className="w-full min-h-screen bg-background text-foreground"
   325→    >
   326→      <PageLayout spaced>
   327→        {/* ── Header ── */}

   329→          <div>
   330→            <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   331→              Tags
   332→            </h1>
   333→            <p className="text-[16px] text-muted-foreground">
   334→              Organize your work with custom tags for quick filtering and categorization
   335→            </p>
   336→          </div>
   337→
   338→          <Dialog open={createOpen} onOpenChange={(open) => { setCreateOpen(open); if (!open) resetForm(); }}>
   339→            <DialogTrigger asChild>
   340→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   341→                <Plus className="mr-2 h-4 w-4" />
   342→                Create Tag
   343→              </Button>
   344→            </DialogTrigger>
   345→            <DialogContent>
   346→              <DialogHeader>
   347→                <DialogTitle>Create New Tag</DialogTitle>
   348→                <DialogDescription>
   349→                  Add a tag to categorize and filter your time entries, projects, and evidence.
   350→                </DialogDescription>
   351→              </DialogHeader>
   352→              <div className="space-y-4 py-2">
   353→                <div className="space-y-2">
   354→                  <Label htmlFor="tag-name">Name</Label>
   355→                  <Input
   356→                    id="tag-name"
   357→                    placeholder="e.g. Urgent, Design, Revision"
   358→                    value={formName}
   359→                    onChange={(e) => setFormName(e.target.value)}
   360→                  />
   361→                </div>
   362→                <div className="space-y-2">
   363→                  <Label>Color</Label>
   364→                  <div className="flex items-center gap-2 flex-wrap">
   365→                    {PRESET_COLORS.map((color) => (
   366→                      <button
   367→                        key={color}
   368→                        type="button"
   369→                        className="w-8 h-8 rounded-full border-2 transition-all hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-primary"
   370→                        style={{
   371→                          backgroundColor: color,
   372→                          borderColor: formColor === color ? "var(--foreground)" : "transparent",
   373→                          boxShadow: formColor === color ? `0 0 0 2px ${color}40` : "none",
   374→                        }}
   375→                        onClick={() => setFormColor(color)}
   376→                      />
   377→                    ))}
   378→                    <div className="flex items-center gap-1 ml-1">
   379→                      <Palette className="h-4 w-4 text-muted-foreground" />
   380→                      <input
   381→                        type="color"
   382→                        value={formColor}
   383→                        onChange={(e) => setFormColor(e.target.value)}
   384→                        className="w-8 h-8 rounded cursor-pointer border border-border"
   385→                      />
   386→                    </div>
   387→                  </div>
   388→                </div>
   389→                <div className="space-y-2">
   390→                  <Label>Category</Label>
   391→                  <div className="flex flex-wrap gap-2">
   392→                    {["general", "client", "project", "evidence"].map((cat) => (
   393→                      <button key={cat} type="button" onClick={() => setFormCategory(cat)}>
   394→                        <Badge
   395→                          variant={formCategory === cat ? "default" : "outline"}
   396→                          className="cursor-pointer transition-colors capitalize"
   397→                        >
   398→                          {cat}
   399→                        </Badge>
   400→                      </button>
   401→                    ))}
   402→                  </div>
   403→                </div>
   404→              </div>
   405→              <DialogFooter>
   406→                <Button variant="outline" onClick={() => { setCreateOpen(false); resetForm(); }} disabled={isCreating}>
   407→                  Cancel
   408→                </Button>
   409→                <Button className="bg-primary hover:bg-primary/90" onClick={handleCreate} disabled={isCreating}>
   410→                  {isCreating ? (
   411→                    <>
   412→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   413→                      Creating...
   414→                    </>
   415→                  ) : (
   416→                    "Create Tag"
   417→                  )}
   418→                </Button>
   419→              </DialogFooter>
   420→            </DialogContent>
   421→          </Dialog>
   422→        </div>
   423→
   424→        {/* ── Demo mode banner ── */}
   425→        {isDemoMode && (
   426→          <div className="flex items-center gap-3 p-3 bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800 rounded-lg">
   427→            <Info className="h-5 w-5 text-amber-600 dark:text-amber-400 shrink-0" />
   428→            <div className="text-sm text-amber-800 dark:text-amber-200">
   429→              <span className="font-semibold">Demo Mode</span> — You're viewing sample data.{" "}
   430→              <a href="/auth" className="underline font-medium hover:text-amber-900 dark:hover:text-amber-100">
   431→                Sign in
   432→              </a>{" "}
   433→              to manage your real tags.
   434→            </div>
   435→          </div>
   436→        )}
   437→
   438→        {/* ── Loading state ── */}
   439→        {isLoading ? (
   440→          <div className="space-y-4">
   441→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   442→              <Skeleton className="h-[76px] rounded-xl" />
   443→              <Skeleton className="h-[76px] rounded-xl" />
   444→              <Skeleton className="h-[76px] rounded-xl" />
   445→            </div>
   446→            <Skeleton className="h-10 rounded-lg" />
   447→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   448→              {[1, 2, 3, 4, 5, 6].map((i) => (
   449→                <Skeleton key={i} className="h-[160px] rounded-xl" />
   450→              ))}
   451→            </div>
   452→          </div>
   453→        ) : (
   454→          <>
   455→            {/* ── Stats Cards ── */}
   456→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   457→              <Card>
   458→                <CardContent className="p-6">
   459→                  <div className="flex items-center gap-3">
   460→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-primary/10">
   461→                      <Hash className="h-5 w-5 text-primary" />
   462→                    </div>
   463→                    <div>
   464→                      <p className="text-sm text-muted-foreground">Total Tags</p>
   465→                      <p className="text-2xl font-bold">{totalTags}</p>
   466→                    </div>
   467→                  </div>
   468→                </CardContent>
   469→              </Card>
   470→              <Card>
   471→                <CardContent className="p-6">
   472→                  <div className="flex items-center gap-3">
   473→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-orange-500/10">
   474→                      <Tag className="h-5 w-5 text-orange-500" />
   475→                    </div>
   476→                    <div>
   477→                      <p className="text-sm text-muted-foreground">Most Used</p>
   478→                      <p className="text-2xl font-bold">
   479→                        {mostUsedTag ? mostUsedTag.name : "—"}
   480→                      </p>
   481→                      {mostUsedTag && (
   482→                        <p className="text-xs text-muted-foreground">
   483→                          {mostUsedTag.usageCount} {mostUsedTag.usageCount === 1 ? "entry" : "entries"}
   484→                        </p>
   485→                      )}
   486→                    </div>
   487→                  </div>
   488→                </CardContent>
   489→              </Card>
   490→              <Card>
   491→                <CardContent className="p-6">
   492→                  <div className="flex items-center gap-3">
   493→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-emerald-500/10">
   494→                      <Hash className="h-5 w-5 text-emerald-500" />
   495→                    </div>
   496→                    <div>
   497→                      <p className="text-sm text-muted-foreground">Categories</p>
   498→                      <p className="text-2xl font-bold">
   499→                        {new Set(tags.map((t: any) => t.category ?? "uncategorized")).size}
   500→                      </p>
   501→                    </div>
   502→                  </div>
   503→                </CardContent>
   504→              </Card>
   505→            </div>
   506→
   507→            {/* ── Quick Filters + Search ── */}
   508→            <div className="space-y-3">
   509→              <div className="flex items-center gap-2">
   510→                <div className="relative flex-1">
   511→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   512→                  <Input
   513→                    placeholder="Search tags..."
   514→                    value={searchQuery}
   515→                    onChange={(e) => setSearchQuery(e.target.value)}
   516→                    className="pl-9"
   517→                  />
   518→                </div>
   519→                {activeFilter && (
   520→                  <Button
   521→                    variant="outline"
   522→                    size="sm"
   523→                    onClick={() => setActiveFilter(null)}
   524→                    className="shrink-0"
   525→                  >
   526→                    <X className="mr-1 h-3 w-3" />
   527→                    Clear filter
   528→                  </Button>
   529→                )}
   530→              </div>
   531→              <div className="flex items-center gap-2 flex-wrap">
   532→                <span className="text-sm text-muted-foreground shrink-0">Quick filter:</span>
   533→                {tags.slice(0, 5).map((tag: any) => (
   534→                  <button
   535→                    key={tag._id}

   537→                    onClick={() => setActiveFilter(activeFilter === tag._id ? null : tag._id)}
   538→                    className="focus:outline-none"
   539→                  >
   540→                    <Badge
   541→                      variant={activeFilter === tag._id ? "default" : "outline"}
   542→                      className="cursor-pointer transition-colors"
   543→                      style={
   544→                        activeFilter === tag._id
   545→                          ? { backgroundColor: tag.color, borderColor: tag.color }
   546→                          : { borderColor: tag.color, color: tag.color }
   547→                      }
   548→                    >
   549→                      <span
   550→                        className="inline-block w-2 h-2 rounded-full mr-1"
   551→                        style={{ backgroundColor: tag.color }}
   552→                      />
   553→                      {tag.name}
   554→                    </Badge>
   555→                  </button>
   556→                ))}
   557→              </div>
   558→            </div>
   559→
   560→            <Separator />
   561→
   562→            {/* ── Tags Grid ── */}
   563→            {filteredTags.length === 0 ? (
   564→              <Card>
   565→                <CardContent className="py-12">
   566→                  <div className="text-center text-muted-foreground">
   567→                    <Tag className="h-12 w-12 mx-auto mb-4 opacity-50" />
   568→                    <p className="text-lg font-medium">No tags found</p>
   569→                    <p className="text-sm">
   570→                      {searchQuery
   571→                        ? "Try adjusting your search query"
   572→                        : "Create your first tag to start organizing your work"}
   573→                    </p>
   574→                    {!searchQuery && (
   575→                      <Button

   649→          </>

   653→      {/* ── Edit Tag Dialog ── */}

   729→            >

   731→            </Button>

   735→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   736→                  Saving...
   737→                </>
   738→              ) : (
   739→                "Save Changes"
   740→              )}
   741→            </Button>
   742→          </DialogFooter>
   743→        </DialogContent>
   744→      </Dialog>

   746→      {/* ── Delete Tag Dialog ── */}
   747→      <Dialog
   748→        open={deleteOpen}
   749→        onOpenChange={(open) => {

   751→          if (!open) setDeletingTag(null);

   755→          <DialogHeader>
   756→            <DialogTitle>Delete Tag</DialogTitle>

   762→          </DialogHeader>

   769→              }}
   770→              disabled={isDeleting}
   771→            >

   773→            </Button>

   776→                <>
   777→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   778→                  Deleting...
   779→                </>
   780→              ) : (
