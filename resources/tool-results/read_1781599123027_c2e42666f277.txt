   131→function formatCurrency(amount: number, currency: string = "USD"): string {
   132→  try {
   133→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   134→  } catch {
   135→    return `$${amount.toFixed(2)}`;
   136→  }
   137→}
   138→
   139→function formatDate(ts: number): string {
   140→  return new Date(ts).toLocaleDateString("en-US", {
   141→    month: "short",
   142→    day: "numeric",
   143→    year: "numeric",
   144→  });
   145→}
   146→
   147→function daysOverdue(dueDate: number): number {
   148→  const now = Date.now();
   149→  return Math.max(0, Math.floor((now - dueDate) / (1000 * 60 * 60 * 24)));
   150→}
