   322→function formatDate(ts: number | undefined) {
   323→  if (!ts) return "—";
   324→  return new Date(ts).toLocaleDateString("en-US", {
   325→    year: "numeric",
   326→    month: "short",
   327→    day: "numeric",
   328→  });
   329→}
   330→
   331→function formatCurrency(amount: number, currency = "USD") {
   332→  return new Intl.NumberFormat("en-US", {
   333→    style: "currency",
   334→    currency,
   335→    minimumFractionDigits: 0,
   336→    maximumFractionDigits: 2,
   337→  }).format(amount);
   338→}
   339→
   340→function relativeTime(ts: number | undefined) {
   341→  if (!ts) return "Never";
   342→  const diff = Date.now() - ts;
   343→  const mins = Math.floor(diff / 60000);
   344→  if (mins < 1) return "Just now";
   345→  if (mins < 60) return `${mins}m ago`;
   346→  const hours = Math.floor(mins / 60);
   347→  if (hours < 24) return `${hours}h ago`;
   348→  const days = Math.floor(hours / 24);
   349→  if (days < 30) return `${days}d ago`;
   350→  return formatDate(ts);
   351→}

   353→function getInitials(name: string) {
   354→  return name
   355→    .split(" ")
   356→    .map((w) => w[0])
   357→    .join("")
   358→    .toUpperCase()
   359→    .slice(0, 2);
   360→}

   362→function getAvatarColor(name: string) {
   363→  const colors = [
   364→    "bg-violet-500",
   365→    "bg-blue-500",
   366→    "bg-emerald-500",
   367→    "bg-amber-500",
   368→    "bg-rose-500",
   369→    "bg-cyan-500",
   370→    "bg-pink-500",
   371→    "bg-teal-500",
   372→  ];
   373→  let hash = 0;
   374→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   375→  return colors[Math.abs(hash) % colors.length];
   376→}
