   320→function invoiceToBillingRecord(inv: ConvexInvoice): BillingRecord {
   321→  // Build description from line items or default to client name
   322→  const firstItem = inv.lineItems?.[0];
   323→  const desc = firstItem
   324→    ? inv.lineItems.length > 1
   325→      ? `${firstItem.description} + ${inv.lineItems.length - 1} more`
   326→      : firstItem.description
   327→    : inv.clientName
   328→    ? `Invoice for ${inv.clientName}`
   329→    : "Invoice";
   330→
   331→  return {
   332→    id: inv.invoiceNumber,
   333→    date: new Date(inv.issueDate).toISOString(),
   334→    description: desc,
   335→    amount: inv.total,
   336→    status: inv.status,
   337→    invoice: inv,
   338→  };
   339→}

   405→function downloadReceipt(record: BillingRecord) {
   406→  const doc = new jsPDF({ unit: "mm", format: "a4" });
   407→  const margin = 20;
   408→  const pageWidth = doc.internal.pageSize.getWidth();
   409→  const pageHeight = doc.internal.pageSize.getHeight();
   410→
   411→  // Title
   412→  doc.setFont("helvetica", "bold");
   413→  doc.setFontSize(20);
   414→  doc.setTextColor(26, 26, 46);
   415→  doc.text("Receipt", margin, 25);
   416→
   417→  // Invoice ID
   418→  doc.setFont("helvetica", "normal");
   419→  doc.setFontSize(10);
   420→  doc.setTextColor(100, 116, 139);
   421→  doc.text(record.id, margin, 32);
   422→
   423→  // Separator
   424→  doc.setDrawColor(226, 232, 240);
   425→  doc.line(margin, 37, pageWidth - margin, 37);
   426→
   427→  let y = 46;
   428→  const addRow = (label: string, value: string) => {
   429→    doc.setFont("helvetica", "normal");
   430→    doc.setFontSize(9);
   431→    doc.setTextColor(148, 163, 184);
   432→    doc.text(label, margin, y);
   433→    doc.setFont("helvetica", "bold");
   434→    doc.setFontSize(11);
   435→    doc.setTextColor(30, 41, 59);
   436→    doc.text(value, margin + 50, y);
   437→    y += 10;
   438→  };
   439→
   440→  addRow("Description", record.description);
   441→  addRow("Date", new Date(record.date).toLocaleDateString("en-US", { year: "numeric", month: "long", day: "numeric" }));
   442→
   443→  // Map invoice status to a display-friendly label
   444→  const statusLabel = record.status.charAt(0).toUpperCase() + record.status.slice(1);
   445→  addRow("Status", statusLabel);
   446→
   447→  const currency = record.invoice?.currency || "USD";
   448→  addRow("Amount", `${currency} $${record.amount.toFixed(2)}`);
   449→
   450→  // Add client info if available
   451→  if (record.invoice?.clientName) {
   452→    addRow("Client", record.invoice.clientName);
   453→  }
   454→
   455→  // Footer
   456→  doc.setFont("helvetica", "normal");
   457→  doc.setFontSize(7);
   458→  doc.setTextColor(203, 213, 225);
   459→  doc.text(`Generated by Axia \u2022 ${new Date().toLocaleString()}`, margin, pageHeight - 12);
   460→
   461→  doc.save(`${record.id}.pdf`);
   462→  toast.success("Receipt downloaded");
   463→}

   497→  return (
   498→    <div className="w-full h-2 rounded-full bg-muted overflow-hidden">

   503→        style={{ width: `${pct}%` }}
   504→      />
   505→    </div>
   506→  );

   553→}

   557→function ChangePlanDialog({

   559→  onClose,
