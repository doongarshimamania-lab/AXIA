   194→   194→function fmtCurrency(n: number): string {
   195→   195→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
   196→   196→}

   198→   198→function fmtRelative(ts: number): string {
   199→   199→  const diff = Date.now() - ts;
   200→   200→  const mins = Math.floor(diff / 60000);
   201→   201→  if (mins < 1) return "just now";
   202→   202→  if (mins < 60) return `${mins}m ago`;
   203→   203→  const hrs = Math.floor(mins / 60);
   204→   204→  if (hrs < 24) return `${hrs}h ago`;
   205→   205→  const days = Math.floor(hrs / 24);
   206→   206→  if (days < 30) return `${days}d ago`;
   207→   207→  return new Date(ts).toLocaleDateString();
   208→   208→}
   209→   209→
   210→   210→function fmtCompactCurrency(n: number): string {
   211→   211→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
   212→   212→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
   213→   213→  return fmtCurrency(n);
   214→   214→}
