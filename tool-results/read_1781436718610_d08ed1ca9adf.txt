   228→   228→
   229→   229→  // ── Library date range (±30 days for monitoring) ──

   231→   231→    const now = new Date();
   232→   232→    const start = new Date(now);
   233→   233→    start.setDate(start.getDate() - 30);
   234→   234→    const end = new Date(now);

   248→   248→

   255→   255→  // Library data with ±30 day window for monitoring

   262→   262→    } : "skip"
   263→   263→  );
   264→   264→
   265→   265→  // Export data with full year window for comprehensive exports
   266→   266→  const exportData = useQuery(
   267→   267→    hasEvidenceApi ? evidenceApi.getEvidenceLibraryData : "skip",
   268→   268→    hasEvidenceApi ? {
   269→   269→      view: "date" as const,
   270→   270→      startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   271→   271→      endDate: Date.now(),
   272→   272→    } : "skip"
   273→   273→  ) as any | undefined;
   274→   274→
   275→   275→  const timelineData = useQuery(
   276→   276→    hasTimelineApi ? evidenceApi.getEvidenceTimeline : "skip",
   277→   277→    hasTimelineApi ? { date: startOfDay } : "skip"

   279→   279→
   280→   280→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   281→   281→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   282→   282→
   283→   283→  const isLoading = evidenceData === undefined || timelineData === undefined;
   284→   284→  const exportLoading = exportData === undefined || clients === undefined;
   285→   285→  const timedOut = useQueryTimeout(isLoading, 5000);
   286→   286→  const showLoading = isLoading && !timedOut && !isDisconnected;
   287→   287→
   288→   288→  // ── Safe fallbacks ──

   290→   290→    totalCount: 0,
   291→   291→    disputeSuccessRate: 0,
   292→   292→    contentQualityScore: 0,
   293→   293→    gapPrediction: {
   294→   294→      status: "no_gaps" as const,
   295→   295→      message: "No Gaps Predicted",
   296→   296→      description: "Start collecting evidence to build your protection.",
   297→   297→      nextGapTime: null,
   298→   298→      missingTypes: [],
   299→   299→    },
   300→   300→    evidenceItems: [],
   301→   301→    healthScore: null,
   302→   302→    disputeData: null,
   303→   303→    contentData: null,
   304→   304→  };
   305→   305→
   306→   306→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   307→   307→
   308→   308→  // ── Export State ──
   309→   309→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   310→   310→  const [selectedProject, setSelectedProject] = useState("all");
   311→   311→  const [selectedClient, setSelectedClient] = useState("all");
   312→   312→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   313→   313→  const [isExporting, setIsExporting] = useState(false);
   314→   314→  const [previewOpen, setPreviewOpen] = useState(false);
   315→   315→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);

   319→   319→    if (!exportData) return [];
   320→   320→    const items = exportData.evidenceItems ?? [];
   321→   321→    const typeCounts: Record<string, number> = {};
   322→   322→    const typeLabels: Record<string, string> = {
   323→   323→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   324→   324→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   325→   325→    };
   326→   326→    const typeIcons: Record<string, React.ReactNode> = {
   327→   327→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,

   363→   363→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;

   365→   365→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;

   375→   375→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),

   378→   378→
   379→   379→  const isFormatLocked = (format: ExportFormat): boolean => {

   385→   385→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]

   388→   388→
   389→   389→  const handleExport = async () => {
   390→   390→    if (selectedEvidenceTypes.length === 0) {
   391→   391→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   392→   392→      return;
   393→   393→    }
   394→   394→    if (isFormatLocked(selectedFormat)) {
   395→   395→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   396→   396→      return;
   397→   397→    }
   398→   398→    setIsExporting(true);
   399→   399→    try {
   400→   400→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   401→   401→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   402→   402→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   403→   403→
   404→   404→      exportEvidence(selectedFormat, {
   405→   405→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   406→   406→        dateRange: `${dateFrom} to ${dateTo}`,
   407→   407→        project: selectedProjectLabel,
   408→   408→        client: selectedClientLabel,
   409→   409→        complianceScore,
   410→   410→        totalItems: totalEvidenceItems,
   411→   411→      });
   412→   412→
   413→   413→      const newExport: RecentExport = {
   414→   414→        id: `exp_${Date.now()}`,
   415→   415→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   416→   416→        format: selectedFormat,
   417→   417→        status: "completed",
   418→   418→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   419→   419→        date: new Date().toISOString().split("T")[0],
   420→   420→        itemCount: totalEvidenceItems,
   421→   421→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   422→   422→        project: selectedProjectLabel,
   423→   423→      };
   424→   424→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   425→   425→
   426→   426→      toast.success("Export complete", {
   427→   427→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   428→   428→      });
   429→   429→
   430→   430→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   431→   431→    } catch {
   432→   432→      toast.error("Export failed", { description: "An error occurred while generating the export file." });
   433→   433→    }
   434→   434→    setIsExporting(false);
   435→   435→  };
   436→   436→
   437→   437→  const handleFormatSelect = (format: ExportFormat) => {
   438→   438→    if (isFormatLocked(format)) {
