   322→function invoiceToBillingRecord(inv: ConvexInvoice): BillingRecord {
   323→  // Build description from line items or default to client name
   324→  const firstItem = inv.lineItems?.[0];
   325→  const desc = firstItem
   326→    ? inv.lineItems.length > 1
   327→      ? `${firstItem.description} + ${inv.lineItems.length - 1} more`
   328→      : firstItem.description
   329→    : inv.clientName
   330→    ? `Invoice for ${inv.clientName}`
   331→    : "Invoice";
   332→
   333→  return {
   334→    id: inv.invoiceNumber,
   335→    date: new Date(inv.issueDate).toISOString(),
   336→    description: desc,
   337→    amount: inv.total,
   338→    status: inv.status,
   339→    invoice: inv,
   340→  };
   341→}

   407→function downloadReceipt(record: BillingRecord) {
   408→  const doc = new jsPDF({ unit: "mm", format: "a4" });
   409→  const margin = 20;
   410→  const pageWidth = doc.internal.pageSize.getWidth();
   411→  const pageHeight = doc.internal.pageSize.getHeight();
   412→
   413→  // Title
   414→  doc.setFont("helvetica", "bold");
   415→  doc.setFontSize(20);
   416→  doc.setTextColor(26, 26, 46);
   417→  doc.text("Receipt", margin, 25);
   418→
   419→  // Invoice ID
   420→  doc.setFont("helvetica", "normal");
   421→  doc.setFontSize(10);
   422→  doc.setTextColor(100, 116, 139);
   423→  doc.text(record.id, margin, 32);
   424→
   425→  // Separator
   426→  doc.setDrawColor(226, 232, 240);
   427→  doc.line(margin, 37, pageWidth - margin, 37);
   428→
   429→  let y = 46;
   430→  const addRow = (label: string, value: string) => {
   431→    doc.setFont("helvetica", "normal");
   432→    doc.setFontSize(9);
   433→    doc.setTextColor(148, 163, 184);
   434→    doc.text(label, margin, y);
   435→    doc.setFont("helvetica", "bold");
   436→    doc.setFontSize(11);
   437→    doc.setTextColor(30, 41, 59);
   438→    doc.text(value, margin + 50, y);
   439→    y += 10;
   440→  };
   441→
   442→  addRow("Description", record.description);
   443→  addRow("Date", new Date(record.date).toLocaleDateString("en-US", { year: "numeric", month: "long", day: "numeric" }));
   444→
   445→  // Map invoice status to a display-friendly label
   446→  const statusLabel = record.status.charAt(0).toUpperCase() + record.status.slice(1);
   447→  addRow("Status", statusLabel);
   448→
   449→  const currency = record.invoice?.currency || "USD";
   450→  addRow("Amount", `${currency} $${record.amount.toFixed(2)}`);
   451→
   452→  // Add client info if available
   453→  if (record.invoice?.clientName) {
   454→    addRow("Client", record.invoice.clientName);
   455→  }
   456→
   457→  // Footer
   458→  doc.setFont("helvetica", "normal");
   459→  doc.setFontSize(7);
   460→  doc.setTextColor(203, 213, 225);
   461→  doc.text(`Generated by Axia \u2022 ${new Date().toLocaleString()}`, margin, pageHeight - 12);
   462→
   463→  doc.save(`${record.id}.pdf`);
   464→  toast.success("Receipt downloaded");
   465→}

   488→function UsageBar({ used, limit }: { used: number; limit: number | null }) {

   490→    return (
   491→      <div className="w-full h-2 rounded-full bg-emerald-100 dark:bg-emerald-950/40 overflow-hidden">

   494→    );
   495→  }
   496→  const pct = Math.min((used / limit) * 100, 100);
   497→  const isHigh = pct >= 80;
   498→  const isFull = pct >= 100;
   499→  return (
   500→    <div className="w-full h-2 rounded-full bg-muted overflow-hidden">
   501→      <div
   502→        className={`h-full rounded-full transition-all duration-500 ${
   503→          isFull ? "bg-destructive" : isHigh ? "bg-amber-500" : "bg-emerald-500"
   504→        }`}
   505→        style={{ width: `${pct}%` }}
   506→      />
   507→    </div>
   508→  );

   549→            </AnimatePresence>
   550→          </motion.div>
   551→        );
   552→      })}
   553→    </div>
   554→  );
   555→}

   559→function ChangePlanDialog({
