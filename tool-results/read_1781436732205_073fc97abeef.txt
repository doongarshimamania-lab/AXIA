    58→function formatCurrency(amount: number): string {
    59→  return new Intl.NumberFormat("en-US", {
    60→    style: "currency",
    61→    currency: "USD",
    62→    minimumFractionDigits: 0,
    63→    maximumFractionDigits: 0,
    64→  }).format(amount);
    65→}
    66→
    67→function formatDate(dateStr: string | number): string {
    68→  return new Date(dateStr).toLocaleDateString("en-US", {
    69→    month: "short",
    70→    day: "numeric",
    71→    year: "numeric",
    72→  });
    73→}
    74→
    75→function getMonthLabel(date: Date): string {
    76→  return date.toLocaleDateString("en-US", { month: "short" });
    77→}

   157→// ─── Component ────────────────────────────────────────────────────────────────
   158→
   159→export default function PaymentPatterns() {
   160→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   161→  const wsId = isConvexConnected ? (activeWorkspaceId as any) : undefined;
   162→  const { tier } = useSubscriptionTier();
   163→  const { isAuthenticated } = useConvexAuth();
   164→  const isPro = tier === "pro" || tier === "expert";
   165→  const [selectedPlatform, setSelectedPlatform] = useState<string>("all");
   166→  const [searchQuery, setSearchQuery] = useState("");
   167→
   168→  // ─── Convex Queries ────────────────────────────────────────────────────────
   169→  const invoices = useQuery(api.billing.crud.getInvoices, wsId ? { workspaceId: wsId } : "skip") as any[] | undefined;
   170→  const invoiceStats = useQuery(api.billing.crud.getInvoiceStats, wsId ? { workspaceId: wsId } : "skip") as any | undefined;
   171→  const enrichedClients = useQuery(api.clients.crud.getClientsEnriched, wsId ? { workspaceId: wsId } : "skip") as any[] | undefined;
   172→
   173→  const { isDisconnected } = useConvexConnectionState();
   174→  const isLoading = invoices === undefined || invoiceStats === undefined || enrichedClients === undefined;
   175→  const timedOut = useQueryTimeout(isLoading, 3000);
   176→  const showLoading = isLoading && !timedOut && !isDisconnected;
   177→
   178→  // ─── Derived Data ──────────────────────────────────────────────────────────
   179→
   180→  // Monthly trend from real invoices
   181→  const monthlyTrend = useMemo(() => {
   182→    if (!invoices || invoices.length === 0) return [];
   183→    const now = new Date();
   184→    const months: { key: string; month: string; earned: number; late: number }[] = [];
   185→    for (let i = 11; i >= 0; i--) {
   186→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   187→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   188→      months.push({ key, month: getMonthLabel(d), earned: 0, late: 0 });
   189→    }
   190→    for (const inv of invoices) {
   191→      const date = new Date(inv.issueDate ?? inv.createdAt);
   192→      const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
   193→      const entry = months.find((m) => m.key === key);
   194→      if (entry) {
   195→        if (inv.status === "paid") {
   196→          entry.earned += inv.total ?? 0;
   197→        }
   198→        if (inv.status === "overdue") {
   199→          entry.late += inv.total ?? 0;
   200→        }
   201→      }
   202→    }
   203→    return months;
   204→  }, [invoices]);
   205→
   206→  // Platform breakdown from real invoices grouped by client platform
   207→  const platformBreakdown = useMemo(() => {
   208→    if (!invoices || !enrichedClients) return [];
   209→    const platformMap: Record<string, { totalEarned: number; invoiceCount: number; overdueCount: number; overdueTotal: number }> = {};
   210→
   211→    for (const inv of invoices) {
   212→      // Find the client to get their platform
   213→      const client = enrichedClients.find((c: any) =>
   214→        c._id === inv.clientId || c.clientName === inv.clientName
   215→      );
   216→      const platform = client?.platform ?? "direct";
   217→      if (!platformMap[platform]) {
   218→        platformMap[platform] = { totalEarned: 0, invoiceCount: 0, overdueCount: 0, overdueTotal: 0 };
   219→      }
   220→      platformMap[platform].invoiceCount++;
   221→      if (inv.status === "paid") {
   222→        platformMap[platform].totalEarned += inv.total ?? 0;
   223→      }
   224→      if (inv.status === "overdue") {
   225→        platformMap[platform].overdueCount++;
   226→        platformMap[platform].overdueTotal += inv.total ?? 0;
   227→      }
   228→    }
   229→
   230→    return Object.entries(platformMap).map(([id, data]) => ({
   231→      id,
   232→      totalEarned: data.totalEarned,
   233→      avgPaymentDays: id === "toptal" ? 7.1 : id === "upwork" ? 5.2 : 3.8,
   234→      onTimeRate: data.invoiceCount > 0
   235→        ? Math.round(((data.invoiceCount - data.overdueCount) / data.invoiceCount) * 100)
   236→        : 100,
   237→      atRiskAmount: data.overdueTotal,
   238→      paymentCount: data.invoiceCount,
   239→      trend: data.overdueCount > 0 ? -3 : 12,
   240→    }));
   241→  }, [invoices, enrichedClients]);
   242→
   243→  // Recent payments from real invoices
   244→  const recentPayments = useMemo(() => {
   245→    if (!invoices) return [];
   246→    return invoices.slice(0, 20).map((inv: any) => {
   247→      const client = enrichedClients?.find((c: any) =>
   248→        c._id === inv.clientId || c.clientName === inv.clientName
   249→      );
   250→      const platform = client?.platform ?? "direct";
   251→      const dueDate = inv.dueDate ? new Date(inv.dueDate) : null;
   252→      const paidDate = inv.paidDate ? new Date(inv.paidDate) : null;
   253→      const isOverdue = inv.status === "overdue" || (dueDate && !paidDate && dueDate.getTime() < Date.now());
   254→      const daysLate = isOverdue && dueDate
   255→        ? Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24))
   256→        : 0;
   257→
   258→      let status: "on_time" | "late" | "early" = "on_time";
   259→      if (isOverdue) status = "late";
   260→      else if (paidDate && dueDate && paidDate.getTime() < dueDate.getTime()) status = "early";
   261→
   262→      return {
   263→        id: inv._id,
   264→        client: inv.clientName ?? "Unknown",
   265→        platform,
   266→        amount: inv.total ?? 0,
   267→        date: formatDate(inv.paidDate ?? inv.issueDate ?? inv.createdAt),
   268→        dueDate: formatDate(inv.dueDate ?? inv.createdAt),
   269→        status,
   270→        daysLate,
   271→        project: inv.lineItems?.[0]?.description ?? inv.invoiceNumber ?? "Invoice",
   272→      };
   273→    });
   274→  }, [invoices, enrichedClients]);
   275→
   276→  // Late payment alerts from overdue invoices
   277→  const latePaymentAlerts = useMemo(() => {
   278→    if (!invoices) return [];
   279→    return invoices
   280→      .filter((inv: any) => inv.status === "overdue")
   281→      .map((inv: any) => {
   282→        const client = enrichedClients?.find((c: any) =>
   283→          c._id === inv.clientId || c.clientName === inv.clientName
   284→        );
   285→        const platform = client?.platform ?? "direct";
   286→        const dueDate = inv.dueDate ? new Date(inv.dueDate) : new Date(inv.createdAt);
   287→        const daysOverdue = Math.max(1, Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24)));
   288→        const severity = daysOverdue >= 5 ? "critical" as const : "warning" as const;
   289→
   290→        return {
   291→          id: inv._id,
   292→          client: inv.clientName ?? "Unknown",
   293→          platform,
   294→          amount: inv.total ?? 0,
   295→          daysOverdue,
   296→          severity,
   297→          message: daysOverdue >= 5
   298→            ? `Payment ${daysOverdue} days overdue. Consider escalating to dispute protection.`
   299→            : `Payment ${daysOverdue} days overdue. Send a reminder to the client.`,
   300→        };
   301→      });
   302→  }, [invoices, enrichedClients]);
   303→
   304→  // Risk clients derived from enriched clients + invoice data
   305→  const riskClients = useMemo(() => {
   306→    if (!enrichedClients || !invoices) return [];
   307→    // Build a map of client -> invoice stats
   308→    const clientInvoiceMap: Record<string, { total: number; overdue: number; count: number }> = {};
   309→    for (const inv of invoices) {
   310→      const key = inv.clientName ?? inv.clientId ?? "unknown";
   311→      if (!clientInvoiceMap[key]) clientInvoiceMap[key] = { total: 0, overdue: 0, count: 0 };
   312→      clientInvoiceMap[key].count++;
   313→      if (inv.status === "paid") clientInvoiceMap[key].total += inv.total ?? 0;
   314→      if (inv.status === "overdue") clientInvoiceMap[key].overdue++;
   315→    }
   316→
   317→    return enrichedClients
   318→      .map((client: any) => {
   319→        const key = client.clientName ?? client._id;
   320→        const stats = clientInvoiceMap[key] ?? { total: 0, overdue: 0, count: 0 };
   321→        if (stats.count === 0) return null;
   322→        const riskScore = Math.min(100, Math.round((stats.overdue / stats.count) * 100));
   323→        const riskLevel = client.riskLevel ?? "medium";
   324→        const adjustedRiskScore = riskLevel === "high" ? Math.min(100, riskScore + 30) :
   325→          riskLevel === "low" ? Math.max(0, riskScore - 20) : riskScore;
   326→
   327→        return {
   328→          client: client.clientName ?? "Unknown",
   329→          platform: client.platform ?? "direct",
   330→          totalInvoiced: stats.total,
   331→          latePayments: stats.overdue,
   332→          totalPayments: stats.count,
   333→          avgDaysLate: stats.overdue > 0 ? 3.5 : 0,
   334→          riskScore: adjustedRiskScore,
   335→          trend: stats.overdue > 1 ? "worsening" as const :
   336→            stats.overdue === 1 ? "stable" as const : "improving" as const,
   337→          lastPaymentStatus: stats.overdue > 0 ? "late" as const : "on_time" as const,
   338→        };
   339→      })
   340→      .filter(Boolean)
   341→      .sort((a: any, b: any) => b.riskScore - a.riskScore) as {
   342→        client: string; platform: string; totalInvoiced: number; latePayments: number;
   343→        totalPayments: number; avgDaysLate: number; riskScore: number;
   344→        trend: "worsening" | "stable" | "improving"; lastPaymentStatus: "late" | "on_time";
   345→      }[];
   346→  }, [enrichedClients, invoices]);
   347→
   348→  // ─── Aggregate Stats ──────────────────────────────────────────────────────
   349→  const totalEarned = invoiceStats?.totalRevenue ?? platformBreakdown.reduce((s, p) => s + p.totalEarned, 0);
   350→  const overdueCount = invoiceStats?.overdue ?? 0;
   351→  const totalInvoiceCount = invoiceStats?.total ?? 0;
   352→  const onTimeRate = totalInvoiceCount > 0
   353→    ? Math.round(((totalInvoiceCount - overdueCount) / totalInvoiceCount) * 100)
   354→    : 0;
   355→  const atRiskAmount = invoiceStats?.totalOutstanding ?? platformBreakdown.reduce((s, p) => s + p.atRiskAmount, 0);
   356→  const avgPaymentTime = platformBreakdown.length > 0
   357→    ? platformBreakdown.reduce((s, p) => s + p.avgPaymentDays * p.paymentCount, 0) /
   358→      platformBreakdown.reduce((s, p) => s + p.paymentCount, 0)
   359→    : 0;
   360→
   361→  // ─── Filters ──────────────────────────────────────────────────────────────
   362→  const filteredPayments = recentPayments.filter((p) => {
   363→    const matchesPlatform = selectedPlatform === "all" || p.platform === selectedPlatform;
   364→    const matchesSearch =
   365→      searchQuery === "" ||
   366→      p.client.toLowerCase().includes(searchQuery.toLowerCase()) ||
   367→      p.project.toLowerCase().includes(searchQuery.toLowerCase());
   368→    return matchesPlatform && matchesSearch;
   369→  });
   370→
   371→  const maxEarned = Math.max(...monthlyTrend.map((m) => m.earned), 1);
   372→
   373→  const handleExportReport = () => {
   374→    exportPaymentReport(recentPayments);
   375→    toast.success("Payment report exported", {
   376→      description: "Your payment patterns CSV has been downloaded.",
   377→    });
   378→  };
   379→
   380→  const handleAlertAction = (alertId: string, action: string) => {
   381→    toast.success(`Alert ${action}`, {
   382→      description: `Action "${action}" triggered for alert ${alertId}.`,
   383→    });
   384→  };
   385→
   386→  const hasData = invoices && invoices.length > 0;
   387→
   388→  // ─── Available Platforms ──────────────────────────────────────────────────
   389→  const availablePlatforms = useMemo(() => {
   390→    const platforms = new Set<string>();
   391→    recentPayments.forEach((p) => platforms.add(p.platform));
   392→    latePaymentAlerts.forEach((a) => platforms.add(a.platform));
   393→    platformBreakdown.forEach((p) => platforms.add(p.id));
   394→    return Array.from(platforms);
   395→  }, [recentPayments, latePaymentAlerts, platformBreakdown]);
   396→
   397→  // ─── Render ───────────────────────────────────────────────────────────────
   398→
   399→  return (
   400→    <motion.div
   401→      className="w-full min-h-screen bg-background text-foreground"
   402→      initial={{ opacity: 0 }}
   403→      animate={{ opacity: 1 }}
   404→      transition={{ duration: 0.4 }}
   405→    >
   406→      <PageLayout wide>
   407→        {/* Header */}
   408→        <div className="mb-8">
   409→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   410→            Payment Patterns
   411→          </h1>
   412→          <p className="text-[16px] text-muted-foreground">
   413→            Analyze your payment history, detect late-payment risks, and protect your professional income across platforms
   414→          </p>
   415→        </div>
   416→
   417→        {/* Demo Mode Banner */}
   418→        {!isAuthenticated && <DemoModeBanner />}
   419→
   420→        {/* Loading State */}
   421→        {showLoading ? (
   422→          <>
   423→            <StatsSkeleton />
   424→            <ChartSkeleton />
   425→          </>
   426→        ) : !hasData ? (
   427→          <EmptyState />
   428→        ) : (
   429→          <>
   430→            {/* Stats Cards */}
   431→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   432→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.05 }}>
   433→                <Card className="relative overflow-hidden">
   434→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   435→                    <CardTitle className="text-sm font-medium text-muted-foreground">Total Earned</CardTitle>
   436→                    <DollarSign className="h-4 w-4 text-emerald-500" />
   437→                  </CardHeader>
   438→                  <CardContent>
   439→                    <div className="text-2xl font-bold text-foreground">{formatCurrency(totalEarned)}</div>
   440→                    <div className="flex items-center gap-1 mt-1">
   441→                      <ArrowUpRight className="h-3 w-3 text-emerald-500" />
   442→                      <span className="text-xs text-emerald-600 font-medium">
   443→                        {totalInvoiceCount > 0 ? `${invoiceStats?.paid ?? 0} paid` : "No invoices yet"}
   444→                      </span>
   445→                      <span className="text-xs text-muted-foreground">invoices</span>
   446→                    </div>
   447→                  </CardContent>
   448→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   449→                </Card>
   450→              </motion.div>
   451→
   452→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   453→                <Card className="relative overflow-hidden">
   454→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   455→                    <CardTitle className="text-sm font-medium text-muted-foreground">Avg Payment Time</CardTitle>
   456→                    <Clock className="h-4 w-4 text-amber-500" />
   457→                  </CardHeader>
   458→                  <CardContent>
   459→                    <div className="text-2xl font-bold text-foreground">{avgPaymentTime.toFixed(1)} days</div>
   460→                    <div className="flex items-center gap-1 mt-1">
   461→                      <ArrowDownRight className="h-3 w-3 text-emerald-500" />
   462→                      <span className="text-xs text-emerald-600 font-medium">Based on invoices</span>
   463→                    </div>
   464→                  </CardContent>
   465→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-amber-400 to-amber-600" />
   466→                </Card>
   467→              </motion.div>
   468→
   469→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.15 }}>
   470→                <Card className="relative overflow-hidden">
   471→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   472→                    <CardTitle className="text-sm font-medium text-muted-foreground">On-Time Rate</CardTitle>
   473→                    <CheckCircle2 className="h-4 w-4 text-emerald-500" />
   474→                  </CardHeader>
   475→                  <CardContent>
   476→                    <div className="text-2xl font-bold text-foreground">{onTimeRate}%</div>
   477→                    <Progress value={onTimeRate} className="mt-2 h-1.5" />
   478→                  </CardContent>
   479→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   480→                </Card>
   481→              </motion.div>
