    73→// Authentication hook
    74→function useOwnerAuth() {
    75→  const [isAuthenticated, setIsAuthenticated] = useState(false);
    76→  const [password, setPassword] = useState("");
    77→  const [failedAttempts, setFailedAttempts] = useState(0);
    78→  const [showError, setShowError] = useState(false);
    79→  const [lastActivity, setLastActivity] = useState(Date.now());
    80→  const timeoutRef = useRef<number | null>(null);
    81→
    82→  const CORRECT_PASSWORD = "@@@@HHH$";
    83→  const SESSION_TIMEOUT = 10 * 60 * 1000; // 10 minutes
    84→
    85→  const logout = useCallback((message?: string) => {
    86→    setIsAuthenticated(false);
    87→    localStorage.removeItem("ownerSessionActive");
    88→    localStorage.removeItem("ownerLastActivity");
    89→    if (timeoutRef.current) {
    90→      window.clearTimeout(timeoutRef.current);
    91→    }
    92→    if (message) {
    93→      toast.error(message);
    94→    }
    95→  }, []);
    96→
    97→  const resetActivityTimer = useCallback(() => {
    98→    const now = Date.now();
    99→    setLastActivity(now);
   100→    localStorage.setItem("ownerLastActivity", now.toString());
   101→
   102→    if (timeoutRef.current) {
   103→      window.clearTimeout(timeoutRef.current);
   104→    }
   105→
   106→    timeoutRef.current = window.setTimeout(() => {
   107→      logout("Session expired");
   108→    }, SESSION_TIMEOUT);
   109→  }, [logout]);
   110→
   111→  const login = (inputPassword: string) => {
   112→    if (inputPassword === CORRECT_PASSWORD) {
   113→      setIsAuthenticated(true);
   114→      localStorage.setItem("ownerSessionActive", "true");
   115→      resetActivityTimer();
   116→      setFailedAttempts(0);
   117→      setShowError(false);
   118→    } else {
   119→      const newFailedAttempts = failedAttempts + 1;
   120→      setFailedAttempts(newFailedAttempts);
   121→      
   122→      if (newFailedAttempts >= 3) {
   123→        setShowError(true);
   124→        setTimeout(() => setShowError(false), 5000);
   125→      }
   126→    }
   127→    setPassword("");
   128→  };
   129→
   130→  useEffect(() => {
   131→    // Check existing session
   132→    const sessionActive = localStorage.getItem("ownerSessionActive") === "true";
   133→    const storedLastActivity = localStorage.getItem("ownerLastActivity");
   134→    
   135→    if (sessionActive && storedLastActivity) {
   136→      const timeSinceLastActivity = Date.now() - parseInt(storedLastActivity);
   137→      if (timeSinceLastActivity < SESSION_TIMEOUT) {
   138→        setIsAuthenticated(true);
   139→        resetActivityTimer();
   140→      } else {
   141→        logout("Session expired");
   142→      }
   143→    }
   144→
   145→    // Activity listeners
   146→    const activities = ['mousemove', 'keydown', 'click', 'touchstart'];
   147→    const handleActivity = () => {
   148→      if (isAuthenticated) {
   149→        resetActivityTimer();
   150→      }
   151→    };
   152→
   153→    activities.forEach(event => {
   154→      document.addEventListener(event, handleActivity);
   155→    });
   156→
   157→    return () => {
   158→      activities.forEach(event => {
   159→        document.removeEventListener(event, handleActivity);
   160→      });
   161→      if (timeoutRef.current) {
   162→        // Use browser-specific API to ensure the type is number
   163→        window.clearTimeout(timeoutRef.current);
   164→      }
   165→    };
   166→  }, [resetActivityTimer]);
   167→
   168→  return {
   169→    isAuthenticated,
   170→    password,
   171→    setPassword,
   172→    login,
   173→    logout,
   174→    showError,
   175→    failedAttempts
   176→  };
   177→}

   179→// Reusable ThemeToggle component (persists and applies global theme)
   180→function ThemeToggle() {
   181→  const [isDark, setIsDark] = useState<boolean>(() => {
   182→    try {
   183→      const saved = localStorage.getItem("theme");
   184→      if (saved) return saved === "dark";
   185→      return document.documentElement.classList.contains("dark");
   186→    } catch {
   187→      return false;
   188→    }
   189→  });
   190→
   191→  useEffect(() => {
   192→    if (isDark) {
   193→      document.documentElement.classList.add("dark");
   194→      localStorage.setItem("theme", "dark");
   195→    } else {
   196→      document.documentElement.classList.remove("dark");
   197→      localStorage.setItem("theme", "light");
   198→    }
   199→  }, [isDark]);
   200→
   201→  return (
   202→    <div className="flex items-center gap-2">
   203→      <span className="text-sm text-muted-foreground">Dark</span>
   204→      <Switch checked={isDark} onCheckedChange={setIsDark} />
   205→    </div>
   206→  );
   207→}

   270→
   271→            {/* Tick marks and labels */}
   272→            {[0, 100, 200, 300, 400, 500].map((value) => {
   273→              const angle = (value / 500) * 180; // 0..180
   274→              const radian = (angle * Math.PI) / 180;
   275→              // Map so 0 -> left (-90°), 180 -> right (+90°)
   276→              const theta = Math.PI - radian; // matches prior logic
   277→              const x1 = CENTER_X + TICK_INNER * Math.cos(theta);
   278→              const y1 = CENTER_Y - TICK_INNER * Math.sin(theta);
   279→              const x2 = CENTER_X + TICK_OUTER * Math.cos(theta);

   296→                </g>
   297→              );
   298→            })}
   299→
   300→            {/* Animated needle - changed to pivot at true center using translate */}
   301→            <g transform={`translate(${CENTER_X} ${CENTER_Y}) rotate(${needleAngle})`}>
   302→              <line
   303→                x1={0}
   304→                y1={0}
   305→                x2={0}
