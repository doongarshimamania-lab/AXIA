    43→// 1. CREATE
    44→// ─────────────────────────────────────────────
    45→export const create = mutation({
    46→  args: {
    47→    workspaceId: v.optional(v.id("workspaces")),
    48→    teamId: v.optional(v.id("teams")),
    49→    clientId: v.id("clients"),
    50→    projectId: v.optional(v.id("projects")),
    51→    proposalId: v.optional(v.id("proposals")),
    52→    lineItems: v.array(lineItemValidator),
    53→    subtotal: v.number(),
    54→    taxRate: v.optional(v.number()),
    55→    taxAmount: v.optional(v.number()),
    56→    discountAmount: v.optional(v.number()),
    57→    total: v.number(),
    58→    currency: v.optional(v.string()),
    59→    dueDate: v.number(),
    60→    notes: v.optional(v.string()),
    61→    terms: v.optional(v.string()),
    62→    customFields: v.optional(v.any()),
    63→  },
    64→  handler: async (ctx, args) => {
    65→    const userId = await getAuthUserId(ctx);
    66→    if (!userId) throw new ConvexError("Not authenticated");
    67→
    68→    // If workspaceId provided, verify membership
    69→    if (args.workspaceId) {
    70→      await requireWorkspaceAccess(ctx, args.workspaceId, "member");
    71→    }
    72→
    73→    // Validate client access
    74→    const client = await ctx.db.get(args.clientId);
    75→    if (!client) throw new ConvexError("Client not found");
    76→    if (client.workspaceId) {
    77→      const access = await getRecordAccess(ctx, client, userId);
    78→      if (!access) throw new ConvexError("Client does not belong to this workspace");
    79→    } else if (client.userId !== userId) {
    80→      throw new ConvexError("Client does not belong to this user");
    81→    }
    82→
    83→    // Validate project access if provided
    84→    if (args.projectId) {
    85→      const project = await ctx.db.get(args.projectId);
    86→      if (!project) throw new ConvexError("Project not found");
    87→      if (project.workspaceId) {
    88→        const access = await getRecordAccess(ctx, project, userId);
    89→        if (!access) throw new ConvexError("Project does not belong to this workspace");
    90→      } else if (project.userId !== userId) {
    91→        throw new ConvexError("Project does not belong to this user");
    92→      }
    93→    }
    94→
    95→    // Validate proposal access if provided
    96→    if (args.proposalId) {
    97→      const proposal = await ctx.db.get(args.proposalId);
    98→      if (!proposal) throw new ConvexError("Proposal not found");
    99→      if (proposal.workspaceId) {
   100→        const access = await getRecordAccess(ctx, proposal, userId);
   101→        if (!access) throw new ConvexError("Proposal does not belong to this workspace");
   102→      } else if (proposal.userId !== userId) {
   103→        throw new ConvexError("Proposal does not belong to this user");
   104→      }
   105→    }
   106→
   107→    // Validate line items are not empty
   108→    if (args.lineItems.length === 0) {
   109→      throw new ConvexError("Invoice must have at least one line item");
   110→    }
   111→
   112→    // Validate subtotal is non-negative
   113→    if (args.subtotal < 0) {
   114→      throw new ConvexError("Subtotal cannot be negative");
   115→    }
   116→
   117→    // Validate total is non-negative
   118→    if (args.total < 0) {
   119→      throw new ConvexError("Total cannot be negative");
   120→    }
   121→
   122→    // Validate taxRate is between 0 and 100 if provided
   123→    if (args.taxRate !== undefined && (args.taxRate < 0 || args.taxRate > 100)) {
   124→      throw new ConvexError("Tax rate must be between 0 and 100");
   125→    }
   126→
   127→    // Validate dueDate is in the future
   128→    if (args.dueDate < Date.now()) {
   129→      throw new ConvexError("Due date must be in the future");
   130→    }
   131→
   132→    // Auto-generate invoice number
   133→    const invoiceNumber = await getNextInvoiceNumberInternal(ctx, userId);
   134→
   135→    // Generate public token
   136→    const publicToken = crypto.randomUUID();
   137→
   138→    const now = Date.now();
   139→    const { workspaceId, teamId, customFields, ...rest } = args;
   140→
   141→    const invoiceId = await ctx.db.insert("invoices", {
   142→      userId,
   143→      workspaceId: workspaceId ?? undefined,
   144→      createdBy: userId,
   145→      teamId: teamId ?? undefined,
   146→      customFields: customFields ?? undefined,
   147→      ...rest,
   148→      invoiceNumber,
   149→      status: "draft",
   150→      publicToken,
   151→      createdAt: now,
   152→      updatedAt: now,
   153→    });
   154→
   155→    return invoiceId;
   156→  },
   157→});

   346→// ─────────────────────────────────────────────
   347→// 5. MARK PAID
   348→// ─────────────────────────────────────────────
   349→export const markPaid = mutation({
   350→  args: {
   351→    invoiceId: v.id("invoices"),
   352→    paidAmount: v.optional(v.number()),
   353→  },
   354→  handler: async (ctx, args) => {
   355→    const userId = await getAuthUserId(ctx);
   356→    if (!userId) throw new ConvexError("Not authenticated");
   357→
   358→    const invoice = await ctx.db.get(args.invoiceId);
   359→    if (!invoice) throw new ConvexError("Invoice not found");
   360→
   361→    // Check workspace access or direct ownership
   362→    if (invoice.workspaceId) {
   363→      await requireRecordAccess(ctx, invoice, "collaborate");
   364→    } else if (invoice.userId !== userId) {
   365→      throw new ConvexError("Not authorized to update this invoice");
   366→    }
   367→
   368→    // Only sent, viewed, or overdue invoices can be marked as paid
   369→    if (!["sent", "viewed", "overdue", "partial"].includes(invoice.status)) {
   370→      throw new ConvexError("This invoice cannot be marked as paid in its current state");
   371→    }
   372→
   373→    // Validate paidAmount is non-negative if provided
   374→    if (args.paidAmount !== undefined && args.paidAmount < 0) {
   375→      throw new ConvexError("Paid amount cannot be negative");
   376→    }
   377→
   378→    const now = Date.now();
   379→    const paidAmount = args.paidAmount ?? invoice.total;
   380→
   381→    // Determine status: paid in full or partial
   382→    const newStatus = paidAmount >= invoice.total ? "paid" : "partial";
   383→
   384→    await ctx.db.patch(args.invoiceId, {
   385→      status: newStatus,
   386→      paidAt: now,
   387→      paidAmount,
   388→      updatedAt: now,
   389→    });
   390→
   391→    // Cancel all pending reminders for this invoice
   392→    await cancelRemindersInternal(ctx, args.invoiceId);
   393→
   394→    // Update client payment behavior
   395→    const client = await ctx.db.get(invoice.clientId);
   396→    if (client) {
   397→      const canUpdate = client.workspaceId
   398→        ? !!(await getRecordAccess(ctx, client, userId))
   399→        : client.userId === userId;
   400→
   401→      if (canUpdate) {
   402→        const totalPaid = (client.totalPaid ?? 0) + paidAmount;
   403→        const totalInvoiced = (client.totalInvoiced ?? 0) + invoice.total;
   404→
   405→        // Calculate average payment days
   406→        let avgPaymentDays = client.avgPaymentDays ?? 0;
   407→        if (invoice.sentAt) {
   408→          const paymentDays = (now - invoice.sentAt) / (1000 * 60 * 60 * 24);
   409→          const previousPaymentCount = client.avgPaymentDays ? Math.round((client.totalPaid ?? 0) / Math.max(invoice.total, 1)) : 0;
   410→          const totalCount = previousPaymentCount + 1;
   411→          avgPaymentDays = ((avgPaymentDays * previousPaymentCount) + paymentDays) / totalCount;
   412→        }
   413→
   414→        // Determine if on time (paid before or on due date)
   415→        const isOnTime = now <= invoice.dueDate;
   416→        const currentOnTimeRate = client.onTimeRate ?? 0;
   417→        const previousPaymentCount = client.totalPaid ? Math.round(client.totalPaid / Math.max(invoice.total, 1)) : 0;
   418→        const newOnTimeRate = previousPaymentCount === 0
   419→          ? (isOnTime ? 1 : 0)
   420→          : ((currentOnTimeRate * previousPaymentCount) + (isOnTime ? 1 : 0)) / (previousPaymentCount + 1);
   421→
   422→        await ctx.db.patch(invoice.clientId, {
   423→          totalPaid,
   424→          totalInvoiced,
   425→          avgPaymentDays: Math.round(avgPaymentDays * 10) / 10,
   426→          onTimeRate: Math.round(newOnTimeRate * 100) / 100,
   427→          lastPaymentAt: now,
   428→          updatedAt: now,
   429→        });
   430→      }
   431→    }
   432→
   433→    return args.invoiceId;
   434→  },
   435→});

   438→// 6. MARK OVERDUE (batch — check overdue invoices)
   439→// ─────────────────────────────────────────────
   440→export const markOverdue = mutation({
   441→  args: {},
   442→  handler: async (ctx) => {
   443→    const now = Date.now();
   444→
   445→    // Find all sent and viewed invoices
   446→    const sentInvoices = await ctx.db
   447→      .query("invoices")
   448→      .filter((q) => q.eq(q.field("status"), "sent"))
   449→      .collect();
   450→
   451→    const viewedInvoices = await ctx.db
   452→      .query("invoices")
   453→      .filter((q) => q.eq(q.field("status"), "viewed"))
   454→      .collect();
   455→
   456→    const overdueEligible = [...sentInvoices, ...viewedInvoices];
   457→    let overdueCount = 0;
   458→
   459→    for (const invoice of overdueEligible) {
   460→      if (invoice.dueDate < now) {
   461→        await ctx.db.patch(invoice._id, {
   462→          status: "overdue",
   463→          updatedAt: now,
   464→        });
   465→        overdueCount++;
   466→      }
   467→    }
   468→
   469→    return { overdueCount };
   470→  },
   471→});

   539→// ─────────────────────────────────────────────
   540→// 9. LIST (workspace-aware, with optional status filter)
   541→// ─────────────────────────────────────────────
   542→export const list = query({
   543→  args: {
   544→    workspaceId: v.optional(v.id("workspaces")),
   545→    status: v.optional(invoiceStatusValidator),
   546→  },
   547→  handler: async (ctx, args) => {
   548→    const userId = await getAuthUserId(ctx);
   549→    if (!userId) throw new ConvexError("Not authenticated");
   550→
   551→    // If workspaceId provided, filter by workspace with team-aware visibility
   552→    if (args.workspaceId) {
   553→      const visibility = await getUserVisibility(ctx, args.workspaceId);
   554→      if (!visibility) throw new ConvexError("Not a member of this workspace");
   555→
   556→      const allInvoices = await ctx.db
   557→        .query("invoices")
   558→        .withIndex("by_workspace", (q) => q.eq("workspaceId", args.workspaceId!))
   559→        .collect();
   560→
   561→      let visible = allInvoices.filter((inv) => isRecordVisible(inv, visibility));
   562→
   563→      if (args.status) {
   564→        visible = visible.filter((inv) => inv.status === args.status);
   565→      }
   566→
   567→      return visible.sort((a, b) => b.updatedAt - a.updatedAt);
   568→    }
   569→
   570→    // Backward compat: no workspaceId
   571→    if (args.status) {
   572→      const invoices = await ctx.db
   573→        .query("invoices")
   574→        .withIndex("by_user_and_status", (q) =>
   575→          q.eq("userId", userId).eq("status", args.status!)

  1098→/**
  1099→ * Generate the next sequential invoice number for a user.
  1100→ * Format: INV-001, INV-002, etc.
  1101→ */
  1102→async function getNextInvoiceNumberInternal(
  1103→  ctx: any,
  1104→  userId: string
  1105→): Promise<string> {
  1106→  // Get all invoices for this user ordered by invoice number
  1107→  const existingInvoices = await ctx.db
  1108→    .query("invoices")
  1109→    .withIndex("by_user_and_number", (q: any) => q.eq("userId", userId))
  1110→    .collect();
  1111→
  1112→  // Find the highest existing number
  1113→  let maxNumber = 0;
  1114→  for (const inv of existingInvoices) {
  1115→    const match = inv.invoiceNumber.match(/^INV-(\d+)$/);
  1116→    if (match) {
  1117→      const num = parseInt(match[1], 10);
  1118→      if (num > maxNumber) maxNumber = num;
  1119→    }
  1120→  }
  1121→
  1122→  const nextNumber = maxNumber + 1;
  1123→  return `INV-${String(nextNumber).padStart(3, "0")}`;
  1124→}
