   114→const TAB_FILTERS = [
   115→  { key: "all", label: "All" },
   116→  { key: "active", label: "Active" },

   121→type TabFilter = (typeof TAB_FILTERS)[number]["key"];
   122→
   123→function filterReports(reports: DisputeReport[], tab: TabFilter): DisputeReport[] {
   124→  switch (tab) {
   125→    case "all":
   126→      return reports;
   127→    case "active":
   128→      return reports.filter((r) => r.status === "generated" || r.status === "sent" || r.status === "viewed");
   129→    case "resolved":
   130→      return reports.filter((r) => r.status === "resolved");
   131→    case "appealed":
   132→      return reports.filter((r) => r.status === "appealed");
   133→    default:
   134→      return reports;
   135→  }
   136→}
   137→
   138→function formatDate(ts: number): string {
   139→  return new Date(ts).toLocaleDateString("en-US", {
   140→    month: "short",
   141→    day: "numeric",
   142→    year: "numeric",
   143→  });
   144→}
   145→
   146→function formatCurrency(amount: number): string {
   147→  return new Intl.NumberFormat("en-US", {
   148→    style: "currency",
   149→    currency: "USD",
   150→    minimumFractionDigits: 0,
   151→    maximumFractionDigits: 2,
   152→  }).format(amount);
   153→}
