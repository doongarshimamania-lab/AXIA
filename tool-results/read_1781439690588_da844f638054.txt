   132→function formatCurrency(amount: number, currency: string = "USD"): string {
   133→  try {
   134→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   135→  } catch {
   136→    return `$${amount.toFixed(2)}`;
   137→  }
   138→}
   139→
   140→function formatDate(ts: number): string {
   141→  return new Date(ts).toLocaleDateString("en-US", {
   142→    month: "short",
   143→    day: "numeric",
   144→    year: "numeric",
   145→  });
   146→}
   147→
   148→function daysOverdue(dueDate: number): number {
   149→  const now = Date.now();
   150→  return Math.max(0, Math.floor((now - dueDate) / (1000 * 60 * 60 * 24)));
   151→}
