   322→function formatDate(ts: number | undefined) {
   323→  if (!ts) return "—";
   324→  return new Date(ts).toLocaleDateString("en-US", {
   325→    year: "numeric",
   326→    month: "short",
   327→    day: "numeric",
   328→  });
   329→}
   330→
   331→function formatCurrency(amount: number, currency = "USD") {
   332→  return new Intl.NumberFormat("en-US", {
   333→    style: "currency",
   334→    currency,
   335→    minimumFractionDigits: 0,
   336→    maximumFractionDigits: 2,
   337→  }).format(amount);
   338→}
   339→
   340→function relativeTime(ts: number | undefined) {
   341→  if (!ts) return "Never";
   342→  const diff = Date.now() - ts;
   343→  const mins = Math.floor(diff / 60000);
   344→  if (mins < 1) return "Just now";
   345→  if (mins < 60) return `${mins}m ago`;
   346→  const hours = Math.floor(mins / 60);
   347→  if (hours < 24) return `${hours}h ago`;
   348→  const days = Math.floor(hours / 24);
   349→  if (days < 30) return `${days}d ago`;
   350→  return formatDate(ts);
   351→}

   353→function getInitials(name: string) {
   354→  return name
   355→    .split(" ")
   356→    .map((w) => w[0])
   357→    .join("")
   358→    .toUpperCase()
   359→    .slice(0, 2);
   360→}

   362→function getAvatarColor(name: string) {
   363→  const colors = [
   364→    "bg-violet-500",
   365→    "bg-blue-500",
   366→    "bg-emerald-500",
   367→    "bg-amber-500",
   368→    "bg-rose-500",
   369→    "bg-cyan-500",
   370→    "bg-pink-500",
   371→    "bg-teal-500",
   372→  ];
   373→  let hash = 0;
   374→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   375→  return colors[Math.abs(hash) % colors.length];
   376→}

   389→
   390→  // Check if this is a demo token
   391→  const isDemo = token?.startsWith("demo-") ?? false;
   392→
   393→  // Validate token (skip for demo tokens)
   394→  const validation = useQuery(
   395→    api.clients.clientWorkspace.validateWorkspaceToken,
   396→    (!isDemo && token) ? { token } : "skip"
   397→  );
   398→
   399→  // Fetch data (only if token is valid OR if demo)
   400→  const shouldFetch = isDemo || validation?.valid;
   401→
   402→  const projects = useQuery(
   403→    api.clients.clientWorkspace.getClientProjects,
   404→    shouldFetch && !isDemo ? { token: token! } : "skip"
   405→  ) as ProjectData[] | undefined;
   406→
   407→  const proposals = useQuery(
   408→    api.clients.clientWorkspace.getClientProposals,
   409→    shouldFetch && !isDemo ? { token: token! } : "skip"
   410→  ) as ProposalData[] | undefined;
   411→
   412→  const invoices = useQuery(
   413→    api.clients.clientWorkspace.getClientInvoices,
   414→    shouldFetch && !isDemo ? { token: token! } : "skip"
   415→  ) as InvoiceData[] | undefined;
   416→
   417→  const team = useQuery(
   418→    api.clients.clientWorkspace.getClientTeamMembers,
   419→    shouldFetch && !isDemo ? { token: token! } : "skip"
   420→  ) as TeamMember[] | undefined;
   421→
   422→  // For demo mode, use static mock data
   423→  const effectiveProjects = isDemo ? DEMO_PROJECTS : projects;
   424→  const effectiveProposals = isDemo ? DEMO_PROPOSALS : proposals;
   425→  const effectiveInvoices = isDemo ? DEMO_INVOICES : invoices;
   426→  const effectiveTeam = isDemo ? DEMO_TEAM : team;
   427→
   428→  // Record access (only for real tokens)
   429→  const recordAccess = useMutation(api.clients.clientWorkspace.recordWorkspaceAccess);
   430→  useEffect(() => {
   431→    if (!isDemo && validation?.valid && token) {
   432→      recordAccess({ token });
   433→    }
   434→  }, [isDemo, validation?.valid, token]);
   435→
   436→  // Mark proposals/invoices as viewed (only for real tokens)
   437→  const markProposalViewed = useMutation(api.clients.clientWorkspace.markProposalViewedByClient);
   438→  const markInvoiceViewed = useMutation(api.clients.clientWorkspace.markInvoiceViewedByClient);
   439→
   440→  useEffect(() => {
   441→    if (!isDemo && proposals) {
   442→      proposals.forEach((p) => {
   443→        if (p.status === "sent") {
   444→          markProposalViewed({ proposalId: p._id as any });
   445→        }
   446→      });
   447→    }
   448→  }, [proposals]);
   449→
   450→  useEffect(() => {
   451→    if (!isDemo && invoices) {
   452→      invoices.forEach((inv) => {
   453→        if (inv.status === "sent") {
   454→          markInvoiceViewed({ invoiceId: inv._id as any });
   455→        }
   456→      });
   457→    }
   458→  }, [invoices]);
   459→
   460→  const copyLink = () => {
   461→    const url = `${window.location.origin}/workspace/${token}`;
   462→    navigator.clipboard.writeText(url);
   463→    setCopied(true);
   464→    setTimeout(() => setCopied(false), 2000);
   465→  };
   466→
   467→  // ---- Loading & Error States ----
   468→
   469→  // Timeout fallback for validation query
   470→  const validationLoading = !isDemo && validation === undefined;
   471→  const validationTimedOut = useQueryTimeout(validationLoading, 6000);
   472→
   473→  // Demo tokens are always valid
   474→  if (validationLoading && !validationTimedOut) {
   475→    return (
   476→      <div className="min-h-screen bg-gradient-to-br from-slate-50 to-violet-50 dark:from-gray-950 dark:to-gray-900 flex items-center justify-center">
   477→        <div className="flex flex-col items-center gap-4">
   478→          <div className="w-10 h-10 border-3 border-violet-500 border-t-transparent rounded-full animate-spin" />
   479→          <p className="text-gray-500 dark:text-gray-400 text-sm">Loading workspace...</p>
   480→        </div>
   481→      </div>
   482→    );
   483→  }
   484→
   485→  // Show error state if validation timed out
   486→  if (validationTimedOut && !isDemo && validation === undefined) {
   487→    return (
   488→      <div className="min-h-screen bg-gradient-to-br from-slate-50 to-violet-50 dark:from-gray-950 dark:to-gray-900 flex items-center justify-center p-6">
   489→        <div className="bg-white dark:bg-gray-900 rounded-2xl shadow-xl border border-gray-200 dark:border-gray-800 p-10 max-w-md text-center">
   490→          <div className="w-16 h-16 bg-amber-100 dark:bg-amber-900/30 rounded-full flex items-center justify-center mx-auto mb-5">
   491→            <AlertTriangle className="w-8 h-8 text-amber-500" />
   492→          </div>
   493→          <h2 className="text-xl font-bold text-gray-900 dark:text-white mb-2">
   494→            Loading Timed Out
   495→          </h2>
   496→          <p className="text-gray-500 dark:text-gray-400 text-sm mb-4">
   497→            Could not load workspace data. Please try refreshing the page.
   498→          </p>
   499→          <button
   500→            onClick={() => window.location.reload()}
   501→            className="px-4 py-2 bg-violet-600 text-white rounded-lg hover:bg-violet-700 transition-colors"
   502→          >
   503→            Refresh
   504→          </button>
   505→        </div>
   506→      </div>
