   319→function invoiceToBillingRecord(inv: ConvexInvoice): BillingRecord {
   320→  // Build description from line items or default to client name
   321→  const firstItem = inv.lineItems?.[0];
   322→  const desc = firstItem
   323→    ? inv.lineItems.length > 1
   324→      ? `${firstItem.description} + ${inv.lineItems.length - 1} more`
   325→      : firstItem.description
   326→    : inv.clientName
   327→    ? `Invoice for ${inv.clientName}`
   328→    : "Invoice";
   329→
   330→  return {
   331→    id: inv.invoiceNumber,
   332→    date: new Date(inv.issueDate).toISOString(),
   333→    description: desc,
   334→    amount: inv.total,
   335→    status: inv.status,
   336→    invoice: inv,
   337→  };
   338→}

   404→function downloadReceipt(record: BillingRecord) {
   405→  const doc = new jsPDF({ unit: "mm", format: "a4" });
   406→  const margin = 20;
   407→  const pageWidth = doc.internal.pageSize.getWidth();
   408→  const pageHeight = doc.internal.pageSize.getHeight();
   409→
   410→  // Title
   411→  doc.setFont("helvetica", "bold");
   412→  doc.setFontSize(20);
   413→  doc.setTextColor(26, 26, 46);
   414→  doc.text("Receipt", margin, 25);
   415→
   416→  // Invoice ID
   417→  doc.setFont("helvetica", "normal");
   418→  doc.setFontSize(10);
   419→  doc.setTextColor(100, 116, 139);
   420→  doc.text(record.id, margin, 32);
   421→
   422→  // Separator
   423→  doc.setDrawColor(226, 232, 240);
   424→  doc.line(margin, 37, pageWidth - margin, 37);
   425→
   426→  let y = 46;
   427→  const addRow = (label: string, value: string) => {
   428→    doc.setFont("helvetica", "normal");
   429→    doc.setFontSize(9);
   430→    doc.setTextColor(148, 163, 184);
   431→    doc.text(label, margin, y);
   432→    doc.setFont("helvetica", "bold");
   433→    doc.setFontSize(11);
   434→    doc.setTextColor(30, 41, 59);
   435→    doc.text(value, margin + 50, y);
   436→    y += 10;
   437→  };
   438→
   439→  addRow("Description", record.description);
   440→  addRow("Date", new Date(record.date).toLocaleDateString("en-US", { year: "numeric", month: "long", day: "numeric" }));
   441→
   442→  // Map invoice status to a display-friendly label
   443→  const statusLabel = record.status.charAt(0).toUpperCase() + record.status.slice(1);
   444→  addRow("Status", statusLabel);
   445→
   446→  const currency = record.invoice?.currency || "USD";
   447→  addRow("Amount", `${currency} $${record.amount.toFixed(2)}`);
   448→
   449→  // Add client info if available
   450→  if (record.invoice?.clientName) {
   451→    addRow("Client", record.invoice.clientName);
   452→  }
   453→
   454→  // Footer
   455→  doc.setFont("helvetica", "normal");
   456→  doc.setFontSize(7);
   457→  doc.setTextColor(203, 213, 225);
   458→  doc.text(`Generated by Axia \u2022 ${new Date().toLocaleString()}`, margin, pageHeight - 12);
   459→
   460→  doc.save(`${record.id}.pdf`);
   461→  toast.success("Receipt downloaded");
   462→}

   485→function UsageBar({ used, limit }: { used: number; limit: number | null }) {

   494→  const isHigh = pct >= 80;

   496→  return (
   497→    <div className="w-full h-2 rounded-full bg-muted overflow-hidden">

   499→        className={`h-full rounded-full transition-all duration-500 ${

   501→        }`}
   502→        style={{ width: `${pct}%` }}
   503→      />
   504→    </div>
   505→  );

   546→            </AnimatePresence>
   547→          </motion.div>
   548→        );
   549→      })}
   550→    </div>
   551→  );
   552→}

   556→function ChangePlanDialog({
   557→  isOpen,
   558→  onClose,
