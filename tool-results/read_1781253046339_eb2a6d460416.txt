    95→function getTierLevel(tier: string): number {
    96→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
    97→  return levels[tier] ?? 0;
    98→}
    99→
   100→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   101→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   102→}

   240→// --- Main Component ---
   241→export default function EvidenceExport() {
   242→  const { tier } = useSubscriptionTier();
   243→  const { isAuthenticated } = useConvexAuth();
   244→
   245→  // ─── Convex Queries ────────────────────────────────────────────────────────
   246→  const evidenceData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   247→    view: "date" as const,
   248→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   249→    endDate: Date.now(),
   250→  }) as any | undefined;
   251→
   252→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   253→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   254→
   255→  const { isDisconnected } = useConvexConnectionState();
   256→  const isLoading = evidenceData === undefined || clients === undefined;
   257→  const timedOut = useQueryTimeout(isLoading, 3000);
   258→  const showLoading = isLoading && !timedOut && !isDisconnected;
   259→
   260→  // ─── Derived Data ──────────────────────────────────────────────────────────
   261→
   262→  // Build evidence type options from real data
   263→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   264→    if (!evidenceData) return [];
   265→    const items = evidenceData.evidenceItems ?? [];
   266→    const typeCounts: Record<string, number> = {};
   267→    const typeLabels: Record<string, string> = {
   268→      screenshot_ref: "Screenshots",
   269→      memo: "Work Memos",
   270→      url: "URL Visits",
   271→      keyboard: "Keyboard Activity",
   272→      mouse: "Mouse Activity",
   273→      platform_status: "Platform Status",
   274→    };
   275→    const typeIcons: Record<string, React.ReactNode> = {
   276→      screenshot_ref: <Eye className="h-4 w-4" />,
   277→      memo: <FileText className="h-4 w-4" />,
   278→      url: <FileJson className="h-4 w-4" />,
   279→      keyboard: <FileText className="h-4 w-4" />,
   280→      mouse: <FileText className="h-4 w-4" />,
   281→      platform_status: <Shield className="h-4 w-4" />,
   282→    };
   283→
   284→    for (const item of items) {
   285→      const kind = item.type ?? "unknown";
   286→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   287→    }
   288→
   289→    return Object.entries(typeCounts).map(([id, count]) => ({
   290→      id,
   291→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   292→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   293→      count,
   294→    }));
   295→  }, [evidenceData]);
   296→
   297→  // Build project options from real scopes
   298→  const projectOptions = useMemo(() => {
   299→    const all = [{ id: "all", label: "All Projects" }];
   300→    if (scopeDefinitions) {
   301→      for (const scope of scopeDefinitions) {
   302→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   303→      }
   304→    }
   305→    return all;
   306→  }, [scopeDefinitions]);
   307→
   308→  // Build client options from real clients
   309→  const clientOptions = useMemo(() => {
   310→    const all = [{ id: "all", label: "All Clients" }];
   311→    if (clients) {
   312→      for (const client of clients) {
   313→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   314→      }
   315→    }
   316→    return all;
   317→  }, [clients]);
   318→
   319→  // Compliance score from real data
   320→  const complianceScore = evidenceData?.healthScore?.score ?? 0;
   321→  const totalItems = evidenceData?.totalCount ?? 0;
   322→  const disputeSuccessRate = evidenceData?.disputeSuccessRate ?? 0;
   323→
   324→  // State
   325→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   326→  const [dateFrom, setDateFrom] = useState(() => {
   327→    const d = new Date();
   328→    d.setMonth(d.getMonth() - 3);
   329→    return d.toISOString().split("T")[0];
   330→  });
   331→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   332→  const [selectedProject, setSelectedProject] = useState("all");
   333→  const [selectedClient, setSelectedClient] = useState("all");
   334→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   335→  const [isExporting, setIsExporting] = useState(false);
   336→  const [previewOpen, setPreviewOpen] = useState(false);
   337→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   338→
   339→  // Auto-select all evidence types when they load
   340→  useEffect(() => {
   341→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   342→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   343→    }
   344→  }, [evidenceTypes]);
   345→
   346→  // Derived
   347→  const totalEvidenceItems = useMemo(
   348→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   349→    [evidenceTypes, selectedEvidenceTypes]
   350→  );
   351→
   352→  const isFormatLocked = (format: ExportFormat): boolean => {
   353→    const requiredTier = FORMAT_CONFIG[format].tierRequired;
   354→    return !hasTierAccess(tier, requiredTier);
   355→  };
   356→
   357→  const toggleEvidenceType = (id: string) => {
   358→    setSelectedEvidenceTypes((prev) =>
   359→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   360→    );
   361→  };
   362→
   363→  const handleExport = async () => {
   364→    if (selectedEvidenceTypes.length === 0) {
   365→      toast.error("No evidence types selected", {
   366→        description: "Select at least one evidence type to export.",
   367→      });
   368→      return;
   369→    }
   370→
   371→    if (isFormatLocked(selectedFormat)) {
   372→      toast.error("Upgrade required", {
   373→        description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.`,
   374→      });
   375→      return;
   376→    }
   377→
   378→    setIsExporting(true);
   379→
   380→    try {
   381→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   382→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   383→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   384→
   385→      exportEvidence(selectedFormat, {
   386→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   387→        dateRange: `${dateFrom} to ${dateTo}`,
   388→        project: selectedProjectLabel,
   389→        client: selectedClientLabel,
   390→        complianceScore,
   391→        totalItems: totalEvidenceItems,
   392→      });
   393→
   394→      // Add to recent exports
   395→      const newExport: RecentExport = {
   396→        id: `exp_${Date.now()}`,
   397→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   398→        format: selectedFormat,
   399→        status: "completed",
   400→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   401→        date: new Date().toISOString().split("T")[0],
   402→        itemCount: totalEvidenceItems,
   403→        complianceVerified: hasTierAccess(tier, "pro"),
   404→        project: selectedProjectLabel,
   405→      };
   406→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   407→
   408→      toast.success("Export complete", {
   409→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   410→      });
   411→    } catch {
   412→      toast.error("Export failed", {
   413→        description: "An error occurred while generating the export file.",
   414→      });
   415→    }
   416→
   417→    setIsExporting(false);
   418→  };
   419→
   420→  const handleFormatSelect = (format: ExportFormat) => {
   421→    if (isFormatLocked(format)) {
   422→      toast.error("Upgrade required", {
   423→        description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.`,
   424→      });
   425→      return;
   426→    }
   427→    setSelectedFormat(format);
   428→  };
   429→
   430→  const previewItems = useMemo(() => {
   431→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   432→  }, [evidenceTypes, selectedEvidenceTypes]);
   433→
   434→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   435→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   436→
   437→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   438→
   439→  // ─── Render ───────────────────────────────────────────────────────────────
   440→
   441→  return (
   442→    <motion.div
   443→      initial={{ opacity: 0 }}
   444→      animate={{ opacity: 1 }}
   445→      className="w-full min-h-screen bg-background"
   446→    >
   447→      <div className="p-6 md:p-8 space-y-8">
   448→        {/* Header */}
   449→        <div>
   450→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">Evidence Export</h1>
   451→          <p className="text-[16px] text-muted-foreground">
   452→            Package and download your work evidence in court-ready formats with compliance verification.
   453→          </p>
   454→        </div>
   455→
   456→        {/* Demo Mode Banner */}
   457→        {!isAuthenticated && <DemoModeBanner />}
   458→
   459→        {/* Loading / Timeout State */}
   460→        {showLoading ? (
   461→          <>
   462→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   463→              {[0, 1, 2, 3].map((i) => <SummarySkeleton key={i} />)}
   464→            </div>
   465→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   466→              <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   467→              <div><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   468→            </div>
   469→          </>
   470→        ) : !hasData ? (
   471→          <EmptyEvidenceState />
   472→        ) : (
   473→          <>
   474→            {/* Summary Dashboard */}
   475→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   476→              <SummaryCard
   477→                title="Total Items"
   478→                value={totalItems}
   479→                subtitle="Across all projects"
   480→                icon={<HardDrive className="h-5 w-5 text-primary-foreground" />}
   481→                accent="bg-primary"
   482→              />
   483→              <SummaryCard
   484→                title="Evidence Types"
   485→                value={evidenceTypes.length}
   486→                subtitle="Different categories"
   487→                icon={<Clock className="h-5 w-5 text-amber-700" />}
   488→                accent="bg-amber-100 dark:bg-amber-900/30"
   489→              />
   490→              <SummaryCard
   491→                title="Compliance Score"
   492→                value={`${complianceScore}%`}
   493→                subtitle={complianceScore >= 80 ? "Court-admissible ready" : "Needs improvement"}
   494→                icon={<ShieldCheck className="h-5 w-5 text-emerald-700" />}
   495→                accent="bg-emerald-100 dark:bg-emerald-900/30"
   496→              />
   497→              <SummaryCard
   498→                title="Dispute Success"
   499→                value={`${disputeSuccessRate}%`}
   500→                subtitle="Based on evidence quality"
   501→                icon={<Shield className="h-5 w-5 text-sky-700" />}
   502→                accent="bg-sky-100 dark:bg-sky-900/30"
   503→              />
   504→            </div>
   505→
   506→            {/* Export Configuration */}
   507→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   508→              {/* Left Column: Format + Filters */}
   509→              <div className="lg:col-span-2 space-y-6">
   510→                {/* Format Selection */}
   511→                <Card>
   512→                  <CardHeader>
   513→                    <CardTitle className="flex items-center gap-2">
   514→                      <Download className="h-5 w-5 text-primary" />
   515→                      Export Format
   516→                    </CardTitle>
   517→                    <CardDescription>Choose the output format for your evidence package.</CardDescription>
   518→                  </CardHeader>
   519→                  <CardContent>
   520→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   521→                      {(Object.keys(FORMAT_CONFIG) as ExportFormat[]).map((formatKey) => (
   522→                        <FormatOption
   523→                          key={formatKey}
   524→                          formatKey={formatKey}
   525→                          selected={selectedFormat === formatKey}
   526→                          onSelect={() => handleFormatSelect(formatKey)}
   527→                          locked={isFormatLocked(formatKey)}
   528→                        />
   529→                      ))}
   530→                    </div>
   531→                    {isFormatLocked(selectedFormat) && (
   532→                      <div className="mt-4 flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-900/20 dark:border-amber-800 p-3">
   533→                        <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400 shrink-0" />
   534→                        <p className="text-sm text-amber-800 dark:text-amber-200">
   535→                          <span className="font-semibold">Upgrade to {FORMAT_CONFIG[selectedFormat].tierRequired}</span> to unlock{" "}
   536→                          {FORMAT_CONFIG[selectedFormat].label} exports with compliance verification.
   537→                        </p>
   538→                      </div>
   539→                    )}
   540→                  </CardContent>
   541→                </Card>
   542→
   543→                {/* Date Range & Filters */}
   544→                <Card>
   545→                  <CardHeader>
