   502→    { value: "abandoned", label: "Abandoned" },
   503→  ];

   507→    <motion.div
   508→      initial={{ opacity: 0 }}
   509→      animate={{ opacity: 1 }}
   510→      className="w-full min-h-screen bg-background text-foreground"
   511→    >
   512→      <div className="container mx-auto px-4 py-6 space-y-6">
   513→        {/* ── Header ── */}
   514→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

   523→
   524→          <Dialog

   526→            onOpenChange={(open) => {

   528→              if (!open) resetForm();
   529→            }}
   530→          >
   531→            <DialogTrigger asChild>
   532→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   533→                <Plus className="mr-2 h-4 w-4" />
   534→                Create Goal
   535→              </Button>
   536→            </DialogTrigger>
   537→            <DialogContent className="sm:max-w-lg">
   538→              <DialogHeader>
   539→                <DialogTitle>Create New Goal</DialogTitle>
   540→                <DialogDescription>
   541→                  Define a measurable goal with a deadline to keep your professional growth on track.
   542→                </DialogDescription>
   543→              </DialogHeader>
   544→              <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   545→                <div className="space-y-2">
   546→                  <Label htmlFor="goal-title">Title</Label>

   553→                </div>
