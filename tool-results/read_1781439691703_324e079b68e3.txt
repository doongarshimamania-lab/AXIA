   297→function formatCurrency(amount: number, currency: string = "USD"): string {
   298→  try {
   299→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   300→  } catch {
   301→    return `$${amount.toFixed(2)}`;
   302→  }
   303→}
   304→
   305→function formatDate(ts: number): string {
   306→  return new Date(ts).toLocaleDateString("en-US", {
   307→    month: "short",
   308→    day: "numeric",
   309→    year: "numeric",
   310→  });
   311→}
   312→
   313→function formatRelativeTime(ts: number): string {
   314→  const diff = Date.now() - ts;
   315→  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   316→  if (days === 0) return "Today";
   317→  if (days === 1) return "Yesterday";
   318→  if (days < 7) return `${days} days ago`;
   319→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   320→  return `${Math.floor(days / 30)}mo ago`;
   321→}
