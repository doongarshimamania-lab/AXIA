    62→    62→function fmtRelative(ts: number): string {
    63→    63→  const diff = Date.now() - ts;
    64→    64→  const mins = Math.floor(diff / 60000);
    65→    65→  if (mins < 1) return "just now";
    66→    66→  if (mins < 60) return `${mins}m ago`;
    67→    67→  const hrs = Math.floor(mins / 60);
    68→    68→  if (hrs < 24) return `${hrs}h ago`;
    69→    69→  const days = Math.floor(hrs / 24);
    70→    70→  if (days < 30) return `${days}d ago`;
    71→    71→  return new Date(ts).toLocaleDateString();
    72→    72→}

    75→    75→function AnimatedNumber({ value, duration = 1200, prefix = "", suffix = "", decimals = 0 }: {
    76→    76→  value: number; duration?: number; prefix?: string; suffix?: string; decimals?: number;
    77→    77→}) {
    78→    78→  const [display, setDisplay] = useState(0);
    79→    79→  const prevValue = useRef(0);
    80→    80→  const rafRef = useRef<number>();
    81→    81→
    82→    82→  useEffect(() => {
    83→    83→    const start = prevValue.current;
    84→    84→    const end = value;
    85→    85→    const startTime = performance.now();
    86→    86→
    87→    87→    const animate = (now: number) => {
    88→    88→      const elapsed = now - startTime;
    89→    89→      const progress = Math.min(elapsed / duration, 1);
    90→    90→      // Ease out cubic
    91→    91→      const eased = 1 - Math.pow(1 - progress, 3);
    92→    92→      const current = start + (end - start) * eased;
    93→    93→      setDisplay(current);
    94→    94→      if (progress < 1) {
    95→    95→        rafRef.current = requestAnimationFrame(animate);
    96→    96→      } else {
    97→    97→        prevValue.current = end;
    98→    98→      }
    99→    99→    };
   100→   100→
   101→   101→    rafRef.current = requestAnimationFrame(animate);
   102→   102→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
   103→   103→  }, [value, duration]);
   104→   104→
   105→   105→  const formatted = decimals > 0
   106→   106→    ? display.toFixed(decimals)
   107→   107→    : Math.round(display).toLocaleString();
   108→   108→
   109→   109→  return <span>{prefix}{formatted}{suffix}</span>;
   110→   110→}

   113→   113→function Sparkline({ data, color = "#8B5CF6", height = 32, width = 80 }: {
   114→   114→  data: number[]; color?: string; height?: number; width?: number;
   115→   115→}) {
   116→   116→  if (!data || data.length < 2) return null;
   117→   117→
   118→   118→  const min = Math.min(...data);
   119→   119→  const max = Math.max(...data);
   120→   120→  const range = max - min || 1;
   121→   121→  const padding = 2;
   122→   122→
   123→   123→  const points = data.map((v, i) => {
   124→   124→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   125→   125→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   126→   126→    return `${x},${y}`;
   127→   127→  }).join(" ");
   128→   128→
   129→   129→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   130→   130→
   131→   131→  return (
   132→   132→    <svg width={width} height={height} className="overflow-visible">
   133→   133→      <defs>
   134→   134→        <linearGradient id={`spark-grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
   135→   135→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   136→   136→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   137→   137→        </linearGradient>
   138→   138→      </defs>
   139→   139→      <polygon points={areaPoints} fill={`url(#spark-grad-${color.replace("#", "")})`} />
   140→   140→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   141→   141→      {/* Last point dot */}
   142→   142→      {data.length > 0 && (
   143→   143→        <circle
   144→   144→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   145→   145→          cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)}
   146→   146→          r="2.5"
   147→   147→          fill={color}
   148→   148→          className="animate-pulse"
   149→   149→        />
   150→   150→      )}
   151→   151→    </svg>
   152→   152→  );
   153→   153→}

   156→   156→function ProgressRing({ value, size = 56, strokeWidth = 4, color = "#8B5CF6", label }: {
   157→   157→  value: number; size?: number; strokeWidth?: number; color?: string; label?: string;
   158→   158→}) {
   159→   159→  const radius = (size - strokeWidth) / 2;
   160→   160→  const circumference = 2 * Math.PI * radius;
   161→   161→  const offset = circumference - (value / 100) * circumference;
   162→   162→
   163→   163→  return (
   164→   164→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   165→   165→      <svg width={size} height={size} className="-rotate-90">
   166→   166→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   167→   167→        <motion.circle
   168→   168→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
   169→   169→          strokeLinecap="round"
   170→   170→          strokeDasharray={circumference}
   171→   171→          initial={{ strokeDashoffset: circumference }}
   172→   172→          animate={{ strokeDashoffset: offset }}
   173→   173→          transition={{ duration: 1.2, ease: "easeOut" }}
   174→   174→        />
   175→   175→      </svg>
   176→   176→      <span className="absolute text-[10px] font-bold" style={{ color }}>{value}%</span>
   177→   177→    </div>
   178→   178→  );
   179→   179→}
