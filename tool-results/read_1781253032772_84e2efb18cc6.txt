   506→                  window.dispatchEvent(event);
   507→                }} className="w-full text-left">
   508→                  <NavItem icon={Activity} label="Work Timeline" isExpanded={true} />
   509→                </button>
   510→                {/* Sign Out */}
   511→                <button onClick={async (e) => {
   512→                  e.stopPropagation();
   513→                  try {
   514→                    await signOut();
   515→                    navigate("/auth");
   516→                  } catch {
   517→                    toast.error("Failed to sign out");
   518→                  }
   519→                }} className="w-full text-left">
   520→                  <NavItem icon={LogOut} label="Sign Out" isExpanded={true} />
   521→                </button>
   522→              </motion.div>
   523→            )}
   524→          </AnimatePresence>
   525→          {/* Collapsed theme toggle icon */}
   526→          {!isExpanded && (
   527→            <div className="flex flex-col items-center gap-4 py-2">
   528→              <button
   529→                onClick={(e) => {
   530→                  e.stopPropagation();
   531→                  handleToggleTheme();
   532→                }}
   533→                title={theme === "dark" ? "Light Mode" : "Dark Mode"}
   534→                className="p-2 hover:bg-sidebar-accent rounded-lg transition-colors"
   535→              >
   536→                {theme === "dark" ? (
   537→                  <Sun className="w-5 h-5 text-sidebar-foreground/60 hover:text-sidebar-foreground" />
   538→                ) : (
   539→                  <Moon className="w-5 h-5 text-sidebar-foreground/60 hover:text-sidebar-foreground" />
   540→                )}
   541→              </button>
   542→            </div>
   543→          )}
   544→        </div>
   545→      </motion.div>
   546→    </>
   547→  );
   548→}

   590→        <span className="font-medium text-[12px]">{title}</span>
