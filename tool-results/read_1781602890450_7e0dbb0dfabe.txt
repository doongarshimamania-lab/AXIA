   144→
   145→  // ── Permissions ──

   147→
   148→  const [activeFilter, setActiveFilter] = useState<"all" | ProposalStatus>("all");
   149→  const [searchQuery, setSearchQuery] = useState("");
   150→  const [deleteDialogId, setDeleteDialogId] = useState<string | null>(null);
   151→  const [isSeeding, setIsSeeding] = useState(false);
   152→  const [showShareDialog, setShowShareDialog] = useState(false);
   153→  const [shareProposalId, setShareProposalId] = useState<string | null>(null);
   154→  const [sharingRecord, setSharingRecord] = useState<{id: string, type: string, sharing: any[]} | null>(null);
   155→
   156→  // ── Convex mutations for sharing ──
   157→  const shareRecordMutation = useMutation(api.permissions.shareRecord.shareRecord);

   159→

   162→  const convexStats = useQuery(api.proposals.crud.getProposalStats, workspaceId ? { workspaceId } : "skip") as {
   163→    total: number;
   164→    sent: number;
   165→    signed: number;
   166→    draft: number;
   167→    signatureRate: number;
   168→    totalValue: number;

   170→

   172→  const sendProposal = useMutation(api.proposals.crud.sendProposal);
   173→  const duplicateProposal = useMutation(api.proposals.crud.duplicateProposal);
   174→  const deleteProposal = useMutation(api.proposals.crud.deleteProposal);
   175→  const seedMockProposals = useMutation(api.seedNew.seedMockProposals);
   176→

   178→  const createClientMutation = useMutation(api.clients.crud.createClient);
   179→  const addProjectMutation = useMutation(api.projects.projectProtectionSimple.addProject);
   180→  const moveDealMutation = useMutation(api.pipeline.crud.moveDeal);
   181→  const updateProposalMutation = useMutation(api.proposals.crud.updateProposal);
   182→
   183→  // Fetch pipeline stages (to find "Won" stage for deal conversion)

   185→

   187→  const existingClients = useQuery(api.clients.crud.getClients, workspaceId ? { workspaceId } : "skip") as { _id: string; clientName: string; contactEmail?: string }[] | undefined;
   188→
   189→  // Filter counts (from all proposals for the tab badges)
   190→  const convexAllProposals = useQuery(api.proposals.crud.getProposals, workspaceId ? { workspaceId } : "skip") as Proposal[] | undefined;
   191→
   192→  const { isDisconnected } = useConvexConnectionState();
   193→  const isLoading = convexProposals === undefined || convexStats === undefined;
   194→  const timedOut = useQueryTimeout(isLoading, 3000);
   195→  const showLoading = isLoading && !timedOut && !isDisconnected;
   196→
   197→  // Use Convex data only
   198→  const proposals = useMemo(() => {
   199→    return convexProposals ?? [];
   200→  }, [convexProposals]);
   201→
   202→  const stats = useMemo(() => {
   203→    return convexStats ?? { total: 0, sent: 0, signed: 0, draft: 0, signatureRate: 0, totalValue: 0 };
   204→  }, [convexStats]);
   205→
   206→  const allProposals = useMemo(() => {
   207→    return convexAllProposals ?? [];
   208→  }, [convexAllProposals]);
   209→
   210→  const filterCounts = useMemo(() => {
   211→    const counts: Record<string, number> = { all: allProposals.length };
   212→    for (const p of allProposals) {
   213→      counts[p.status] = (counts[p.status] || 0) + 1;
   214→    }
   215→    return counts;
   216→  }, [allProposals]);
   217→
   218→  // Filtered + searched proposals
   219→  const filteredProposals = useMemo(() => {
   220→    if (!proposals) return [];
   221→    return proposals.filter((p) => {
   222→      if (searchQuery === "") return true;
   223→      const q = searchQuery.toLowerCase();
   224→      return (
   225→        p.title.toLowerCase().includes(q) ||
   226→        (p.clientName && p.clientName.toLowerCase().includes(q))
   227→      );
   228→    });
   229→  }, [proposals, searchQuery]);
   230→
   231→  const formatCurrency = (amount: number) =>
   232→    new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(amount);
   233→
   234→  const formatDate = (timestamp: number) => {
   235→    const d = new Date(timestamp);
   236→    return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
   237→  };
   238→
   239→  const handleSeed = async () => {
   240→    setIsSeeding(true);
   241→    try {
   242→      const result = await seedMockProposals({});
   243→      if (result && typeof result === "object" && "seeded" in result) {
   244→        if ((result as any).seeded) {
   245→          toast.success("Mock proposals seeded!", {
   246→            description: `${(result as any).count} proposals created.`,
   247→          });
   248→        } else {
   249→          toast.info("Proposals already exist", {
   250→            description: `${(result as any).count} proposals found.`,
   251→          });
   252→        }
   253→      }
   254→    } catch (err: any) {
   255→      toast.error("Failed to seed proposals", { description: err.message });
   256→    } finally {
   257→      setIsSeeding(false);
   258→    }
   259→  };
   260→
   261→  const handleSend = async (proposalId: string) => {
   262→    try {
   263→      await sendProposal({ proposalId: proposalId as any });
   264→      toast.success("Proposal sent!", {
   265→        description: "Follow-ups have been scheduled automatically.",
   266→      });
   267→    } catch (err: any) {
   268→      toast.error("Failed to send proposal", { description: err.message });
   269→    }
   270→  };

   273→    try {
   274→      const newId = await duplicateProposal({ proposalId: proposalId as any });
   275→      toast.success("Proposal duplicated!", {
   276→        description: "A draft copy has been created.",
   277→      });
   278→    } catch (err: any) {
   279→      toast.error("Failed to duplicate proposal", { description: err.message });
   280→    }
   281→  };
   282→
   283→  const handleDelete = async (proposalId: string) => {
   284→    try {
   285→      await deleteProposal({ proposalId: proposalId as any });
   286→      setDeleteDialogId(null);
   287→      toast.success("Proposal deleted");
   288→    } catch (err: any) {
   289→      toast.error("Failed to delete proposal", { description: err.message });
   290→    }
   291→  };
   292→
   293→  const handleConvertToProject = async (proposal: Proposal) => {
   294→    try {
   295→      // Step 1: Find or create the client
   296→      const clientName = proposal.clientName || "Unknown Client";
   297→      const clientEmail = proposal.clientEmail;
   298→      let clientId: string | undefined;
   299→
   300→      // Check if a client with this name already exists
   301→      const existingClient = existingClients?.find(
   302→        (c) => c.clientName.toLowerCase() === clientName.toLowerCase()
   303→      );
   304→
   305→      if (existingClient) {
   306→        clientId = existingClient._id;
   307→      } else {
   308→        // Create a new client
   309→        const newClientId = await createClientMutation({
   310→          clientName,
   311→          platform: "direct" as const,
   312→          hourlyRate: proposal.totalValue > 0 ? proposal.totalValue / 40 : 50,
   313→          contractType: "hourly" as const,
   314→          contactEmail: clientEmail,
   315→          contactName: clientName,
   316→          workspaceId,
   317→          notes: `Created from proposal: ${proposal.title}`,
   318→        });
   319→        clientId = newClientId as unknown as string;
   320→      }
   321→
   322→      if (!clientId) {
   323→        throw new Error("Failed to resolve client ID");
   324→      }
   325→
   326→      // Step 2: Create the project
   327→      await addProjectMutation({
   328→        projectName: proposal.title,
   329→        clientId: clientId as any,
   330→        hourlyRate: proposal.totalValue > 0 ? proposal.totalValue / 40 : 50,
   331→        projectType: "ongoing" as const,
   332→        protectionLevel: "enhanced" as const,
   333→        workspaceId,
   334→      });
   335→
   336→      // Step 3: If proposal has a dealId, move the deal to "Won" stage
   337→      const dealId = (proposal as any).dealId;
   338→      if (dealId && pipelineStages) {
   339→        const wonStage = pipelineStages.find(
   340→          (s) => s.name.toLowerCase() === "won"
   341→        );
   342→        if (wonStage) {
   343→          await moveDealMutation({
   344→            dealId: dealId as any,
   345→            stageId: wonStage._id as any,
   346→          });
   347→        }
   348→      }
   349→
   350→      // Step 4: Update proposal notes to indicate conversion
   351→      const existingNotes = proposal.notes || "";
   352→      await updateProposalMutation({
   353→        proposalId: proposal._id as any,
   354→        notes: `${existingNotes ? existingNotes + "\n" : ""}[Converted to project — ${new Date().toLocaleDateString()}]`,
   355→      });
   356→
   357→      toast.success("Project created!", {
   358→        description: `"${proposal.title}" has been converted to a project.`,
   359→      });
   360→    } catch (err: any) {
   361→      toast.error("Failed to convert to project", { description: err.message });
   362→    }
   363→  };
   364→
   365→  return (
   366→    <motion.div
   367→      className="flex-1 min-h-screen bg-background text-foreground transition-colors"
   368→      initial={{ opacity: 0 }}
   369→      animate={{ opacity: 1 }}
   370→      transition={{ duration: 0.5 }}
   371→    >
   372→      <PageLayout wide>
   373→        {/* Header */}
   374→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 mb-6">
   375→          <div>
   376→            <h1 className="text-2xl md:text-[32px] font-bold text-foreground tracking-tight mb-1" style={{ fontFamily: "Space Grotesk, sans-serif" }}>
   377→              Proposals
   378→            </h1>
   379→            <p className="text-[16px] text-muted-foreground">
   380→              Create, send, and track proposals with automated follow-ups
   381→            </p>
   382→          </div>
   383→          <div className="flex items-center gap-2">
   384→            <Button
   385→              variant="outline"
   386→              size="sm"
   387→              className="gap-2 text-muted-foreground"
   388→              onClick={handleSeed}
   389→              disabled={isSeeding}
   390→            >
   391→              {isSeeding ? (
   392→                <Loader2 className="h-4 w-4 animate-spin" />
   393→              ) : (
   394→                <Sparkles className="h-4 w-4" />
   395→              )}
   396→              Seed Data
   397→            </Button>
   398→            <Button
   399→              className="bg-[#8B5CF6] hover:bg-[#7C3AED] text-white gap-2"
   400→              onClick={() => navigate("/proposals/new")}
   401→            >
   402→              <Plus className="h-4 w-4" />
   403→              Create Proposal
   404→            </Button>
   405→          </div>
   406→        </div>
   407→
   408→        {/* Loading State */}
   409→        {showLoading ? (
   410→          <div className="space-y-6">
   411→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   412→              {[1, 2, 3, 4, 5, 6].map((i) => (
   413→                <Card key={i}>
   414→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-1.5 px-4 pt-4">
   415→                    <Skeleton className="h-4 w-16" />
   416→                    <Skeleton className="h-4 w-4" />
   417→                  </CardHeader>
   418→                  <CardContent className="px-4 pb-4">
   419→                    <Skeleton className="h-6 w-12 mb-1" />
   420→                  </CardContent>
   421→                </Card>
   422→              ))}
   423→            </div>
   424→            <Skeleton className="h-10 rounded-lg" />
   425→            <div className="space-y-3">
   426→              {[1, 2, 3].map((i) => (
   427→                <Skeleton key={i} className="h-[100px] rounded-xl" />
   428→              ))}
   429→            </div>
   430→          </div>
   431→        ) : (
   432→          <>
   433→        {/* Stats Bar */}
   434→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3 mb-6">
   435→          {[
   436→            {
   437→              title: "Total",
   438→              value: stats?.total ?? 0,
   439→              icon: FileText,
   440→              color: "text-[#8B5CF6]",
   441→              bgColor: "bg-[#8B5CF6]/10",
   442→              format: (v: number) => String(v),
   443→            },
   444→            {
   445→              title: "Sent",
   446→              value: stats?.sent ?? 0,
   447→              icon: Send,
   448→              color: "text-blue-600",
   449→              bgColor: "bg-blue-500/10",
   450→              format: (v: number) => String(v),
   451→            },
   452→            {
   453→              title: "Signed",
   454→              value: stats?.signed ?? 0,
   455→              icon: CheckCircle2,
   456→              color: "text-emerald-600",
   457→              bgColor: "bg-emerald-500/10",
   458→              format: (v: number) => String(v),
   459→            },
   460→            {
   461→              title: "Draft",
   462→              value: stats?.draft ?? 0,
   463→              icon: FilePenLine,
   464→              color: "text-slate-600 dark:text-slate-400",
   465→              bgColor: "bg-slate-500/10",
   466→              format: (v: number) => String(v),
   467→            },
   468→            {
   469→              title: "Signature Rate",
   470→              value: stats?.signatureRate ?? 0,
   471→              icon: TrendingUp,
   472→              color: "text-amber-600",
   473→              bgColor: "bg-amber-500/10",
   474→              format: (v: number) => `${v}%`,
   475→            },
   476→            {
   477→              title: "Total Value",
   478→              value: stats?.totalValue ?? 0,
   479→              icon: DollarSign,
   480→              color: "text-emerald-600",
   481→              bgColor: "bg-emerald-500/10",
   482→              format: (v: number) => formatCurrency(v),
   483→            },
   484→          ].map((stat, idx) => (
   485→            <motion.div
   486→              key={stat.title}
   487→              initial={{ opacity: 0, y: 20 }}
   488→              animate={{ opacity: 1, y: 0 }}
   489→              transition={{ duration: 0.4, delay: idx * 0.05 }}
   490→            >
   491→              <Card>
   492→                <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-1.5 px-4 pt-4">
   493→                  <CardTitle className="text-[12px] font-medium text-muted-foreground">
   494→                    {stat.title}
   495→                  </CardTitle>
   496→                  <div className={`h-7 w-7 rounded-md ${stat.bgColor} flex items-center justify-center`}>
   497→                    <stat.icon className={`h-3.5 w-3.5 ${stat.color}`} />

   499→                </CardHeader>
   500→                <CardContent className="px-4 pb-3">
   501→                  <div className="text-[20px] font-bold text-foreground">

   503→                  </div>
   504→                </CardContent>

   506→            </motion.div>
   507→          ))}
   508→        </div>
   509→
   510→        {/* Action Bar: Search */}
   511→        <div className="flex items-center gap-3 mb-4">
   512→          <div className="relative flex-1 max-w-sm">
   513→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   514→            <Input
   515→              placeholder="Search by title or client..."
   516→              value={searchQuery}
   517→              onChange={(e) => setSearchQuery(e.target.value)}
   518→              className="pl-9"
   519→            />
   520→          </div>
   521→        </div>
   522→
   523→        {/* Filter Tabs */}
   524→        <div className="flex gap-1 mb-6 border-b border-border overflow-x-auto">
   525→          {filterTabs.map((tab) => {
   526→            const isActive = activeFilter === tab.key;

   531→                className={`pb-2 text-sm rounded-t-md px-3 transition-colors relative whitespace-nowrap ${

   555→        </div>
   556→
   557→        {/* Proposal Cards Grid */}
   558→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   559→          <AnimatePresence mode="popLayout">
   560→            {filteredProposals.length === 0 ? (
   561→              <motion.div
   562→                initial={{ opacity: 0 }}
   563→                animate={{ opacity: 1 }}
   564→                exit={{ opacity: 0 }}
   565→                className="col-span-full"
   566→              >
   567→                <Card>
   568→                  <CardContent className="py-16">
   569→                    <div className="text-center text-muted-foreground">
   570→                      <FileText className="h-12 w-12 mx-auto mb-4 opacity-40" />
   571→                      <p className="text-[16px] font-medium mb-1">No proposals found</p>
   572→                      <p className="text-[14px] mb-4">
   573→                        {searchQuery
   574→                          ? "Try adjusting your search terms"
   575→                          : "Create your first proposal or seed sample data"}
   576→                      </p>
   577→                      {!searchQuery && (
   578→                        <div className="flex items-center justify-center gap-3">
   579→                          <Button
   580→                            variant="outline"
   581→                            className="gap-2"
   582→                            onClick={handleSeed}
   583→                            disabled={isSeeding}
   584→                          >
   585→                            <Sparkles className="h-4 w-4" />
   586→                            Seed Sample Data
   587→                          </Button>
   588→                          <Button
   589→                            className="bg-[#8B5CF6] hover:bg-[#7C3AED] text-white gap-2"
   590→                            onClick={() => navigate("/proposals/new")}
   591→                          >
   592→                            <Plus className="h-4 w-4" />
   593→                            Create Proposal
   594→                          </Button>
   595→                        </div>
   596→                      )}
   597→                    </div>
   598→                  </CardContent>
   599→                </Card>
   600→              </motion.div>
   601→            ) : (
   602→              filteredProposals.map((proposal, idx) => (
   603→                <ProposalCard
   604→                  key={proposal._id}
   605→                  proposal={proposal}
   606→                  idx={idx}
   607→                  isMock={false}
   608→                  formatCurrency={formatCurrency}
   609→                  formatDate={formatDate}
   610→                  onSend={handleSend}
   611→                  onDuplicate={handleDuplicate}
   612→                  onDelete={canDeleteRecords ? setDeleteDialogId : () => {}}
   613→                  onView={() => navigate(`/proposals/new?edit=${proposal._id}`)}
   614→                  onShare={canShareRecords ? (id: string) => {
   615→                    const p = proposals.find((p: any) => p._id === id);
   616→                    if (canShareRecords) {
   617→                      setSharingRecord({
   618→                        id,
   619→                        type: "proposal",
   620→                        sharing: (p as any)?.sharing || [],
   621→                      });
   622→                      setShareProposalId(id);
   623→                      setShowShareDialog(true);
   624→                    }
   625→                  } : () => {}}
   626→                  canDelete={canDeleteRecords}
   627→                  canShare={canShareRecords}
   628→                  onConvertToProject={handleConvertToProject}
   629→                />
   630→              ))
   631→            )}
   632→          </AnimatePresence>
   633→        </div>
   634→        </>
   635→        )}
   636→      </PageLayout>
   637→
   638→      {/* Delete Confirmation Dialog */}
   639→      <Dialog open={!!deleteDialogId} onOpenChange={() => setDeleteDialogId(null)}>
   640→        <DialogContent className="sm:max-w-md">
   641→          <DialogHeader>
   642→            <DialogTitle>Delete Proposal</DialogTitle>
   643→            <DialogDescription>
   644→              Are you sure you want to delete this proposal? This action cannot be undone. All associated follow-ups will also be removed.
   645→            </DialogDescription>
   646→          </DialogHeader>
   647→          <DialogFooter className="gap-2">
   648→            <Button variant="outline" onClick={() => setDeleteDialogId(null)}>
   649→              Cancel
   650→            </Button>
   651→            <Button
   652→              variant="destructive"
   653→              onClick={() => deleteDialogId && handleDelete(deleteDialogId)}
   654→            >
   655→              Delete
   656→            </Button>
   657→          </DialogFooter>
   658→        </DialogContent>
   659→      </Dialog>
   660→
   661→      {/* Share Dialog */}
   662→      <ShareDialog
   663→        open={showShareDialog}
   664→        onOpenChange={setShowShareDialog}
   665→        recordId={sharingRecord?.id || shareProposalId || ""}
   666→        recordType={sharingRecord?.type || "proposal"}
   667→        currentSharing={sharingRecord?.sharing || []}
   668→        onShare={async (args) => {
   669→          try {
   670→            if (shareRecordMutation) {
   671→              await shareRecordMutation({
   672→                recordId: sharingRecord?.id || shareProposalId,
   673→                recordType: sharingRecord?.type || "proposal",
   674→                ...args,
   675→              });
   676→            }
   677→            toast.success("Record shared successfully");
   678→          } catch (err: any) {
   679→            toast.error(err?.message || "Failed to share record");
   680→          }
   681→        }}
   682→        onUnshare={async (args) => {
   683→          try {
   684→            if (unshareRecordMutation) {
   685→              await unshareRecordMutation({
   686→                recordId: sharingRecord?.id || shareProposalId,
   687→                recordType: sharingRecord?.type || "proposal",
   688→                ...args,
   689→              });
   690→            }
   691→            toast.success("Access removed");
   692→          } catch (err: any) {
   693→            toast.error(err?.message || "Failed to remove access");
   694→          }
   695→        }}
   696→      />
   697→    </motion.div>
   698→  );
   699→}
