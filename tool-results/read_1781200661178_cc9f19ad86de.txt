   216→const TAB_FILTERS = [
   217→  { key: "all", label: "All" },
   218→  { key: "active", label: "Active" },

   223→type TabFilter = (typeof TAB_FILTERS)[number]["key"];
   224→
   225→function filterReports(reports: DisputeReport[], tab: TabFilter): DisputeReport[] {
   226→  switch (tab) {
   227→    case "all":
   228→      return reports;
   229→    case "active":
   230→      return reports.filter((r) => r.status === "generated" || r.status === "sent" || r.status === "viewed");
   231→    case "resolved":
   232→      return reports.filter((r) => r.status === "resolved");
   233→    case "appealed":
   234→      return reports.filter((r) => r.status === "appealed");
   235→    default:
   236→      return reports;
   237→  }
   238→}
   239→
   240→function formatDate(ts: number): string {
   241→  return new Date(ts).toLocaleDateString("en-US", {
   242→    month: "short",
   243→    day: "numeric",
   244→    year: "numeric",
   245→  });
   246→}
   247→
   248→function formatCurrency(amount: number): string {
   249→  return new Intl.NumberFormat("en-US", {
   250→    style: "currency",
   251→    currency: "USD",
   252→    minimumFractionDigits: 0,
   253→    maximumFractionDigits: 2,
   254→  }).format(amount);
   255→}
