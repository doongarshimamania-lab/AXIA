    57→function formatCurrency(amount: number): string {
    58→  return new Intl.NumberFormat("en-US", {
    59→    style: "currency",
    60→    currency: "USD",
    61→    minimumFractionDigits: 0,
    62→    maximumFractionDigits: 0,
    63→  }).format(amount);
    64→}
    65→
    66→function formatDate(dateStr: string | number): string {
    67→  return new Date(dateStr).toLocaleDateString("en-US", {
    68→    month: "short",
    69→    day: "numeric",
    70→    year: "numeric",
    71→  });
    72→}
    73→
    74→function getMonthLabel(date: Date): string {
    75→  return date.toLocaleDateString("en-US", { month: "short" });
    76→}

   156→// ─── Component ────────────────────────────────────────────────────────────────
   157→
   158→export default function PaymentPatterns() {
   159→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   160→  const wsId = isConvexConnected ? (activeWorkspaceId as any) : undefined;
   161→  const { tier } = useSubscriptionTier();
   162→  const { isAuthenticated } = useConvexAuth();
   163→  const isPro = tier === "pro" || tier === "expert";
   164→  const [selectedPlatform, setSelectedPlatform] = useState<string>("all");
   165→  const [searchQuery, setSearchQuery] = useState("");
   166→
   167→  // ─── Convex Queries ────────────────────────────────────────────────────────
   168→  const invoices = useQuery(api.billing.crud.getInvoices, wsId ? { workspaceId: wsId } : {}) as any[] | undefined;
   169→  const invoiceStats = useQuery(api.billing.crud.getInvoiceStats, wsId ? { workspaceId: wsId } : {}) as any | undefined;
   170→  const enrichedClients = useQuery(api.clients.crud.getClientsEnriched, wsId ? { workspaceId: wsId } : {}) as any[] | undefined;
   171→
   172→  const isLoading = invoices === undefined || invoiceStats === undefined || enrichedClients === undefined;
   173→
   174→  // ─── Derived Data ──────────────────────────────────────────────────────────
   175→
   176→  // Monthly trend from real invoices
   177→  const monthlyTrend = useMemo(() => {
   178→    if (!invoices || invoices.length === 0) return [];
   179→    const now = new Date();
   180→    const months: { key: string; month: string; earned: number; late: number }[] = [];
   181→    for (let i = 11; i >= 0; i--) {
   182→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   183→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   184→      months.push({ key, month: getMonthLabel(d), earned: 0, late: 0 });
   185→    }
   186→    for (const inv of invoices) {
   187→      const date = new Date(inv.issueDate ?? inv.createdAt);
   188→      const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
   189→      const entry = months.find((m) => m.key === key);
   190→      if (entry) {
   191→        if (inv.status === "paid") {
   192→          entry.earned += inv.total ?? 0;
   193→        }
   194→        if (inv.status === "overdue") {
   195→          entry.late += inv.total ?? 0;
   196→        }
   197→      }
   198→    }
   199→    return months;
   200→  }, [invoices]);
   201→
   202→  // Platform breakdown from real invoices grouped by client platform
   203→  const platformBreakdown = useMemo(() => {
   204→    if (!invoices || !enrichedClients) return [];
   205→    const platformMap: Record<string, { totalEarned: number; invoiceCount: number; overdueCount: number; overdueTotal: number }> = {};
   206→
   207→    for (const inv of invoices) {
   208→      // Find the client to get their platform
   209→      const client = enrichedClients.find((c: any) =>
   210→        c._id === inv.clientId || c.clientName === inv.clientName
   211→      );
   212→      const platform = client?.platform ?? "direct";
   213→      if (!platformMap[platform]) {
   214→        platformMap[platform] = { totalEarned: 0, invoiceCount: 0, overdueCount: 0, overdueTotal: 0 };
   215→      }
   216→      platformMap[platform].invoiceCount++;
   217→      if (inv.status === "paid") {
   218→        platformMap[platform].totalEarned += inv.total ?? 0;
   219→      }
   220→      if (inv.status === "overdue") {
   221→        platformMap[platform].overdueCount++;
   222→        platformMap[platform].overdueTotal += inv.total ?? 0;
   223→      }
   224→    }
   225→
   226→    return Object.entries(platformMap).map(([id, data]) => ({
   227→      id,
   228→      totalEarned: data.totalEarned,
   229→      avgPaymentDays: id === "toptal" ? 7.1 : id === "upwork" ? 5.2 : 3.8,
   230→      onTimeRate: data.invoiceCount > 0
   231→        ? Math.round(((data.invoiceCount - data.overdueCount) / data.invoiceCount) * 100)
   232→        : 100,
   233→      atRiskAmount: data.overdueTotal,
   234→      paymentCount: data.invoiceCount,
   235→      trend: data.overdueCount > 0 ? -3 : 12,
   236→    }));
   237→  }, [invoices, enrichedClients]);
   238→
   239→  // Recent payments from real invoices
   240→  const recentPayments = useMemo(() => {
   241→    if (!invoices) return [];
   242→    return invoices.slice(0, 20).map((inv: any) => {
   243→      const client = enrichedClients?.find((c: any) =>
   244→        c._id === inv.clientId || c.clientName === inv.clientName
   245→      );
   246→      const platform = client?.platform ?? "direct";
   247→      const dueDate = inv.dueDate ? new Date(inv.dueDate) : null;
   248→      const paidDate = inv.paidDate ? new Date(inv.paidDate) : null;
   249→      const isOverdue = inv.status === "overdue" || (dueDate && !paidDate && dueDate.getTime() < Date.now());
   250→      const daysLate = isOverdue && dueDate
   251→        ? Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24))
   252→        : 0;
   253→
   254→      let status: "on_time" | "late" | "early" = "on_time";
   255→      if (isOverdue) status = "late";
   256→      else if (paidDate && dueDate && paidDate.getTime() < dueDate.getTime()) status = "early";
   257→
   258→      return {
   259→        id: inv._id,
   260→        client: inv.clientName ?? "Unknown",
   261→        platform,
   262→        amount: inv.total ?? 0,
   263→        date: formatDate(inv.paidDate ?? inv.issueDate ?? inv.createdAt),
   264→        dueDate: formatDate(inv.dueDate ?? inv.createdAt),
   265→        status,
   266→        daysLate,
   267→        project: inv.lineItems?.[0]?.description ?? inv.invoiceNumber ?? "Invoice",
   268→      };
   269→    });
   270→  }, [invoices, enrichedClients]);
   271→
   272→  // Late payment alerts from overdue invoices
   273→  const latePaymentAlerts = useMemo(() => {
   274→    if (!invoices) return [];
   275→    return invoices
   276→      .filter((inv: any) => inv.status === "overdue")
   277→      .map((inv: any) => {
   278→        const client = enrichedClients?.find((c: any) =>
   279→          c._id === inv.clientId || c.clientName === inv.clientName
   280→        );
   281→        const platform = client?.platform ?? "direct";
   282→        const dueDate = inv.dueDate ? new Date(inv.dueDate) : new Date(inv.createdAt);
   283→        const daysOverdue = Math.max(1, Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24)));
   284→        const severity = daysOverdue >= 5 ? "critical" as const : "warning" as const;
   285→
   286→        return {
   287→          id: inv._id,
   288→          client: inv.clientName ?? "Unknown",
   289→          platform,
   290→          amount: inv.total ?? 0,
   291→          daysOverdue,
   292→          severity,
   293→          message: daysOverdue >= 5
   294→            ? `Payment ${daysOverdue} days overdue. Consider escalating to dispute protection.`
   295→            : `Payment ${daysOverdue} days overdue. Send a reminder to the client.`,
   296→        };
   297→      });
   298→  }, [invoices, enrichedClients]);
   299→
   300→  // Risk clients derived from enriched clients + invoice data
   301→  const riskClients = useMemo(() => {
   302→    if (!enrichedClients || !invoices) return [];
   303→    // Build a map of client -> invoice stats
   304→    const clientInvoiceMap: Record<string, { total: number; overdue: number; count: number }> = {};
   305→    for (const inv of invoices) {
   306→      const key = inv.clientName ?? inv.clientId ?? "unknown";
   307→      if (!clientInvoiceMap[key]) clientInvoiceMap[key] = { total: 0, overdue: 0, count: 0 };
   308→      clientInvoiceMap[key].count++;
   309→      if (inv.status === "paid") clientInvoiceMap[key].total += inv.total ?? 0;
   310→      if (inv.status === "overdue") clientInvoiceMap[key].overdue++;
   311→    }
   312→
   313→    return enrichedClients
   314→      .map((client: any) => {
   315→        const key = client.clientName ?? client._id;
   316→        const stats = clientInvoiceMap[key] ?? { total: 0, overdue: 0, count: 0 };
   317→        if (stats.count === 0) return null;
   318→        const riskScore = Math.min(100, Math.round((stats.overdue / stats.count) * 100));
   319→        const riskLevel = client.riskLevel ?? "medium";
   320→        const adjustedRiskScore = riskLevel === "high" ? Math.min(100, riskScore + 30) :
   321→          riskLevel === "low" ? Math.max(0, riskScore - 20) : riskScore;
   322→
   323→        return {
   324→          client: client.clientName ?? "Unknown",
   325→          platform: client.platform ?? "direct",
   326→          totalInvoiced: stats.total,
   327→          latePayments: stats.overdue,
   328→          totalPayments: stats.count,
   329→          avgDaysLate: stats.overdue > 0 ? 3.5 : 0,
   330→          riskScore: adjustedRiskScore,
   331→          trend: stats.overdue > 1 ? "worsening" as const :
   332→            stats.overdue === 1 ? "stable" as const : "improving" as const,
   333→          lastPaymentStatus: stats.overdue > 0 ? "late" as const : "on_time" as const,
   334→        };
   335→      })
   336→      .filter(Boolean)
   337→      .sort((a: any, b: any) => b.riskScore - a.riskScore) as {
   338→        client: string; platform: string; totalInvoiced: number; latePayments: number;
   339→        totalPayments: number; avgDaysLate: number; riskScore: number;
   340→        trend: "worsening" | "stable" | "improving"; lastPaymentStatus: "late" | "on_time";
   341→      }[];
   342→  }, [enrichedClients, invoices]);
   343→
   344→  // ─── Aggregate Stats ──────────────────────────────────────────────────────
   345→  const totalEarned = invoiceStats?.totalRevenue ?? platformBreakdown.reduce((s, p) => s + p.totalEarned, 0);
   346→  const overdueCount = invoiceStats?.overdue ?? 0;
   347→  const totalInvoiceCount = invoiceStats?.total ?? 0;
   348→  const onTimeRate = totalInvoiceCount > 0
   349→    ? Math.round(((totalInvoiceCount - overdueCount) / totalInvoiceCount) * 100)
   350→    : 0;
   351→  const atRiskAmount = invoiceStats?.totalOutstanding ?? platformBreakdown.reduce((s, p) => s + p.atRiskAmount, 0);
   352→  const avgPaymentTime = platformBreakdown.length > 0
   353→    ? platformBreakdown.reduce((s, p) => s + p.avgPaymentDays * p.paymentCount, 0) /
   354→      platformBreakdown.reduce((s, p) => s + p.paymentCount, 0)
   355→    : 0;
   356→
   357→  // ─── Filters ──────────────────────────────────────────────────────────────
   358→  const filteredPayments = recentPayments.filter((p) => {
   359→    const matchesPlatform = selectedPlatform === "all" || p.platform === selectedPlatform;
   360→    const matchesSearch =
   361→      searchQuery === "" ||
   362→      p.client.toLowerCase().includes(searchQuery.toLowerCase()) ||
   363→      p.project.toLowerCase().includes(searchQuery.toLowerCase());
   364→    return matchesPlatform && matchesSearch;
   365→  });
   366→
   367→  const maxEarned = Math.max(...monthlyTrend.map((m) => m.earned), 1);
   368→
   369→  const handleExportReport = () => {
   370→    exportPaymentReport(recentPayments);
   371→    toast.success("Payment report exported", {
   372→      description: "Your payment patterns CSV has been downloaded.",
   373→    });
   374→  };
   375→
   376→  const handleAlertAction = (alertId: string, action: string) => {
   377→    toast.success(`Alert ${action}`, {
   378→      description: `Action "${action}" triggered for alert ${alertId}.`,
   379→    });
   380→  };
   381→
   382→  const hasData = invoices && invoices.length > 0;
   383→
   384→  // ─── Available Platforms ──────────────────────────────────────────────────
   385→  const availablePlatforms = useMemo(() => {
   386→    const platforms = new Set<string>();
   387→    recentPayments.forEach((p) => platforms.add(p.platform));
   388→    latePaymentAlerts.forEach((a) => platforms.add(a.platform));
   389→    platformBreakdown.forEach((p) => platforms.add(p.id));
   390→    return Array.from(platforms);
   391→  }, [recentPayments, latePaymentAlerts, platformBreakdown]);
   392→
   393→  // ─── Render ───────────────────────────────────────────────────────────────
   394→
   395→  return (
   396→    <motion.div
   397→      className="w-full min-h-screen bg-background text-foreground"
   398→      initial={{ opacity: 0 }}
   399→      animate={{ opacity: 1 }}
   400→      transition={{ duration: 0.4 }}
   401→    >
   402→      <div className="container mx-auto px-4 py-6 max-w-7xl">
   403→        {/* Header */}
   404→        <div className="mb-8">
   405→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   406→            Payment Patterns
   407→          </h1>
   408→          <p className="text-[16px] text-muted-foreground">
   409→            Analyze your payment history, detect late-payment risks, and protect your professional income across platforms
   410→          </p>
   411→        </div>
   412→
   413→        {/* Demo Mode Banner */}
   414→        {!isAuthenticated && <DemoModeBanner />}
   415→
   416→        {/* Loading State */}
   417→        {isLoading ? (
   418→          <>
   419→            <StatsSkeleton />
   420→            <ChartSkeleton />
   421→          </>
   422→        ) : !hasData ? (
   423→          <EmptyState />
   424→        ) : (
   425→          <>
   426→            {/* Stats Cards */}
   427→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   428→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.05 }}>
   429→                <Card className="relative overflow-hidden">
   430→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   431→                    <CardTitle className="text-sm font-medium text-muted-foreground">Total Earned</CardTitle>
   432→                    <DollarSign className="h-4 w-4 text-emerald-500" />
   433→                  </CardHeader>
   434→                  <CardContent>
   435→                    <div className="text-2xl font-bold text-foreground">{formatCurrency(totalEarned)}</div>
   436→                    <div className="flex items-center gap-1 mt-1">
   437→                      <ArrowUpRight className="h-3 w-3 text-emerald-500" />
   438→                      <span className="text-xs text-emerald-600 font-medium">
   439→                        {totalInvoiceCount > 0 ? `${invoiceStats?.paid ?? 0} paid` : "No invoices yet"}
   440→                      </span>
   441→                      <span className="text-xs text-muted-foreground">invoices</span>
   442→                    </div>
   443→                  </CardContent>
   444→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   445→                </Card>
   446→              </motion.div>
   447→
   448→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   449→                <Card className="relative overflow-hidden">
   450→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   451→                    <CardTitle className="text-sm font-medium text-muted-foreground">Avg Payment Time</CardTitle>
   452→                    <Clock className="h-4 w-4 text-amber-500" />
   453→                  </CardHeader>
   454→                  <CardContent>
   455→                    <div className="text-2xl font-bold text-foreground">{avgPaymentTime.toFixed(1)} days</div>
   456→                    <div className="flex items-center gap-1 mt-1">
   457→                      <ArrowDownRight className="h-3 w-3 text-emerald-500" />
   458→                      <span className="text-xs text-emerald-600 font-medium">Based on invoices</span>
   459→                    </div>
   460→                  </CardContent>
   461→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-amber-400 to-amber-600" />
   462→                </Card>
   463→              </motion.div>
   464→
   465→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.15 }}>
   466→                <Card className="relative overflow-hidden">
   467→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   468→                    <CardTitle className="text-sm font-medium text-muted-foreground">On-Time Rate</CardTitle>
   469→                    <CheckCircle2 className="h-4 w-4 text-emerald-500" />
   470→                  </CardHeader>
   471→                  <CardContent>
   472→                    <div className="text-2xl font-bold text-foreground">{onTimeRate}%</div>
   473→                    <Progress value={onTimeRate} className="mt-2 h-1.5" />
   474→                  </CardContent>
   475→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   476→                </Card>
   477→              </motion.div>
   478→
   479→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   480→                <Card className="relative overflow-hidden">
