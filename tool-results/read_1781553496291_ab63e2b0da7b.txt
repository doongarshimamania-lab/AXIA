     6→// Get all clients for the current user
     7→export const getMyClients = query({
     8→  args: {},
     9→  handler: async (ctx) => {
    10→    let userId;
    11→    try {
    12→      userId = await getAuthUserId(ctx);
    13→    } catch (error) {
    14→      // User not authenticated
    15→      return [];
    16→    }
    17→    if (!userId) return [];
    18→
    19→    const clients = await ctx.db
    20→      .query("clients")
    21→      .withIndex("by_user", (q) => q.eq("userId", userId))
    22→      .collect();
    23→
    24→    // Enrich with protection metrics
    25→    const enrichedClients = await Promise.all(
    26→      clients.map(async (client) => {
    27→        // Get sessions for this client
    28→        const sessions = await ctx.db
    29→          .query("workSessions")
    30→          .withIndex("by_user", (q) => q.eq("userId", userId))
    31→          .filter((q) => q.eq(q.field("clientName"), client.clientName))
    32→          .collect();
    33→
    34→        const totalHours = sessions.reduce((sum, s) => sum + (s.totalMinutes || 0) / 60, 0);
    35→        const activeSession = sessions.find((s) => !s.endTime);
    36→
    37→        // Get time blocks for compliance
    38→        const timeBlocks = await ctx.db
    39→          .query("timeBlocks")
    40→          .withIndex("by_user", (q) => q.eq("userId", userId))
    41→          .collect();
    42→
    43→        const clientBlocks = timeBlocks.filter((block) => {
    44→          const session = sessions.find((s) => s._id === block.sessionId);
    45→          return session !== undefined;
    46→        });
    47→
    48→        const compliantBlocks = clientBlocks.filter((b) => b.complianceStatus === "compliant").length;
    49→        const totalBlocks = clientBlocks.length;
    50→        const protectionScore = totalBlocks > 0 ? Math.round((compliantBlocks / totalBlocks) * 100) : 100;
    51→
    52→        return {
    53→          ...client,
    54→          totalHours: Math.round(totalHours * 10) / 10,
    55→          protectionScore,
    56→          activeSession: activeSession ? true : false,
    57→          totalValue: Math.round(totalHours * client.hourlyRate * 100) / 100,
    58→        };
    59→      })
    60→    );
    61→
    62→    return enrichedClients;
    63→  },
    64→});

    66→// Add a new client
    67→export const addClient = mutation({
    68→  args: {
    69→    clientName: v.string(),
    70→    platform: v.union(
    71→      v.literal("upwork"),
    72→      v.literal("fiverr"),
    73→      v.literal("toptal"),
    74→      v.literal("freelancer"),
    75→      v.literal("direct")
    76→    ),
    77→    hourlyRate: v.number(),
    78→    contractType: v.union(v.literal("hourly"), v.literal("fixed")),
    79→    riskLevel: v.union(v.literal("low"), v.literal("medium"), v.literal("high")),
    80→  },
    81→  handler: async (ctx, args) => {
    82→    let userId;
    83→    try {
    84→      userId = await getAuthUserId(ctx);
    85→    } catch (error) {
    86→      return { success: false, error: "Not authenticated" };
    87→    }
    88→    if (!userId) return { success: false, error: "Not authenticated" };
    89→
    90→    // Check if client already exists
    91→    const existing = await ctx.db
    92→      .query("clients")
    93→      .withIndex("by_user_and_name", (q) =>
    94→        q.eq("userId", userId).eq("clientName", args.clientName)
    95→      )
    96→      .first();
    97→
    98→    if (existing) {
    99→      throw new Error("Client already exists");
   100→    }
   101→
   102→    const clientId = await ctx.db.insert("clients", {
   103→      userId,
   104→      clientName: args.clientName,
   105→      platform: args.platform,
   106→      hourlyRate: args.hourlyRate,
   107→      contractType: args.contractType,
   108→      riskLevel: args.riskLevel,
   109→      addedAt: Date.now(),
   110→      lastActivityAt: Date.now(),
   111→    });
   112→
   113→    return { clientId, success: true };
   114→  },
   115→});

   376→        currentRate: 0,
   377→        simulatedRate: 0,
   378→        improvement: 0,

   387→        currentRate: 0,

   392→      };
   393→    }

   396→

   398→    let simClientFilter: string | null = null;
   399→    if (args.clientId) {
   400→      const client = await ctx.db.get(args.clientId);
   401→      if (client) simClientFilter = client.clientName;
   402→    }
   403→
   404→    const simAllSessions = await ctx.db
   405→      .query("workSessions")
   406→      .withIndex("by_user", (q) => q.eq("userId", userId))
   407→      .collect();
   408→
   409→    const simSessions = simClientFilter
   410→      ? simAllSessions.filter((s) => s.clientName === simClientFilter)
   411→      : simAllSessions;
   412→
   413→    // Calculate base rate from evidence quality
   414→    let baseRate = 75;
   415→    if (simSessions.length > 0) {
   416→      const sessionIds = simSessions.map((s) => s._id);
   417→      let totalEvidence = 0;
   418→      let workContextEvidence = 0;
   419→
   420→      for (const sessionId of sessionIds) {
   421→        const evidenceSession = await ctx.db
   422→          .query("evidenceSessions")
   423→          .withIndex("by_session", (q) => q.eq("sessionId", sessionId))
   424→          .first();

   426→        if (evidenceSession) {

   429→            .withIndex("by_session_and_time", (q) => q.eq("evidenceSessionId", evidenceSession._id))

   431→
   432→          totalEvidence += events.length;
   433→          workContextEvidence += events.filter((e) => e.kind === "url" || e.kind === "memo").length;
   434→        }
   435→      }
   436→
   437→      const clientContextScore = totalEvidence > 0 ? Math.min(100, Math.round((workContextEvidence / totalEvidence) * 150)) : 0;
   438→      const now = Date.now();
   439→      const thirtyDaysAgo = now - 30 * 24 * 60 * 60 * 1000;
   440→      const recentSessions = simSessions.filter((s) => s.startTime >= thirtyDaysAgo);
   441→      const evidenceConsistency = recentSessions.length > 0 ? Math.min(100, recentSessions.length * 10) : 0;
   442→      
   443→      baseRate = Math.round(clientContextScore * 0.4 + evidenceConsistency * 0.3 + 80 * 0.3);
   444→    }
   445→
   446→    // Calculate improvement from adding memos
   447→    const memoImprovement = (args.addMemos || 0) * 0.12;

   449→
   450→    // Calculate protected value this month

   452→    const thirtyDaysAgo = now - 30 * 24 * 60 * 60 * 1000;

   470→    const totalHours = filteredSessions.reduce((sum, s) => sum + (s.totalMinutes || 0) / 60, 0);

   472→

   475→      simulatedRate: Math.round(simulatedRate),
   476→      improvement: Math.round(simulatedRate - baseRate),

   479→    };

   483→// Predict client-specific evidence gaps
   484→export const getClientGapPrediction = query({
   485→  args: {
   486→    clientId: v.optional(v.id("clients")),
   487→  },
   488→  handler: async (ctx, args) => {
   489→    let userId;
   490→    try {
   491→      userId = await getAuthUserId(ctx);
   492→    } catch (error) {
   493→      return {
   494→        gapProbability: 0,
   495→        predictedTime: "N/A",
   496→        confidence: "Low" as const,
   497→        reason: "Please log in to view gap predictions",
   498→        hasGapPrediction: false,
   499→      };
   500→    }
   501→    if (!userId) {
   502→      return {
   503→        gapProbability: 0,
   504→        predictedTime: "N/A",
   505→        confidence: "Low" as const,
   506→        reason: "Please log in to view gap predictions",
   507→        hasGapPrediction: false,
   508→      };
   509→    }
   510→
   511→    // Get historical gap patterns
   512→    const allSessions = await ctx.db
   513→      .query("workSessions")
   514→      .withIndex("by_user", (q) => q.eq("userId", userId))
   515→      .collect();
   516→
   517→    let clientFilter: string | null = null;
   518→    if (args.clientId) {
   519→      const client = await ctx.db.get(args.clientId);
   520→      if (client) clientFilter = client.clientName;
   521→    }
   522→
   523→    const sessions = clientFilter
   524→      ? allSessions.filter((s) => s.clientName === clientFilter)
   525→      : allSessions;
   526→
   527→    // Analyze time patterns for gaps
   528→    const hourlyGaps: Record<number, number> = {};
   529→    for (let i = 0; i < 24; i++) {
   530→      hourlyGaps[i] = 0;
   531→    }
   532→
   533→    for (const session of sessions) {
   534→      const evidenceSession = await ctx.db
   535→        .query("evidenceSessions")
   536→        .withIndex("by_session", (q) => q.eq("sessionId", session._id))
   537→        .first();
   538→
   539→      if (evidenceSession) {
   540→        const events = await ctx.db
   541→          .query("evidenceEvents")
   542→          .withIndex("by_session_and_time", (q) => q.eq("evidenceSessionId", evidenceSession._id))
   543→          .collect();
   544→
   545→        const eventsByHour: Record<number, number> = {};
   546→        events.forEach((e) => {
   547→          const hour = new Date(e.t).getHours();
   548→          eventsByHour[hour] = (eventsByHour[hour] || 0) + 1;
   549→        });
   550→
   551→        Object.keys(eventsByHour).forEach((hourStr) => {
   552→          const hour = parseInt(hourStr);
   553→          if (eventsByHour[hour] < 3) {
   554→            hourlyGaps[hour]++;
   555→          }
   556→        });
   557→      }
   558→    }
   559→
   560→    // Find hour with most gaps
   561→    let maxGapHour = 15;
   562→    let maxGaps = 0;
   563→    Object.entries(hourlyGaps).forEach(([hour, gaps]) => {
   564→      if (gaps > maxGaps) {
   565→        maxGaps = gaps;
   566→        maxGapHour = parseInt(hour);
   567→      }
   568→    });
   569→
   570→    // Calculate gap probability
   571→    const clientHistoricalGaps = sessions.length > 0 ? (maxGaps / sessions.length) * 100 : 0;
   572→    const currentInactivity = 60;
   573→    const gapProbability = clientHistoricalGaps * 0.7 + currentInactivity * 0.3;
   574→
   575→    const predictedTime = `${maxGapHour % 12 || 12}:${Math.floor(Math.random() * 60).toString().padStart(2, "0")} ${maxGapHour >= 12 ? "PM" : "AM"}`;
   576→
   577→    return {
   578→      gapProbability: Math.round(gapProbability),
   579→      predictedTime,
   580→      confidence: gapProbability > 70 ? "High" : gapProbability > 50 ? "Medium" : "Low",
   581→      reason: "Based on your pattern, you typically have evidence gaps during client meetings",
   582→      hasGapPrediction: gapProbability > 30, // Lowered threshold from 70 to 30 to show predictions more often
   583→    };
   584→  },
   585→});
