   313→   313→function WeekView({ data, currentDate, onDayClick }: { data: TimeBlock[]; currentDate: Date; onDayClick: (d: Date) => void }) {
   314→   314→  const weekStart = new Date(currentDate);
   315→   315→  weekStart.setDate(currentDate.getDate() - currentDate.getDay());
   316→   316→  const days = Array.from({ length: 7 }, (_, i) => {
   317→   317→    const d = new Date(weekStart);
   318→   318→    d.setDate(weekStart.getDate() + i);
   319→   319→    return d;
   320→   320→  });
   321→   321→
   322→   322→  const getDaySummary = (date: Date) => {
   323→   323→    const dayBlocks = data.filter((b) => new Date(b.startTime).toDateString() === date.toDateString());
   324→   324→    const compliant = dayBlocks.filter((b) => b.complianceStatus === "compliant").length;
   325→   325→    const atRisk = dayBlocks.filter((b) => b.complianceStatus === "at_risk").length;
   326→   326→    const rejected = dayBlocks.filter((b) => b.complianceStatus === "rejected").length;
   327→   327→    const totalHours = dayBlocks.length;
   328→   328→    const earnings = totalHours * 25;
   329→   329→    return { compliant, atRisk, rejected, totalHours, earnings };
   330→   330→  };
   331→   331→
   332→   332→  const today = new Date();
   333→   333→
   334→   334→  return (
   335→   335→    <div className="h-full p-3 flex flex-col">
   336→   336→      {/* Day headers */}
   337→   337→      <div className="grid grid-cols-7 gap-1 mb-1.5 flex-shrink-0">
   338→   338→        {days.map((day, i) => {
   339→   339→          const isToday = day.toDateString() === today.toDateString();
   340→   340→          return (
   341→   341→            <div key={i} className="text-center py-1">
   342→   342→              <div className="text-[10px] text-muted-foreground font-medium">{DAY_NAMES[day.getDay()]}</div>
   343→   343→              <div className={`text-sm font-semibold mt-0.5 w-6 h-6 rounded-full flex items-center justify-center mx-auto ${isToday ? "bg-primary text-primary-foreground" : "text-foreground"}`}>
   344→   344→                {day.getDate()}
   345→   345→              </div>
   346→   346→            </div>
   347→   347→          );
   348→   348→        })}
   349→   349→      </div>
   350→   350→
   351→   351→      {/* Day cells */}
   352→   352→      <div className="grid grid-cols-7 gap-1 flex-1 min-h-0">
   353→   353→        {days.map((day, i) => {
   354→   354→          const summary = getDaySummary(day);
   355→   355→          const hasData = summary.totalHours > 0;
   356→   356→          const isToday = day.toDateString() === today.toDateString();
   357→   357→
   358→   358→          return (
   359→   359→            <button
   360→   360→              key={i}
   361→   361→              onClick={() => onDayClick(day)}
   362→   362→              className={`rounded-lg border transition-all hover:shadow-md flex flex-col items-center justify-center p-1.5 min-h-0 ${
   363→   363→                isToday ? "border-primary/50 bg-primary/5" : "border-border bg-card hover:bg-muted/50"
   364→   364→              }`}
   365→   365→            >
   366→   366→              {hasData ? (
   367→   367→                <>
   368→   368→                  <div className={`text-sm font-bold ${summary.rejected > 0 ? "text-red-600" : summary.atRisk > 0 ? "text-amber-600" : "text-emerald-600"}`}>
   369→   369→                    +${summary.earnings}
   370→   370→                  </div>
   371→   371→                  <div className="text-[9px] text-muted-foreground mt-0.5">{summary.totalHours}h</div>
   372→   372→                  <div className="flex gap-0.5 mt-1">
   373→   373→                    {summary.compliant > 0 && <div className="w-1.5 h-1.5 rounded-full bg-blue-500" />}
   374→   374→                    {summary.atRisk > 0 && <div className="w-1.5 h-1.5 rounded-full bg-amber-500" />}
   375→   375→                    {summary.rejected > 0 && <div className="w-1.5 h-1.5 rounded-full bg-red-500" />}
   376→   376→                  </div>
   377→   377→                </>
   378→   378→              ) : (
   379→   379→                <div className="text-xs text-muted-foreground">$0</div>
   380→   380→              )}
   381→   381→            </button>
   382→   382→          );
   383→   383→        })}
   384→   384→      </div>
   385→   385→    </div>
   386→   386→  );
   387→   387→}

   391→   391→function MonthView({ data, currentDate, onDayClick }: { data: TimeBlock[]; currentDate: Date; onDayClick: (d: Date) => void }) {
   392→   392→  const year = currentDate.getFullYear();
   393→   393→  const month = currentDate.getMonth();
   394→   394→  const firstDay = new Date(year, month, 1).getDay();
   395→   395→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   396→   396→  const today = new Date();
   397→   397→
   398→   398→  const cells: (Date | null)[] = [];
   399→   399→  for (let i = 0; i < firstDay; i++) cells.push(null);
   400→   400→  for (let d = 1; d <= daysInMonth; d++) cells.push(new Date(year, month, d));
   401→   401→  while (cells.length % 7 !== 0) cells.push(null);
   402→   402→
   403→   403→  const getDaySummary = (date: Date | null) => {
   404→   404→    if (!date) return { totalHours: 0, earnings: 0, rejected: 0, atRisk: 0 };
   405→   405→    const dayBlocks = data.filter((b) => new Date(b.startTime).toDateString() === date.toDateString());

   457→   457→function YearView({ data, currentDate, onMonthClick }: { data: TimeBlock[]; currentDate: Date; onMonthClick: (d: Date) => void }) {

   614→   614→function formatHourShort(hour: number): string {
   615→   615→  if (hour === 12) return "12 PM";
   616→   616→  if (hour > 12) return `${hour - 12} PM`;
   617→   617→  return `${hour} AM`;
   618→   618→}
   619→   619→
   620→   620→function formatHour(h: number): string {
   621→   621→  const hour = Math.floor(h);
   622→   622→  const min = Math.round((h - hour) * 60);
   623→   623→  const suffix = hour >= 12 ? "PM" : "AM";
   624→   624→  const display = hour > 12 ? hour - 12 : hour === 0 ? 12 : hour;
   625→   625→  return min > 0 ? `${display}:${min.toString().padStart(2, "0")} ${suffix}` : `${display} ${suffix}`;
   626→   626→}

   628→   628→function formatPeriodLabel(date: Date, view: TimeView): string {
   629→   629→  switch (view) {
   630→   630→    case "day":
   631→   631→      return date.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" });
   632→   632→    case "week": {
   633→   633→      const ws = new Date(date);
   634→   634→      ws.setDate(date.getDate() - date.getDay());
   635→   635→      const we = new Date(ws);
   636→   636→      we.setDate(ws.getDate() + 6);
   637→   637→      return `${ws.toLocaleDateString("en-US", { month: "short", day: "numeric" })} – ${we.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })}`;
   638→   638→    }
   639→   639→    case "month":
   640→   640→      return date.toLocaleDateString("en-US", { month: "long", year: "numeric" });
   641→   641→    case "year":
   642→   642→      return String(date.getFullYear());
   643→   643→  }
   644→   644→}
