   117→export default function TeamManagement() {
   118→  const { activeWorkspaceId, isTeamMode, isOwner, canManageTeam, activeWorkspace } = useWorkspaceContext();
   119→
   120→  // ─── Convex Queries ────────────────────────────────────────────────────────
   121→  const hasRealWorkspaceId = activeWorkspaceId && !activeWorkspaceId.startsWith("ws_");
   122→
   123→  const convexMembers = useQuery(
   124→    hasRealWorkspaceId ? api.workspaces.members.getMembers : "skip",
   125→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   126→  ) as any[] | undefined;
   127→
   128→  const convexStats = useQuery(
   129→    hasRealWorkspaceId ? api.workspaces.crud.getWorkspaceStats : "skip",
   130→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   131→  ) as any | undefined;
   132→
   133→  const convexInvitations = useQuery(
   134→    hasRealWorkspaceId ? api.workspaces.invitations.getInvitations : "skip",
   135→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any, status: "pending" as const } : "skip"
   136→  ) as any[] | undefined;
   137→
   138→  // ─── Teams queries ─────────────────────────────────────────────────────────
   139→  const teamsApi = (api as any).teams?.crud;
   140→  const hasTeamsApi = !!teamsApi?.getTeams;
   141→
   142→  const convexTeams = useQuery(
   143→    hasTeamsApi && hasRealWorkspaceId ? teamsApi.getTeams : "skip",
   144→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   145→  ) as any[] | undefined;
   146→
   147→  // ─── Team Members per team ──────────────────────────────────────────────────
   148→  const hasTeamMembersApi = !!teamsApi?.getTeamMembers;
   149→
   150→  // ─── Mutations ─────────────────────────────────────────────────────────────
   151→  const inviteMemberMutation = useMutation(api.workspaces.invitations.createInvitation);
   152→  const removeMemberMutation = useMutation(api.workspaces.members.removeMember);
   153→  const updateRoleMutation = useMutation(api.workspaces.members.updateMemberRole);
   154→  const cancelInvitationMutation = useMutation(api.workspaces.invitations.cancelInvitation);
   155→
   156→  // Team mutations
   157→  const createTeamMutation = useMutation(hasTeamsApi ? teamsApi.createTeam : null);
   158→  const updateTeamMutation = useMutation(hasTeamsApi ? teamsApi.updateTeam : null);
   159→  const deleteTeamMutation = useMutation(hasTeamsApi ? teamsApi.deleteTeam : null);
   160→  const addTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.addTeamMember : null);
   161→  const removeTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.removeTeamMember : null);
   162→
   163→  // ─── State ─────────────────────────────────────────────────────────────────
   164→  const [activeTab, setActiveTab] = useState<string>("members");
   165→
   166→  // Invite dialog
   167→  const [showInviteDialog, setShowInviteDialog] = useState(false);
   168→  const [inviteEmail, setInviteEmail] = useState("");
   169→  const [inviteRole, setInviteRole] = useState<"manager" | "member">("member");
   170→  const [inviteMessage, setInviteMessage] = useState("");
   171→  const [isInviting, setIsInviting] = useState(false);
   172→
   173→  // Remove/change role
   174→  const [memberToRemove, setMemberToRemove] = useState<any>(null);
   175→  const [memberToChangeRole, setMemberToChangeRole] = useState<any>(null);
   176→  const [newRole, setNewRole] = useState<"manager" | "member">("member");
   177→
   178→  // Team CRUD state
   179→  const [showCreateTeamDialog, setShowCreateTeamDialog] = useState(false);
   180→  const [teamName, setTeamName] = useState("");
   181→  const [teamColor, setTeamColor] = useState(TEAM_COLORS[0]);
   182→  const [teamDescription, setTeamDescription] = useState("");
   183→  const [teamIsCrossTeam, setTeamIsCrossTeam] = useState(false);
   184→  const [isCreatingTeam, setIsCreatingTeam] = useState(false);
   185→
   186→  // Edit team
   187→  const [editingTeam, setEditingTeam] = useState<any>(null);
   188→  const [editTeamName, setEditTeamName] = useState("");
   189→  const [editTeamColor, setEditTeamColor] = useState("");
   190→  const [editTeamDescription, setEditTeamDescription] = useState("");
   191→  const [isUpdatingTeam, setIsUpdatingTeam] = useState(false);
   192→
   193→  // Delete team
   194→  const [teamToDelete, setTeamToDelete] = useState<any>(null);
   195→  const [isDeletingTeam, setIsDeletingTeam] = useState(false);
   196→
   197→  // Add member to team
   198→  const [teamToAddMember, setTeamToAddMember] = useState<any>(null);
   199→  const [addMemberUserId, setAddMemberUserId] = useState("");
   200→  const [addMemberRole, setAddMemberRole] = useState<"lead" | "member">("member");
   201→  const [isAddingTeamMember, setIsAddingTeamMember] = useState(false);
   202→
   203→  // Remove member from team
   204→  const [teamMemberToRemove, setTeamMemberToRemove] = useState<any>(null);
   205→  const [teamMemberRemoveTeamId, setTeamMemberRemoveTeamId] = useState<string>("");
   206→
   207→  // ─── Map Convex data to UI shape ───────────────────────────────────────────
   208→  const members = useMemo(() => {
   209→    if (convexMembers && convexMembers.length > 0) {
   210→      return convexMembers.map((m: any) => ({
   211→        _id: m._id,
   212→        userId: m.userId,
   213→        name: m.userName ?? m.userEmail ?? "Unknown",
   214→        displayName: m.userName ?? m.userEmail ?? "Unknown",
   215→        email: m.userEmail ?? "",
   216→        image: m.userImage ?? "",
   217→        role: m.role,
   218→        status: m.status,
   219→        joinedAt: m.joinedAt,
   220→        lastActiveAt: m.lastActiveAt ?? null,
   221→        projectsAssigned: 0,
   222→        hoursThisWeek: 0,
   223→      }));
   224→    }
   225→    return [];
   226→  }, [convexMembers]);
   227→
   228→  const teams = useMemo(() => {
   229→    if (convexTeams && convexTeams.length > 0) {
   230→      return convexTeams;
   231→    }
   232→    return [];
   233→  }, [convexTeams]);
   234→
   235→  const stats = useMemo(() => {
   236→    if (convexStats) {
   237→      return {
   238→        memberCount: convexStats.memberCount ?? 0,
   239→        clientCount: convexStats.clientCount ?? 0,
   240→        activeProjectCount: convexStats.activeProjectCount ?? 0,
   241→        pendingInvoiceCount: 0,
   242→        totalRevenue: 0,
   243→        totalHoursThisWeek: 0,
   244→        protectionScore: 0,
   245→      };
   246→    }
   247→    return null;
   248→  }, [convexStats]);
   249→
   250→  const invitations = useMemo(() => {
   251→    if (convexInvitations && convexInvitations.length > 0) {
   252→      return convexInvitations.map((inv: any) => ({
   253→        _id: inv._id,
   254→        name: inv.email,
   255→        displayName: inv.email,
   256→        email: inv.email,
   257→        role: inv.role,
   258→        invitedAt: inv.createdAt,
   259→        invitedBy: inv.inviterName,
   260→      }));
   261→    }
   262→    return [];
   263→  }, [convexInvitations]);
   264→
   265→  const { isDisconnected } = useConvexConnectionState();
   266→  const isLoading = hasRealWorkspaceId && (convexMembers === undefined);
   267→  const isDemoMode = !hasRealWorkspaceId;
   268→  const loadingTimedOut = useQueryTimeout(isLoading, 3000);
   269→  const showLoading = isLoading && !loadingTimedOut && !isDisconnected;
   270→
   271→  // ─── Handlers ──────────────────────────────────────────────────────────────
   272→  const handleInvite = async () => {
   273→    if (!inviteEmail.trim() || !activeWorkspaceId) return;
   274→    setIsInviting(true);
   275→    try {
   276→      if (hasRealWorkspaceId) {
   277→        await inviteMemberMutation({
   278→          workspaceId: activeWorkspaceId as any,
   279→          email: inviteEmail.trim(),
   280→          role: inviteRole,
   281→        });
   282→      }
   283→      toast.success(`Invitation sent to ${inviteEmail}`);
   284→      setInviteEmail("");
   285→      setInviteMessage("");
   286→      setShowInviteDialog(false);
   287→    } catch (e: any) {
   288→      toast.error(e.message || "Failed to send invitation");
   289→    } finally {
   290→      setIsInviting(false);
   291→    }
   292→  };
   293→
   294→  const handleRemoveMember = async () => {
   295→    if (!memberToRemove) return;
   296→    try {
   297→      await removeMemberMutation({ memberId: memberToRemove._id });
   298→      toast.success(`${memberToRemove.name} removed from workspace`);
   299→      setMemberToRemove(null);
   300→    } catch (e: any) {
   301→      toast.error(e.message || "Failed to remove member");
   302→    }
   303→  };
   304→
   305→  const handleChangeRole = async () => {
   306→    if (!memberToChangeRole) return;
   307→    try {
   308→      await updateRoleMutation({ memberId: memberToChangeRole._id, role: newRole });
   309→      toast.success(`Role updated to ${newRole}`);
   310→      setMemberToChangeRole(null);
   311→    } catch (e: any) {
   312→      toast.error(e.message || "Failed to update role");
   313→    }
   314→  };
   315→
   316→  const handleCancelInvitation = async (invitation: any) => {
   317→    try {
   318→      await cancelInvitationMutation({ invitationId: invitation._id });
   319→      toast.success(`Invitation to ${invitation.email || invitation.name} cancelled`);
   320→    } catch (e: any) {
   321→      toast.error(e.message || "Failed to cancel invitation");
   322→    }
   323→  };
   324→
   325→  // ─── Team CRUD Handlers ────────────────────────────────────────────────────
   326→  const handleCreateTeam = async () => {
   327→    if (!teamName.trim() || !activeWorkspaceId) return;
   328→    setIsCreatingTeam(true);
   329→    try {
   330→      if (createTeamMutation && hasRealWorkspaceId) {
   331→        await createTeamMutation({
   332→          workspaceId: activeWorkspaceId as any,
   333→          name: teamName.trim(),
   334→          color: teamColor,
   335→          description: teamDescription.trim() || undefined,
   336→          isCrossTeam: teamIsCrossTeam || undefined,
   337→        });
   338→      }
   339→      toast.success(`Team "${teamName}" created!`);
   340→      setShowCreateTeamDialog(false);
   341→      setTeamName("");
   342→      setTeamColor(TEAM_COLORS[0]);
   343→      setTeamDescription("");
   344→      setTeamIsCrossTeam(false);
   345→    } catch (e: any) {
   346→      toast.error(e.message || "Failed to create team");
   347→    } finally {
   348→      setIsCreatingTeam(false);
   349→    }
   350→  };
   351→
   352→  const handleUpdateTeam = async () => {
   353→    if (!editingTeam) return;
   354→    setIsUpdatingTeam(true);
   355→    try {
   356→      if (updateTeamMutation) {
   357→        await updateTeamMutation({
   358→          teamId: editingTeam._id,
   359→          name: editTeamName.trim() || undefined,
   360→          color: editTeamColor || undefined,
   361→          description: editTeamDescription.trim() || undefined,
   362→        });
   363→      }
   364→      toast.success("Team updated!");
   365→      setEditingTeam(null);
   366→    } catch (e: any) {
   367→      toast.error(e.message || "Failed to update team");
   368→    } finally {
   369→      setIsUpdatingTeam(false);
   370→    }
   371→  };
   372→
   373→  const handleDeleteTeam = async () => {
   374→    if (!teamToDelete) return;
   375→    setIsDeletingTeam(true);
   376→    try {
   377→      if (deleteTeamMutation) {
   378→        await deleteTeamMutation({ teamId: teamToDelete._id });
   379→      }
   380→      toast.success(`Team "${teamToDelete.name}" deleted`);
   381→      setTeamToDelete(null);
   382→    } catch (e: any) {
   383→      toast.error(e.message || "Failed to delete team");
   384→    } finally {
   385→      setIsDeletingTeam(false);
   386→    }
   387→  };
   388→
   389→  const handleAddTeamMember = async () => {
   390→    if (!teamToAddMember || !addMemberUserId) return;
   391→    setIsAddingTeamMember(true);
   392→    try {
   393→      if (addTeamMemberMutation) {
   394→        await addTeamMemberMutation({
   395→          teamId: teamToAddMember._id,
   396→          userId: addMemberUserId as any,
   397→          role: addMemberRole,
   398→        });
   399→      }
   400→      toast.success("Member added to team!");
   401→      setTeamToAddMember(null);
   402→      setAddMemberUserId("");
   403→      setAddMemberRole("member");
   404→    } catch (e: any) {
   405→      toast.error(e.message || "Failed to add member to team");
   406→    } finally {
   407→      setIsAddingTeamMember(false);
   408→    }
   409→  };
   410→
   411→  const handleRemoveTeamMember = async () => {
   412→    if (!teamMemberToRemove || !teamMemberRemoveTeamId) return;
   413→    try {
   414→      if (removeTeamMemberMutation) {
   415→        await removeTeamMemberMutation({
   416→          teamId: teamMemberRemoveTeamId as any,
   417→          userId: teamMemberToRemove.userId as any,
   418→        });
   419→      }
   420→      toast.success("Member removed from team");
   421→      setTeamMemberToRemove(null);
   422→      setTeamMemberRemoveTeamId("");
   423→    } catch (e: any) {
   424→      toast.error(e.message || "Failed to remove member from team");
   425→    }
   426→  };
   427→
   428→  // ─── Not in team mode: Show upgrade prompt ────────────────
   429→  if (!isTeamMode) {
   430→    return <SoloModePrompt />;
   431→  }
   432→
   433→  const activeMembers = members.filter((m: any) => m.status === "active");
   434→  const ownerCount = activeMembers.filter((m: any) => m.role === "owner").length;
   435→  const managerCount = activeMembers.filter((m: any) => m.role === "manager").length;
   436→  const memberCount = activeMembers.filter((m: any) => m.role === "member").length;
   437→
   438→  const roleIcon = (role: string) => {
   439→    switch (role) {
   440→      case "owner": return <Crown className="w-3.5 h-3.5 text-amber-500" />;
   441→      case "manager": return <Shield className="w-3.5 h-3.5 text-blue-500" />;
   442→      case "lead": return <Shield className="w-3.5 h-3.5 text-emerald-500" />;
   443→      case "member": return <Eye className="w-3.5 h-3.5 text-gray-500" />;
   444→      default: return null;
   445→    }
   446→  };
   447→
   448→  const roleBadge = (role: string) => {
   449→    const variants: Record<string, any> = {
   450→      owner: { className: "bg-amber-500/10 text-amber-600 border-amber-500/20", label: "Owner" },
   451→      manager: { className: "bg-blue-500/10 text-blue-600 border-blue-500/20", label: "Manager" },
   452→      lead: { className: "bg-emerald-500/10 text-emerald-600 border-emerald-500/20", label: "Lead" },
   453→      member: { className: "bg-gray-500/10 text-gray-600 border-gray-500/20", label: "Member" },
   454→    };
   455→    const v = variants[role] || variants.member;
   456→    return (
   457→      <Badge variant="outline" className={`gap-1 ${v.className}`}>
   458→        {roleIcon(role)}
   459→        {v.label}
   460→      </Badge>
   461→    );
   462→  };
   463→
   464→  const formatLastActive = (timestamp: number | null) => {
   465→    if (!timestamp) return "Never";
   466→    const diffMs = Date.now() - timestamp;
   467→    const diffMins = Math.floor(diffMs / 60000);
   468→    if (diffMins < 5) return "Just now";
   469→    if (diffMins < 60) return `${diffMins}m ago`;
   470→    const diffHours = Math.floor(diffMins / 60);
   471→    if (diffHours < 24) return `${diffHours}h ago`;
   472→    const diffDays = Math.floor(diffHours / 24);
   473→    return `${diffDays}d ago`;
   474→  };
   475→
   476→  const formatJoined = (timestamp: number | null) => {
   477→    if (!timestamp) return "";
   478→    const diffMs = Date.now() - timestamp;
   479→    const diffDays = Math.floor(diffMs / (24 * 60 * 60 * 1000));
   480→    if (diffDays < 1) return "Joined today";
   481→    if (diffDays < 30) return `Joined ${diffDays}d ago`;
   482→    const diffMonths = Math.floor(diffDays / 30);
   483→    return `Joined ${diffMonths}mo ago`;
   484→  };
   485→
   486→  return (
   487→    <div className="p-8">
   488→      <div className="max-w-6xl mx-auto space-y-6">
   489→        {/* Header */}
   490→        <div className="flex items-center justify-between">
   491→          <div>
   492→            <h1 className="text-3xl font-bold text-foreground mb-2 flex items-center gap-3">
   493→              <Building2 className="w-8 h-8 text-purple-500" />
   494→              {activeWorkspace?.name || "Team Workspace"}
   495→            </h1>
   496→            <p className="text-muted-foreground">
   497→              Manage your team members, teams, and invitations
   498→            </p>
   499→          </div>
   500→          {canManageTeam && (
   501→            <div className="flex gap-2">
   502→              <Button className="gap-2" onClick={() => setShowInviteDialog(true)}>
   503→                <UserPlus className="w-4 h-4" />
   504→                Invite Member
   505→              </Button>
   506→            </div>
   507→          )}
   508→        </div>
   509→
   510→        {/* Demo Mode Banner */}
   511→        {isDemoMode && <DemoModeBanner />}
   512→
   513→        {/* Loading / Error State */}
   514→        {showLoading ? (
   515→          <StatsSkeleton />
   516→        ) : (
   517→          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
   518→            <Card className="p-5">
   519→              <div className="flex items-center gap-3">
   520→                <div className="p-2.5 bg-purple-500/10 rounded-lg"><Users className="w-5 h-5 text-purple-500" /></div>
   521→                <div>
   522→                  <p className="text-sm text-muted-foreground">Members</p>
   523→                  <p className="text-2xl font-bold">{stats?.memberCount ?? activeMembers.length}</p>
   524→                </div>
   525→              </div>
   526→            </Card>
   527→            <Card className="p-5">
   528→              <div className="flex items-center gap-3">
   529→                <div className="p-2.5 bg-emerald-500/10 rounded-lg"><Shield className="w-5 h-5 text-emerald-500" /></div>
   530→                <div>
   531→                  <p className="text-sm text-muted-foreground">Teams</p>
   532→                  <p className="text-2xl font-bold">{teams.length}</p>
   533→                </div>
   534→              </div>
   535→            </Card>
