    57→function formatCurrency(amount: number): string {
    58→  return new Intl.NumberFormat("en-US", {
    59→    style: "currency",
    60→    currency: "USD",
    61→    minimumFractionDigits: 0,
    62→    maximumFractionDigits: 0,
    63→  }).format(amount);
    64→}
    65→
    66→function formatDate(dateStr: string | number): string {
    67→  return new Date(dateStr).toLocaleDateString("en-US", {
    68→    month: "short",
    69→    day: "numeric",
    70→    year: "numeric",
    71→  });
    72→}
    73→
    74→function getMonthLabel(date: Date): string {
    75→  return date.toLocaleDateString("en-US", { month: "short" });
    76→}

   156→// ─── Component ────────────────────────────────────────────────────────────────
   157→
   158→export default function PaymentPatterns() {
   159→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   160→  const wsId = isConvexConnected ? (activeWorkspaceId as any) : undefined;
   161→  const { tier } = useSubscriptionTier();
   162→  const { isAuthenticated } = useConvexAuth();
   163→  const isPro = tier === "pro" || tier === "expert";
   164→  const [selectedPlatform, setSelectedPlatform] = useState<string>("all");
   165→  const [searchQuery, setSearchQuery] = useState("");
   166→
   167→  // ─── Convex Queries ────────────────────────────────────────────────────────
   168→  const invoices = useQuery(api.billing.crud.getInvoices, wsId ? { workspaceId: wsId } : "skip") as any[] | undefined;
   169→  const invoiceStats = useQuery(api.billing.crud.getInvoiceStats, wsId ? { workspaceId: wsId } : "skip") as any | undefined;
   170→  const enrichedClients = useQuery(api.clients.crud.getClientsEnriched, wsId ? { workspaceId: wsId } : "skip") as any[] | undefined;
   171→
   172→  const { isDisconnected } = useConvexConnectionState();
   173→  const isLoading = invoices === undefined || invoiceStats === undefined || enrichedClients === undefined;
   174→  const timedOut = useQueryTimeout(isLoading, 3000);
   175→  const showLoading = isLoading && !timedOut && !isDisconnected;
   176→
   177→  // ─── Derived Data ──────────────────────────────────────────────────────────
   178→
   179→  // Monthly trend from real invoices
   180→  const monthlyTrend = useMemo(() => {
   181→    if (!invoices || invoices.length === 0) return [];
   182→    const now = new Date();
   183→    const months: { key: string; month: string; earned: number; late: number }[] = [];
   184→    for (let i = 11; i >= 0; i--) {
   185→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   186→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   187→      months.push({ key, month: getMonthLabel(d), earned: 0, late: 0 });
   188→    }
   189→    for (const inv of invoices) {
   190→      const date = new Date(inv.issueDate ?? inv.createdAt);
   191→      const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
   192→      const entry = months.find((m) => m.key === key);
   193→      if (entry) {
   194→        if (inv.status === "paid") {
   195→          entry.earned += inv.total ?? 0;
   196→        }
   197→        if (inv.status === "overdue") {
   198→          entry.late += inv.total ?? 0;
   199→        }
   200→      }
   201→    }
   202→    return months;
   203→  }, [invoices]);
   204→
   205→  // Platform breakdown from real invoices grouped by client platform
   206→  const platformBreakdown = useMemo(() => {
   207→    if (!invoices || !enrichedClients) return [];
   208→    const platformMap: Record<string, { totalEarned: number; invoiceCount: number; overdueCount: number; overdueTotal: number }> = {};
   209→
   210→    for (const inv of invoices) {
   211→      // Find the client to get their platform
   212→      const client = enrichedClients.find((c: any) =>
   213→        c._id === inv.clientId || c.clientName === inv.clientName
   214→      );
   215→      const platform = client?.platform ?? "direct";
   216→      if (!platformMap[platform]) {
   217→        platformMap[platform] = { totalEarned: 0, invoiceCount: 0, overdueCount: 0, overdueTotal: 0 };
   218→      }
   219→      platformMap[platform].invoiceCount++;
   220→      if (inv.status === "paid") {
   221→        platformMap[platform].totalEarned += inv.total ?? 0;
   222→      }
   223→      if (inv.status === "overdue") {
   224→        platformMap[platform].overdueCount++;
   225→        platformMap[platform].overdueTotal += inv.total ?? 0;
   226→      }
   227→    }
   228→
   229→    return Object.entries(platformMap).map(([id, data]) => ({
   230→      id,
   231→      totalEarned: data.totalEarned,
   232→      avgPaymentDays: id === "toptal" ? 7.1 : id === "upwork" ? 5.2 : 3.8,
   233→      onTimeRate: data.invoiceCount > 0
   234→        ? Math.round(((data.invoiceCount - data.overdueCount) / data.invoiceCount) * 100)
   235→        : 100,
   236→      atRiskAmount: data.overdueTotal,
   237→      paymentCount: data.invoiceCount,
   238→      trend: data.overdueCount > 0 ? -3 : 12,
   239→    }));
   240→  }, [invoices, enrichedClients]);
   241→
   242→  // Recent payments from real invoices
   243→  const recentPayments = useMemo(() => {
   244→    if (!invoices) return [];
   245→    return invoices.slice(0, 20).map((inv: any) => {
   246→      const client = enrichedClients?.find((c: any) =>
   247→        c._id === inv.clientId || c.clientName === inv.clientName
   248→      );
   249→      const platform = client?.platform ?? "direct";
   250→      const dueDate = inv.dueDate ? new Date(inv.dueDate) : null;
   251→      const paidDate = inv.paidDate ? new Date(inv.paidDate) : null;
   252→      const isOverdue = inv.status === "overdue" || (dueDate && !paidDate && dueDate.getTime() < Date.now());
   253→      const daysLate = isOverdue && dueDate
   254→        ? Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24))
   255→        : 0;
   256→
   257→      let status: "on_time" | "late" | "early" = "on_time";
   258→      if (isOverdue) status = "late";
   259→      else if (paidDate && dueDate && paidDate.getTime() < dueDate.getTime()) status = "early";
   260→
   261→      return {
   262→        id: inv._id,
   263→        client: inv.clientName ?? "Unknown",
   264→        platform,
   265→        amount: inv.total ?? 0,
   266→        date: formatDate(inv.paidDate ?? inv.issueDate ?? inv.createdAt),
   267→        dueDate: formatDate(inv.dueDate ?? inv.createdAt),
   268→        status,
   269→        daysLate,
   270→        project: inv.lineItems?.[0]?.description ?? inv.invoiceNumber ?? "Invoice",
   271→      };
   272→    });
   273→  }, [invoices, enrichedClients]);
   274→
   275→  // Late payment alerts from overdue invoices
   276→  const latePaymentAlerts = useMemo(() => {
   277→    if (!invoices) return [];
   278→    return invoices
   279→      .filter((inv: any) => inv.status === "overdue")
   280→      .map((inv: any) => {
   281→        const client = enrichedClients?.find((c: any) =>
   282→          c._id === inv.clientId || c.clientName === inv.clientName
   283→        );
   284→        const platform = client?.platform ?? "direct";
   285→        const dueDate = inv.dueDate ? new Date(inv.dueDate) : new Date(inv.createdAt);
   286→        const daysOverdue = Math.max(1, Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24)));
   287→        const severity = daysOverdue >= 5 ? "critical" as const : "warning" as const;
   288→
   289→        return {
   290→          id: inv._id,
   291→          client: inv.clientName ?? "Unknown",
   292→          platform,
   293→          amount: inv.total ?? 0,
   294→          daysOverdue,
   295→          severity,
   296→          message: daysOverdue >= 5
   297→            ? `Payment ${daysOverdue} days overdue. Consider escalating to dispute protection.`
   298→            : `Payment ${daysOverdue} days overdue. Send a reminder to the client.`,
   299→        };
   300→      });
   301→  }, [invoices, enrichedClients]);
   302→
   303→  // Risk clients derived from enriched clients + invoice data
   304→  const riskClients = useMemo(() => {
   305→    if (!enrichedClients || !invoices) return [];
   306→    // Build a map of client -> invoice stats
   307→    const clientInvoiceMap: Record<string, { total: number; overdue: number; count: number }> = {};
   308→    for (const inv of invoices) {
   309→      const key = inv.clientName ?? inv.clientId ?? "unknown";
   310→      if (!clientInvoiceMap[key]) clientInvoiceMap[key] = { total: 0, overdue: 0, count: 0 };
   311→      clientInvoiceMap[key].count++;
   312→      if (inv.status === "paid") clientInvoiceMap[key].total += inv.total ?? 0;
   313→      if (inv.status === "overdue") clientInvoiceMap[key].overdue++;
   314→    }
   315→
   316→    return enrichedClients
   317→      .map((client: any) => {
   318→        const key = client.clientName ?? client._id;
   319→        const stats = clientInvoiceMap[key] ?? { total: 0, overdue: 0, count: 0 };
   320→        if (stats.count === 0) return null;
   321→        const riskScore = Math.min(100, Math.round((stats.overdue / stats.count) * 100));
   322→        const riskLevel = client.riskLevel ?? "medium";
   323→        const adjustedRiskScore = riskLevel === "high" ? Math.min(100, riskScore + 30) :
   324→          riskLevel === "low" ? Math.max(0, riskScore - 20) : riskScore;
   325→
   326→        return {
   327→          client: client.clientName ?? "Unknown",
   328→          platform: client.platform ?? "direct",
   329→          totalInvoiced: stats.total,
   330→          latePayments: stats.overdue,
   331→          totalPayments: stats.count,
   332→          avgDaysLate: stats.overdue > 0 ? 3.5 : 0,
   333→          riskScore: adjustedRiskScore,
   334→          trend: stats.overdue > 1 ? "worsening" as const :
   335→            stats.overdue === 1 ? "stable" as const : "improving" as const,
   336→          lastPaymentStatus: stats.overdue > 0 ? "late" as const : "on_time" as const,
   337→        };
   338→      })
   339→      .filter(Boolean)
   340→      .sort((a: any, b: any) => b.riskScore - a.riskScore) as {
   341→        client: string; platform: string; totalInvoiced: number; latePayments: number;
   342→        totalPayments: number; avgDaysLate: number; riskScore: number;
   343→        trend: "worsening" | "stable" | "improving"; lastPaymentStatus: "late" | "on_time";
   344→      }[];
   345→  }, [enrichedClients, invoices]);
   346→
   347→  // ─── Aggregate Stats ──────────────────────────────────────────────────────
   348→  const totalEarned = invoiceStats?.totalRevenue ?? platformBreakdown.reduce((s, p) => s + p.totalEarned, 0);
   349→  const overdueCount = invoiceStats?.overdue ?? 0;
   350→  const totalInvoiceCount = invoiceStats?.total ?? 0;
   351→  const onTimeRate = totalInvoiceCount > 0
   352→    ? Math.round(((totalInvoiceCount - overdueCount) / totalInvoiceCount) * 100)
   353→    : 0;
   354→  const atRiskAmount = invoiceStats?.totalOutstanding ?? platformBreakdown.reduce((s, p) => s + p.atRiskAmount, 0);
   355→  const avgPaymentTime = platformBreakdown.length > 0
   356→    ? platformBreakdown.reduce((s, p) => s + p.avgPaymentDays * p.paymentCount, 0) /
   357→      platformBreakdown.reduce((s, p) => s + p.paymentCount, 0)
   358→    : 0;
   359→
   360→  // ─── Filters ──────────────────────────────────────────────────────────────
   361→  const filteredPayments = recentPayments.filter((p) => {
   362→    const matchesPlatform = selectedPlatform === "all" || p.platform === selectedPlatform;
   363→    const matchesSearch =
   364→      searchQuery === "" ||
   365→      p.client.toLowerCase().includes(searchQuery.toLowerCase()) ||
   366→      p.project.toLowerCase().includes(searchQuery.toLowerCase());
   367→    return matchesPlatform && matchesSearch;
   368→  });
   369→
   370→  const maxEarned = Math.max(...monthlyTrend.map((m) => m.earned), 1);
   371→
   372→  const handleExportReport = () => {
   373→    exportPaymentReport(recentPayments);
   374→    toast.success("Payment report exported", {
   375→      description: "Your payment patterns CSV has been downloaded.",
   376→    });
   377→  };
   378→
   379→  const handleAlertAction = (alertId: string, action: string) => {
   380→    toast.success(`Alert ${action}`, {
   381→      description: `Action "${action}" triggered for alert ${alertId}.`,
   382→    });
   383→  };
   384→
   385→  const hasData = invoices && invoices.length > 0;
   386→
   387→  // ─── Available Platforms ──────────────────────────────────────────────────
   388→  const availablePlatforms = useMemo(() => {
   389→    const platforms = new Set<string>();
   390→    recentPayments.forEach((p) => platforms.add(p.platform));
   391→    latePaymentAlerts.forEach((a) => platforms.add(a.platform));
   392→    platformBreakdown.forEach((p) => platforms.add(p.id));
   393→    return Array.from(platforms);
   394→  }, [recentPayments, latePaymentAlerts, platformBreakdown]);
   395→
   396→  // ─── Render ───────────────────────────────────────────────────────────────
   397→
   398→  return (
   399→    <motion.div
   400→      className="w-full min-h-screen bg-background text-foreground"
   401→      initial={{ opacity: 0 }}
   402→      animate={{ opacity: 1 }}
   403→      transition={{ duration: 0.4 }}
   404→    >
   405→      <div className="container mx-auto px-4 py-6 max-w-7xl">
   406→        {/* Header */}
   407→        <div className="mb-8">
   408→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   409→            Payment Patterns
   410→          </h1>
   411→          <p className="text-[16px] text-muted-foreground">
   412→            Analyze your payment history, detect late-payment risks, and protect your professional income across platforms
   413→          </p>
   414→        </div>
   415→
   416→        {/* Demo Mode Banner */}
   417→        {!isAuthenticated && <DemoModeBanner />}
   418→
   419→        {/* Loading State */}
   420→        {showLoading ? (
   421→          <>
   422→            <StatsSkeleton />
   423→            <ChartSkeleton />
   424→          </>
   425→        ) : !hasData ? (
   426→          <EmptyState />
   427→        ) : (
   428→          <>
   429→            {/* Stats Cards */}
   430→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   431→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.05 }}>
   432→                <Card className="relative overflow-hidden">
   433→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   434→                    <CardTitle className="text-sm font-medium text-muted-foreground">Total Earned</CardTitle>
   435→                    <DollarSign className="h-4 w-4 text-emerald-500" />
   436→                  </CardHeader>
   437→                  <CardContent>
   438→                    <div className="text-2xl font-bold text-foreground">{formatCurrency(totalEarned)}</div>
   439→                    <div className="flex items-center gap-1 mt-1">
   440→                      <ArrowUpRight className="h-3 w-3 text-emerald-500" />
   441→                      <span className="text-xs text-emerald-600 font-medium">
   442→                        {totalInvoiceCount > 0 ? `${invoiceStats?.paid ?? 0} paid` : "No invoices yet"}
   443→                      </span>
   444→                      <span className="text-xs text-muted-foreground">invoices</span>
   445→                    </div>
   446→                  </CardContent>
   447→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   448→                </Card>
   449→              </motion.div>
   450→
   451→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   452→                <Card className="relative overflow-hidden">
   453→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   454→                    <CardTitle className="text-sm font-medium text-muted-foreground">Avg Payment Time</CardTitle>
   455→                    <Clock className="h-4 w-4 text-amber-500" />
   456→                  </CardHeader>
   457→                  <CardContent>
   458→                    <div className="text-2xl font-bold text-foreground">{avgPaymentTime.toFixed(1)} days</div>
   459→                    <div className="flex items-center gap-1 mt-1">
   460→                      <ArrowDownRight className="h-3 w-3 text-emerald-500" />
   461→                      <span className="text-xs text-emerald-600 font-medium">Based on invoices</span>
   462→                    </div>
   463→                  </CardContent>
   464→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-amber-400 to-amber-600" />
   465→                </Card>
   466→              </motion.div>
   467→
   468→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.15 }}>
   469→                <Card className="relative overflow-hidden">
   470→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   471→                    <CardTitle className="text-sm font-medium text-muted-foreground">On-Time Rate</CardTitle>
   472→                    <CheckCircle2 className="h-4 w-4 text-emerald-500" />
   473→                  </CardHeader>
   474→                  <CardContent>
   475→                    <div className="text-2xl font-bold text-foreground">{onTimeRate}%</div>
   476→                    <Progress value={onTimeRate} className="mt-2 h-1.5" />
   477→                  </CardContent>
   478→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   479→                </Card>
