   207→   207→  if (!timestamp) return "—";
   208→   208→  return new Intl.DateTimeFormat("en-US", {
   209→   209→    month: "short",
   210→   210→    day: "numeric",
   211→   211→    year: "numeric",
   212→   212→  }).format(new Date(timestamp));
   213→   213→}

   215→   215→function daysUntilClose(timestamp: number | undefined): number | null {
   216→   216→  if (!timestamp) return null;
   217→   217→  const diff = timestamp - Date.now();
   218→   218→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   219→   219→}
