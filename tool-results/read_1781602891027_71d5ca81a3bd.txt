    96→function getTierLevel(tier: string): number {
    97→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
    98→  return levels[tier] ?? 0;
    99→}
   100→
   101→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   102→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   103→}

   241→// --- Main Component ---
   242→export default function EvidenceExport() {
   243→  const { tier } = useSubscriptionTier();
   244→  const { isAuthenticated } = useConvexAuth();
   245→
   246→  // ─── Convex Queries ────────────────────────────────────────────────────────
   247→  const evidenceData = useQuery(
   248→    api.evidence.library.getEvidenceLibraryData,
   249→    {
   250→      view: "date" as const,
   251→      startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   252→      endDate: Date.now(),
   253→    }
   254→  ) as any | undefined;
   255→
   256→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   257→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   258→
   259→  const { isDisconnected } = useConvexConnectionState();
   260→  const isLoading = evidenceData === undefined || clients === undefined;
   261→  const timedOut = useQueryTimeout(isLoading, 3000);
   262→  const showLoading = isLoading && !timedOut && !isDisconnected;
   263→
   264→  // ─── Derived Data ──────────────────────────────────────────────────────────
   265→
   266→  // Build evidence type options from real data
   267→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   268→    if (!evidenceData) return [];
   269→    const items = evidenceData.evidenceItems ?? [];
   270→    const typeCounts: Record<string, number> = {};
   271→    const typeLabels: Record<string, string> = {
   272→      screenshot_ref: "Screenshots",
   273→      memo: "Work Memos",
   274→      url: "URL Visits",
   275→      keyboard: "Keyboard Activity",
   276→      mouse: "Mouse Activity",
   277→      platform_status: "Platform Status",
   278→    };
   279→    const typeIcons: Record<string, React.ReactNode> = {
   280→      screenshot_ref: <Eye className="h-4 w-4" />,
   281→      memo: <FileText className="h-4 w-4" />,
   282→      url: <FileJson className="h-4 w-4" />,
   283→      keyboard: <FileText className="h-4 w-4" />,
   284→      mouse: <FileText className="h-4 w-4" />,
   285→      platform_status: <Shield className="h-4 w-4" />,
   286→    };
   287→
   288→    for (const item of items) {
   289→      const kind = item.type ?? "unknown";
   290→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   291→    }
   292→
   293→    return Object.entries(typeCounts).map(([id, count]) => ({
   294→      id,
   295→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   296→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   297→      count,
   298→    }));
   299→  }, [evidenceData]);
   300→
   301→  // Build project options from real scopes
   302→  const projectOptions = useMemo(() => {
   303→    const all = [{ id: "all", label: "All Projects" }];
   304→    if (scopeDefinitions) {
   305→      for (const scope of scopeDefinitions) {
   306→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   307→      }
   308→    }
   309→    return all;
   310→  }, [scopeDefinitions]);
   311→
   312→  // Build client options from real clients
   313→  const clientOptions = useMemo(() => {
   314→    const all = [{ id: "all", label: "All Clients" }];
   315→    if (clients) {
   316→      for (const client of clients) {
   317→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   318→      }
   319→    }
   320→    return all;
   321→  }, [clients]);
   322→
   323→  // Compliance score from real data
   324→  const complianceScore = evidenceData?.healthScore?.score ?? 0;
   325→  const totalItems = evidenceData?.totalCount ?? 0;
   326→  const disputeSuccessRate = evidenceData?.disputeSuccessRate ?? 0;
   327→
   328→  // State
   329→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   330→  const [dateFrom, setDateFrom] = useState(() => {
   331→    const d = new Date();
   332→    d.setMonth(d.getMonth() - 3);
   333→    return d.toISOString().split("T")[0];
   334→  });
   335→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   336→  const [selectedProject, setSelectedProject] = useState("all");
   337→  const [selectedClient, setSelectedClient] = useState("all");
   338→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   339→  const [isExporting, setIsExporting] = useState(false);
   340→  const [previewOpen, setPreviewOpen] = useState(false);
   341→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   342→
   343→  // Auto-select all evidence types when they load
   344→  useEffect(() => {
   345→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   346→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   347→    }
   348→  }, [evidenceTypes]);
   349→
   350→  // Derived
   351→  const totalEvidenceItems = useMemo(
   352→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   353→    [evidenceTypes, selectedEvidenceTypes]
   354→  );
   355→
   356→  const isFormatLocked = (format: ExportFormat): boolean => {
   357→    const requiredTier = FORMAT_CONFIG[format].tierRequired;
   358→    return !hasTierAccess(tier, requiredTier);
   359→  };
   360→
   361→  const toggleEvidenceType = (id: string) => {
   362→    setSelectedEvidenceTypes((prev) =>
   363→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   364→    );
   365→  };
   366→
   367→  const handleExport = async () => {
   368→    if (selectedEvidenceTypes.length === 0) {
   369→      toast.error("No evidence types selected", {
   370→        description: "Select at least one evidence type to export.",
   371→      });
   372→      return;
   373→    }
   374→
   375→    if (isFormatLocked(selectedFormat)) {
   376→      toast.error("Upgrade required", {
   377→        description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.`,
   378→      });
   379→      return;
   380→    }
   381→
   382→    setIsExporting(true);
   383→
   384→    try {
   385→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   386→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   387→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   388→
   389→      exportEvidence(selectedFormat, {
   390→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   391→        dateRange: `${dateFrom} to ${dateTo}`,
   392→        project: selectedProjectLabel,
   393→        client: selectedClientLabel,
   394→        complianceScore,
   395→        totalItems: totalEvidenceItems,
   396→      });
   397→
   398→      // Add to recent exports
   399→      const newExport: RecentExport = {
   400→        id: `exp_${Date.now()}`,
   401→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   402→        format: selectedFormat,
   403→        status: "completed",
   404→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   405→        date: new Date().toISOString().split("T")[0],
   406→        itemCount: totalEvidenceItems,
   407→        complianceVerified: hasTierAccess(tier, "pro"),
   408→        project: selectedProjectLabel,
   409→      };
   410→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   411→
   412→      toast.success("Export complete", {
   413→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   414→      });
   415→    } catch {
   416→      toast.error("Export failed", {
   417→        description: "An error occurred while generating the export file.",
   418→      });
   419→    }
   420→
   421→    setIsExporting(false);
   422→  };
   423→
   424→  const handleFormatSelect = (format: ExportFormat) => {
   425→    if (isFormatLocked(format)) {
   426→      toast.error("Upgrade required", {
   427→        description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.`,
   428→      });
   429→      return;
   430→    }
   431→    setSelectedFormat(format);
   432→  };
   433→
   434→  const previewItems = useMemo(() => {
   435→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   436→  }, [evidenceTypes, selectedEvidenceTypes]);
   437→
   438→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   439→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   440→
   441→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   442→
   443→  // ─── Render ───────────────────────────────────────────────────────────────
   444→
   445→  return (
   446→    <motion.div
   447→      initial={{ opacity: 0 }}
   448→      animate={{ opacity: 1 }}
   449→      className="w-full min-h-screen bg-background"
   450→    >
   451→      <PageLayout spaced>
   452→        {/* Header */}
   453→        <div>
   454→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">Evidence Export</h1>
   455→          <p className="text-[16px] text-muted-foreground">
   456→            Package and download your work evidence in court-ready formats with compliance verification.
   457→          </p>
   458→        </div>
   459→
   460→        {/* Demo Mode Banner */}
   461→        {!isAuthenticated && <DemoModeBanner />}
   462→
   463→        {/* Loading / Timeout State */}
   464→        {showLoading ? (
   465→          <>
   466→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   467→              {[0, 1, 2, 3].map((i) => <SummarySkeleton key={i} />)}
   468→            </div>
   469→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   470→              <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   471→              <div><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   472→            </div>
   473→          </>
   474→        ) : !hasData ? (
   475→          <EmptyEvidenceState />
   476→        ) : (
   477→          <>
   478→            {/* Summary Dashboard */}
   479→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   480→              <SummaryCard
   481→                title="Total Items"
   482→                value={totalItems}
   483→                subtitle="Across all projects"
   484→                icon={<HardDrive className="h-5 w-5 text-primary-foreground" />}
   485→                accent="bg-primary"
   486→              />
   487→              <SummaryCard
   488→                title="Evidence Types"
   489→                value={evidenceTypes.length}
   490→                subtitle="Different categories"
   491→                icon={<Clock className="h-5 w-5 text-amber-700" />}
   492→                accent="bg-amber-100 dark:bg-amber-900/30"
   493→              />
   494→              <SummaryCard
   495→                title="Compliance Score"
   496→                value={`${complianceScore}%`}
   497→                subtitle={complianceScore >= 80 ? "Court-admissible ready" : "Needs improvement"}
   498→                icon={<ShieldCheck className="h-5 w-5 text-emerald-700" />}
   499→                accent="bg-emerald-100 dark:bg-emerald-900/30"
   500→              />
   501→              <SummaryCard
   502→                title="Dispute Success"
   503→                value={`${disputeSuccessRate}%`}
   504→                subtitle="Based on evidence quality"
   505→                icon={<Shield className="h-5 w-5 text-sky-700" />}
   506→                accent="bg-sky-100 dark:bg-sky-900/30"
   507→              />
   508→            </div>
   509→
   510→            {/* Export Configuration */}
   511→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   512→              {/* Left Column: Format + Filters */}
   513→              <div className="lg:col-span-2 space-y-6">
   514→                {/* Format Selection */}
   515→                <Card>
   516→                  <CardHeader>
   517→                    <CardTitle className="flex items-center gap-2">
   518→                      <Download className="h-5 w-5 text-primary" />
   519→                      Export Format
   520→                    </CardTitle>
   521→                    <CardDescription>Choose the output format for your evidence package.</CardDescription>
   522→                  </CardHeader>
   523→                  <CardContent>
   524→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   525→                      {(Object.keys(FORMAT_CONFIG) as ExportFormat[]).map((formatKey) => (
   526→                        <FormatOption
   527→                          key={formatKey}
   528→                          formatKey={formatKey}
   529→                          selected={selectedFormat === formatKey}
   530→                          onSelect={() => handleFormatSelect(formatKey)}
   531→                          locked={isFormatLocked(formatKey)}
   532→                        />
   533→                      ))}
   534→                    </div>
   535→                    {isFormatLocked(selectedFormat) && (
   536→                      <div className="mt-4 flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-900/20 dark:border-amber-800 p-3">
   537→                        <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400 shrink-0" />
   538→                        <p className="text-sm text-amber-800 dark:text-amber-200">
   539→                          <span className="font-semibold">Upgrade to {FORMAT_CONFIG[selectedFormat].tierRequired}</span> to unlock{" "}
   540→                          {FORMAT_CONFIG[selectedFormat].label} exports with compliance verification.
   541→                        </p>
   542→                      </div>
   543→                    )}
   544→                  </CardContent>
   545→                </Card>
   546→
   547→                {/* Date Range & Filters */}
   548→                <Card>
