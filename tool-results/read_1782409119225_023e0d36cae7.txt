    26→function formatTime(seconds: number) {
    27→  const h = Math.floor(seconds / 3600);
    28→  const m = Math.floor((seconds % 3600) / 60);
    29→  const s = seconds % 60;
    30→  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    31→}
    32→
    33→function formatDuration(minutes: number | undefined) {
    34→  if (!minutes) return "0h 0m";
    35→  const hours = Math.floor(minutes / 60);
    36→  const mins = minutes % 60;
    37→  return `${hours}h ${mins}m`;
    38→}
    39→
    40→function formatDate(ts: number) {
    41→  return new Date(ts).toLocaleDateString("en-US", {
    42→    month: "short",
    43→    day: "numeric",
    44→    year: "numeric",
    45→  });
    46→}
