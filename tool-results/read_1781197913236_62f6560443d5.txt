    91→function formatDate(ts: number): string {
    92→  return new Date(ts).toLocaleDateString("en-US", {

    99→function timeAgo(ts: number): string {
   100→  const diff = Date.now() - ts;
   101→  const minutes = Math.floor(diff / 60000);
   102→  const hours = Math.floor(minutes / 60);
   103→  const days = Math.floor(hours / 24);
   104→  if (days > 0) return `${days}d ago`;
   105→  if (hours > 0) return `${hours}h ago`;
   106→  return `${minutes}m ago`;
   107→}

   159→function calculateScopeHealth(scope: ScopeDefinition, changeOrders: ScopeChangeOrder[]) {
   160→  const scopeChanges = changeOrders.filter(co => co.scopeId === scope._id);
   161→  const pendingChanges = scopeChanges.filter(co => co.status === "pending" || co.status === "auto_generated");
   162→  const revisionUsage = scope.revisionLimit > 0 ? (scope.revisionCount / scope.revisionLimit) * 100 : 0;
   163→  const totalHoursAdded = scopeChanges.reduce((sum, co) => sum + co.impact.hoursAdded, 0);
   164→  const totalCostImpact = scopeChanges.reduce((sum, co) => sum + co.impact.costImpact, 0);
   165→  const hoursOverrun = scope.totalEstimatedHours ? (totalHoursAdded / scope.totalEstimatedHours) * 100 : 0;
   166→
   167→  let riskLevel: "low" | "medium" | "high" = "low";
   168→  if (revisionUsage >= 100 || hoursOverrun > 30 || pendingChanges.length >= 3) riskLevel = "high";
   169→  else if (revisionUsage >= 66 || hoursOverrun > 15 || pendingChanges.length >= 2) riskLevel = "medium";
   170→
   171→  const completedDeliverables = scope.deliverables.filter(d => d.status === "completed").length;
   172→  const progressPercent = scope.deliverables.length > 0 ? (completedDeliverables / scope.deliverables.length) * 100 : 0;
   173→
   174→  return {
   175→    riskLevel,
   176→    revisionUsage,
   177→    totalHoursAdded,
   178→    totalCostImpact,
   179→    hoursOverrun,
   180→    pendingChanges: pendingChanges.length,
   181→    completedDeliverables,
   182→    totalDeliverables: scope.deliverables.length,
   183→    progressPercent,
   184→  };
   185→}

   191→  return (
   192→    <div className="space-y-2">
   193→      <div className="flex items-center justify-between text-xs">
   194→        <span className="text-muted-foreground">Scope Health</span>
   195→        <span className={`font-semibold ${health.riskLevel === "high" ? "text-red-600 dark:text-red-400" : health.riskLevel === "medium" ? "text-amber-600 dark:text-amber-400" : "text-green-600 dark:text-green-400"}`}>
   196→          {health.riskLevel === "high" ? "High Risk" : health.riskLevel === "medium" ? "Moderate Risk" : "Healthy"}
   197→        </span>
   198→      </div>
   199→      <div className="h-2 bg-muted rounded-full overflow-hidden">
   200→        <div className={`h-full rounded-full transition-all ${barColor}`} style={{ width: `${Math.max(5, 100 - health.hoursOverrun - health.revisionUsage * 0.3)}%` }} />
   201→      </div>
   202→    </div>
   203→  );
   204→}

   435→// ─── ScopeCard with its own change orders query ─────────────────────────────
   436→
   437→function ScopeCardContainer({
   438→  scope,
   439→  isSelected,
   440→  onSelect,
   441→  onDelete,
   442→  deleting,
   443→  onMetricsReady,
   444→}: {
   445→  scope: ScopeDefinition;
   446→  isSelected: boolean;
   447→  onSelect: () => void;
   448→  onDelete: () => void;
   449→  deleting: boolean;
   450→  onMetricsReady: (scopeId: string, metrics: { pendingChanges: number; costAtRisk: number; hoursAtRisk: number }) => void;
   451→}) {
   452→  const changeOrders = useQuery(api.scope.crud.getChangeOrders, { scopeId: scope._id as any });
   453→
   454→  const prevMetricsRef = useRef<string>("");
   455→
   456→  useEffect(() => {
   457→    if (changeOrders && Array.isArray(changeOrders)) {
   458→      const pending = (changeOrders as ScopeChangeOrder[]).filter(co => co.status === "pending" || co.status === "auto_generated");
   459→      const costAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.costImpact, 0);
   460→      const hoursAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.hoursAdded, 0);
   461→      const key = `${pending.length}-${costAtRisk}-${hoursAtRisk}`;
   462→      if (key !== prevMetricsRef.current) {
   463→        prevMetricsRef.current = key;
   464→        onMetricsReady(scope._id, { pendingChanges: pending.length, costAtRisk, hoursAtRisk });
   465→      }
   466→    }
   467→  }, [changeOrders, scope._id, onMetricsReady]);
   468→
   469→  return (
   470→    <ScopeCard
   471→      scope={scope}
   472→      changeOrders={(changeOrders as ScopeChangeOrder[]) ?? []}
   473→      isSelected={isSelected}
   474→      onSelect={onSelect}
   475→      onDelete={onDelete}
   476→      deleting={deleting}
   477→    />
   478→  );
