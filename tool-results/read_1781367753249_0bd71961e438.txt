   610→interface NavItemProps {

   619→    <div
   620→      className={`flex items-center gap-2 px-3 py-2 rounded-lg cursor-pointer transition-colors text-sm relative ${
   621→        isActive
   622→          ? "bg-primary/15 text-primary font-semibold border-l-2 border-l-primary -ml-0.5 pl-[10px]"

   626→      <Icon className={`w-4 h-4 flex-shrink-0 ${isActive ? "text-primary" : ""}`} />
   627→      {isExpanded && (
   628→        <span className="font-medium text-[13px] whitespace-nowrap overflow-hidden">
