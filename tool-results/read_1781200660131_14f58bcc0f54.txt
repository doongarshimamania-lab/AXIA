   107→// ─── Helper ─────────────────────────────────────────────────────────────────
   108→
   109→function generateId(): string {
   110→  return Math.random().toString(36).substring(2, 10);
   111→}
   112→
   113→function createEmptySection(type: SectionType): ProposalSection {
   114→  const section: ProposalSection = {
   115→    id: generateId(),
   116→    type,
   117→    content: "",
   118→  };
   119→  if (type === "pricing") {
   120→    section.content = "Pricing Package";
   121→    section.metadata = { items: [{ name: "", price: 0 }] };
   122→  }
   123→  if (type === "milestone") {
   124→    section.content = "Project Milestones";
   125→    section.metadata = { milestones: [{ name: "", weeks: 1 }] };
   126→  }
   127→  if (type === "heading") {
   128→    section.content = "New Section";
   129→  }
   130→  if (type === "text" || type === "summary" || type === "scope_of_work") {
   131→    section.content = "";
   132→  }
   133→  if (type === "terms") {
   134→    section.content = "Payment terms and conditions...";
   135→  }
   136→  if (type === "client_info" || type === "sender_info") {
   137→    section.content = "Contact details";
   138→  }
   139→  return section;
   140→}
   141→
   142→function calculateTotal(sections: ProposalSection[]): number {
   143→  return sections.reduce((total, section) => {
   144→    if (section.type === "pricing" && section.metadata?.items) {
   145→      return total + section.metadata.items.reduce((sum, item) => sum + (item.price || 0), 0);
   146→    }
   147→    return total;
   148→  }, 0);
   149→}
   150→
   151→function formatCurrency(amount: number): string {
   152→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(amount);
   153→}

   272→      const newSections = [...prev];
   273→      const swapIdx = direction === "up" ? idx - 1 : idx + 1;
   274→      [newSections[idx], newSections[swapIdx]] = [newSections[swapIdx], newSections[idx]];

   305→    },
   306→    []
   307→  );
   308→
   309→  const removePricingItem = useCallback((sectionId: string, itemIdx: number) => {

   311→      prev.map((s) => {

   316→    );
   317→  }, []);
   318→

   322→      prev.map((s) => {

   326→      })

   328→  }, []);

   333→        prev.map((s) => {

   336→          milestones[itemIdx] = { ...milestones[itemIdx], [field]: value };

   339→      );
   340→    },
   341→    []

   344→  const removeMilestoneItem = useCallback((sectionId: string, itemIdx: number) => {
   345→    setSections((prev) =>
   346→      prev.map((s) => {
   347→        if (s.id !== sectionId) return s;
   348→        const milestones = (s.metadata?.milestones || []).filter((_, i) => i !== itemIdx);
   349→        return { ...s, metadata: { ...s.metadata, milestones } };
   350→      })

   352→  }, []);
   353→

   355→  const applyTemplate = useCallback((template: Template) => {

   357→    setShowTemplateDialog(false);

   359→  }, []);
   360→

   368→

   371→      toast.error("Please add a proposal title");

   373→    }

   375→    setIsSaving(true);

   391→          totalValue,

   401→          sections: sectionData,

   403→          clientName: clientName || undefined,

   405→          validUntil: validUntilTimestamp,

   482→  if (editLoading && !editLoadTimedOut) {

   492→
   493→  if (editLoadTimedOut && editLoading) {

   495→      <div className="w-full min-h-screen bg-background text-foreground flex items-center justify-center">

   505→
   506→  return (

   508→      className="flex-1 min-h-screen bg-background text-foreground transition-colors"

   511→      transition={{ duration: 0.4 }}
   512→    >
   513→      <div className="container mx-auto px-4 py-6 max-w-5xl">
   514→        {/* ─── Top Bar ────────────────────────────────────────────────── */}

   521→              onClick={() => navigate("/proposals")}
   522→            >
   523→              <ArrowLeft className="h-4 w-4" />
   524→              Back
   525→            </Button>
   526→            <div className="h-5 w-px bg-border" />
   527→            <h1
   528→              className="text-[22px] font-bold text-foreground"

   530→            >

   532→            </h1>

   534→
   535→          <div className="flex items-center gap-2">
   536→            <Button
   537→              variant="outline"
   538→              size="sm"
   539→              className={`gap-1.5 ${isPreview ? "bg-[#8B5CF6]/10 text-[#8B5CF6] border-[#8B5CF6]/30" : ""}`}

   543→              {isPreview ? "Edit" : "Preview"}
   544→            </Button>
   545→            <Button

   547→              size="sm"

   557→              Save Draft
   558→            </Button>
   559→            <Button

   569→              )}
