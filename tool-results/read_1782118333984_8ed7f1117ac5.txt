   321→      const clientEmail = proposal.clientEmail;

   326→        (c) => c.clientName.toLowerCase() === clientName.toLowerCase()

   330→        clientId = existingClient._id;

   333→        const newClientId = await createClientMutation({
   334→          clientName,
   335→          platform: "direct" as const,
   336→          hourlyRate: proposal.totalValue > 0 ? proposal.totalValue / 40 : 50,
   337→          contractType: "hourly" as const,

   339→          contactName: clientName,

   341→          notes: `Created from proposal: ${proposal.title}`,

   474→              format: (v: number) => String(v),

   691→        recordType={sharingRecord?.type || "proposal"}
   692→        currentSharing={sharingRecord?.sharing || []}
   693→        onShare={async (args) => {

   698→                recordType: sharingRecord?.type || "proposal",
   699→                ...args,
   700→              });

   704→            toast.error(err?.message || "Failed to share record");

   707→        onUnshare={async (args) => {
   708→          try {
   709→            if (unshareRecordMutation) {
   710→              await unshareRecordMutation({
   711→                recordId: sharingRecord?.id || shareProposalId,
   712→                recordType: sharingRecord?.type || "proposal",
   713→                ...args,
   714→              });
   715→            }
   716→            toast.success("Access removed");
   717→          } catch (err: any) {
   718→            toast.error(err?.message || "Failed to remove access");
   719→          }
   720→        }}
   721→      />
   722→
   723→      {/* Phase 1: Manual Send Dialog — opens when user clicks "Mark as sent" or "Log another send" */}
   724→      <ManualSendDialog
   725→        open={!!manualSendProposal}
   726→        onOpenChange={(open) => {
   727→          if (!open) setManualSendProposal(null);
   728→        }}
   729→        entityType="proposal"
   730→        entityId={manualSendProposal?.id ?? ""}
   731→        entityTitle={manualSendProposal?.title ?? ""}
   732→        defaultRecipient={manualSendProposal?.defaultRecipient}
   733→        onSuccess={() => {
   734→          toast.success("Send logged", {
   735→            description: "Status updated, follow-ups scheduled, audit trail recorded.",
   736→          });
   737→        }}
   738→      />
   739→    </motion.div>
   740→  );
   741→}
