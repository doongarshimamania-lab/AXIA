   131→// ─── Helpers ────────────────────────────────────────────────────────────────

   133→function formatCurrency(amount: number, currency: string = "USD"): string {
   134→  try {
   135→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   136→  } catch {
   137→    return `$${amount.toFixed(2)}`;
   138→  }
   139→}
   140→
   141→function formatDate(ts: number): string {
   142→  return new Date(ts).toLocaleDateString("en-US", {
   143→    month: "short",
   144→    day: "numeric",
   145→    year: "numeric",
   146→  });
   147→}
   148→
   149→function daysOverdue(dueDate: number): number {
   150→  const now = Date.now();
   151→  return Math.max(0, Math.floor((now - dueDate) / (1000 * 60 * 60 * 24)));
   152→}
