    26→    26→function formatTime(seconds: number) {
    27→    27→  const h = Math.floor(seconds / 3600);
    28→    28→  const m = Math.floor((seconds % 3600) / 60);
    29→    29→  const s = seconds % 60;
    30→    30→  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    31→    31→}
    32→    32→
    33→    33→function formatDuration(minutes: number | undefined) {
    34→    34→  if (!minutes) return "0h 0m";
    35→    35→  const hours = Math.floor(minutes / 60);
    36→    36→  const mins = minutes % 60;
    37→    37→  return `${hours}h ${mins}m`;
    38→    38→}
    39→    39→
    40→    40→function formatDate(ts: number) {
    41→    41→  return new Date(ts).toLocaleDateString("en-US", {
    42→    42→    month: "short",
    43→    43→    day: "numeric",
    44→    44→    year: "numeric",
    45→    45→  });
    46→    46→}
