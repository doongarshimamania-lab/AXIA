   135→function formatCurrency(amount: number, currency: string = "USD"): string {
   136→  try {
   137→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   138→  } catch {
   139→    return `$${amount.toFixed(2)}`;
   140→  }
   141→}
   142→
   143→function formatDate(ts: number): string {
   144→  return new Date(ts).toLocaleDateString("en-US", {
   145→    month: "short",
   146→    day: "numeric",
   147→    year: "numeric",
   148→  });
   149→}
   150→
   151→function daysOverdue(dueDate: number): number {
   152→  const now = Date.now();
   153→  return Math.max(0, Math.floor((now - dueDate) / (1000 * 60 * 60 * 24)));
   154→}
