    55→// ─── Format helpers ─────────────────────────────────────────────────────
    56→function fmtCurrency(n: number): string {
    57→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
    58→}
    59→function fmtCompactCurrency(n: number): string {
    60→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
    61→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
    62→  return fmtCurrency(n);
    63→}
    64→function fmtRelative(ts: number): string {
    65→  const diff = Date.now() - ts;
    66→  const mins = Math.floor(diff / 60000);
    67→  if (mins < 1) return "just now";
    68→  if (mins < 60) return `${mins}m ago`;
    69→  const hrs = Math.floor(mins / 60);
    70→  if (hrs < 24) return `${hrs}h ago`;
    71→  const days = Math.floor(hrs / 24);
    72→  if (days < 30) return `${days}d ago`;
    73→  return new Date(ts).toLocaleDateString();
    74→}

    76→// ─── Animated Number Counter ──────────────────────────────────────────────
    77→function AnimatedNumber({ value, duration = 1200, prefix = "", suffix = "", decimals = 0 }: {
    78→  value: number; duration?: number; prefix?: string; suffix?: string; decimals?: number;
    79→}) {
    80→  const [display, setDisplay] = useState(0);
    81→  const prevValue = useRef(0);
    82→  const rafRef = useRef<number>();
    83→
    84→  useEffect(() => {
    85→    const start = prevValue.current;
    86→    const end = value;
    87→    const startTime = performance.now();
    88→
    89→    const animate = (now: number) => {
    90→      const elapsed = now - startTime;
    91→      const progress = Math.min(elapsed / duration, 1);
    92→      // Ease out cubic
    93→      const eased = 1 - Math.pow(1 - progress, 3);
    94→      const current = start + (end - start) * eased;
    95→      setDisplay(current);
    96→      if (progress < 1) {
    97→        rafRef.current = requestAnimationFrame(animate);
    98→      } else {
    99→        prevValue.current = end;
   100→      }
   101→    };
   102→
   103→    rafRef.current = requestAnimationFrame(animate);
   104→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
   105→  }, [value, duration]);
   106→
   107→  const formatted = decimals > 0
   108→    ? display.toFixed(decimals)
   109→    : Math.round(display).toLocaleString();
   110→
   111→  return <span>{prefix}{formatted}{suffix}</span>;
   112→}

   114→// ─── Mini Sparkline Chart (SVG) ──────────────────────────────────────────
   115→function Sparkline({ data, color = "#8B5CF6", height = 32, width = 80 }: {
   116→  data: number[]; color?: string; height?: number; width?: number;
   117→}) {
   118→  if (!data || data.length < 2) return null;
   119→
   120→  const min = Math.min(...data);
   121→  const max = Math.max(...data);
   122→  const range = max - min || 1;
   123→  const padding = 2;
   124→
   125→  const points = data.map((v, i) => {
   126→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   127→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   128→    return `${x},${y}`;
   129→  }).join(" ");
   130→
   131→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   132→
   133→  return (
   134→    <svg width={width} height={height} className="overflow-visible">
   135→      <defs>
   136→        <linearGradient id={`spark-grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
   137→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   138→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   139→        </linearGradient>
   140→      </defs>
   141→      <polygon points={areaPoints} fill={`url(#spark-grad-${color.replace("#", "")})`} />
   142→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   143→      {/* Last point dot */}
   144→      {data.length > 0 && (
   145→        <circle
   146→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   147→          cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)}
   148→          r="2.5"
   149→          fill={color}
   150→          className="animate-pulse"
   151→        />
   152→      )}
   153→    </svg>
   154→  );
   155→}

   157→// ─── Circular Progress Ring ──────────────────────────────────────────────
   158→function ProgressRing({ value, size = 56, strokeWidth = 4, color = "#8B5CF6", label }: {
   159→  value: number; size?: number; strokeWidth?: number; color?: string; label?: string;
   160→}) {
   161→  const radius = (size - strokeWidth) / 2;
   162→  const circumference = 2 * Math.PI * radius;
   163→  const offset = circumference - (value / 100) * circumference;
   164→
   165→  return (
   166→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   167→      <svg width={size} height={size} className="-rotate-90">
   168→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   169→        <motion.circle
   170→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
   171→          strokeLinecap="round"
   172→          strokeDasharray={circumference}
   173→          initial={{ strokeDashoffset: circumference }}
   174→          animate={{ strokeDashoffset: offset }}
   175→          transition={{ duration: 1.2, ease: "easeOut" }}
   176→        />
   177→      </svg>
   178→      <span className="absolute text-[10px] font-bold" style={{ color }}>{value}%</span>
   179→    </div>
   180→  );
   181→}
