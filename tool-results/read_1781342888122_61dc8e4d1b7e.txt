    54→function fmtCurrency(n: number): string {
    55→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
    56→}
    57→function fmtCompactCurrency(n: number): string {
    58→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
    59→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
    60→  return fmtCurrency(n);
    61→}
    62→function fmtRelative(ts: number): string {
    63→  const diff = Date.now() - ts;
    64→  const mins = Math.floor(diff / 60000);
    65→  if (mins < 1) return "just now";
    66→  if (mins < 60) return `${mins}m ago`;
    67→  const hrs = Math.floor(mins / 60);
    68→  if (hrs < 24) return `${hrs}h ago`;
    69→  const days = Math.floor(hrs / 24);
    70→  if (days < 30) return `${days}d ago`;
    71→  return new Date(ts).toLocaleDateString();
    72→}

    74→// ─── Animated Number Counter ──────────────────────────────────────────────
    75→function AnimatedNumber({ value, duration = 1200, prefix = "", suffix = "", decimals = 0 }: {
    76→  value: number; duration?: number; prefix?: string; suffix?: string; decimals?: number;
    77→}) {
    78→  const [display, setDisplay] = useState(0);
    79→  const prevValue = useRef(0);
    80→  const rafRef = useRef<number>();
    81→
    82→  useEffect(() => {
    83→    const start = prevValue.current;
    84→    const end = value;
    85→    const startTime = performance.now();
    86→
    87→    const animate = (now: number) => {
    88→      const elapsed = now - startTime;
    89→      const progress = Math.min(elapsed / duration, 1);
    90→      // Ease out cubic
    91→      const eased = 1 - Math.pow(1 - progress, 3);
    92→      const current = start + (end - start) * eased;
    93→      setDisplay(current);
    94→      if (progress < 1) {
    95→        rafRef.current = requestAnimationFrame(animate);
    96→      } else {
    97→        prevValue.current = end;
    98→      }
    99→    };
   100→
   101→    rafRef.current = requestAnimationFrame(animate);
   102→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
   103→  }, [value, duration]);
   104→
   105→  const formatted = decimals > 0
   106→    ? display.toFixed(decimals)
   107→    : Math.round(display).toLocaleString();
   108→
   109→  return <span>{prefix}{formatted}{suffix}</span>;
   110→}

   112→// ─── Mini Sparkline Chart (SVG) ──────────────────────────────────────────
   113→function Sparkline({ data, color = "#8B5CF6", height = 32, width = 80 }: {
   114→  data: number[]; color?: string; height?: number; width?: number;
   115→}) {
   116→  if (!data || data.length < 2) return null;
   117→
   118→  const min = Math.min(...data);
   119→  const max = Math.max(...data);
   120→  const range = max - min || 1;
   121→  const padding = 2;
   122→
   123→  const points = data.map((v, i) => {
   124→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   125→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   126→    return `${x},${y}`;
   127→  }).join(" ");
   128→
   129→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   130→
   131→  return (
   132→    <svg width={width} height={height} className="overflow-visible">
   133→      <defs>
   134→        <linearGradient id={`spark-grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
   135→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   136→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   137→        </linearGradient>
   138→      </defs>
   139→      <polygon points={areaPoints} fill={`url(#spark-grad-${color.replace("#", "")})`} />
   140→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   141→      {/* Last point dot */}
   142→      {data.length > 0 && (
   143→        <circle
   144→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   145→          cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)}
   146→          r="2.5"
   147→          fill={color}
   148→          className="animate-pulse"
   149→        />
   150→      )}
   151→    </svg>
   152→  );
   153→}

   155→// ─── Circular Progress Ring ──────────────────────────────────────────────
   156→function ProgressRing({ value, size = 56, strokeWidth = 4, color = "#8B5CF6", label }: {
   157→  value: number; size?: number; strokeWidth?: number; color?: string; label?: string;
   158→}) {
   159→  const radius = (size - strokeWidth) / 2;
   160→  const circumference = 2 * Math.PI * radius;
   161→  const offset = circumference - (value / 100) * circumference;
   162→
   163→  return (
   164→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   165→      <svg width={size} height={size} className="-rotate-90">
   166→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   167→        <motion.circle
   168→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
   169→          strokeLinecap="round"
   170→          strokeDasharray={circumference}
   171→          initial={{ strokeDashoffset: circumference }}
   172→          animate={{ strokeDashoffset: offset }}
   173→          transition={{ duration: 1.2, ease: "easeOut" }}
   174→        />
   175→      </svg>
   176→      <span className="absolute text-[10px] font-bold" style={{ color }}>{value}%</span>
   177→    </div>
   178→  );
   179→}
