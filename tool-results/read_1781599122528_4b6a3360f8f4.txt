   255→  const [statusFilter, setStatusFilter] = useState<string>("all");
   256→  const [createOpen, setCreateOpen] = useState(false);
   257→  const [editOpen, setEditOpen] = useState(false);
   258→  const [editingGoal, setEditingGoal] = useState<any>(null);

   260→  const [deletingGoal, setDeletingGoal] = useState<any>(null);
   261→

   264→  const [formDescription, setFormDescription] = useState("");
   265→  const [formType, setFormType] = useState("custom");

   270→  const [formStatus, setFormStatus] = useState("not_started");
   271→

   273→  const [isCreating, setIsCreating] = useState(false);
   274→  const [isUpdating, setIsUpdating] = useState(false);
   275→  const [isDeleting, setIsDeleting] = useState(false);
   276→  const [isCompleting, setIsCompleting] = useState<string | null>(null);
   277→
   278→  // ─── Loading timeout pattern ────────────────────────────────────────────
   279→  const { isDisconnected } = useConvexConnectionState();

   281→
   282→  const isLoading = !authLoading && goalsData === undefined && !timedOut && !isDisconnected;
   283→
   284→  // ─── Demo mode ──────────────────────────────────────────────────────────
   285→  const isDemoMode = !authLoading && !isAuthenticated;
   286→

   288→  const goals = isDemoMode ? MOCK_GOALS : (goalsData ?? []);
   289→
   290→  // ── Computed Stats ──

   292→  const completedGoals = goals.filter((g: any) => g.status === "completed").length;
   293→  const completionRate =
   294→    goals.length > 0 ? Math.round((completedGoals / goals.length) * 100) : 0;
   295→  const longestStreak = goals.reduce((max: number, g: any) => Math.max(max, g.streak ?? 0), 0);
   296→
   297→  // ── Filtering ──
   298→  const filteredGoals =
   299→    statusFilter === "all"
   300→      ? goals
   301→      : goals.filter((g: any) => g.status === statusFilter);
   302→

   304→  const resetForm = () => {
   305→    setFormTitle("");
   306→    setFormDescription("");
   307→    setFormType("custom");
   308→    setFormTarget("");
   309→    setFormCurrent("0");
   310→    setFormUnit("%");

   312→    setFormStatus("not_started");
   313→  };

   316→    if (goal.target === 0) return goal.status === "completed" ? 100 : 0;
   317→    return Math.min(Math.round((goal.current / goal.target) * 100), 100);
   318→  };

   321→  const handleCreate = async () => {
   322→    if (!formTitle.trim()) {
   323→      toast.error("Goal title is required");
   324→      return;
   325→    }
   326→    if (!formTarget) {
   327→      toast.error("Target value is required");
   328→      return;

   332→    const current = Number(formCurrent) || 0;
   333→    const deadline = formDeadline ? new Date(formDeadline).getTime() : undefined;

   345→        title: formTitle.trim(),

   347→        type: formType,
   348→        target,
   349→        current,
   350→        unit: formUnit,

   352→        status: formStatus,

   358→      toast.error(err?.message || "Failed to create goal");
   359→    } finally {
   360→      setIsCreating(false);
   361→    }
   362→  };

   366→    setEditingGoal(goal);
   367→    setFormTitle(goal.title);
   368→    setFormDescription(goal.description ?? "");
   369→    setFormType(goal.type);
   370→    setFormTarget(String(goal.target));
   371→    setFormCurrent(String(goal.current));
   372→    setFormUnit(goal.unit);
   373→    setFormDeadline(goal.deadline ? new Date(goal.deadline).toISOString().split("T")[0] : "");
   374→    setFormStatus(goal.status);
   375→    setEditOpen(true);
   376→  };
   377→
   378→  const handleEdit = async () => {
   379→    if (!editingGoal) return;
   380→    if (!formTitle.trim()) {
   381→      toast.error("Goal title is required");
   382→      return;
   383→    }
   384→
   385→    const target = Number(formTarget);
   386→    const deadline = formDeadline ? new Date(formDeadline).getTime() : undefined;
   387→
   388→    if (isDemoMode) {
   389→      toast.success(`Goal "${formTitle.trim()}" updated! (Demo mode)`);
   390→      resetForm();
   391→      setEditOpen(false);
   392→      setEditingGoal(null);
   393→      return;
   394→    }
   395→
   396→    setIsUpdating(true);
   397→    try {
   398→      await updateGoalMutation({
   399→        goalId: editingGoal._id,
   400→        title: formTitle.trim(),
   401→        description: formDescription.trim() || undefined,
   402→        type: formType,
   403→        target,
   404→        current: Number(formCurrent) || 0,
   405→        unit: formUnit,
   406→        deadline,
   407→        status: formStatus,
   408→      });
   409→      resetForm();
   410→      setEditOpen(false);
   411→      setEditingGoal(null);
   412→      toast.success(`Goal "${formTitle.trim()}" updated`);
   413→    } catch (err: any) {
   414→      toast.error(err?.message || "Failed to update goal");
   415→    } finally {
   416→      setIsUpdating(false);
   417→    }
   418→  };
   419→
   420→  // ── Delete ──
   421→  const openDelete = (goal: any) => {
   422→    setDeletingGoal(goal);
   423→    setDeleteOpen(true);
   424→  };
   425→
   426→  const handleDelete = async () => {
   427→    if (!deletingGoal) return;
   428→
   429→    if (isDemoMode) {
   430→      toast.success(`Goal "${deletingGoal.title}" deleted! (Demo mode)`);
   431→      setDeleteOpen(false);
   432→      setDeletingGoal(null);
   433→      return;
   434→    }
   435→
   436→    setIsDeleting(true);
   437→    try {
   438→      await deleteGoalMutation({ goalId: deletingGoal._id });
   439→      setDeleteOpen(false);
   440→      setDeletingGoal(null);
   441→      toast.success(`Goal "${deletingGoal.title}" deleted`);
   442→    } catch (err: any) {
   443→      toast.error(err?.message || "Failed to delete goal");
   444→    } finally {
   445→      setIsDeleting(false);
   446→    }
   447→  };

   450→  const handleMarkComplete = async (goalId: any) => {

   455→      toast.success(`Goal "${goal.title}" marked as complete! (Demo mode)`);

   459→    setIsCompleting(goalId);

   461→      await markGoalCompleteMutation({ goalId });

   463→    } catch (err: any) {
   464→      toast.error(err?.message || "Failed to mark goal complete");
   465→    } finally {
   466→      setIsCompleting(null);
   467→    }
   468→  };

   473→      toast.success(currentCompleted ? "Milestone unmarked! (Demo mode)" : "Milestone completed! (Demo mode)");
   474→      return;

   476→
   477→    try {
   478→      await updateMilestoneMutation({
   479→        goalId,
   480→        milestoneId,
   481→        completed: !currentCompleted,
   482→      });
   483→    } catch (err: any) {
   484→      toast.error(err?.message || "Failed to update milestone");
   485→    }
   486→  };
   487→
   488→  // ── Map status filter labels ──
   489→  const FILTER_OPTIONS = [
   490→    { value: "all", label: "All" },
   491→    { value: "in_progress", label: "In Progress" },
   492→    { value: "not_started", label: "Not Started" },
   493→    { value: "completed", label: "Completed" },
   494→    { value: "abandoned", label: "Abandoned" },
   495→  ];
   496→
   497→  // ── Render ──
   498→  return (
   499→    <motion.div
   500→      initial={{ opacity: 0 }}
   501→      animate={{ opacity: 1 }}
   502→      className="w-full min-h-screen bg-background text-foreground"
   503→    >
   504→      <div className="container mx-auto px-4 py-6 space-y-6">
   505→        {/* ── Header ── */}
   506→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   507→          <div>
   508→            <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   509→              Goals
   510→            </h1>
   511→            <p className="text-[16px] text-muted-foreground">
   512→              Set and track professional goals to protect your income and grow your agency
   513→            </p>
   514→          </div>
   515→
   516→          <Dialog
   517→            open={createOpen}
   518→            onOpenChange={(open) => {
   519→              setCreateOpen(open);
   520→              if (!open) resetForm();
   521→            }}
   522→          >
   523→            <DialogTrigger asChild>
   524→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   525→                <Plus className="mr-2 h-4 w-4" />
   526→                Create Goal
   527→              </Button>
   528→            </DialogTrigger>
   529→            <DialogContent className="sm:max-w-lg">
   530→              <DialogHeader>
   531→                <DialogTitle>Create New Goal</DialogTitle>
   532→                <DialogDescription>
   533→                  Define a measurable goal with a deadline to keep your professional growth on track.
   534→                </DialogDescription>
   535→              </DialogHeader>
   536→              <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   537→                <div className="space-y-2">
   538→                  <Label htmlFor="goal-title">Title</Label>
   539→                  <Input
   540→                    id="goal-title"
   541→                    placeholder="e.g. Reach $10K monthly revenue"
   542→                    value={formTitle}
   543→                    onChange={(e) => setFormTitle(e.target.value)}
   544→                  />
   545→                </div>
   546→                <div className="space-y-2">
   547→                  <Label htmlFor="goal-description">Description</Label>
   548→                  <Textarea
   549→                    id="goal-description"
