    39→    39→
    40→    40→const PRESET_COLORS = [
    41→    41→  "#ef4444", "#f97316", "#f59e0b", "#84cc16", "#22c55e",
    42→    42→  "#14b8a6", "#06b6d4", "#6366f1", "#a855f7", "#ec4899",

    45→    45→
    46→    46→// ─── Component ───────────────────────────────────────────────────────────────
    47→    47→
    48→    48→export default function Tags() {
    49→    49→  const { isAuthenticated, isLoading: authLoading } = useAuth();
    50→    50→  // ponytail: extract active workspace id so tags are scoped per-workspace (prevents cross-workspace data leak)
    51→    51→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
    52→    52→  const workspaceId = isConvexConnected ? activeWorkspaceId : undefined;
    53→    53→
    54→    54→  // ─── Convex queries & mutations ──────────────────────────────────────────
    55→    55→  // ponytail: pass workspaceId to scope the by_workspace index (falls back to by_user when undefined)
    56→    56→  const tagsData = useQuery(api.tags.crud.getTags, { workspaceId: workspaceId as any });
    57→    57→  const createTagMutation = useMutation(api.tags.crud.createTag);
    58→    58→  const updateTagMutation = useMutation(api.tags.crud.updateTag);
    59→    59→  const deleteTagMutation = useMutation(api.tags.crud.deleteTag);
    60→    60→
    61→    61→  // ─── Local state ────────────────────────────────────────────────────────
    62→    62→  const [searchQuery, setSearchQuery] = useState("");
    63→    63→  const [activeFilter, setActiveFilter] = useState<string | null>(null);
    64→    64→  const [createOpen, setCreateOpen] = useState(false);
    65→    65→  const [editOpen, setEditOpen] = useState(false);
    66→    66→  const [editingTag, setEditingTag] = useState<any>(null);
    67→    67→  const [deleteOpen, setDeleteOpen] = useState(false);
    68→    68→  const [deletingTag, setDeletingTag] = useState<any>(null);
    69→    69→
    70→    70→  // Form state
    71→    71→  const [formName, setFormName] = useState("");
    72→    72→  const [formColor, setFormColor] = useState(PRESET_COLORS[0]);
    73→    73→  const [formCategory, setFormCategory] = useState<string>("general");
    74→    74→
    75→    75→  // Mutation loading state
    76→    76→  const [isCreating, setIsCreating] = useState(false);
    77→    77→  const [isUpdating, setIsUpdating] = useState(false);
    78→    78→  const [isDeleting, setIsDeleting] = useState(false);
    79→    79→
    80→    80→  // ─── Loading timeout pattern ────────────────────────────────────────────
    81→    81→  const { isDisconnected } = useConvexConnectionState();
    82→    82→  const timedOut = useQueryTimeout(!authLoading && tagsData === undefined, 3000);
    83→    83→
    84→    84→  const isLoading = !authLoading && tagsData === undefined && !timedOut && !isDisconnected;
    85→    85→
    86→    86→  // ─── Demo mode ──────────────────────────────────────────────────────────
    87→    87→  const isDemoMode = !authLoading && !isAuthenticated;
    88→    88→
    89→    89→  // ─── Data resolution ────────────────────────────────────────────────────
    90→    90→  const realTags = (tagsData ?? []).map((t: any) => ({
    91→    91→    _id: t._id,
    92→    92→    name: t.name,
    93→    93→    color: t.color,
    94→    94→    category: t.category ?? null,
    95→    95→    usageCount: t.usageCount ?? 0,
    96→    96→    createdAt: t.createdAt,
    97→    97→  }));
    98→    98→
    99→    99→  const tags = realTags;
   100→   100→
   101→   101→  // ── Computed Stats ──
   102→   102→  const totalTags = tags.length;
   103→   103→  const mostUsedTag =
   104→   104→    tags.length > 0
   105→   105→      ? tags.reduce((prev: any, curr: any) => (curr.usageCount > prev.usageCount ? curr : prev))
   106→   106→      : null;
   107→   107→
   108→   108→  // ── Filtering ──

   111→   111→      tag.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   112→   112→      (tag.category ?? "").toLowerCase().includes(searchQuery.toLowerCase());
   113→   113→    const matchesFilter = activeFilter ? tag._id === activeFilter : true;
   114→   114→    return matchesSearch && matchesFilter;
   115→   115→  });

   121→   121→    setFormCategory("general");
   122→   122→  };

   125→   125→    const date = new Date(timestamp);
   126→   126→    return date.toLocaleDateString("en-US", {
   127→   127→      month: "short",
   128→   128→      day: "numeric",
   129→   129→      year: "numeric",
   130→   130→    });
   131→   131→  };
   132→   132→

   135→   135→    if (!formName.trim()) {
   136→   136→      toast.error("Tag name is required");
   137→   137→      return;
   138→   138→    }
   139→   139→
   140→   140→    if (isDemoMode) {
   141→   141→      toast.success(`Tag "${formName.trim()}" created successfully! (Demo mode)`);
   142→   142→      resetForm();
   143→   143→      setCreateOpen(false);
   144→   144→      return;

   146→   146→
   147→   147→    setIsCreating(true);

   149→   149→      await createTagMutation({
   150→   150→        workspaceId: workspaceId as any, // ponytail: stamp the new tag with the active workspace
   151→   151→        name: formName.trim(),
   152→   152→        color: formColor,
   153→   153→        category: formCategory,
   154→   154→      });
   155→   155→      resetForm();
   156→   156→      setCreateOpen(false);
   157→   157→      toast.success(`Tag "${formName.trim()}" created successfully`);
   158→   158→    } catch (err: any) {
   159→   159→      toast.error(err?.message || "Failed to create tag");
   160→   160→    } finally {
   161→   161→      setIsCreating(false);
   162→   162→    }
   163→   163→  };
   164→   164→
   165→   165→  // ── Edit ──
   166→   166→  const openEdit = (tag: any) => {
   167→   167→    setEditingTag(tag);
   168→   168→    setFormName(tag.name);
   169→   169→    setFormColor(tag.color);
   170→   170→    setFormCategory(tag.category ?? "general");
   171→   171→    setEditOpen(true);
   172→   172→  };

   174→   174→  const handleEdit = async () => {
   175→   175→    if (!editingTag) return;
   176→   176→    if (!formName.trim()) {
   177→   177→      toast.error("Tag name is required");
   178→   178→      return;
   179→   179→    }
   180→   180→
   181→   181→    if (isDemoMode) {
   182→   182→      toast.success(`Tag "${formName.trim()}" updated! (Demo mode)`);
   183→   183→      resetForm();
   184→   184→      setEditOpen(false);
   185→   185→      setEditingTag(null);
   186→   186→      return;
   187→   187→    }
   188→   188→
   189→   189→    setIsUpdating(true);
   190→   190→    try {
   191→   191→      await updateTagMutation({
   192→   192→        tagId: editingTag._id,
   193→   193→        name: formName.trim(),
   194→   194→        color: formColor,
   195→   195→        category: formCategory,
   196→   196→      });
   197→   197→      resetForm();
   198→   198→      setEditOpen(false);
   199→   199→      setEditingTag(null);
   200→   200→      toast.success(`Tag "${formName.trim()}" updated`);
   201→   201→    } catch (err: any) {
   202→   202→      toast.error(err?.message || "Failed to update tag");
   203→   203→    } finally {
   204→   204→      setIsUpdating(false);
   205→   205→    }
   206→   206→  };
   207→   207→
   208→   208→  // ── Delete ──
   209→   209→  const openDelete = (tag: any) => {
   210→   210→    setDeletingTag(tag);
   211→   211→    setDeleteOpen(true);
   212→   212→  };
   213→   213→
   214→   214→  const handleDelete = async () => {
   215→   215→    if (!deletingTag) return;
   216→   216→
   217→   217→    if (isDemoMode) {
   218→   218→      toast.success(`Tag "${deletingTag.name}" deleted! (Demo mode)`);
   219→   219→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   220→   220→      setDeleteOpen(false);
   221→   221→      setDeletingTag(null);
   222→   222→      return;
   223→   223→    }
   224→   224→
   225→   225→    setIsDeleting(true);
   226→   226→    try {
   227→   227→      await deleteTagMutation({ tagId: deletingTag._id });
   228→   228→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   229→   229→      setDeleteOpen(false);
   230→   230→      setDeletingTag(null);
   231→   231→      toast.success(`Tag "${deletingTag.name}" deleted`);
   232→   232→    } catch (err: any) {
   233→   233→      toast.error(err?.message || "Failed to delete tag");
   234→   234→    } finally {
   235→   235→      setIsDeleting(false);
   236→   236→    }
   237→   237→  };
   238→   238→
   239→   239→  // ── Render ──
   240→   240→  return (
   241→   241→    <motion.div
   242→   242→      initial={{ opacity: 0 }}
   243→   243→      animate={{ opacity: 1 }}
   244→   244→      className="w-full min-h-screen bg-background text-foreground"
   245→   245→    >
   246→   246→      <PageLayout spaced>
   247→   247→        {/* ── Header ── */}
   248→   248→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   249→   249→          <div>
   250→   250→            <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   251→   251→              Tags
   252→   252→            </h1>
   253→   253→            <p className="text-[16px] text-muted-foreground">
   254→   254→              Organize your work with custom tags for quick filtering and categorization
   255→   255→            </p>
   256→   256→          </div>
   257→   257→
   258→   258→          <Dialog open={createOpen} onOpenChange={(open) => { setCreateOpen(open); if (!open) resetForm(); }}>
   259→   259→            <DialogTrigger asChild>
   260→   260→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   261→   261→                <Plus className="mr-2 h-4 w-4" />
   262→   262→                Create Tag
   263→   263→              </Button>
   264→   264→            </DialogTrigger>
   265→   265→            <DialogContent className="sm:max-w-md">
   266→   266→              <DialogHeader>
   267→   267→                <DialogTitle>Create New Tag</DialogTitle>
   268→   268→                <DialogDescription>
   269→   269→                  Add a tag to categorize and filter your time entries, projects, and evidence.
   270→   270→                </DialogDescription>
   271→   271→              </DialogHeader>
   272→   272→              <div className="space-y-4 py-2">
   273→   273→                <div className="space-y-2">
   274→   274→                  <Label htmlFor="tag-name">Name</Label>
   275→   275→                  <Input
   276→   276→                    id="tag-name"
   277→   277→                    placeholder="e.g. Urgent, Design, Revision"
   278→   278→                    value={formName}
   279→   279→                    onChange={(e) => setFormName(e.target.value)}
   280→   280→                  />
   281→   281→                </div>
   282→   282→                <div className="space-y-2">
   283→   283→                  <Label>Color</Label>
   284→   284→                  <div className="flex items-center gap-2 flex-wrap">
   285→   285→                    {PRESET_COLORS.map((color) => (
   286→   286→                      <button
   287→   287→                        key={color}
   288→   288→                        type="button"
   289→   289→                        className="w-8 h-8 rounded-full border-2 transition-all hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-primary"
   290→   290→                        style={{
   291→   291→                          backgroundColor: color,
   292→   292→                          borderColor: formColor === color ? "var(--foreground)" : "transparent",
   293→   293→                          boxShadow: formColor === color ? `0 0 0 2px ${color}40` : "none",
   294→   294→                        }}
   295→   295→                        onClick={() => setFormColor(color)}
   296→   296→                      />
   297→   297→                    ))}
   298→   298→                    <div className="flex items-center gap-1 ml-1">
   299→   299→                      <Palette className="h-4 w-4 text-muted-foreground" />
   300→   300→                      <input
   301→   301→                        type="color"
   302→   302→                        value={formColor}
   303→   303→                        onChange={(e) => setFormColor(e.target.value)}
   304→   304→                        className="w-8 h-8 rounded cursor-pointer border border-border"
   305→   305→                      />
   306→   306→                    </div>
   307→   307→                  </div>
   308→   308→                </div>
   309→   309→                <div className="space-y-2">
   310→   310→                  <Label>Category</Label>
   311→   311→                  <div className="flex flex-wrap gap-2">
   312→   312→                    {["general", "client", "project", "evidence"].map((cat) => (
   313→   313→                      <button key={cat} type="button" onClick={() => setFormCategory(cat)}>
   314→   314→                        <Badge
   315→   315→                          variant={formCategory === cat ? "default" : "outline"}
   316→   316→                          className="cursor-pointer transition-colors capitalize"
   317→   317→                        >
   318→   318→                          {cat}
   319→   319→                        </Badge>
   320→   320→                      </button>
   321→   321→                    ))}
   322→   322→                  </div>
   323→   323→                </div>
   324→   324→              </div>
   325→   325→              <DialogFooter>
   326→   326→                <Button variant="outline" onClick={() => { setCreateOpen(false); resetForm(); }} disabled={isCreating}>
   327→   327→                  Cancel
   328→   328→                </Button>
   329→   329→                <Button className="bg-primary hover:bg-primary/90" onClick={handleCreate} disabled={isCreating}>
   330→   330→                  {isCreating ? (
   331→   331→                    <>
   332→   332→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   333→   333→                      Creating...
   334→   334→                    </>
   335→   335→                  ) : (
   336→   336→                    "Create Tag"
   337→   337→                  )}
   338→   338→                </Button>
   339→   339→              </DialogFooter>
   340→   340→            </DialogContent>
   341→   341→          </Dialog>
   342→   342→        </div>
   343→   343→
   344→   344→        {/* ── Demo mode empty state ── */}
   345→   345→        {isDemoMode && (
   346→   346→          <Card className="p-8 bg-card rounded-xl border border-border">
   347→   347→            <div className="text-center space-y-4">
   348→   348→              <div className="mx-auto h-16 w-16 rounded-full bg-primary/10 flex items-center justify-center">
   349→   349→                <Tag className="h-8 w-8 text-primary" />
   350→   350→              </div>
   351→   351→              <div>
   352→   352→                <h3 className="text-lg font-semibold text-foreground">Sign in to see your tags</h3>
   353→   353→                <p className="text-sm text-muted-foreground mt-1">
   354→   354→                  Connect your account to organize your work with custom tags.
   355→   355→                </p>
   356→   356→              </div>
   357→   357→              <Button asChild>
   358→   358→                <a href="/auth">Sign In</a>
   359→   359→              </Button>
   360→   360→            </div>
   361→   361→          </Card>
   362→   362→        )}
   363→   363→
   364→   364→        {/* ── Loading state ── */}
   365→   365→        {isLoading ? (
   366→   366→          <div className="space-y-4">
   367→   367→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   368→   368→              <Skeleton className="h-[76px] rounded-xl" />
   369→   369→              <Skeleton className="h-[76px] rounded-xl" />
   370→   370→              <Skeleton className="h-[76px] rounded-xl" />
   371→   371→            </div>
   372→   372→            <Skeleton className="h-10 rounded-lg" />
   373→   373→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   374→   374→              {[1, 2, 3, 4, 5, 6].map((i) => (
   375→   375→                <Skeleton key={i} className="h-[160px] rounded-xl" />
   376→   376→              ))}
   377→   377→            </div>
   378→   378→          </div>
   379→   379→        ) : (
   380→   380→          <>
   381→   381→            {/* ── Stats Cards ── */}
   382→   382→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   383→   383→              <Card>
   384→   384→                <CardContent className="p-6">
   385→   385→                  <div className="flex items-center gap-3">
   386→   386→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-primary/10">
   387→   387→                      <Hash className="h-5 w-5 text-primary" />
   388→   388→                    </div>
   389→   389→                    <div>
   390→   390→                      <p className="text-sm text-muted-foreground">Total Tags</p>
   391→   391→                      <p className="text-2xl font-bold">{totalTags}</p>
   392→   392→                    </div>
   393→   393→                  </div>
   394→   394→                </CardContent>
   395→   395→              </Card>
   396→   396→              <Card>
   397→   397→                <CardContent className="p-6">
   398→   398→                  <div className="flex items-center gap-3">
   399→   399→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-orange-500/10">
   400→   400→                      <Tag className="h-5 w-5 text-orange-500" />
   401→   401→                    </div>
   402→   402→                    <div>
   403→   403→                      <p className="text-sm text-muted-foreground">Most Used</p>
   404→   404→                      <p className="text-2xl font-bold">
   405→   405→                        {mostUsedTag ? mostUsedTag.name : "—"}
   406→   406→                      </p>
   407→   407→                      {mostUsedTag && (
   408→   408→                        <p className="text-xs text-muted-foreground">
   409→   409→                          {mostUsedTag.usageCount} {mostUsedTag.usageCount === 1 ? "entry" : "entries"}
   410→   410→                        </p>
   411→   411→                      )}
   412→   412→                    </div>
   413→   413→                  </div>
   414→   414→                </CardContent>
   415→   415→              </Card>
   416→   416→              <Card>
   417→   417→                <CardContent className="p-6">
   418→   418→                  <div className="flex items-center gap-3">
   419→   419→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-emerald-500/10">
   420→   420→                      <Hash className="h-5 w-5 text-emerald-500" />
   421→   421→                    </div>
   422→   422→                    <div>
   423→   423→                      <p className="text-sm text-muted-foreground">Categories</p>
   424→   424→                      <p className="text-2xl font-bold">
   425→   425→                        {new Set(tags.map((t: any) => t.category ?? "uncategorized")).size}
   426→   426→                      </p>
   427→   427→                    </div>
   428→   428→                  </div>
   429→   429→                </CardContent>
   430→   430→              </Card>
   431→   431→            </div>
   432→   432→
   433→   433→            {/* ── Quick Filters + Search ── */}
   434→   434→            <div className="space-y-3">
   435→   435→              <div className="flex items-center gap-2">
   436→   436→                <div className="relative flex-1">
   437→   437→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   438→   438→                  <Input
   439→   439→                    placeholder="Search tags..."
   440→   440→                    value={searchQuery}
   441→   441→                    onChange={(e) => setSearchQuery(e.target.value)}
   442→   442→                    className="pl-9"
   443→   443→                  />
   444→   444→                </div>
   445→   445→                {activeFilter && (
   446→   446→                  <Button
   447→   447→                    variant="outline"
   448→   448→                    size="sm"
   449→   449→                    onClick={() => setActiveFilter(null)}
   450→   450→                    className="shrink-0"

   453→   453→                    Clear filter
   454→   454→                  </Button>
   455→   455→                )}
   456→   456→              </div>
   457→   457→              <div className="flex items-center gap-2 flex-wrap">
   458→   458→                <span className="text-sm text-muted-foreground shrink-0">Quick filter:</span>
   459→   459→                {tags.slice(0, 5).map((tag: any) => (
   460→   460→                  <button
   461→   461→                    key={tag._id}
   462→   462→                    type="button"
   463→   463→                    onClick={() => setActiveFilter(activeFilter === tag._id ? null : tag._id)}
   464→   464→                    className="focus:outline-none"
