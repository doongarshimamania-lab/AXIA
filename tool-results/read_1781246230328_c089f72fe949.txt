   267→function formatDate(timestamp: number | undefined): string {
   268→  if (!timestamp) return "—";
   269→  return new Intl.DateTimeFormat("en-US", {
   270→    month: "short",
   271→    day: "numeric",
   272→    year: "numeric",
   273→  }).format(new Date(timestamp));
   274→}

   276→function daysUntilClose(timestamp: number | undefined): number | null {
   277→  if (!timestamp) return null;
   278→  const diff = timestamp - Date.now();
   279→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   280→}
