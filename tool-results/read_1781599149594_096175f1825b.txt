   106→   106→
   107→   107→export function BulkImportDialog({
   108→   108→  open,
   109→   109→  onOpenChange,
   110→   110→  tableName,
   111→   111→  onImportComplete,
   112→   112→}: BulkImportDialogProps) {
   113→   113→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   114→   114→  const workspaceId = activeWorkspaceId;
   115→   115→
   116→   116→  // ─── State ──────────────────────────────────────────────────────────────
   117→   117→  const [step, setStep] = useState<ImportStep>("upload");
   118→   118→  const [rawData, setRawData] = useState<Record<string, string>[]>([]);
   119→   119→  const [headers, setHeaders] = useState<string[]>([]);
   120→   120→  const [columnMapping, setColumnMapping] = useState<Record<string, string>>({});
   121→   121→  const [skipDuplicates, setSkipDuplicates] = useState(true);
   122→   122→  const [isDragging, setIsDragging] = useState(false);
   123→   123→  const [isExcelFile, setIsExcelFile] = useState(false);
   124→   124→  const [importProgress, setImportProgress] = useState(0);
   125→   125→  const fileInputRef = useRef<HTMLInputElement>(null);
   126→   126→
   127→   127→  // Import results
   128→   128→  const [importResults, setImportResults] = useState<{
   129→   129→    imported: number;
   130→   130→    skipped: number;
   131→   131→    errors: string[];
   132→   132→  } | null>(null);
   133→   133→
   134→   134→  // ─── Query DB fields for mapping ────────────────────────────────────────
   135→   135→  // For clients, query the bulkImport API to get available fields
   136→   136→  const hasBulkImportApi = !!(api as any).clients?.bulkImport?.getClientImportFields;
   137→   137→  const dbFields = useQuery(
   138→   138→    hasBulkImportApi && tableName === "clients" && isConvexConnected
   139→   139→      ? (api as any).clients.bulkImport.getClientImportFields
   140→   140→      : "skip",
   141→   141→    {}
   142→   142→  ) as any[] | undefined;
   143→   143→
   144→   144→  // Also fetch custom field definitions for this table
   145→   145→  const hasCustomFieldsApi = !!(api as any).customFields?.crud?.getFields;
   146→   146→  const customFields = useQuery(
   147→   147→    hasCustomFieldsApi && isConvexConnected && workspaceId
   148→   148→      ? (api as any).customFields.crud.getFields
   149→   149→      : "skip",
   150→   150→    workspaceId ? { workspaceId: workspaceId as any, tableName } : "skip"
   151→   151→  ) as any[] | undefined;
   152→   152→
   153→   153→  // ─── Mutations ──────────────────────────────────────────────────────────
   154→   154→  const importMutation = useMutation(
   155→   155→    (api as any).clients?.bulkImport?.importClients ?? null
   156→   156→  );
   157→   157→  const legacyImportMutation = useMutation(
   158→   158→    (api as any).clients?.bulkImport?.bulkImportClients ?? null
   159→   159→  );
   160→   160→
   161→   161→  // ─── Build field options for mapping dropdowns ──────────────────────────
   162→   162→  const coreFieldOptions = useMemo(() => {
   163→   163→    const options: { value: string; label: string; required?: boolean; group: string }[] = [
   164→   164→      { value: "_skip", label: "— Skip this column —", group: "skip" },
   165→   165→    ];
   166→   166→
   167→   167→    // If we got DB fields from the API, use those
   168→   168→    if (dbFields && dbFields.length > 0) {
   169→   169→      for (const f of dbFields) {
   170→   170→        options.push({
   171→   171→          value: f.key,
   172→   172→          label: f.label + (f.required ? " *" : ""),
   173→   173→          required: f.required,
   174→   174→          group: "core",
   175→   175→        });
   176→   176→      }
   177→   177→    } else {
   178→   178→      // Fallback defaults based on tableName
   179→   179→      const defaults: Record<string, { key: string; label: string; required?: boolean }[]> = {
   180→   180→        clients: [
   181→   181→          { key: "clientName", label: "Client Name", required: true },
   182→   182→          { key: "platform", label: "Platform" },
   183→   183→          { key: "hourlyRate", label: "Hourly Rate" },
   184→   184→          { key: "contractType", label: "Contract Type" },
   185→   185→          { key: "riskLevel", label: "Risk Level" },
   186→   186→          { key: "contactEmail", label: "Contact Email" },
   187→   187→          { key: "contactName", label: "Contact Name" },
   188→   188→          { key: "notes", label: "Notes" },
   189→   189→        ],
   190→   190→        deals: [
   191→   191→          { key: "title", label: "Deal Title", required: true },
   192→   192→          { key: "value", label: "Value" },
   193→   193→          { key: "stage", label: "Stage" },
   194→   194→          { key: "probability", label: "Probability" },
   195→   195→          { key: "expectedCloseDate", label: "Expected Close Date" },
   196→   196→        ],
   197→   197→        projects: [
   198→   198→          { key: "name", label: "Project Name", required: true },
   199→   199→          { key: "status", label: "Status" },
   200→   200→          { key: "budget", label: "Budget" },
   201→   201→          { key: "deadline", label: "Deadline" },
   202→   202→          { key: "startDate", label: "Start Date" },
   203→   203→        ],
   204→   204→      };
   205→   205→      const fields = defaults[tableName] || defaults.clients;
   206→   206→      for (const f of fields) {
   207→   207→        options.push({
   208→   208→          value: f.key,
   209→   209→          label: f.label + (f.required ? " *" : ""),
   210→   210→          required: f.required,
   211→   211→          group: "core",
   212→   212→        });
   213→   213→      }
   214→   214→    }
   215→   215→
   216→   216→    // Add custom fields as mapping targets
   217→   217→    if (customFields && customFields.length > 0) {
   218→   218→      for (const cf of customFields) {
   219→   219→        options.push({
   220→   220→          value: `custom:${cf.fieldName}`,
   221→   221→          label: `${cf.label} (custom)`,
   222→   222→          group: "custom",
   223→   223→        });
   224→   224→      }
   225→   225→    }
   226→   226→
   227→   227→    return options;
   228→   228→  }, [dbFields, customFields, tableName]);
   229→   229→
   230→   230→  // ─── Parse CSV ──────────────────────────────────────────────────────────
   231→   231→  const parseCSV = useCallback((text: string) => {
   232→   232→    const lines = text.trim().split("\n");
   233→   233→    if (lines.length < 2) {
   234→   234→      toast.error("CSV must have at least a header row and one data row");
   235→   235→      return;
   236→   236→    }
   237→   237→
   238→   238→    // Detect delimiter
   239→   239→    const firstLine = lines[0];
   240→   240→    const commaCount = (firstLine.match(/,/g) || []).length;
   241→   241→    const tabCount = (firstLine.match(/\t/g) || []).length;
   242→   242→    const semicolonCount = (firstLine.match(/;/g) || []).length;
   243→   243→
   244→   244→    let delimiter = ",";
   245→   245→    if (tabCount > commaCount) delimiter = "\t";
   246→   246→    else if (semicolonCount > commaCount) delimiter = ";";
   247→   247→
   248→   248→    const csvHeaders = lines[0]
   249→   249→      .split(delimiter)
   250→   250→      .map((h) => h.trim().replace(/^["']|["']$/g, ""));
   251→   251→    const rows: Record<string, string>[] = [];
   252→   252→
   253→   253→    for (let i = 1; i < lines.length; i++) {
   254→   254→      const line = lines[i].trim();
   255→   255→      if (!line) continue; // skip empty lines
   256→   256→      const values = line
   257→   257→        .split(delimiter)
   258→   258→        .map((v) => v.trim().replace(/^["']|["']$/g, ""));
   259→   259→      const row: Record<string, string> = {};
   260→   260→      csvHeaders.forEach((header, idx) => {
   261→   261→        row[header] = values[idx] || "";
   262→   262→      });
   263→   263→      rows.push(row);
   264→   264→    }
   265→   265→
   266→   266→    if (rows.length === 0) {
   267→   267→      toast.error("No data rows found in the file");
   268→   268→      return;
   269→   269→    }
   270→   270→
   271→   271→    setHeaders(csvHeaders);
   272→   272→    setRawData(rows);
   273→   273→    setIsExcelFile(false);
   274→   274→
   275→   275→    // Auto-detect column mapping
   276→   276→    const mapping: Record<string, string> = {};
   277→   277→    for (const header of csvHeaders) {
   278→   278→      const headerLower = header.toLowerCase().trim();
   279→   279→      if (AUTO_DETECT_MAP[headerLower]) {
   280→   280→        mapping[header] = AUTO_DETECT_MAP[headerLower];
   281→   281→      } else {
   282→   282→        mapping[header] = "_skip";
   283→   283→      }
   284→   284→    }
   285→   285→    setColumnMapping(mapping);
   286→   286→    setStep("mapping");
   287→   287→  }, []);
   288→   288→
   289→   289→  // ─── File handling ──────────────────────────────────────────────────────
   290→   290→  const handleFileUpload = useCallback(
   291→   291→    async (file: File) => {
   292→   292→      const ext = file.name.split(".").pop()?.toLowerCase();
   293→   293→
   294→   294→      if (ext === "csv" || ext === "tsv" || ext === "txt") {
   295→   295→        const text = await file.text();
   296→   296→        parseCSV(text);
   297→   297→      } else if (ext === "xlsx" || ext === "xls") {
   298→   298→        // Show CSV recommendation, then attempt to parse
   299→   299→        setIsExcelFile(true);
   300→   300→
   301→   301→        try {
   302→   302→          const XLSX = await import("xlsx");
   303→   303→          const arrayBuffer = await file.arrayBuffer();
   304→   304→          const workbook = XLSX.read(arrayBuffer, { type: "array" });
   305→   305→          const sheetName = workbook.SheetNames[0];
   306→   306→          const sheet = workbook.Sheets[sheetName];
   307→   307→          const jsonData = XLSX.utils.sheet_to_json<Record<string, any>>(sheet, {
   308→   308→            defval: "",
   309→   309→          });
   310→   310→
   311→   311→          if (jsonData.length === 0) {
   312→   312→            toast.error("Spreadsheet is empty");
   313→   313→            return;
   314→   314→          }
   315→   315→
   316→   316→          const csvHeaders = Object.keys(jsonData[0]);
   317→   317→          const rows = jsonData.map((row) => {
   318→   318→            const stringRow: Record<string, string> = {};
   319→   319→            for (const key of csvHeaders) {
   320→   320→              stringRow[key] = String(row[key] ?? "");
   321→   321→            }
   322→   322→            return stringRow;
   323→   323→          });
   324→   324→
   325→   325→          setHeaders(csvHeaders);
   326→   326→          setRawData(rows);
   327→   327→
   328→   328→          // Auto-detect mapping
   329→   329→          const mapping: Record<string, string> = {};
   330→   330→          for (const header of csvHeaders) {
   331→   331→            const headerLower = header.toLowerCase().trim();
   332→   332→            if (AUTO_DETECT_MAP[headerLower]) {
   333→   333→              mapping[header] = AUTO_DETECT_MAP[headerLower];
   334→   334→            } else {
   335→   335→              mapping[header] = "_skip";
   336→   336→            }
   337→   337→          }
   338→   338→          setColumnMapping(mapping);
   339→   339→          setStep("mapping");
   340→   340→        } catch (err: any) {
   341→   341→          toast.error(
   342→   342→            "Failed to parse spreadsheet. We recommend using CSV format for best results. Error: " +
   343→   343→              (err.message || "Unknown error")
   344→   344→          );
   345→   345→        }
   346→   346→      } else {
   347→   347→        toast.error("Unsupported file type. Please upload CSV, XLSX, or XLS files.");
   348→   348→      }
   349→   349→    },
   350→   350→    [parseCSV]
   351→   351→  );
   352→   352→
   353→   353→  const handleDrop = useCallback(
   354→   354→    (e: React.DragEvent) => {
   355→   355→      e.preventDefault();
   356→   356→      setIsDragging(false);
   357→   357→      const file = e.dataTransfer.files[0];
   358→   358→      if (file) handleFileUpload(file);
   359→   359→    },
   360→   360→    [handleFileUpload]
   361→   361→  );
   362→   362→
   363→   363→  const handleFileInput = useCallback(
   364→   364→    (e: React.ChangeEvent<HTMLInputElement>) => {
   365→   365→      const file = e.target.files?.[0];
   366→   366→      if (file) handleFileUpload(file);
   367→   367→    },
   368→   368→    [handleFileUpload]
   369→   369→  );
   370→   370→
   371→   371→  // ─── Import ─────────────────────────────────────────────────────────────
   372→   372→  const handleImport = async () => {
   373→   373→    if (!workspaceId) {
   374→   374→      toast.error("No workspace selected");
   375→   375→      return;
   376→   376→    }
   377→   377→
   378→   378→    // Validate: at least one required field must be mapped
   379→   379→    const requiredFields = coreFieldOptions.filter((o) => o.required);
   380→   380→    const hasRequiredMapping = requiredFields.some((rf) =>
   381→   381→      Object.values(columnMapping).some((v) => v === rf.value)
   382→   382→    );
   383→   383→    if (!hasRequiredMapping) {
   384→   384→      toast.error("You must map at least one column to a required field");
   385→   385→      return;
   386→   386→    }
   387→   387→
   388→   388→    setStep("importing");
   389→   389→    setImportProgress(0);
   390→   390→
   391→   391→    try {
   392→   392→      // Map raw data using the column mapping
   393→   393→      const records = rawData.map((row) => {
   394→   394→        const mapped: Record<string, any> = {};
   395→   395→        for (const [csvCol, targetField] of Object.entries(columnMapping)) {
   396→   396→          if (targetField === "_skip") continue;
   397→   397→          const value = row[csvCol];
   398→   398→          if (value !== undefined && value !== "") {
   399→   399→            mapped[targetField] = value;
   400→   400→          }
   401→   401→        }
   402→   402→        return mapped;
   403→   403→      });
   404→   404→
   405→   405→      // Simulate progress
   406→   406→      setImportProgress(20);
   407→   407→
   408→   408→      let results;
   409→   409→      if (importMutation) {
   410→   410→        results = await importMutation({
   411→   411→          workspaceId,
   412→   412→          records,
   413→   413→          fieldMapping: columnMapping,
   414→   414→          skipDuplicates,
   415→   415→        });
   416→   416→      } else if (legacyImportMutation && tableName === "clients") {
   417→   417→        // Legacy format: transform records to bulkImportClients format
   418→   418→        setImportProgress(40);
   419→   419→        const clients = records.map((r) => ({
   420→   420→          clientName: r.clientName || "",
   421→   421→          platform: r.platform as any,
   422→   422→          hourlyRate: r.hourlyRate ? Number(r.hourlyRate) : undefined,
   423→   423→          contractType: r.contractType as any,
   424→   424→          riskLevel: r.riskLevel as any,
   425→   425→          contactEmail: r.contactEmail,
   426→   426→          contactName: r.contactName,
   427→   427→          notes: r.notes,
   428→   428→        }));
   429→   429→        results = await legacyImportMutation({
   430→   430→          workspaceId,
   431→   431→          clients,
   432→   432→          skipDuplicates,
   433→   433→        });
   434→   434→      } else {
   435→   435→        // No API available - simulate
   436→   436→        setImportProgress(60);
   437→   437→        await new Promise((resolve) => setTimeout(resolve, 800));
   438→   438→        results = { imported: records.length, skipped: 0, errors: [] };
   439→   439→      }
   440→   440→
   441→   441→      setImportProgress(100);
   442→   442→
   443→   443→      setImportResults({
   444→   444→        imported: (results as any)?.imported ?? (results as any)?.created ?? 0,
   445→   445→        skipped: (results as any)?.skipped ?? 0,
   446→   446→        errors: (results as any)?.errors ?? [],
   447→   447→      });
   448→   448→      setStep("results");
   449→   449→
   450→   450→      // Notify parent
   451→   451→      onImportComplete();
   452→   452→    } catch (err: any) {
   453→   453→      toast.error("Import failed: " + (err.message || "Unknown error"));
   454→   454→      setStep("mapping");
   455→   455→    }
   456→   456→  };
   457→   457→
   458→   458→  // ─── Reset ──────────────────────────────────────────────────────────────
   459→   459→  const resetDialog = () => {
   460→   460→    setStep("upload");
   461→   461→    setRawData([]);
   462→   462→    setHeaders([]);
   463→   463→    setColumnMapping({});
   464→   464→    setImportResults(null);
   465→   465→    setIsExcelFile(false);
   466→   466→    setImportProgress(0);
   467→   467→    if (fileInputRef.current) {
   468→   468→      fileInputRef.current.value = "";
   469→   469→    }
   470→   470→  };
   471→   471→
   472→   472→  const handleClose = (open: boolean) => {
   473→   473→    if (!open) {
   474→   474→      resetDialog();
   475→   475→    }
   476→   476→    onOpenChange(open);
   477→   477→  };
   478→   478→
   479→   479→  // ─── Preview rows (first 3) ─────────────────────────────────────────────
   480→   480→  const previewRows = rawData.slice(0, 3);
   481→   481→  const mappedHeaders = headers.filter(
   482→   482→    (h) => columnMapping[h] && columnMapping[h] !== "_skip"
   483→   483→  );
   484→   484→
   485→   485→  // ─── Display name for table ─────────────────────────────────────────────
   486→   486→  const tableDisplayName =
   487→   487→    tableName.charAt(0).toUpperCase() + tableName.slice(1);
   488→   488→
   489→   489→  return (
   490→   490→    <Dialog open={open} onOpenChange={handleClose}>
   491→   491→      <DialogContent className="sm:max-w-3xl max-h-[85vh] overflow-y-auto">
   492→   492→        <DialogHeader>
   493→   493→          <DialogTitle className="flex items-center gap-2">
   494→   494→            <Upload className="w-5 h-5 text-purple-500" />
   495→   495→            Bulk Import {tableDisplayName}
