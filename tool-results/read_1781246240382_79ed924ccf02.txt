   215→const TAB_FILTERS = [
   216→  { key: "all", label: "All" },
   217→  { key: "active", label: "Active" },

   222→type TabFilter = (typeof TAB_FILTERS)[number]["key"];
   223→
   224→function filterReports(reports: DisputeReport[], tab: TabFilter): DisputeReport[] {
   225→  switch (tab) {
   226→    case "all":
   227→      return reports;
   228→    case "active":
   229→      return reports.filter((r) => r.status === "generated" || r.status === "sent" || r.status === "viewed");
   230→    case "resolved":
   231→      return reports.filter((r) => r.status === "resolved");
   232→    case "appealed":
   233→      return reports.filter((r) => r.status === "appealed");
   234→    default:
   235→      return reports;
   236→  }
   237→}
   238→
   239→function formatDate(ts: number): string {
   240→  return new Date(ts).toLocaleDateString("en-US", {
   241→    month: "short",
   242→    day: "numeric",
   243→    year: "numeric",
   244→  });
   245→}
   246→
   247→function formatCurrency(amount: number): string {
   248→  return new Intl.NumberFormat("en-US", {
   249→    style: "currency",
   250→    currency: "USD",
   251→    minimumFractionDigits: 0,
   252→    maximumFractionDigits: 2,
   253→  }).format(amount);
   254→}
