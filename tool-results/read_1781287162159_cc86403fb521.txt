    90→function formatDate(ts: number): string {
    91→  return new Date(ts).toLocaleDateString("en-US", {

    93→    day: "numeric",
    94→    year: "numeric",

    98→function timeAgo(ts: number): string {
    99→  const diff = Date.now() - ts;
   100→  const minutes = Math.floor(diff / 60000);
   101→  const hours = Math.floor(minutes / 60);
   102→  const days = Math.floor(hours / 24);
   103→  if (days > 0) return `${days}d ago`;
   104→  if (hours > 0) return `${hours}h ago`;
   105→  return `${minutes}m ago`;
   106→}

   158→function calculateScopeHealth(scope: ScopeDefinition, changeOrders: ScopeChangeOrder[]) {
   159→  const scopeChanges = changeOrders.filter(co => co.scopeId === scope._id);
   160→  const pendingChanges = scopeChanges.filter(co => co.status === "pending" || co.status === "auto_generated");
   161→  const revisionUsage = scope.revisionLimit > 0 ? (scope.revisionCount / scope.revisionLimit) * 100 : 0;
   162→  const totalHoursAdded = scopeChanges.reduce((sum, co) => sum + co.impact.hoursAdded, 0);
   163→  const totalCostImpact = scopeChanges.reduce((sum, co) => sum + co.impact.costImpact, 0);
   164→  const hoursOverrun = scope.totalEstimatedHours ? (totalHoursAdded / scope.totalEstimatedHours) * 100 : 0;
   165→
   166→  let riskLevel: "low" | "medium" | "high" = "low";
   167→  if (revisionUsage >= 100 || hoursOverrun > 30 || pendingChanges.length >= 3) riskLevel = "high";
   168→  else if (revisionUsage >= 66 || hoursOverrun > 15 || pendingChanges.length >= 2) riskLevel = "medium";
   169→
   170→  const completedDeliverables = scope.deliverables.filter(d => d.status === "completed").length;
   171→  const progressPercent = scope.deliverables.length > 0 ? (completedDeliverables / scope.deliverables.length) * 100 : 0;
   172→
   173→  return {
   174→    riskLevel,
   175→    revisionUsage,
   176→    totalHoursAdded,
   177→    totalCostImpact,
   178→    hoursOverrun,
   179→    pendingChanges: pendingChanges.length,
   180→    completedDeliverables,
   181→    totalDeliverables: scope.deliverables.length,
   182→    progressPercent,
   183→  };
   184→}

   189→  const barColor = health.riskLevel === "high" ? "bg-red-500" : health.riskLevel === "medium" ? "bg-amber-500" : "bg-green-500";
   190→  return (
   191→    <div className="space-y-2">
   192→      <div className="flex items-center justify-between text-xs">
   193→        <span className="text-muted-foreground">Scope Health</span>
   194→        <span className={`font-semibold ${health.riskLevel === "high" ? "text-red-600 dark:text-red-400" : health.riskLevel === "medium" ? "text-amber-600 dark:text-amber-400" : "text-green-600 dark:text-green-400"}`}>
   195→          {health.riskLevel === "high" ? "High Risk" : health.riskLevel === "medium" ? "Moderate Risk" : "Healthy"}
   196→        </span>
   197→      </div>
   198→      <div className="h-2 bg-muted rounded-full overflow-hidden">
   199→        <div className={`h-full rounded-full transition-all ${barColor}`} style={{ width: `${Math.max(5, 100 - health.hoursOverrun - health.revisionUsage * 0.3)}%` }} />
   200→      </div>
   201→    </div>
   202→  );
   203→}

   434→// ─── ScopeCard with its own change orders query ─────────────────────────────
   435→
   436→function ScopeCardContainer({
   437→  scope,
   438→  isSelected,
   439→  onSelect,
   440→  onDelete,
   441→  deleting,
   442→  onMetricsReady,
   443→}: {
   444→  scope: ScopeDefinition;
   445→  isSelected: boolean;
   446→  onSelect: () => void;
   447→  onDelete: () => void;
   448→  deleting: boolean;
   449→  onMetricsReady: (scopeId: string, metrics: { pendingChanges: number; costAtRisk: number; hoursAtRisk: number }) => void;
   450→}) {
   451→  const changeOrders = useQuery(api.scope.crud.getChangeOrders, { scopeId: scope._id as any });
   452→
   453→  const prevMetricsRef = useRef<string>("");
   454→
   455→  useEffect(() => {
   456→    if (changeOrders && Array.isArray(changeOrders)) {
   457→      const pending = (changeOrders as ScopeChangeOrder[]).filter(co => co.status === "pending" || co.status === "auto_generated");
   458→      const costAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.costImpact, 0);
   459→      const hoursAtRisk = pending.reduce((sum: number, co: ScopeChangeOrder) => sum + co.impact.hoursAdded, 0);
   460→      const key = `${pending.length}-${costAtRisk}-${hoursAtRisk}`;
   461→      if (key !== prevMetricsRef.current) {
   462→        prevMetricsRef.current = key;
   463→        onMetricsReady(scope._id, { pendingChanges: pending.length, costAtRisk, hoursAtRisk });
   464→      }
   465→    }
   466→  }, [changeOrders, scope._id, onMetricsReady]);
   467→
   468→  return (
   469→    <ScopeCard
   470→      scope={scope}
   471→      changeOrders={(changeOrders as ScopeChangeOrder[]) ?? []}
   472→      isSelected={isSelected}
   473→      onSelect={onSelect}
   474→      onDelete={onDelete}
   475→      deleting={deleting}
   476→    />
   477→  );
