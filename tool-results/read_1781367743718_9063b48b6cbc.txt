   118→export default function TeamManagement() {
   119→  const { activeWorkspaceId, isTeamMode, isOwner, canManageTeam, activeWorkspace } = useWorkspaceContext();
   120→
   121→  // ─── Convex Queries ────────────────────────────────────────────────────────
   122→  const hasRealWorkspaceId = activeWorkspaceId && !activeWorkspaceId.startsWith("ws_");
   123→
   124→  const convexMembers = useQuery(
   125→    hasRealWorkspaceId ? api.workspaces.members.getMembers : "skip",
   126→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   127→  ) as any[] | undefined;
   128→
   129→  const convexStats = useQuery(
   130→    hasRealWorkspaceId ? api.workspaces.crud.getWorkspaceStats : "skip",
   131→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   132→  ) as any | undefined;
   133→
   134→  const convexInvitations = useQuery(
   135→    hasRealWorkspaceId ? api.workspaces.invitations.getInvitations : "skip",
   136→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any, status: "pending" as const } : "skip"
   137→  ) as any[] | undefined;
   138→
   139→  // ─── Teams queries ─────────────────────────────────────────────────────────
   140→  const teamsApi = (api as any).teams?.crud;
   141→  const hasTeamsApi = !!teamsApi?.getTeams;
   142→
   143→  const convexTeams = useQuery(
   144→    hasTeamsApi && hasRealWorkspaceId ? teamsApi.getTeams : "skip",
   145→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   146→  ) as any[] | undefined;
   147→
   148→  // ─── Team Members per team ──────────────────────────────────────────────────
   149→  const hasTeamMembersApi = !!teamsApi?.getTeamMembers;
   150→
   151→  // ─── Mutations ─────────────────────────────────────────────────────────────
   152→  const inviteMemberMutation = useMutation(api.workspaces.invitations.createInvitation);
   153→  const removeMemberMutation = useMutation(api.workspaces.members.removeMember);
   154→  const updateRoleMutation = useMutation(api.workspaces.members.updateMemberRole);
   155→  const cancelInvitationMutation = useMutation(api.workspaces.invitations.cancelInvitation);
   156→
   157→  // Team mutations
   158→  const createTeamMutation = useMutation(hasTeamsApi ? teamsApi.createTeam : null);
   159→  const updateTeamMutation = useMutation(hasTeamsApi ? teamsApi.updateTeam : null);
   160→  const deleteTeamMutation = useMutation(hasTeamsApi ? teamsApi.deleteTeam : null);
   161→  const addTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.addTeamMember : null);
   162→  const removeTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.removeTeamMember : null);
   163→
   164→  // ─── State ─────────────────────────────────────────────────────────────────
   165→  const [activeTab, setActiveTab] = useState<string>("members");
   166→
   167→  // Invite dialog
   168→  const [showInviteDialog, setShowInviteDialog] = useState(false);
   169→  const [inviteEmail, setInviteEmail] = useState("");
   170→  const [inviteRole, setInviteRole] = useState<"manager" | "member">("member");
   171→  const [inviteMessage, setInviteMessage] = useState("");
   172→  const [isInviting, setIsInviting] = useState(false);
   173→
   174→  // Remove/change role
   175→  const [memberToRemove, setMemberToRemove] = useState<any>(null);
   176→  const [memberToChangeRole, setMemberToChangeRole] = useState<any>(null);
   177→  const [newRole, setNewRole] = useState<"manager" | "member">("member");
   178→
   179→  // Team CRUD state
   180→  const [showCreateTeamDialog, setShowCreateTeamDialog] = useState(false);
   181→  const [teamName, setTeamName] = useState("");
   182→  const [teamColor, setTeamColor] = useState(TEAM_COLORS[0]);
   183→  const [teamDescription, setTeamDescription] = useState("");
   184→  const [teamIsCrossTeam, setTeamIsCrossTeam] = useState(false);
   185→  const [isCreatingTeam, setIsCreatingTeam] = useState(false);
   186→
   187→  // Edit team
   188→  const [editingTeam, setEditingTeam] = useState<any>(null);
   189→  const [editTeamName, setEditTeamName] = useState("");
   190→  const [editTeamColor, setEditTeamColor] = useState("");
   191→  const [editTeamDescription, setEditTeamDescription] = useState("");
   192→  const [isUpdatingTeam, setIsUpdatingTeam] = useState(false);
   193→
   194→  // Delete team
   195→  const [teamToDelete, setTeamToDelete] = useState<any>(null);
   196→  const [isDeletingTeam, setIsDeletingTeam] = useState(false);
   197→
   198→  // Add member to team
   199→  const [teamToAddMember, setTeamToAddMember] = useState<any>(null);
   200→  const [addMemberUserId, setAddMemberUserId] = useState("");
   201→  const [addMemberRole, setAddMemberRole] = useState<"lead" | "member">("member");
   202→  const [isAddingTeamMember, setIsAddingTeamMember] = useState(false);
   203→
   204→  // Remove member from team
   205→  const [teamMemberToRemove, setTeamMemberToRemove] = useState<any>(null);
   206→  const [teamMemberRemoveTeamId, setTeamMemberRemoveTeamId] = useState<string>("");
   207→
   208→  // ─── Map Convex data to UI shape ───────────────────────────────────────────
   209→  const members = useMemo(() => {
   210→    if (convexMembers && convexMembers.length > 0) {
   211→      return convexMembers.map((m: any) => ({
   212→        _id: m._id,
   213→        userId: m.userId,
   214→        name: m.userName ?? m.userEmail ?? "Unknown",
   215→        displayName: m.userName ?? m.userEmail ?? "Unknown",
   216→        email: m.userEmail ?? "",
   217→        image: m.userImage ?? "",
   218→        role: m.role,
   219→        status: m.status,
   220→        joinedAt: m.joinedAt,
   221→        lastActiveAt: m.lastActiveAt ?? null,
   222→        projectsAssigned: 0,
   223→        hoursThisWeek: 0,
   224→      }));
   225→    }
   226→    return [];
   227→  }, [convexMembers]);
   228→
   229→  const teams = useMemo(() => {
   230→    if (convexTeams && convexTeams.length > 0) {
   231→      return convexTeams;
   232→    }
   233→    return [];
   234→  }, [convexTeams]);
   235→
   236→  const stats = useMemo(() => {
   237→    if (convexStats) {
   238→      return {
   239→        memberCount: convexStats.memberCount ?? 0,
   240→        clientCount: convexStats.clientCount ?? 0,
   241→        activeProjectCount: convexStats.activeProjectCount ?? 0,
   242→        pendingInvoiceCount: 0,
   243→        totalRevenue: 0,
   244→        totalHoursThisWeek: 0,
   245→        protectionScore: 0,
   246→      };
   247→    }
   248→    return null;
   249→  }, [convexStats]);
   250→
   251→  const invitations = useMemo(() => {
   252→    if (convexInvitations && convexInvitations.length > 0) {
   253→      return convexInvitations.map((inv: any) => ({
   254→        _id: inv._id,
   255→        name: inv.email,
   256→        displayName: inv.email,
   257→        email: inv.email,
   258→        role: inv.role,
   259→        invitedAt: inv.createdAt,
   260→        invitedBy: inv.inviterName,
   261→      }));
   262→    }
   263→    return [];
   264→  }, [convexInvitations]);
   265→
   266→  const { isDisconnected } = useConvexConnectionState();
   267→  const isLoading = hasRealWorkspaceId && (convexMembers === undefined);
   268→  const isDemoMode = !hasRealWorkspaceId;
   269→  const loadingTimedOut = useQueryTimeout(isLoading, 3000);
   270→  const showLoading = isLoading && !loadingTimedOut && !isDisconnected;
   271→
   272→  // ─── Handlers ──────────────────────────────────────────────────────────────
   273→  const handleInvite = async () => {
   274→    if (!inviteEmail.trim() || !activeWorkspaceId) return;
   275→    setIsInviting(true);
   276→    try {
   277→      if (hasRealWorkspaceId) {
   278→        await inviteMemberMutation({
   279→          workspaceId: activeWorkspaceId as any,
   280→          email: inviteEmail.trim(),
   281→          role: inviteRole,
   282→        });
   283→      }
   284→      toast.success(`Invitation sent to ${inviteEmail}`);
   285→      setInviteEmail("");
   286→      setInviteMessage("");
   287→      setShowInviteDialog(false);
   288→    } catch (e: any) {
   289→      toast.error(e.message || "Failed to send invitation");
   290→    } finally {
   291→      setIsInviting(false);
   292→    }
   293→  };
   294→
   295→  const handleRemoveMember = async () => {
   296→    if (!memberToRemove) return;
   297→    try {
   298→      await removeMemberMutation({ memberId: memberToRemove._id });
   299→      toast.success(`${memberToRemove.name} removed from workspace`);
   300→      setMemberToRemove(null);
   301→    } catch (e: any) {
   302→      toast.error(e.message || "Failed to remove member");
   303→    }
   304→  };
   305→
   306→  const handleChangeRole = async () => {
   307→    if (!memberToChangeRole) return;
   308→    try {
   309→      await updateRoleMutation({ memberId: memberToChangeRole._id, role: newRole });
   310→      toast.success(`Role updated to ${newRole}`);
   311→      setMemberToChangeRole(null);
   312→    } catch (e: any) {
   313→      toast.error(e.message || "Failed to update role");
   314→    }
   315→  };
   316→
   317→  const handleCancelInvitation = async (invitation: any) => {
   318→    try {
   319→      await cancelInvitationMutation({ invitationId: invitation._id });
   320→      toast.success(`Invitation to ${invitation.email || invitation.name} cancelled`);
   321→    } catch (e: any) {
   322→      toast.error(e.message || "Failed to cancel invitation");
   323→    }
   324→  };
   325→
   326→  // ─── Team CRUD Handlers ────────────────────────────────────────────────────
   327→  const handleCreateTeam = async () => {
   328→    if (!teamName.trim() || !activeWorkspaceId) return;
   329→    setIsCreatingTeam(true);
   330→    try {
   331→      if (createTeamMutation && hasRealWorkspaceId) {
   332→        await createTeamMutation({
   333→          workspaceId: activeWorkspaceId as any,
   334→          name: teamName.trim(),
   335→          color: teamColor,
   336→          description: teamDescription.trim() || undefined,
   337→          isCrossTeam: teamIsCrossTeam || undefined,
   338→        });
   339→      }
   340→      toast.success(`Team "${teamName}" created!`);
   341→      setShowCreateTeamDialog(false);
   342→      setTeamName("");
   343→      setTeamColor(TEAM_COLORS[0]);
   344→      setTeamDescription("");
   345→      setTeamIsCrossTeam(false);
   346→    } catch (e: any) {
   347→      toast.error(e.message || "Failed to create team");
   348→    } finally {
   349→      setIsCreatingTeam(false);
   350→    }
   351→  };
   352→
   353→  const handleUpdateTeam = async () => {
   354→    if (!editingTeam) return;
   355→    setIsUpdatingTeam(true);
   356→    try {
   357→      if (updateTeamMutation) {
   358→        await updateTeamMutation({
   359→          teamId: editingTeam._id,
   360→          name: editTeamName.trim() || undefined,
   361→          color: editTeamColor || undefined,
   362→          description: editTeamDescription.trim() || undefined,
   363→        });
   364→      }
   365→      toast.success("Team updated!");
   366→      setEditingTeam(null);
   367→    } catch (e: any) {
   368→      toast.error(e.message || "Failed to update team");
   369→    } finally {
   370→      setIsUpdatingTeam(false);
   371→    }
   372→  };
   373→
   374→  const handleDeleteTeam = async () => {
   375→    if (!teamToDelete) return;
   376→    setIsDeletingTeam(true);
   377→    try {
   378→      if (deleteTeamMutation) {
   379→        await deleteTeamMutation({ teamId: teamToDelete._id });
   380→      }
   381→      toast.success(`Team "${teamToDelete.name}" deleted`);
   382→      setTeamToDelete(null);
   383→    } catch (e: any) {
   384→      toast.error(e.message || "Failed to delete team");
   385→    } finally {
   386→      setIsDeletingTeam(false);
   387→    }
   388→  };
   389→
   390→  const handleAddTeamMember = async () => {
   391→    if (!teamToAddMember || !addMemberUserId) return;
   392→    setIsAddingTeamMember(true);
   393→    try {
   394→      if (addTeamMemberMutation) {
   395→        await addTeamMemberMutation({
   396→          teamId: teamToAddMember._id,
   397→          userId: addMemberUserId as any,
   398→          role: addMemberRole,
   399→        });
   400→      }
   401→      toast.success("Member added to team!");
   402→      setTeamToAddMember(null);
   403→      setAddMemberUserId("");
   404→      setAddMemberRole("member");
   405→    } catch (e: any) {
   406→      toast.error(e.message || "Failed to add member to team");
   407→    } finally {
   408→      setIsAddingTeamMember(false);
   409→    }
   410→  };
   411→
   412→  const handleRemoveTeamMember = async () => {
   413→    if (!teamMemberToRemove || !teamMemberRemoveTeamId) return;
   414→    try {
   415→      if (removeTeamMemberMutation) {
   416→        await removeTeamMemberMutation({
   417→          teamId: teamMemberRemoveTeamId as any,
   418→          userId: teamMemberToRemove.userId as any,
   419→        });
   420→      }
   421→      toast.success("Member removed from team");
   422→      setTeamMemberToRemove(null);
   423→      setTeamMemberRemoveTeamId("");
   424→    } catch (e: any) {
   425→      toast.error(e.message || "Failed to remove member from team");
   426→    }
   427→  };
   428→
   429→  // ─── Not in team mode: Show upgrade prompt ────────────────
   430→  if (!isTeamMode) {
   431→    return <SoloModePrompt />;
   432→  }
   433→
   434→  const activeMembers = members.filter((m: any) => m.status === "active");
   435→  const ownerCount = activeMembers.filter((m: any) => m.role === "owner").length;
   436→  const managerCount = activeMembers.filter((m: any) => m.role === "manager").length;
   437→  const memberCount = activeMembers.filter((m: any) => m.role === "member").length;
   438→
   439→  const roleIcon = (role: string) => {
   440→    switch (role) {
   441→      case "owner": return <Crown className="w-3.5 h-3.5 text-amber-500" />;
   442→      case "manager": return <Shield className="w-3.5 h-3.5 text-blue-500" />;
   443→      case "lead": return <Shield className="w-3.5 h-3.5 text-emerald-500" />;
   444→      case "member": return <Eye className="w-3.5 h-3.5 text-gray-500" />;
   445→      default: return null;
   446→    }
   447→  };
   448→
   449→  const roleBadge = (role: string) => {
   450→    const variants: Record<string, any> = {
   451→      owner: { className: "bg-amber-500/10 text-amber-600 border-amber-500/20", label: "Owner" },
   452→      manager: { className: "bg-blue-500/10 text-blue-600 border-blue-500/20", label: "Manager" },
   453→      lead: { className: "bg-emerald-500/10 text-emerald-600 border-emerald-500/20", label: "Lead" },
   454→      member: { className: "bg-gray-500/10 text-gray-600 border-gray-500/20", label: "Member" },
   455→    };
   456→    const v = variants[role] || variants.member;
   457→    return (
   458→      <Badge variant="outline" className={`gap-1 ${v.className}`}>
   459→        {roleIcon(role)}
   460→        {v.label}
   461→      </Badge>
   462→    );
   463→  };
   464→
   465→  const formatLastActive = (timestamp: number | null) => {
   466→    if (!timestamp) return "Never";
   467→    const diffMs = Date.now() - timestamp;
   468→    const diffMins = Math.floor(diffMs / 60000);
   469→    if (diffMins < 5) return "Just now";
   470→    if (diffMins < 60) return `${diffMins}m ago`;
   471→    const diffHours = Math.floor(diffMins / 60);
   472→    if (diffHours < 24) return `${diffHours}h ago`;
   473→    const diffDays = Math.floor(diffHours / 24);
   474→    return `${diffDays}d ago`;
   475→  };
   476→
   477→  const formatJoined = (timestamp: number | null) => {
   478→    if (!timestamp) return "";
   479→    const diffMs = Date.now() - timestamp;
   480→    const diffDays = Math.floor(diffMs / (24 * 60 * 60 * 1000));
   481→    if (diffDays < 1) return "Joined today";
   482→    if (diffDays < 30) return `Joined ${diffDays}d ago`;
   483→    const diffMonths = Math.floor(diffDays / 30);
   484→    return `Joined ${diffMonths}mo ago`;
   485→  };
   486→
   487→  return (
   488→    <PageLayout>
   489→      <div className="w-full space-y-6">
   490→        {/* Header */}
   491→        <div className="flex items-center justify-between">
   492→          <div>
   493→            <h1 className="text-3xl font-bold text-foreground mb-2 flex items-center gap-3">
   494→              <Building2 className="w-8 h-8 text-purple-500" />
   495→              {activeWorkspace?.name || "Team Workspace"}
   496→            </h1>
   497→            <p className="text-muted-foreground">
   498→              Manage your team members, teams, and invitations
   499→            </p>
   500→          </div>
   501→          {canManageTeam && (
   502→            <div className="flex gap-2">
   503→              <Button className="gap-2" onClick={() => setShowInviteDialog(true)}>
   504→                <UserPlus className="w-4 h-4" />
   505→                Invite Member
   506→              </Button>
   507→            </div>
   508→          )}
   509→        </div>
   510→
   511→        {/* Demo Mode Banner */}
   512→        {isDemoMode && <DemoModeBanner />}
   513→
   514→        {/* Loading / Error State */}
   515→        {showLoading ? (
   516→          <StatsSkeleton />
   517→        ) : (
   518→          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
   519→            <Card className="p-5">
   520→              <div className="flex items-center gap-3">
   521→                <div className="p-2.5 bg-purple-500/10 rounded-lg"><Users className="w-5 h-5 text-purple-500" /></div>
   522→                <div>
   523→                  <p className="text-sm text-muted-foreground">Members</p>
   524→                  <p className="text-2xl font-bold">{stats?.memberCount ?? activeMembers.length}</p>
   525→                </div>
   526→              </div>
   527→            </Card>
   528→            <Card className="p-5">
   529→              <div className="flex items-center gap-3">
   530→                <div className="p-2.5 bg-emerald-500/10 rounded-lg"><Shield className="w-5 h-5 text-emerald-500" /></div>
   531→                <div>
   532→                  <p className="text-sm text-muted-foreground">Teams</p>
   533→                  <p className="text-2xl font-bold">{teams.length}</p>
   534→                </div>
   535→              </div>
