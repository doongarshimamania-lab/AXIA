    93→function getTierLevel(tier: string): number {
    94→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
    95→  return levels[tier] ?? 0;
    96→}
    97→
    98→function hasTierAccess(userTier: string, requiredTier: string): boolean {
    99→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   100→}

   154→  selected,
   155→  onSelect,

   158→  formatKey: ExportFormat;

   162→}) {
   163→  const config = FORMAT_CONFIG[formatKey];

   166→    <button
   167→      type="button"
   168→      onClick={locked ? undefined : onSelect}
   169→      disabled={locked}
   170→      className={`
   171→        relative flex flex-col items-center gap-2 rounded-xl border-2 p-4 transition-all duration-200
   172→        ${selected && !locked

   175→            ? "border-muted bg-muted/30 opacity-60 cursor-not-allowed"

   178→      `}

   189→          {config.tierRequired}+
   190→        </span>
   191→      )}
   192→      {selected && !locked && (
   193→        <div className="absolute -top-1.5 -right-1.5 rounded-full bg-primary p-0.5">
   194→          <CheckCircle2 className="h-3.5 w-3.5 text-primary-foreground" />
   195→        </div>

   238→// --- Main Component ---
   239→export default function EvidenceExport() {
   240→  const { tier } = useSubscriptionTier();
   241→  const { isAuthenticated } = useConvexAuth();
   242→
   243→  // ─── Convex Queries ────────────────────────────────────────────────────────
   244→  const evidenceData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   245→    view: "date" as const,
   246→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   247→    endDate: Date.now(),
   248→  }) as any | undefined;
   249→
   250→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   251→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   252→
   253→  const isLoading = evidenceData === undefined || clients === undefined;
   254→
   255→  // ─── Derived Data ──────────────────────────────────────────────────────────
   256→
   257→  // Build evidence type options from real data
   258→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   259→    if (!evidenceData) return [];
   260→    const items = evidenceData.evidenceItems ?? [];
   261→    const typeCounts: Record<string, number> = {};
   262→    const typeLabels: Record<string, string> = {
   263→      screenshot_ref: "Screenshots",
   264→      memo: "Work Memos",
   265→      url: "URL Visits",
   266→      keyboard: "Keyboard Activity",
   267→      mouse: "Mouse Activity",
   268→      platform_status: "Platform Status",
   269→    };
   270→    const typeIcons: Record<string, React.ReactNode> = {
   271→      screenshot_ref: <Eye className="h-4 w-4" />,
   272→      memo: <FileText className="h-4 w-4" />,
   273→      url: <FileJson className="h-4 w-4" />,
   274→      keyboard: <FileText className="h-4 w-4" />,
   275→      mouse: <FileText className="h-4 w-4" />,
   276→      platform_status: <Shield className="h-4 w-4" />,
   277→    };
   278→
   279→    for (const item of items) {
   280→      const kind = item.type ?? "unknown";
   281→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   282→    }
   283→
   284→    return Object.entries(typeCounts).map(([id, count]) => ({
   285→      id,
   286→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   287→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   288→      count,
   289→    }));
   290→  }, [evidenceData]);
   291→
   292→  // Build project options from real scopes
   293→  const projectOptions = useMemo(() => {
   294→    const all = [{ id: "all", label: "All Projects" }];
   295→    if (scopeDefinitions) {
   296→      for (const scope of scopeDefinitions) {
   297→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   298→      }
   299→    }
   300→    return all;
   301→  }, [scopeDefinitions]);
   302→
   303→  // Build client options from real clients
   304→  const clientOptions = useMemo(() => {
   305→    const all = [{ id: "all", label: "All Clients" }];
   306→    if (clients) {
   307→      for (const client of clients) {
   308→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   309→      }
   310→    }
   311→    return all;
   312→  }, [clients]);
   313→
   314→  // Compliance score from real data
   315→  const complianceScore = evidenceData?.healthScore?.score ?? 0;
   316→  const totalItems = evidenceData?.totalCount ?? 0;
   317→  const disputeSuccessRate = evidenceData?.disputeSuccessRate ?? 0;
   318→
   319→  // State
   320→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   321→  const [dateFrom, setDateFrom] = useState(() => {
   322→    const d = new Date();
   323→    d.setMonth(d.getMonth() - 3);
   324→    return d.toISOString().split("T")[0];
   325→  });
   326→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   327→  const [selectedProject, setSelectedProject] = useState("all");
   328→  const [selectedClient, setSelectedClient] = useState("all");
   329→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   330→  const [isExporting, setIsExporting] = useState(false);
   331→  const [previewOpen, setPreviewOpen] = useState(false);
   332→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   333→
   334→  // Auto-select all evidence types when they load
   335→  useMemo(() => {
   336→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   337→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   338→    }
   339→  }, [evidenceTypes]);
   340→
   341→  // Derived
   342→  const totalEvidenceItems = useMemo(
   343→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   344→    [evidenceTypes, selectedEvidenceTypes]
   345→  );
   346→
   347→  const isFormatLocked = (format: ExportFormat): boolean => {
   348→    const requiredTier = FORMAT_CONFIG[format].tierRequired;
   349→    return !hasTierAccess(tier, requiredTier);
   350→  };
   351→
   352→  const toggleEvidenceType = (id: string) => {
   353→    setSelectedEvidenceTypes((prev) =>
   354→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   355→    );
   356→  };
   357→
   358→  const handleExport = async () => {
   359→    if (selectedEvidenceTypes.length === 0) {
   360→      toast.error("No evidence types selected", {
   361→        description: "Select at least one evidence type to export.",
   362→      });
   363→      return;
   364→    }
   365→
   366→    if (isFormatLocked(selectedFormat)) {
   367→      toast.error("Upgrade required", {
   368→        description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.`,
   369→      });
   370→      return;
   371→    }
   372→
   373→    setIsExporting(true);
   374→
   375→    try {
   376→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   377→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   378→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   379→
   380→      exportEvidence(selectedFormat, {
   381→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   382→        dateRange: `${dateFrom} to ${dateTo}`,
   383→        project: selectedProjectLabel,
   384→        client: selectedClientLabel,
   385→        complianceScore,
   386→        totalItems: totalEvidenceItems,
   387→      });
   388→
   389→      // Add to recent exports
   390→      const newExport: RecentExport = {
   391→        id: `exp_${Date.now()}`,
   392→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   393→        format: selectedFormat,
   394→        status: "completed",
   395→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   396→        date: new Date().toISOString().split("T")[0],
   397→        itemCount: totalEvidenceItems,
   398→        complianceVerified: hasTierAccess(tier, "pro"),
   399→        project: selectedProjectLabel,
   400→      };
   401→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   402→
   403→      toast.success("Export complete", {
   404→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   405→      });
   406→    } catch {
   407→      toast.error("Export failed", {
   408→        description: "An error occurred while generating the export file.",
   409→      });
   410→    }
   411→
   412→    setIsExporting(false);
   413→  };
   414→
   415→  const handleFormatSelect = (format: ExportFormat) => {
   416→    if (isFormatLocked(format)) {
   417→      toast.error("Upgrade required", {
   418→        description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.`,
   419→      });
   420→      return;
   421→    }
   422→    setSelectedFormat(format);
   423→  };
   424→
   425→  const previewItems = useMemo(() => {
   426→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   427→  }, [evidenceTypes, selectedEvidenceTypes]);
   428→
   429→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   430→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   431→
   432→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   433→
   434→  // ─── Render ───────────────────────────────────────────────────────────────
   435→
   436→  return (
   437→    <motion.div
   438→      initial={{ opacity: 0 }}
   439→      animate={{ opacity: 1 }}
   440→      className="w-full min-h-screen bg-background"
   441→    >
   442→      <div className="p-6 md:p-8 space-y-8">
   443→        {/* Header */}
   444→        <div>
   445→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">Evidence Export</h1>
   446→          <p className="text-[16px] text-muted-foreground">
   447→            Package and download your work evidence in court-ready formats with compliance verification.
   448→          </p>
   449→        </div>
   450→
   451→        {/* Demo Mode Banner */}
   452→        {!isAuthenticated && <DemoModeBanner />}
   453→
   454→        {/* Loading State */}
   455→        {isLoading ? (
   456→          <>
   457→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   458→              {[0, 1, 2, 3].map((i) => <SummarySkeleton key={i} />)}
   459→            </div>
   460→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   461→              <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   462→              <div><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   463→            </div>
   464→          </>
   465→        ) : !hasData ? (
   466→          <EmptyEvidenceState />
   467→        ) : (
   468→          <>
   469→            {/* Summary Dashboard */}
   470→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   471→              <SummaryCard
   472→                title="Total Items"
   473→                value={totalItems}
   474→                subtitle="Across all projects"
   475→                icon={<HardDrive className="h-5 w-5 text-primary-foreground" />}
   476→                accent="bg-primary"
   477→              />
   478→              <SummaryCard
   479→                title="Evidence Types"
   480→                value={evidenceTypes.length}
   481→                subtitle="Different categories"
   482→                icon={<Clock className="h-5 w-5 text-amber-700" />}
   483→                accent="bg-amber-100 dark:bg-amber-900/30"
   484→              />
   485→              <SummaryCard
   486→                title="Compliance Score"
   487→                value={`${complianceScore}%`}
   488→                subtitle={complianceScore >= 80 ? "Court-admissible ready" : "Needs improvement"}
   489→                icon={<ShieldCheck className="h-5 w-5 text-emerald-700" />}
   490→                accent="bg-emerald-100 dark:bg-emerald-900/30"
   491→              />
   492→              <SummaryCard
   493→                title="Dispute Success"
   494→                value={`${disputeSuccessRate}%`}
   495→                subtitle="Based on evidence quality"
   496→                icon={<Shield className="h-5 w-5 text-sky-700" />}
   497→                accent="bg-sky-100 dark:bg-sky-900/30"
   498→              />
   499→            </div>
   500→
   501→            {/* Export Configuration */}
   502→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   503→              {/* Left Column: Format + Filters */}
   504→              <div className="lg:col-span-2 space-y-6">
   505→                {/* Format Selection */}
   506→                <Card>
   507→                  <CardHeader>
   508→                    <CardTitle className="flex items-center gap-2">
   509→                      <Download className="h-5 w-5 text-primary" />
   510→                      Export Format
   511→                    </CardTitle>
   512→                    <CardDescription>Choose the output format for your evidence package.</CardDescription>
   513→                  </CardHeader>
   514→                  <CardContent>
   515→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   516→                      {(Object.keys(FORMAT_CONFIG) as ExportFormat[]).map((formatKey) => (
   517→                        <FormatOption
   518→                          key={formatKey}
   519→                          formatKey={formatKey}
   520→                          selected={selectedFormat === formatKey}
   521→                          onSelect={() => handleFormatSelect(formatKey)}
   522→                          locked={isFormatLocked(formatKey)}
   523→                        />
   524→                      ))}
   525→                    </div>
   526→                    {isFormatLocked(selectedFormat) && (
   527→                      <div className="mt-4 flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-900/20 dark:border-amber-800 p-3">
   528→                        <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400 shrink-0" />
   529→                        <p className="text-sm text-amber-800 dark:text-amber-200">
   530→                          <span className="font-semibold">Upgrade to {FORMAT_CONFIG[selectedFormat].tierRequired}</span> to unlock{" "}
   531→                          {FORMAT_CONFIG[selectedFormat].label} exports with compliance verification.
   532→                        </p>
   533→                      </div>
   534→                    )}
   535→                  </CardContent>
   536→                </Card>
   537→
   538→                {/* Date Range & Filters */}
   539→                <Card>
   540→                  <CardHeader>
   541→                    <CardTitle className="flex items-center gap-2">
   542→                      <Filter className="h-5 w-5 text-primary" />
   543→                      Filters & Date Range
   544→                    </CardTitle>
   545→                    <CardDescription>Refine which evidence to include in the export.</CardDescription>
   546→                  </CardHeader>
   547→                  <CardContent className="space-y-5">
   548→                    {/* Date Range */}
