   106→// ─── Helper ─────────────────────────────────────────────────────────────────
   107→
   108→function generateId(): string {
   109→  return Math.random().toString(36).substring(2, 10);
   110→}
   111→
   112→function createEmptySection(type: SectionType): ProposalSection {
   113→  const section: ProposalSection = {
   114→    id: generateId(),
   115→    type,
   116→    content: "",
   117→  };
   118→  if (type === "pricing") {
   119→    section.content = "Pricing Package";
   120→    section.metadata = { items: [{ name: "", price: 0 }] };
   121→  }
   122→  if (type === "milestone") {
   123→    section.content = "Project Milestones";
   124→    section.metadata = { milestones: [{ name: "", weeks: 1 }] };
   125→  }
   126→  if (type === "heading") {
   127→    section.content = "New Section";
   128→  }
   129→  if (type === "text" || type === "summary" || type === "scope_of_work") {
   130→    section.content = "";
   131→  }
   132→  if (type === "terms") {
   133→    section.content = "Payment terms and conditions...";
   134→  }
   135→  if (type === "client_info" || type === "sender_info") {
   136→    section.content = "Contact details";
   137→  }
   138→  return section;
   139→}
   140→
   141→function calculateTotal(sections: ProposalSection[]): number {
   142→  return sections.reduce((total, section) => {
   143→    if (section.type === "pricing" && section.metadata?.items) {
   144→      return total + section.metadata.items.reduce((sum, item) => sum + (item.price || 0), 0);
   145→    }
   146→    return total;
   147→  }, 0);
   148→}
   149→
   150→function formatCurrency(amount: number): string {
   151→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(amount);
   152→}

   309→
