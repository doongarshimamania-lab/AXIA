    40→    40→    }

   104→   104→        }

   106→   106→        userTemplates = await ctx.db
   107→   107→          .query("proposalTemplates")
   108→   108→          .withIndex("by_user", (q) => q.eq("userId", userId))
   109→   109→          .collect();
   110→   110→      }

   117→   117→export const getProposalStats = query({
   118→   118→  args: { workspaceId: v.optional(v.id("workspaces")) },
   119→   119→  handler: async (ctx, { workspaceId }) => {
   120→   120→    const userId = await getAuthUserId(ctx);
   121→   121→    if (!userId) return { total: 0, sent: 0, signed: 0, draft: 0, signatureRate: 0, totalValue: 0 };
   122→   122→
   123→   123→    let proposals;
   124→   124→    if (workspaceId) {
   125→   125→      const visibility = await getUserVisibility(ctx, workspaceId);
   126→   126→      if (!visibility) return { total: 0, sent: 0, signed: 0, draft: 0, signatureRate: 0, totalValue: 0 };
   127→   127→
   128→   128→      const allProposals = await ctx.db
   129→   129→        .query("proposals")
   130→   130→        .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
   131→   131→        .collect();
   132→   132→      proposals = allProposals.filter((p) => isRecordVisible(p, visibility));
   133→   133→    } else {
   134→   134→      proposals = await ctx.db
   135→   135→        .query("proposals")
   136→   136→        .withIndex("by_user", (q) => q.eq("userId", userId))
   137→   137→        .collect();
   138→   138→    }
   139→   139→
   140→   140→    const sent = proposals.filter(p => p.status === "sent" || p.status === "viewed").length;
   141→   141→    const signed = proposals.filter(p => p.status === "signed").length;
   142→   142→    const draft = proposals.filter(p => p.status === "draft").length;
   143→   143→    const totalValue = proposals.filter(p => p.status === "signed").reduce((sum, p) => sum + p.totalValue, 0);
   144→   144→
   145→   145→    return {
   146→   146→      total: proposals.length,
   147→   147→      sent,
   148→   148→      signed,
   149→   149→      draft,
   150→   150→      signatureRate: sent + signed > 0 ? Math.round((signed / (sent + signed)) * 100) : 0,
   151→   151→      totalValue,
   152→   152→    };
   153→   153→  },
   154→   154→});

   168→   168→function generateToken(): string {
   169→   169→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
   170→   170→  let result = "";
   171→   171→  for (let i = 0; i < 32; i++) {
   172→   172→    result += chars.charAt(Math.floor(Math.random() * chars.length));
   173→   173→  }
   174→   174→  return result;
   175→   175→}

   327→   327→  handler: async (ctx, { publicToken, signatureData }) => {
   328→   328→    // This is a public action via shared link — client signs without being a platform user
   329→   329→    const proposal = await ctx.db
   330→   330→      .query("proposals")
   331→   331→      .withIndex("by_public_token", (q) => q.eq("publicToken", publicToken))
   332→   332→      .first();

   335→   335→    if (proposal.status !== "sent" && proposal.status !== "viewed") {
   336→   336→      throw new Error("Proposal is not in a signable state");
   337→   337→    }
   338→   338→    if (proposal.status === "signed") {
   339→   339→      throw new Error("Proposal has already been signed");
   340→   340→    }
   341→   341→
   342→   342→    await ctx.db.patch(proposal._id, {
   343→   343→      status: "signed",
   344→   344→      signedAt: Date.now(),
   345→   345→      signatureData,
   346→   346→      updatedAt: Date.now(),
   347→   347→    });
   348→   348→
   349→   349→    // Cancel remaining follow-ups
   350→   350→    const followUps = await ctx.db
   351→   351→      .query("proposalFollowUps")
   352→   352→      .withIndex("by_proposal", (q) => q.eq("proposalId", proposal._id))
   353→   353→      .collect();
   354→   354→
   355→   355→    for (const fu of followUps) {
   356→   356→      if (fu.status === "scheduled") {
   357→   357→        await ctx.db.patch(fu._id, { status: "cancelled" });
   358→   358→      }
   359→   359→    }
   360→   360→  },

   545→   545→function getFollowUpSubject(dayNumber, title) {
   546→   546→  if (dayNumber <= 3) return `Following up: ${title}`;
   547→   547→  if (dayNumber <= 10) return `Checking in: ${title}`;
   548→   548→  return `Final reminder: ${title}`;
   549→   549→}
   550→   550→
   551→   551→function getFollowUpBody(dayNumber, title, clientName, sentAt) {
   552→   552→  const name = clientName || "there";
   553→   553→  const sentDate = new Date(sentAt).toLocaleDateString();
   554→   554→  if (dayNumber <= 3) {
   555→   555→    return `Hi ${name},\n\nI wanted to follow up on the proposal "${title}" I sent on ${sentDate}.\n\nI'd love to hear your thoughts and answer any questions you might have.\n\nBest regards`;
   556→   556→  }
   557→   557→  if (dayNumber <= 10) {
   558→   558→    return `Hi ${name},\n\nI'm checking in on the proposal "${title}" I sent on ${sentDate}.\n\nPlease let me know if you need any additional information or would like to discuss any adjustments.\n\nBest regards`;
   559→   559→  }
   560→   560→  return `Hi ${name},\n\nThis is a final reminder regarding the proposal "${title}" I sent on ${sentDate}.\n\nThis proposal will expire soon. Please let me know if you have any questions or if you'd like to proceed.\n\nBest regards`;
   561→   561→}

   563→   563→export const startFollowUps = mutation({
   564→   564→  args: {
   565→   565→    proposalId: v.id("proposals"),
   566→   566→    intervals: v.optional(v.array(v.number())),
   567→   567→  },
   568→   568→  handler: async (ctx, { proposalId, intervals }) => {
   569→   569→    const userId = await getAuthUserId(ctx);
   570→   570→    if (!userId) throw new Error("Not authenticated");
   571→   571→
   572→   572→    const proposal = await ctx.db.get(proposalId);
   573→   573→    if (!proposal) throw new Error("Proposal not found");
   574→   574→
   575→   575→    // Check workspace access or direct ownership
   576→   576→    if (proposal.workspaceId) {
   577→   577→      const access = await getRecordAccess(ctx, proposal, userId);
   578→   578→      if (!access) throw new Error("Not authorized");
   579→   579→    } else if (proposal.userId !== userId) {
   580→   580→      throw new Error("Not authorized");
   581→   581→    }
   582→   582→
   583→   583→    // Only allow for proposals with status "sent" or "viewed"
   584→   584→    if (proposal.status !== "sent" && proposal.status !== "viewed") {
   585→   585→      throw new Error("Follow-ups can only be started for sent or viewed proposals");
   586→   586→    }
   587→   587→
   588→   588→    // Cancel any existing scheduled follow-ups for this proposal
   589→   589→    const existingFollowUps = await ctx.db
   590→   590→      .query("proposalFollowUps")
   591→   591→      .withIndex("by_proposal", (q) => q.eq("proposalId", proposalId))
   592→   592→      .collect();
   593→   593→
   594→   594→    for (const fu of existingFollowUps) {
   595→   595→      if (fu.status === "scheduled") {
   596→   596→        await ctx.db.patch(fu._id, { status: "cancelled" });
   597→   597→      }
   598→   598→    }
   599→   599→
   600→   600→    // Create new follow-ups based on provided intervals or defaults
   601→   601→    const dayIntervals = intervals || [3, 7, 14];
   602→   602→    const sentAt = proposal.sentAt || Date.now();
   603→   603→
   604→   604→    for (const dayNumber of dayIntervals) {
   605→   605→      await ctx.db.insert("proposalFollowUps", {
   606→   606→        userId: proposal.userId,
   607→   607→        workspaceId: proposal.workspaceId ?? undefined,
   608→   608→        proposalId,
   609→   609→        dayNumber,
   610→   610→        subject: getFollowUpSubject(dayNumber, proposal.title),
   611→   611→        body: getFollowUpBody(dayNumber, proposal.title, proposal.clientName, sentAt),
   612→   612→        channel: "email",
   613→   613→        status: "scheduled",
   614→   614→        scheduledAt: sentAt + dayNumber * 24 * 60 * 60 * 1000,
   615→   615→        createdAt: Date.now(),
   616→   616→      });
   617→   617→    }
   618→   618→
   619→   619→    return { success: true, scheduledCount: dayIntervals.length };
   620→   620→  },
   621→   621→});
