    10→// Start evidence collection for a work session

    60→// Record evidence events in bulk
    61→export const recordEvents = mutation({
    62→  args: {
    63→    evidenceSessionId: v.id("evidenceSessions"),
    64→    events: v.array(v.object({
    65→      t: v.number(),
    66→      kind: v.union(
    67→        v.literal("mouse"),
    68→        v.literal("keyboard"),
    69→        v.literal("url"),
    70→        v.literal("screenshot_ref"),
    71→        v.literal("memo"),
    72→        v.literal("platform_status")
    73→      ),
    74→      data: v.any(),
    75→      url: v.optional(v.string()),
    76→    })),
    77→  },
    78→  handler: async (ctx, args) => {
    79→    const identity = await ctx.auth.getUserIdentity();
    80→    if (!identity) throw new Error("Not authenticated");
    81→
    82→    // Validate evidence session exists and is active
    83→    const evidenceSession = await ctx.db.get(args.evidenceSessionId);
    84→    if (!evidenceSession || evidenceSession.status !== "active") {
    85→      throw new Error("Evidence session not found or not active");
    86→    }
    87→
    88→    // Chunk events to stay under Convex limits (max 500 per batch)
    89→    const chunks = [];
    90→    for (let i = 0; i < args.events.length; i += 500) {
    91→      chunks.push(args.events.slice(i, i + 500));
    92→    }
    93→
    94→    let insertedCount = 0;
    95→    for (const chunk of chunks) {
    96→      for (const event of chunk) {
    97→        await ctx.db.insert("evidenceEvents", {
    98→          workspaceId: evidenceSession.workspaceId ?? undefined,
    99→          evidenceSessionId: args.evidenceSessionId,
   100→          t: event.t,
   101→          kind: event.kind,
   102→          data: event.data,
   103→          url: event.url,
   104→        });
   105→        insertedCount++;
   106→      }
   107→    }
   108→
   109→    return { insertedCount };
   110→  },
   111→});

   243→// Generate universal dispute report
   244→export const generateUniversalReport = mutation({
   245→  args: {
   246→    sessionId: v.id("workSessions"),
   247→  },
   248→  handler: async (ctx, args) => {
   249→    const user = await getCurrentUser(ctx);
   250→    if (!user) {
   251→      throw new Error("Not authenticated");
   252→    }
   253→
   254→    // Get work session (must belong to user)
   255→    const workSession = await ctx.db.get(args.sessionId);
   256→    if (!workSession || workSession.userId !== user._id) {
   257→      throw new Error("Work session not found or unauthorized");
   258→    }
   259→
   260→    // Enforce free tier limit (1/month) using generatedAt window like disputeReports.ts
   261→    if (user.subscriptionTier === "free") {
   262→      const now = new Date();
   263→      const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
   264→      const endOfMonth = new Date(
   265→        now.getFullYear(),
   266→        now.getMonth() + 1,
   267→        0,
   268→        23,
   269→        59,
   270→        59,
   271→        999
   272→      ).getTime();
   273→
   274→      const monthlyReports = await ctx.db
   275→        .query("disputeReports")
   276→        .withIndex("by_user", (q) => q.eq("userId", user._id))
   277→        .filter((q) =>
   278→          q.and(
   279→            q.gte(q.field("generatedAt"), startOfMonth),
   280→            q.lte(q.field("generatedAt"), endOfMonth)
   281→          )
   282→        )
   283→        .collect();
   284→
   285→      const sessionHours =
   286→        ((workSession.endTime ?? Date.now()) - workSession.startTime) /
   287→        (1000 * 60 * 60);
   288→      const sessionLoss = Math.max(
   289→        0,
   290→        Math.round(sessionHours * workSession.hourlyRate * 100) / 100
   291→      );
   292→
   293→      if (monthlyReports.length >= 1) {
   294→        const prevLoss = monthlyReports.reduce(
   295→          (sum, r) => sum + r.lostIncome,
   296→          0
   297→        );
   298→        const monthlyLoss =
   299→          Math.round((prevLoss + sessionLoss) * 100) / 100;
   300→        const monthlySavings =
   301→          Math.round(monthlyLoss * 0.83 * 100) / 100;
   302→
   303→        return {
   304→          limited: true,
   305→          monthlyLoss,
   306→          monthlySavings,
   307→        };
   308→      }
   309→    }
   310→
   311→    // Find evidence session for this work session
   312→    const evidenceSession = await ctx.db
   313→      .query("evidenceSessions")
   314→      .withIndex("by_session", (q) => q.eq("sessionId", args.sessionId))
   315→      .first();
   316→
   317→    if (!evidenceSession) {
   318→      throw new Error("No evidence session found for this work session");
   319→    }
   320→
   321→    // Collect all evidence events ordered by time
   322→    const events = await ctx.db
   323→      .query("evidenceEvents")
   324→      .withIndex("by_session_and_time", (q) =>
   325→        q.eq("evidenceSessionId", evidenceSession._id)
   326→      )
   327→      .order("asc")
   328→      .collect();
   329→
   330→    const caseId = `TL-${Date.now()
   331→      .toString(36)
   332→      .toUpperCase()}-${Math.random().toString(36).slice(2, 6).toUpperCase()}`;
   333→
   334→    // Analyze and build report
   335→    const universalCore = analyzeUniversalEvidence(
   336→      events,
   337→      evidenceSession,
   338→      workSession
   339→    );
   340→    const platformAdaptation = adaptToPlatform(
   341→      evidenceSession.platform,
   342→      events,
   343→      universalCore
   344→    );
   345→    const authenticity = generateAuthenticityProof(
   346→      events,
   347→      evidenceSession
   348→    );
   349→    const reportContent = generateReportContent({
   350→      caseId,
   351→      workSession,
   352→      evidenceSession,
   353→      universalCore,
   354→      platformAdaptation,
   355→      authenticity,
   356→      events,
   357→    });
   358→
   359→    // Compute safe values
   360→    const hours =
   361→      ((workSession.endTime ?? Date.now()) - workSession.startTime) /
   362→      (1000 * 60 * 60);
   363→    const lostIncome =
   364→      Math.max(0, Math.round(hours * workSession.hourlyRate * 100) / 100);
   365→
   366→    // Store minimal dispute report per schema
   367→    await ctx.db.insert("disputeReports", {
   368→      userId: user._id,
   369→      sessionId: args.sessionId,
   370→      caseId,
   371→      generatedAt: Date.now(),
   372→      rejectedHours: Math.max(0, Math.round(hours * 10) / 10),
   373→      lostIncome,
   374→      reportContent,
   375→      status: "generated",
   376→    });
   377→
   378→    return { caseId, reportContent, limited: false };
   379→  },
   380→});

   608→// Helper functions
   609→function analyzeUniversalEvidence(events: any[], evidenceSession: any, workSession: any) {
   610→  const mouseEvents = events.filter(e => e.kind === "mouse");
   611→  const keyboardEvents = events.filter(e => e.kind === "keyboard");
   612→  const urlEvents = events.filter(e => e.kind === "url");
   613→
   614→  const sessionDuration = (evidenceSession.endTime || Date.now()) - evidenceSession.startTime;
   615→  const activityDensity = (mouseEvents.length + keyboardEvents.length) / (sessionDuration / 60000); // per minute
   616→
   617→  return {
   618→    continuous_activity: {
   619→      mouseEvents: mouseEvents.length,
   620→      keyboardEvents: keyboardEvents.length,
   621→      activityDensity,
   622→      gapsAnalysis: analyzeActivityGaps(events),
   623→      status: activityDensity > 2 ? "PASS" : "FAIL",
   624→    },
   625→    work_context: {
   626→      sitesVisited: [...new Set(urlEvents.map(e => e.url).filter(Boolean))],
   627→      workSites: urlEvents.filter(e => isWorkSite(e.url || "")).length,
   628→      nonWorkSites: urlEvents.filter(e => !isWorkSite(e.url || "")).length,
   629→      relevanceScore: calculateWorkRelevance(urlEvents, workSession),
   630→      status: calculateWorkRelevance(urlEvents, workSession) > 0.7 ? "PASS" : "FAIL",
   631→    },
   632→    time_consistency: {
   633→      evidencePoints: events.length,
   634→      timeSpan: sessionDuration,
   635→      consistencyCheck: validateTimeConsistency(events, workSession),
   636→      status: validateTimeConsistency(events, workSession) ? "PASS" : "FAIL",
   637→    },
   638→  };
   639→}

   641→function adaptToPlatform(platform: string, events: any[], universalCore: any) {

   796→function analyzeActivityGaps(events: any[]) {
   797→  const sortedEvents = events.filter(e => e.kind === "mouse" || e.kind === "keyboard").sort((a, b) => a.t - b.t);
   798→  let maxGap = 0;
   799→  let gapCount = 0;
   800→
   801→  for (let i = 1; i < sortedEvents.length; i++) {
   802→    const gap = sortedEvents[i].t - sortedEvents[i-1].t;
   803→    if (gap > 60000) { // > 1 minute
   804→      gapCount++;
   805→      maxGap = Math.max(maxGap, gap);
   806→    }
   807→  }
   808→
   809→  return `${gapCount} gaps > 1min (max: ${Math.round(maxGap/1000)}s)`;
   810→}

   821→function calculateWorkRelevance(urlEvents: any[], workSession: any): number {
   822→  if (urlEvents.length === 0) return 0;

   827→function validateTimeConsistency(events: any[], workSession: any): boolean {
   828→  if (events.length === 0) return false;
   829→  const firstEvent = Math.min(...events.map(e => e.t));
   830→  const lastEvent = Math.max(...events.map(e => e.t));
   831→  
   832→  const sessionStart = workSession.startTime;
   833→  const sessionEnd = workSession.endTime || Date.now();
   834→  
   835→  return Math.abs(firstEvent - sessionStart) <= 2000 && Math.abs(lastEvent - sessionEnd) <= 2000;
   836→}
   837→
   838→function calculateTimestampConsistency(events: any[]): string {
   839→  const sortedEvents = events.sort((a, b) => a.t - b.t);
   840→  let inconsistencies = 0;
   841→  
   842→  for (let i = 1; i < sortedEvents.length; i++) {
   843→    if (sortedEvents[i].t < sortedEvents[i-1].t) {
   844→      inconsistencies++;
   845→    }
   846→  }
   847→  
   848→  return inconsistencies === 0 ? "Consistent" : `${inconsistencies} inconsistencies`;
   849→}
   850→
   851→function calculateMaxDrift(events: any[]): number {
   852→  if (events.length < 2) return 0;
   853→  const sortedEvents = events.sort((a, b) => a.t - b.t);
   854→  let maxDrift = 0;
   855→  
   856→  for (let i = 1; i < sortedEvents.length; i++) {
   857→    const drift = Math.abs(sortedEvents[i].t - sortedEvents[i-1].t);
   858→    maxDrift = Math.max(maxDrift, drift);
   859→  }
   860→  
   861→  return maxDrift;
   862→}
