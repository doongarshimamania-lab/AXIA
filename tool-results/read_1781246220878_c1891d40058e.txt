   139→// ─── Status badge colors ─────────────────────────────────────────────────
   140→function statusColor(status: string): "default" | "secondary" | "destructive" | "outline" {
   141→  if (["won", "signed", "paid", "completed"].includes(status)) return "default";
   142→  if (["lost", "declined", "overdue", "expired", "rejected"].includes(status)) return "destructive";
   143→  return "secondary";
   144→}
   145→
   146→function statusIcon(type: string, status: string) {
   147→  if (type === "deal") {
   148→    if (status === "won") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   149→    if (status === "lost") return <AlertCircle className="h-4 w-4 text-red-500" />;
   150→    return <TrendingUp className="h-4 w-4 text-blue-500" />;
   151→  }
   152→  if (type === "proposal") {
   153→    if (status === "signed") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   154→    if (status === "sent" || status === "viewed") return <Send className="h-4 w-4 text-blue-500" />;
   155→    return <FileText className="h-4 w-4 text-muted-foreground" />;
   156→  }
   157→  // invoice
   158→  if (status === "paid") return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   159→  if (status === "overdue") return <AlertCircle className="h-4 w-4 text-red-500" />;
   160→  return <DollarSign className="h-4 w-4 text-amber-500" />;
   161→}
   162→
   163→// ─── Format currency ─────────────────────────────────────────────────────
   164→function fmtCurrency(n: number): string {
   165→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
   166→}
   167→
   168→function fmtRelative(ts: number): string {
   169→  const diff = Date.now() - ts;
   170→  const mins = Math.floor(diff / 60000);
   171→  if (mins < 1) return "just now";
   172→  if (mins < 60) return `${mins}m ago`;
   173→  const hrs = Math.floor(mins / 60);
   174→  if (hrs < 24) return `${hrs}h ago`;
   175→  const days = Math.floor(hrs / 24);
   176→  if (days < 30) return `${days}d ago`;
   177→  return new Date(ts).toLocaleDateString();
   178→}
