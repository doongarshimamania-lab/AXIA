   117→export default function TeamManagement() {
   118→  const { activeWorkspaceId, isTeamMode, isOwner, canManageTeam, activeWorkspace } = useWorkspaceContext();
   119→
   120→  // ─── Convex Queries ────────────────────────────────────────────────────────
   121→  const hasRealWorkspaceId = activeWorkspaceId && !activeWorkspaceId.startsWith("ws_");
   122→
   123→  const convexMembers = useQuery(
   124→    hasRealWorkspaceId ? api.workspaces.members.getMembers : "skip",
   125→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   126→  ) as any[] | undefined;
   127→
   128→  const convexStats = useQuery(
   129→    hasRealWorkspaceId ? api.workspaces.crud.getWorkspaceStats : "skip",
   130→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   131→  ) as any | undefined;
   132→
   133→  const convexInvitations = useQuery(
   134→    hasRealWorkspaceId ? api.workspaces.invitations.getInvitations : "skip",
   135→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any, status: "pending" as const } : "skip"
   136→  ) as any[] | undefined;
   137→
   138→  // ─── Teams queries ─────────────────────────────────────────────────────────
   139→  const teamsApi = (api as any).teams?.crud;
   140→  const hasTeamsApi = !!teamsApi?.getTeams;
   141→
   142→  const convexTeams = useQuery(
   143→    hasTeamsApi && hasRealWorkspaceId ? teamsApi.getTeams : "skip",
   144→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   145→  ) as any[] | undefined;
   146→
   147→  // ─── Team Members per team ──────────────────────────────────────────────────
   148→  const hasTeamMembersApi = !!teamsApi?.getTeamMembers;
   149→
   150→  // ─── Mutations ─────────────────────────────────────────────────────────────
   151→  const inviteMemberMutation = useMutation(api.workspaces.invitations.createInvitation);
   152→  const removeMemberMutation = useMutation(api.workspaces.members.removeMember);
   153→  const updateRoleMutation = useMutation(api.workspaces.members.updateMemberRole);
   154→  const cancelInvitationMutation = useMutation(api.workspaces.invitations.cancelInvitation);
   155→
   156→  // Team mutations
   157→  const createTeamMutation = useMutation(hasTeamsApi ? teamsApi.createTeam : null);
   158→  const updateTeamMutation = useMutation(hasTeamsApi ? teamsApi.updateTeam : null);
   159→  const deleteTeamMutation = useMutation(hasTeamsApi ? teamsApi.deleteTeam : null);
   160→  const addTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.addTeamMember : null);
   161→  const removeTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.removeTeamMember : null);
   162→
   163→  // ─── State ─────────────────────────────────────────────────────────────────
   164→  const [activeTab, setActiveTab] = useState<string>("members");
   165→
   166→  // Invite dialog
   167→  const [showInviteDialog, setShowInviteDialog] = useState(false);
   168→  const [inviteEmail, setInviteEmail] = useState("");
   169→  const [inviteRole, setInviteRole] = useState<"manager" | "member">("member");
   170→  const [inviteMessage, setInviteMessage] = useState("");
   171→  const [isInviting, setIsInviting] = useState(false);
   172→
   173→  // Remove/change role
   174→  const [memberToRemove, setMemberToRemove] = useState<any>(null);
   175→  const [memberToChangeRole, setMemberToChangeRole] = useState<any>(null);
   176→  const [newRole, setNewRole] = useState<"manager" | "member">("member");
   177→
   178→  // Team CRUD state
   179→  const [showCreateTeamDialog, setShowCreateTeamDialog] = useState(false);
   180→  const [teamName, setTeamName] = useState("");
   181→  const [teamColor, setTeamColor] = useState(TEAM_COLORS[0]);
   182→  const [teamDescription, setTeamDescription] = useState("");
   183→  const [teamIsCrossTeam, setTeamIsCrossTeam] = useState(false);
   184→  const [isCreatingTeam, setIsCreatingTeam] = useState(false);
   185→
   186→  // Edit team
   187→  const [editingTeam, setEditingTeam] = useState<any>(null);
   188→  const [editTeamName, setEditTeamName] = useState("");
   189→  const [editTeamColor, setEditTeamColor] = useState("");
   190→  const [editTeamDescription, setEditTeamDescription] = useState("");
   191→  const [isUpdatingTeam, setIsUpdatingTeam] = useState(false);
   192→
   193→  // Delete team
   194→  const [teamToDelete, setTeamToDelete] = useState<any>(null);
   195→  const [isDeletingTeam, setIsDeletingTeam] = useState(false);
   196→
   197→  // Add member to team
   198→  const [teamToAddMember, setTeamToAddMember] = useState<any>(null);
   199→  const [addMemberUserId, setAddMemberUserId] = useState("");
   200→  const [addMemberRole, setAddMemberRole] = useState<"lead" | "member">("member");
   201→  const [isAddingTeamMember, setIsAddingTeamMember] = useState(false);
   202→
   203→  // Remove member from team
   204→  const [teamMemberToRemove, setTeamMemberToRemove] = useState<any>(null);
   205→  const [teamMemberRemoveTeamId, setTeamMemberRemoveTeamId] = useState<string>("");
   206→
   207→  // ─── Map Convex data to UI shape ───────────────────────────────────────────
   208→  const members = useMemo(() => {
   209→    if (convexMembers && convexMembers.length > 0) {
   210→      return convexMembers.map((m: any) => ({
   211→        _id: m._id,
   212→        userId: m.userId,
   213→        name: m.userName ?? m.userEmail ?? "Unknown",
   214→        displayName: m.userName ?? m.userEmail ?? "Unknown",
   215→        email: m.userEmail ?? "",
   216→        image: m.userImage ?? "",
   217→        role: m.role,
   218→        status: m.status,
   219→        joinedAt: m.joinedAt,
   220→        lastActiveAt: m.lastActiveAt ?? null,
   221→        projectsAssigned: 0,
   222→        hoursThisWeek: 0,
   223→      }));
   224→    }
   225→    return [];
   226→  }, [convexMembers]);
   227→
   228→  const teams = useMemo(() => {
   229→    if (convexTeams && convexTeams.length > 0) {
   230→      return convexTeams;
   231→    }
   232→    return [];
   233→  }, [convexTeams]);
   234→
   235→  const stats = useMemo(() => {
   236→    if (convexStats) {
   237→      return {
   238→        memberCount: convexStats.memberCount ?? 0,
   239→        clientCount: convexStats.clientCount ?? 0,
   240→        activeProjectCount: convexStats.activeProjectCount ?? 0,
   241→        pendingInvoiceCount: 0,
   242→        totalRevenue: 0,
   243→        totalHoursThisWeek: 0,
   244→        protectionScore: 0,
   245→      };
   246→    }
   247→    return null;
   248→  }, [convexStats]);
   249→
   250→  const invitations = useMemo(() => {
   251→    if (convexInvitations && convexInvitations.length > 0) {
   252→      return convexInvitations.map((inv: any) => ({
   253→        _id: inv._id,
   254→        name: inv.email,
   255→        displayName: inv.email,
   256→        email: inv.email,
   257→        role: inv.role,
   258→        invitedAt: inv.createdAt,
   259→        invitedBy: inv.inviterName,
   260→      }));
   261→    }
   262→    return [];
   263→  }, [convexInvitations]);
   264→
   265→  const isLoading = hasRealWorkspaceId && (convexMembers === undefined);
   266→  const isDemoMode = !hasRealWorkspaceId;
   267→  const loadingTimedOut = useQueryTimeout(isLoading, 6000);
   268→
   269→  // ─── Handlers ──────────────────────────────────────────────────────────────
   270→  const handleInvite = async () => {
   271→    if (!inviteEmail.trim() || !activeWorkspaceId) return;
   272→    setIsInviting(true);
   273→    try {
   274→      if (hasRealWorkspaceId) {
   275→        await inviteMemberMutation({
   276→          workspaceId: activeWorkspaceId as any,
   277→          email: inviteEmail.trim(),
   278→          role: inviteRole,
   279→        });
   280→      }
   281→      toast.success(`Invitation sent to ${inviteEmail}`);
   282→      setInviteEmail("");
   283→      setInviteMessage("");
   284→      setShowInviteDialog(false);
   285→    } catch (e: any) {
   286→      toast.error(e.message || "Failed to send invitation");
   287→    } finally {
   288→      setIsInviting(false);
   289→    }
   290→  };
   291→
   292→  const handleRemoveMember = async () => {
   293→    if (!memberToRemove) return;
   294→    try {
   295→      await removeMemberMutation({ memberId: memberToRemove._id });
   296→      toast.success(`${memberToRemove.name} removed from workspace`);
   297→      setMemberToRemove(null);
   298→    } catch (e: any) {
   299→      toast.error(e.message || "Failed to remove member");
   300→    }
   301→  };
   302→
   303→  const handleChangeRole = async () => {
   304→    if (!memberToChangeRole) return;
   305→    try {
   306→      await updateRoleMutation({ memberId: memberToChangeRole._id, role: newRole });
   307→      toast.success(`Role updated to ${newRole}`);
   308→      setMemberToChangeRole(null);
   309→    } catch (e: any) {
   310→      toast.error(e.message || "Failed to update role");
   311→    }
   312→  };
   313→
   314→  const handleCancelInvitation = async (invitation: any) => {
   315→    try {
   316→      await cancelInvitationMutation({ invitationId: invitation._id });
   317→      toast.success(`Invitation to ${invitation.email || invitation.name} cancelled`);
   318→    } catch (e: any) {
   319→      toast.error(e.message || "Failed to cancel invitation");
   320→    }
   321→  };
   322→
   323→  // ─── Team CRUD Handlers ────────────────────────────────────────────────────
   324→  const handleCreateTeam = async () => {
   325→    if (!teamName.trim() || !activeWorkspaceId) return;
   326→    setIsCreatingTeam(true);
   327→    try {
   328→      if (createTeamMutation && hasRealWorkspaceId) {
   329→        await createTeamMutation({
   330→          workspaceId: activeWorkspaceId as any,
   331→          name: teamName.trim(),
   332→          color: teamColor,
   333→          description: teamDescription.trim() || undefined,
   334→          isCrossTeam: teamIsCrossTeam || undefined,
   335→        });
   336→      }
   337→      toast.success(`Team "${teamName}" created!`);
   338→      setShowCreateTeamDialog(false);
   339→      setTeamName("");
   340→      setTeamColor(TEAM_COLORS[0]);
   341→      setTeamDescription("");
   342→      setTeamIsCrossTeam(false);
   343→    } catch (e: any) {
   344→      toast.error(e.message || "Failed to create team");
   345→    } finally {
   346→      setIsCreatingTeam(false);
   347→    }
   348→  };
   349→
   350→  const handleUpdateTeam = async () => {
   351→    if (!editingTeam) return;
   352→    setIsUpdatingTeam(true);
   353→    try {
   354→      if (updateTeamMutation) {
   355→        await updateTeamMutation({
   356→          teamId: editingTeam._id,
   357→          name: editTeamName.trim() || undefined,
   358→          color: editTeamColor || undefined,
   359→          description: editTeamDescription.trim() || undefined,
   360→        });
   361→      }
   362→      toast.success("Team updated!");
   363→      setEditingTeam(null);
   364→    } catch (e: any) {
   365→      toast.error(e.message || "Failed to update team");
   366→    } finally {
   367→      setIsUpdatingTeam(false);
   368→    }
   369→  };
   370→
   371→  const handleDeleteTeam = async () => {
   372→    if (!teamToDelete) return;
   373→    setIsDeletingTeam(true);
   374→    try {
   375→      if (deleteTeamMutation) {
   376→        await deleteTeamMutation({ teamId: teamToDelete._id });
   377→      }
   378→      toast.success(`Team "${teamToDelete.name}" deleted`);
   379→      setTeamToDelete(null);
   380→    } catch (e: any) {
   381→      toast.error(e.message || "Failed to delete team");
   382→    } finally {
   383→      setIsDeletingTeam(false);
   384→    }
   385→  };
   386→
   387→  const handleAddTeamMember = async () => {
   388→    if (!teamToAddMember || !addMemberUserId) return;
   389→    setIsAddingTeamMember(true);
   390→    try {
   391→      if (addTeamMemberMutation) {
   392→        await addTeamMemberMutation({
   393→          teamId: teamToAddMember._id,
   394→          userId: addMemberUserId as any,
   395→          role: addMemberRole,
   396→        });
   397→      }
   398→      toast.success("Member added to team!");
   399→      setTeamToAddMember(null);
   400→      setAddMemberUserId("");
   401→      setAddMemberRole("member");
   402→    } catch (e: any) {
   403→      toast.error(e.message || "Failed to add member to team");
   404→    } finally {
   405→      setIsAddingTeamMember(false);
   406→    }
   407→  };
   408→
   409→  const handleRemoveTeamMember = async () => {
   410→    if (!teamMemberToRemove || !teamMemberRemoveTeamId) return;
   411→    try {
   412→      if (removeTeamMemberMutation) {
   413→        await removeTeamMemberMutation({
   414→          teamId: teamMemberRemoveTeamId as any,
   415→          userId: teamMemberToRemove.userId as any,
   416→        });
   417→      }
   418→      toast.success("Member removed from team");
   419→      setTeamMemberToRemove(null);
   420→      setTeamMemberRemoveTeamId("");
   421→    } catch (e: any) {
   422→      toast.error(e.message || "Failed to remove member from team");
   423→    }
   424→  };
   425→
   426→  // ─── Not in team mode: Show upgrade prompt ────────────────
   427→  if (!isTeamMode) {
   428→    return <SoloModePrompt />;
   429→  }
   430→
   431→  const activeMembers = members.filter((m: any) => m.status === "active");
   432→  const ownerCount = activeMembers.filter((m: any) => m.role === "owner").length;
   433→  const managerCount = activeMembers.filter((m: any) => m.role === "manager").length;
   434→  const memberCount = activeMembers.filter((m: any) => m.role === "member").length;
   435→
   436→  const roleIcon = (role: string) => {
   437→    switch (role) {
   438→      case "owner": return <Crown className="w-3.5 h-3.5 text-amber-500" />;
   439→      case "manager": return <Shield className="w-3.5 h-3.5 text-blue-500" />;
   440→      case "lead": return <Shield className="w-3.5 h-3.5 text-emerald-500" />;
   441→      case "member": return <Eye className="w-3.5 h-3.5 text-gray-500" />;
   442→      default: return null;
   443→    }
   444→  };
   445→
   446→  const roleBadge = (role: string) => {
   447→    const variants: Record<string, any> = {
   448→      owner: { className: "bg-amber-500/10 text-amber-600 border-amber-500/20", label: "Owner" },
   449→      manager: { className: "bg-blue-500/10 text-blue-600 border-blue-500/20", label: "Manager" },
   450→      lead: { className: "bg-emerald-500/10 text-emerald-600 border-emerald-500/20", label: "Lead" },
   451→      member: { className: "bg-gray-500/10 text-gray-600 border-gray-500/20", label: "Member" },
   452→    };
   453→    const v = variants[role] || variants.member;
   454→    return (
   455→      <Badge variant="outline" className={`gap-1 ${v.className}`}>
   456→        {roleIcon(role)}
   457→        {v.label}
   458→      </Badge>
   459→    );
   460→  };
   461→
   462→  const formatLastActive = (timestamp: number | null) => {
   463→    if (!timestamp) return "Never";
   464→    const diffMs = Date.now() - timestamp;
   465→    const diffMins = Math.floor(diffMs / 60000);
   466→    if (diffMins < 5) return "Just now";
   467→    if (diffMins < 60) return `${diffMins}m ago`;
   468→    const diffHours = Math.floor(diffMins / 60);
   469→    if (diffHours < 24) return `${diffHours}h ago`;
   470→    const diffDays = Math.floor(diffHours / 24);
   471→    return `${diffDays}d ago`;
   472→  };
   473→
   474→  const formatJoined = (timestamp: number | null) => {
   475→    if (!timestamp) return "";
   476→    const diffMs = Date.now() - timestamp;
   477→    const diffDays = Math.floor(diffMs / (24 * 60 * 60 * 1000));
   478→    if (diffDays < 1) return "Joined today";
   479→    if (diffDays < 30) return `Joined ${diffDays}d ago`;
   480→    const diffMonths = Math.floor(diffDays / 30);
   481→    return `Joined ${diffMonths}mo ago`;
   482→  };
   483→
   484→  return (
   485→    <div className="p-8">
   486→      <div className="max-w-6xl mx-auto space-y-6">
   487→        {/* Header */}
   488→        <div className="flex items-center justify-between">
   489→          <div>
   490→            <h1 className="text-3xl font-bold text-foreground mb-2 flex items-center gap-3">
   491→              <Building2 className="w-8 h-8 text-purple-500" />
   492→              {activeWorkspace?.name || "Team Workspace"}
   493→            </h1>
   494→            <p className="text-muted-foreground">
   495→              Manage your team members, teams, and invitations
   496→            </p>
   497→          </div>
   498→          {canManageTeam && (
   499→            <div className="flex gap-2">
   500→              <Button className="gap-2" onClick={() => setShowInviteDialog(true)}>
   501→                <UserPlus className="w-4 h-4" />
   502→                Invite Member
   503→              </Button>
   504→            </div>
   505→          )}
   506→        </div>
   507→
   508→        {/* Demo Mode Banner */}
   509→        {isDemoMode && <DemoModeBanner />}
   510→
   511→        {/* Loading / Error State */}
   512→        {isLoading && !loadingTimedOut ? (
   513→          <StatsSkeleton />
   514→        ) : loadingTimedOut && isLoading ? (
   515→          <Card className="p-8 text-center border-amber-200 dark:border-amber-800 bg-amber-50/50 dark:bg-amber-900/20">
   516→            <AlertTriangle className="w-8 h-8 text-amber-500 mx-auto mb-3" />
   517→            <h3 className="font-semibold text-foreground mb-1">Loading Timed Out</h3>
   518→            <p className="text-sm text-muted-foreground mb-3">Could not load team data. Showing demo data instead.</p>
   519→            <Button variant="outline" size="sm" onClick={() => window.location.reload()}>Refresh</Button>
   520→          </Card>
   521→        ) : (
   522→          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
   523→            <Card className="p-5">
   524→              <div className="flex items-center gap-3">
   525→                <div className="p-2.5 bg-purple-500/10 rounded-lg"><Users className="w-5 h-5 text-purple-500" /></div>
   526→                <div>
   527→                  <p className="text-sm text-muted-foreground">Members</p>
   528→                  <p className="text-2xl font-bold">{stats?.memberCount ?? activeMembers.length}</p>
   529→                </div>
   530→              </div>
   531→            </Card>
