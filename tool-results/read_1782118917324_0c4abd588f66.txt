     1→import { useState, useEffect } from "react";
     2→import { motion } from "framer-motion";
     3→import { toast } from "sonner";
     4→import {
     5→  Tag,
     6→  Plus,
     7→  Search,
     8→  Pencil,
     9→  Trash2,
    10→  Hash,
    11→  X,
    12→  Palette,
    13→  Loader2,
    14→} from "lucide-react";
    15→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    16→import { Button } from "@/components/ui/button";
    17→import { Badge } from "@/components/ui/badge";
    18→import { Input } from "@/components/ui/input";
    19→import { Label } from "@/components/ui/label";
    20→import { Textarea } from "@/components/ui/textarea";
    21→import { Separator } from "@/components/ui/separator";
    22→import { Skeleton } from "@/components/ui/skeleton";
    23→import {
    24→  Dialog,
    25→  DialogContent,
    26→  DialogDescription,
    27→  DialogFooter,
    28→  DialogHeader,
    29→  DialogTitle,
    30→  DialogTrigger,
    31→} from "@/components/ui/dialog";
    32→import { useAuth } from "@/hooks/use-auth";
    33→import { useQuery, useMutation, useQueryTimeout, useConvexConnectionState } from "@/lib/safe-convex-react";
    34→import { api } from "@/convex/_generated/api";
    35→import { PageLayout } from "@/components/design-system/PageLayout";
    36→
    37→// ─── Constants ───────────────────────────────────────────────────────────────
    38→
    39→const PRESET_COLORS = [
    40→  "#ef4444", "#f97316", "#f59e0b", "#84cc16", "#22c55e",
    41→  "#14b8a6", "#06b6d4", "#6366f1", "#a855f7", "#ec4899",
    42→  "#f43f5e", "#0ea5e9",
    43→];
    44→
    45→// ─── Component ───────────────────────────────────────────────────────────────
    46→
    47→export default function Tags() {
    48→  const { isAuthenticated, isLoading: authLoading } = useAuth();
    49→
    50→  // ─── Convex queries & mutations ──────────────────────────────────────────
    51→  const tagsData = useQuery(api.tags.crud.getTags, {});
    52→  const createTagMutation = useMutation(api.tags.crud.createTag);
    53→  const updateTagMutation = useMutation(api.tags.crud.updateTag);
    54→  const deleteTagMutation = useMutation(api.tags.crud.deleteTag);
    55→
    56→  // ─── Local state ────────────────────────────────────────────────────────
    57→  const [searchQuery, setSearchQuery] = useState("");
    58→  const [activeFilter, setActiveFilter] = useState<string | null>(null);
    59→  const [createOpen, setCreateOpen] = useState(false);
    60→  const [editOpen, setEditOpen] = useState(false);
    61→  const [editingTag, setEditingTag] = useState<any>(null);
    62→  const [deleteOpen, setDeleteOpen] = useState(false);
    63→  const [deletingTag, setDeletingTag] = useState<any>(null);
    64→
    65→  // Form state
    66→  const [formName, setFormName] = useState("");
    67→  const [formColor, setFormColor] = useState(PRESET_COLORS[0]);
    68→  const [formCategory, setFormCategory] = useState<string>("general");
    69→
    70→  // Mutation loading state
    71→  const [isCreating, setIsCreating] = useState(false);
    72→  const [isUpdating, setIsUpdating] = useState(false);
    73→  const [isDeleting, setIsDeleting] = useState(false);
    74→
    75→  // ─── Loading timeout pattern ────────────────────────────────────────────
    76→  const { isDisconnected } = useConvexConnectionState();
    77→  const timedOut = useQueryTimeout(!authLoading && tagsData === undefined, 3000);
    78→
    79→  const isLoading = !authLoading && tagsData === undefined && !timedOut && !isDisconnected;
    80→
    81→  // ─── Demo mode ──────────────────────────────────────────────────────────
    82→  const isDemoMode = !authLoading && !isAuthenticated;
    83→
    84→  // ─── Data resolution ────────────────────────────────────────────────────
    85→  const realTags = (tagsData ?? []).map((t: any) => ({
    86→    _id: t._id,
    87→    name: t.name,
    88→    color: t.color,
    89→    category: t.category ?? null,
    90→    usageCount: t.usageCount ?? 0,
    91→    createdAt: t.createdAt,
    92→  }));
    93→
    94→  const tags = realTags;
    95→
    96→  // ── Computed Stats ──
    97→  const totalTags = tags.length;
    98→  const mostUsedTag =
    99→    tags.length > 0
   100→      ? tags.reduce((prev: any, curr: any) => (curr.usageCount > prev.usageCount ? curr : prev))
   101→      : null;
   102→
   103→  // ── Filtering ──
   104→  const filteredTags = tags.filter((tag: any) => {
   105→    const matchesSearch =
   106→      tag.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   107→      (tag.category ?? "").toLowerCase().includes(searchQuery.toLowerCase());
   108→    const matchesFilter = activeFilter ? tag._id === activeFilter : true;
   109→    return matchesSearch && matchesFilter;
   110→  });
   111→
   112→  // ── Helpers ──
   113→  const resetForm = () => {
   114→    setFormName("");
   115→    setFormColor(PRESET_COLORS[0]);
   116→    setFormCategory("general");
   117→  };
   118→
   119→  const formatDate = (timestamp: number) => {
   120→    const date = new Date(timestamp);
   121→    return date.toLocaleDateString("en-US", {
   122→      month: "short",
   123→      day: "numeric",
   124→      year: "numeric",
   125→    });
   126→  };
   127→
   128→  // ── Create ──
   129→  const handleCreate = async () => {
   130→    if (!formName.trim()) {
   131→      toast.error("Tag name is required");
   132→      return;
   133→    }
   134→
   135→    if (isDemoMode) {
   136→      toast.success(`Tag "${formName.trim()}" created successfully! (Demo mode)`);
   137→      resetForm();
   138→      setCreateOpen(false);
   139→      return;
   140→    }
   141→
   142→    setIsCreating(true);
   143→    try {
   144→      await createTagMutation({
   145→        name: formName.trim(),
   146→        color: formColor,
   147→        category: formCategory,
   148→      });
   149→      resetForm();
   150→      setCreateOpen(false);
   151→      toast.success(`Tag "${formName.trim()}" created successfully`);
   152→    } catch (err: any) {
   153→      toast.error(err?.message || "Failed to create tag");
   154→    } finally {
   155→      setIsCreating(false);
   156→    }
   157→  };
   158→
   159→  // ── Edit ──
   160→  const openEdit = (tag: any) => {
   161→    setEditingTag(tag);
   162→    setFormName(tag.name);
   163→    setFormColor(tag.color);
   164→    setFormCategory(tag.category ?? "general");
   165→    setEditOpen(true);
   166→  };
   167→
   168→  const handleEdit = async () => {
   169→    if (!editingTag) return;
   170→    if (!formName.trim()) {
   171→      toast.error("Tag name is required");
   172→      return;
   173→    }
   174→
   175→    if (isDemoMode) {
   176→      toast.success(`Tag "${formName.trim()}" updated! (Demo mode)`);
   177→      resetForm();
   178→      setEditOpen(false);
   179→      setEditingTag(null);
   180→      return;
   181→    }
   182→
   183→    setIsUpdating(true);
   184→    try {
   185→      await updateTagMutation({
   186→        tagId: editingTag._id,
   187→        name: formName.trim(),
   188→        color: formColor,
   189→        category: formCategory,
   190→      });
   191→      resetForm();
   192→      setEditOpen(false);
   193→      setEditingTag(null);
   194→      toast.success(`Tag "${formName.trim()}" updated`);
   195→    } catch (err: any) {
   196→      toast.error(err?.message || "Failed to update tag");
   197→    } finally {
   198→      setIsUpdating(false);
   199→    }
   200→  };
   201→
   202→  // ── Delete ──
   203→  const openDelete = (tag: any) => {
   204→    setDeletingTag(tag);
   205→    setDeleteOpen(true);
   206→  };
   207→
   208→  const handleDelete = async () => {
   209→    if (!deletingTag) return;
   210→
   211→    if (isDemoMode) {
   212→      toast.success(`Tag "${deletingTag.name}" deleted! (Demo mode)`);
   213→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   214→      setDeleteOpen(false);
   215→      setDeletingTag(null);
   216→      return;
   217→    }
   218→
   219→    setIsDeleting(true);
   220→    try {
   221→      await deleteTagMutation({ tagId: deletingTag._id });
   222→      if (activeFilter === deletingTag._id) setActiveFilter(null);
   223→      setDeleteOpen(false);
   224→      setDeletingTag(null);
   225→      toast.success(`Tag "${deletingTag.name}" deleted`);
   226→    } catch (err: any) {
   227→      toast.error(err?.message || "Failed to delete tag");
   228→    } finally {
   229→      setIsDeleting(false);
   230→    }
   231→  };
   232→
   233→  // ── Render ──
   234→  return (
   235→    <motion.div
   236→      initial={{ opacity: 0 }}
   237→      animate={{ opacity: 1 }}
   238→      className="w-full min-h-screen bg-background text-foreground"
   239→    >
   240→      <PageLayout spaced>
   241→        {/* ── Header ── */}
   242→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   243→          <div>
   244→            <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   245→              Tags
   246→            </h1>
   247→            <p className="text-[16px] text-muted-foreground">
   248→              Organize your work with custom tags for quick filtering and categorization
   249→            </p>
   250→          </div>
   251→
   252→          <Dialog open={createOpen} onOpenChange={(open) => { setCreateOpen(open); if (!open) resetForm(); }}>
   253→            <DialogTrigger asChild>
   254→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   255→                <Plus className="mr-2 h-4 w-4" />
   256→                Create Tag
   257→              </Button>
   258→            </DialogTrigger>
   259→            <DialogContent className="sm:max-w-md">
   260→              <DialogHeader>
   261→                <DialogTitle>Create New Tag</DialogTitle>
   262→                <DialogDescription>
   263→                  Add a tag to categorize and filter your time entries, projects, and evidence.
   264→                </DialogDescription>
   265→              </DialogHeader>
   266→              <div className="space-y-4 py-2">
   267→                <div className="space-y-2">
   268→                  <Label htmlFor="tag-name">Name</Label>
   269→                  <Input
   270→                    id="tag-name"
   271→                    placeholder="e.g. Urgent, Design, Revision"
   272→                    value={formName}
   273→                    onChange={(e) => setFormName(e.target.value)}
   274→                  />
   275→                </div>
   276→                <div className="space-y-2">
   277→                  <Label>Color</Label>
   278→                  <div className="flex items-center gap-2 flex-wrap">
   279→                    {PRESET_COLORS.map((color) => (
   280→                      <button
   281→                        key={color}
   282→                        type="button"
   283→                        className="w-8 h-8 rounded-full border-2 transition-all hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-primary"
   284→                        style={{
   285→                          backgroundColor: color,
   286→                          borderColor: formColor === color ? "var(--foreground)" : "transparent",
   287→                          boxShadow: formColor === color ? `0 0 0 2px ${color}40` : "none",
   288→                        }}
   289→                        onClick={() => setFormColor(color)}
   290→                      />
   291→                    ))}
   292→                    <div className="flex items-center gap-1 ml-1">
   293→                      <Palette className="h-4 w-4 text-muted-foreground" />
   294→                      <input
   295→                        type="color"
   296→                        value={formColor}
   297→                        onChange={(e) => setFormColor(e.target.value)}
   298→                        className="w-8 h-8 rounded cursor-pointer border border-border"
   299→                      />
   300→                    </div>
   301→                  </div>
   302→                </div>
   303→                <div className="space-y-2">
   304→                  <Label>Category</Label>
   305→                  <div className="flex flex-wrap gap-2">
   306→                    {["general", "client", "project", "evidence"].map((cat) => (
   307→                      <button key={cat} type="button" onClick={() => setFormCategory(cat)}>
   308→                        <Badge
   309→                          variant={formCategory === cat ? "default" : "outline"}
   310→                          className="cursor-pointer transition-colors capitalize"
   311→                        >
   312→                          {cat}
   313→                        </Badge>
   314→                      </button>
   315→                    ))}
   316→                  </div>
   317→                </div>
   318→              </div>
   319→              <DialogFooter>
   320→                <Button variant="outline" onClick={() => { setCreateOpen(false); resetForm(); }} disabled={isCreating}>
   321→                  Cancel
   322→                </Button>
   323→                <Button className="bg-primary hover:bg-primary/90" onClick={handleCreate} disabled={isCreating}>
   324→                  {isCreating ? (
   325→                    <>
   326→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   327→                      Creating...
   328→                    </>
   329→                  ) : (
   330→                    "Create Tag"
   331→                  )}
   332→                </Button>
   333→              </DialogFooter>
   334→            </DialogContent>
   335→          </Dialog>
   336→        </div>
   337→
   338→        {/* ── Demo mode empty state ── */}
   339→        {isDemoMode && (
   340→          <Card className="p-8 bg-card rounded-xl border border-border">
   341→            <div className="text-center space-y-4">
   342→              <div className="mx-auto h-16 w-16 rounded-full bg-primary/10 flex items-center justify-center">
   343→                <Tag className="h-8 w-8 text-primary" />
   344→              </div>
   345→              <div>
   346→                <h3 className="text-lg font-semibold text-foreground">Sign in to see your tags</h3>
   347→                <p className="text-sm text-muted-foreground mt-1">
   348→                  Connect your account to organize your work with custom tags.
   349→                </p>
   350→              </div>
   351→              <Button asChild>
   352→                <a href="/auth">Sign In</a>
   353→              </Button>
   354→            </div>
   355→          </Card>
   356→        )}
   357→
   358→        {/* ── Loading state ── */}
   359→        {isLoading ? (
   360→          <div className="space-y-4">
   361→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   362→              <Skeleton className="h-[76px] rounded-xl" />
   363→              <Skeleton className="h-[76px] rounded-xl" />
   364→              <Skeleton className="h-[76px] rounded-xl" />
   365→            </div>
   366→            <Skeleton className="h-10 rounded-lg" />
   367→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   368→              {[1, 2, 3, 4, 5, 6].map((i) => (
   369→                <Skeleton key={i} className="h-[160px] rounded-xl" />
   370→              ))}
   371→            </div>
   372→          </div>
   373→        ) : (
   374→          <>
   375→            {/* ── Stats Cards ── */}
   376→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   377→              <Card>
   378→                <CardContent className="p-6">
   379→                  <div className="flex items-center gap-3">
   380→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-primary/10">
   381→                      <Hash className="h-5 w-5 text-primary" />
   382→                    </div>
   383→                    <div>
   384→                      <p className="text-sm text-muted-foreground">Total Tags</p>
   385→                      <p className="text-2xl font-bold">{totalTags}</p>
   386→                    </div>
   387→                  </div>
   388→                </CardContent>
   389→              </Card>
   390→              <Card>
   391→                <CardContent className="p-6">
   392→                  <div className="flex items-center gap-3">
   393→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-orange-500/10">
   394→                      <Tag className="h-5 w-5 text-orange-500" />
   395→                    </div>
   396→                    <div>
   397→                      <p className="text-sm text-muted-foreground">Most Used</p>
   398→                      <p className="text-2xl font-bold">
   399→                        {mostUsedTag ? mostUsedTag.name : "—"}
   400→                      </p>
   401→                      {mostUsedTag && (
   402→                        <p className="text-xs text-muted-foreground">
   403→                          {mostUsedTag.usageCount} {mostUsedTag.usageCount === 1 ? "entry" : "entries"}
   404→                        </p>
   405→                      )}
   406→                    </div>
   407→                  </div>
   408→                </CardContent>
   409→              </Card>
   410→              <Card>
   411→                <CardContent className="p-6">
   412→                  <div className="flex items-center gap-3">
   413→                    <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-emerald-500/10">
   414→                      <Hash className="h-5 w-5 text-emerald-500" />
   415→                    </div>
   416→                    <div>
   417→                      <p className="text-sm text-muted-foreground">Categories</p>
   418→                      <p className="text-2xl font-bold">
   419→                        {new Set(tags.map((t: any) => t.category ?? "uncategorized")).size}
   420→                      </p>
   421→                    </div>
   422→                  </div>
   423→                </CardContent>
   424→              </Card>
   425→            </div>
   426→
   427→            {/* ── Quick Filters + Search ── */}
   428→            <div className="space-y-3">
   429→              <div className="flex items-center gap-2">
   430→                <div className="relative flex-1">
   431→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   432→                  <Input
   433→                    placeholder="Search tags..."
   434→                    value={searchQuery}
   435→                    onChange={(e) => setSearchQuery(e.target.value)}
   436→                    className="pl-9"
   437→                  />
   438→                </div>
   439→                {activeFilter && (
   440→                  <Button
   441→                    variant="outline"
   442→                    size="sm"
   443→                    onClick={() => setActiveFilter(null)}
   444→                    className="shrink-0"
   445→                  >
   446→                    <X className="mr-1 h-3 w-3" />
   447→                    Clear filter
   448→                  </Button>
   449→                )}
   450→              </div>
   451→              <div className="flex items-center gap-2 flex-wrap">
   452→                <span className="text-sm text-muted-foreground shrink-0">Quick filter:</span>
   453→                {tags.slice(0, 5).map((tag: any) => (
   454→                  <button
   455→                    key={tag._id}
   456→                    type="button"
   457→                    onClick={() => setActiveFilter(activeFilter === tag._id ? null : tag._id)}
   458→                    className="focus:outline-none"
   459→                  >
   460→                    <Badge
   461→                      variant={activeFilter === tag._id ? "default" : "outline"}
   462→                      className="cursor-pointer transition-colors"
   463→                      style={
   464→                        activeFilter === tag._id
   465→                          ? { backgroundColor: tag.color, borderColor: tag.color }
   466→                          : { borderColor: tag.color, color: tag.color }
   467→                      }
   468→                    >
   469→                      <span
   470→                        className="inline-block w-2 h-2 rounded-full mr-1"
   471→                        style={{ backgroundColor: tag.color }}
   472→                      />
   473→                      {tag.name}
   474→                    </Badge>
   475→                  </button>
   476→                ))}
   477→              </div>
   478→            </div>
   479→
   480→            <Separator />
   481→
   482→            {/* ── Tags Grid ── */}
   483→            {filteredTags.length === 0 ? (
   484→              <Card>
   485→                <CardContent className="py-12">
   486→                  <div className="text-center text-muted-foreground">
   487→                    <Tag className="h-12 w-12 mx-auto mb-4 opacity-50" />
   488→                    <p className="text-lg font-medium">No tags found</p>
   489→                    <p className="text-sm">
   490→                      {searchQuery
   491→                        ? "Try adjusting your search query"
   492→                        : "Create your first tag to start organizing your work"}
   493→                    </p>
   494→                    {!searchQuery && (
   495→                      <Button
   496→                        className="mt-4 bg-primary hover:bg-primary/90"
   497→                        onClick={() => setCreateOpen(true)}
   498→                      >
   499→                        <Plus className="mr-2 h-4 w-4" />
   500→                        Create Your First Tag
   501→                      </Button>
   502→                    )}
   503→                  </div>
   504→                </CardContent>
   505→              </Card>
   506→            ) : (
   507→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   508→                {filteredTags.map((tag: any, index: number) => (
   509→                  <motion.div
   510→                    key={tag._id}
   511→                    initial={{ opacity: 0, y: 12 }}
   512→                    animate={{ opacity: 1, y: 0 }}
   513→                    transition={{ delay: index * 0.04, duration: 0.3 }}
   514→                  >
   515→                    <Card className="group hover:shadow-md transition-shadow">
   516→                      <CardHeader className="pb-3">
   517→                        <div className="flex items-start justify-between">
   518→                          <div className="flex items-center gap-3">
   519→                            <span
   520→                              className="w-5 h-5 rounded-full shrink-0 ring-2 ring-offset-2 ring-offset-background"
   521→                              style={{ backgroundColor: tag.color, outlineColor: tag.color }}
   522→                            />
   523→                            <CardTitle className="text-base">{tag.name}</CardTitle>
   524→                          </div>
   525→                          <div className="flex items-center gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
   526→                            <Button
   527→                              variant="ghost"
   528→                              size="icon"
   529→                              className="h-8 w-8"
   530→                              onClick={() => openEdit(tag)}
   531→                            >
   532→                              <Pencil className="h-3.5 w-3.5" />
   533→                            </Button>
   534→                            <Button
   535→                              variant="ghost"
   536→                              size="icon"
   537→                              className="h-8 w-8 text-destructive hover:text-destructive"
   538→                              onClick={() => openDelete(tag)}
   539→                            >
   540→                              <Trash2 className="h-3.5 w-3.5" />
   541→                            </Button>
   542→                          </div>
   543→                        </div>
   544→                      </CardHeader>
   545→                      <CardContent className="pt-0 space-y-3">
   546→                        {tag.category && (
   547→                          <Badge variant="secondary" className="text-xs capitalize">
   548→                            {tag.category}
   549→                          </Badge>
   550→                        )}
   551→                        <Separator />
   552→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   553→                          <span>
   554→                            <span className="font-semibold text-foreground">{tag.usageCount}</span>{" "}
   555→                            {tag.usageCount === 1 ? "entry" : "entries"} tagged
   556→                          </span>
   557→                          <span>
   558→                            {tag.usageCount === 0
   559→                              ? "Never used"
   560→                              : `Created ${formatDate(tag.createdAt)}`}
   561→                          </span>
   562→                        </div>
   563→                      </CardContent>
   564→                    </Card>
   565→                  </motion.div>
   566→                ))}
   567→              </div>
   568→            )}
   569→          </>
   570→        )}
   571→      </PageLayout>
   572→
   573→      {/* ── Edit Tag Dialog ── */}
   574→      <Dialog
