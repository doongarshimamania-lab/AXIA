    96→function getTierLevel(tier: string): number {
    97→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
    98→  return levels[tier] ?? 0;
    99→}
   100→
   101→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   102→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   103→}

   241→// --- Main Component ---
   242→export default function EvidenceExport() {
   243→  const { tier } = useSubscriptionTier();
   244→  const { isAuthenticated } = useConvexAuth();
   245→
   246→  // ─── Convex Queries ────────────────────────────────────────────────────────
   247→  const evidenceData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   248→    view: "date" as const,
   249→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   250→    endDate: Date.now(),
   251→  }) as any | undefined;
   252→
   253→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   254→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   255→
   256→  const isLoading = evidenceData === undefined || clients === undefined;
   257→  const timedOut = useQueryTimeout(isLoading, 6000);
   258→
   259→  // ─── Derived Data ──────────────────────────────────────────────────────────
   260→
   261→  // Build evidence type options from real data
   262→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   263→    if (!evidenceData) return [];
   264→    const items = evidenceData.evidenceItems ?? [];
   265→    const typeCounts: Record<string, number> = {};
   266→    const typeLabels: Record<string, string> = {
   267→      screenshot_ref: "Screenshots",
   268→      memo: "Work Memos",
   269→      url: "URL Visits",
   270→      keyboard: "Keyboard Activity",
   271→      mouse: "Mouse Activity",
   272→      platform_status: "Platform Status",
   273→    };
   274→    const typeIcons: Record<string, React.ReactNode> = {
   275→      screenshot_ref: <Eye className="h-4 w-4" />,
   276→      memo: <FileText className="h-4 w-4" />,
   277→      url: <FileJson className="h-4 w-4" />,
   278→      keyboard: <FileText className="h-4 w-4" />,
   279→      mouse: <FileText className="h-4 w-4" />,
   280→      platform_status: <Shield className="h-4 w-4" />,
   281→    };
   282→
   283→    for (const item of items) {
   284→      const kind = item.type ?? "unknown";
   285→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   286→    }
   287→
   288→    return Object.entries(typeCounts).map(([id, count]) => ({
   289→      id,
   290→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   291→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   292→      count,
   293→    }));
   294→  }, [evidenceData]);
   295→
   296→  // Build project options from real scopes
   297→  const projectOptions = useMemo(() => {
   298→    const all = [{ id: "all", label: "All Projects" }];
   299→    if (scopeDefinitions) {
   300→      for (const scope of scopeDefinitions) {
   301→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   302→      }
   303→    }
   304→    return all;
   305→  }, [scopeDefinitions]);
   306→
   307→  // Build client options from real clients
   308→  const clientOptions = useMemo(() => {
   309→    const all = [{ id: "all", label: "All Clients" }];
   310→    if (clients) {
   311→      for (const client of clients) {
   312→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   313→      }
   314→    }
   315→    return all;
   316→  }, [clients]);
   317→
   318→  // Compliance score from real data
   319→  const complianceScore = evidenceData?.healthScore?.score ?? 0;
   320→  const totalItems = evidenceData?.totalCount ?? 0;
   321→  const disputeSuccessRate = evidenceData?.disputeSuccessRate ?? 0;
   322→
   323→  // State
   324→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   325→  const [dateFrom, setDateFrom] = useState(() => {
   326→    const d = new Date();
   327→    d.setMonth(d.getMonth() - 3);
   328→    return d.toISOString().split("T")[0];
   329→  });
   330→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   331→  const [selectedProject, setSelectedProject] = useState("all");
   332→  const [selectedClient, setSelectedClient] = useState("all");
   333→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   334→  const [isExporting, setIsExporting] = useState(false);
   335→  const [previewOpen, setPreviewOpen] = useState(false);
   336→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   337→
   338→  // Auto-select all evidence types when they load
   339→  useMemo(() => {
   340→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   341→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   342→    }
   343→  }, [evidenceTypes]);
   344→
   345→  // Derived
   346→  const totalEvidenceItems = useMemo(
   347→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   348→    [evidenceTypes, selectedEvidenceTypes]
   349→  );
   350→
   351→  const isFormatLocked = (format: ExportFormat): boolean => {
   352→    const requiredTier = FORMAT_CONFIG[format].tierRequired;
   353→    return !hasTierAccess(tier, requiredTier);
   354→  };
   355→
   356→  const toggleEvidenceType = (id: string) => {
   357→    setSelectedEvidenceTypes((prev) =>
   358→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   359→    );
   360→  };
   361→
   362→  const handleExport = async () => {
   363→    if (selectedEvidenceTypes.length === 0) {
   364→      toast.error("No evidence types selected", {
   365→        description: "Select at least one evidence type to export.",
   366→      });
   367→      return;
   368→    }
   369→
   370→    if (isFormatLocked(selectedFormat)) {
   371→      toast.error("Upgrade required", {
   372→        description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.`,
   373→      });
   374→      return;
   375→    }
   376→
   377→    setIsExporting(true);
   378→
   379→    try {
   380→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   381→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   382→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   383→
   384→      exportEvidence(selectedFormat, {
   385→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   386→        dateRange: `${dateFrom} to ${dateTo}`,
   387→        project: selectedProjectLabel,
   388→        client: selectedClientLabel,
   389→        complianceScore,
   390→        totalItems: totalEvidenceItems,
   391→      });
   392→
   393→      // Add to recent exports
   394→      const newExport: RecentExport = {
   395→        id: `exp_${Date.now()}`,
   396→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   397→        format: selectedFormat,
   398→        status: "completed",
   399→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   400→        date: new Date().toISOString().split("T")[0],
   401→        itemCount: totalEvidenceItems,
   402→        complianceVerified: hasTierAccess(tier, "pro"),
   403→        project: selectedProjectLabel,
   404→      };
   405→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   406→
   407→      toast.success("Export complete", {
   408→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   409→      });
   410→    } catch {
   411→      toast.error("Export failed", {
   412→        description: "An error occurred while generating the export file.",
   413→      });
   414→    }
   415→
   416→    setIsExporting(false);
   417→  };
   418→
   419→  const handleFormatSelect = (format: ExportFormat) => {
   420→    if (isFormatLocked(format)) {
   421→      toast.error("Upgrade required", {
   422→        description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.`,
   423→      });
   424→      return;
   425→    }
   426→    setSelectedFormat(format);
   427→  };
   428→
   429→  const previewItems = useMemo(() => {
   430→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   431→  }, [evidenceTypes, selectedEvidenceTypes]);
   432→
   433→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   434→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   435→
   436→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   437→
   438→  // ─── Render ───────────────────────────────────────────────────────────────
   439→
   440→  return (
   441→    <motion.div
   442→      initial={{ opacity: 0 }}
   443→      animate={{ opacity: 1 }}
   444→      className="w-full min-h-screen bg-background"
   445→    >
   446→      <div className="p-6 md:p-8 space-y-8">
   447→        {/* Header */}
   448→        <div>
   449→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">Evidence Export</h1>
   450→          <p className="text-[16px] text-muted-foreground">
   451→            Package and download your work evidence in court-ready formats with compliance verification.
   452→          </p>
   453→        </div>
   454→
   455→        {/* Demo Mode Banner */}
   456→        {!isAuthenticated && <DemoModeBanner />}
   457→
   458→        {/* Loading / Timeout State */}
   459→        {isLoading && !timedOut ? (
   460→          <>
   461→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   462→              {[0, 1, 2, 3].map((i) => <SummarySkeleton key={i} />)}
   463→            </div>
   464→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   465→              <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   466→              <div><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   467→            </div>
   468→          </>
   469→        ) : isLoading && timedOut ? (
   470→          <div className="flex flex-col items-center justify-center py-20 px-4 text-center">
   471→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mb-4">
   472→              <AlertTriangle className="h-8 w-8 text-destructive" />
   473→            </div>
   474→            <h3 className="text-lg font-semibold text-foreground mb-2">
   475→              Data is taking longer than expected
   476→            </h3>
   477→            <p className="text-sm text-muted-foreground max-w-md mb-6">
   478→              The evidence data could not be loaded. This might be due to a connection issue or the data not being available yet.
   479→            </p>
   480→            <Button variant="outline" onClick={() => window.location.reload()} className="gap-2">
   481→              <RefreshCw className="h-4 w-4" />
   482→              Reload page
   483→            </Button>
   484→          </div>
   485→        ) : !hasData ? (
   486→          <EmptyEvidenceState />
   487→        ) : (
   488→          <>
   489→            {/* Summary Dashboard */}
   490→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   491→              <SummaryCard
   492→                title="Total Items"
   493→                value={totalItems}
   494→                subtitle="Across all projects"
   495→                icon={<HardDrive className="h-5 w-5 text-primary-foreground" />}
   496→                accent="bg-primary"
   497→              />
   498→              <SummaryCard
   499→                title="Evidence Types"
   500→                value={evidenceTypes.length}
   501→                subtitle="Different categories"
   502→                icon={<Clock className="h-5 w-5 text-amber-700" />}
   503→                accent="bg-amber-100 dark:bg-amber-900/30"
   504→              />
   505→              <SummaryCard
   506→                title="Compliance Score"
   507→                value={`${complianceScore}%`}
   508→                subtitle={complianceScore >= 80 ? "Court-admissible ready" : "Needs improvement"}
   509→                icon={<ShieldCheck className="h-5 w-5 text-emerald-700" />}
   510→                accent="bg-emerald-100 dark:bg-emerald-900/30"
   511→              />
   512→              <SummaryCard
   513→                title="Dispute Success"
   514→                value={`${disputeSuccessRate}%`}
   515→                subtitle="Based on evidence quality"
   516→                icon={<Shield className="h-5 w-5 text-sky-700" />}
   517→                accent="bg-sky-100 dark:bg-sky-900/30"
   518→              />
   519→            </div>
   520→
   521→            {/* Export Configuration */}
   522→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   523→              {/* Left Column: Format + Filters */}
   524→              <div className="lg:col-span-2 space-y-6">
   525→                {/* Format Selection */}
   526→                <Card>
   527→                  <CardHeader>
   528→                    <CardTitle className="flex items-center gap-2">
   529→                      <Download className="h-5 w-5 text-primary" />
   530→                      Export Format
   531→                    </CardTitle>
   532→                    <CardDescription>Choose the output format for your evidence package.</CardDescription>
   533→                  </CardHeader>
   534→                  <CardContent>
   535→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   536→                      {(Object.keys(FORMAT_CONFIG) as ExportFormat[]).map((formatKey) => (
   537→                        <FormatOption
   538→                          key={formatKey}
   539→                          formatKey={formatKey}
   540→                          selected={selectedFormat === formatKey}
   541→                          onSelect={() => handleFormatSelect(formatKey)}
   542→                          locked={isFormatLocked(formatKey)}
   543→                        />
   544→                      ))}
