     1→/**
     2→ * Centralized Data Store for Axia
     3→ *
     4→ * All pages share this single source of truth. Every entity cross-references
     5→ * others by ID so the UI can show connected data (e.g. which team members
     6→ * are on a project, which projects belong to a client, etc.).
     7→ *
     8→ * HYBRID DATA SOURCE:
     9→ * Pipeline deals and proposals use Convex as the primary data source when
    10→ * the user is authenticated. When Convex returns no data (unauthenticated),
    11→ * the mock data is used as a fallback. Other entities still use local state.
    12→ */

   127→  contactName: string | null;

   131→  id: string;
   132→  name: string;
   133→  email: string;
   134→  image: string | null;
   135→  role: string;
   136→  title: string | null;
   137→}
   138→
   139→export interface RichProposal {
   140→  id: string;
   141→  title: string;
   142→  status: string;
   143→  totalValue: number;
   144→}
   145→
   146→export interface RichDeal {
   147→  id: string;
   148→  title: string;
   149→  value: number;
   150→  stageName: string;
   151→  stageColor: string;
   152→}
   153→
   154→export interface PipelineDeal {
   155→  id: string;
   156→  title: string;
   157→  clientId: string | null;
   158→  stage: DealStage;
   159→  value: number;
   160→  probability: number;
   161→  source: string;
   162→  contactName: string;
   163→  contactEmail: string;
   164→  expectedCloseDate: number | null;
   165→  assignedMemberId: string | null;
   166→  notes: string;
   167→  createdAt: number;
   168→  proposalId?: string | null;
   169→  description?: string;
   170→  workspaceId?: string | null;
   171→  // Enriched fields (resolved server-side from Convex)
   172→  client?: RichClient | null;
   173→  assignedMember?: RichMember | null;
   174→  linkedProposal?: RichProposal | null;
   175→  stageName?: string;
   176→  stageColor?: string;
   177→}
   178→
   179→export type ProposalStatus = "draft" | "sent" | "viewed" | "signed" | "declined" | "expired";
   180→
   181→export interface Proposal {
   182→  id: string;
   183→  title: string;
   184→  clientId: string | null;
   185→  status: ProposalStatus;
   186→  totalValue: number;
   187→  clientEmail: string;
   188→  assignedMemberId: string | null;
   189→  sentAt: number | null;
   190→  viewedAt: number | null;
   191→  signedAt: number | null;
   192→  validUntil: number | null;
   193→  createdAt: number;
   194→  dealId?: string | null;
   195→  workspaceId?: string | null;
   196→  // Enriched fields (resolved server-side from Convex)
   197→  client?: RichClient | null;
   198→  assignedMember?: RichMember | null;
   199→  linkedDeal?: RichDeal | null;
   200→}

  1870→  // Lookup helpers

  1873→  getMember: (id: string) => TeamMember | undefined;
  1874→  getProjectsForClient: (clientId: string) => Project[];
  1875→  getProjectsForMember: (memberId: string) => Project[];
  1876→  getClientsForMember: (memberId: string) => Client[];
  1877→  getInvoicesForClient: (clientId: string) => Invoice[];
  1878→  getInvoicesForProject: (projectId: string) => Invoice[];
  1879→  getTimeEntriesForProject: (projectId: string) => TimeEntry[];
  1880→  getTimeEntriesForMember: (memberId: string) => TimeEntry[];
  1881→  getDealsForClient: (clientId: string) => PipelineDeal[];
  1882→  getProposalsForClient: (clientId: string) => Proposal[];
  1883→
  1884→  // Mutations (update local state)
  1885→  assignMemberToProject: (projectId: string, memberId: string) => void;
  1886→  unassignMemberFromProject: (projectId: string, memberId: string) => void;
  1887→  assignMemberToClient: (clientId: string, memberId: string) => void;
  1888→  unassignMemberFromClient: (clientId: string, memberId: string) => void;
  1889→  updateProjectStatus: (projectId: string, status: ProjectStatus) => void;
  1890→  updateInvoiceStatus: (invoiceId: string, status: InvoiceStatus) => void;
  1891→  addTimeEntry: (entry: Omit<TimeEntry, "id">) => void;
  1892→  addProject: (project: Omit<Project, "id" | "createdAt" | "lastActivityAt">) => void;
  1893→  addClient: (client: Omit<Client, "id" | "addedAt" | "lastActivityAt">) => void;
  1894→  moveDealToStage: (dealId: string, stage: DealStage) => void;
  1895→  updateProposalStatus: (proposalId: string, status: ProposalStatus) => void;
  1896→}

  1900→export function useAppData() {
  1901→  const ctx = useContext(AppDataCtx);

  1903→  return ctx;
  1904→}
