   145→// ─── Helpers ────────────────────────────────────────────────────────────────
   146→
   147→function generateId(): string {
   148→  return `li_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`;
   149→}
   150→
   151→function formatCurrency(amount: number, currency: string = "USD"): string {
   152→  try {
   153→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   154→  } catch {
   155→    return `$${amount.toFixed(2)}`;
   156→  }
   157→}
   158→
   159→function formatDate(ts: number): string {
   160→  return new Date(ts).toLocaleDateString("en-US", {
   161→    month: "short",
   162→    day: "numeric",
   163→    year: "numeric",
   164→  });
   165→}
   166→
   167→function dateToTimestamp(dateStr: string): number {
   168→  return new Date(dateStr + "T00:00:00").getTime();
   169→}
   170→
   171→function timestampToDate(ts: number): string {
   172→  const d = new Date(ts);
   173→  return d.toISOString().split("T")[0];
   174→}
