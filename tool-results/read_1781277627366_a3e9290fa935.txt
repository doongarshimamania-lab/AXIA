   116→function getTierLevel(tier: string): number {
   117→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
   118→  return levels[tier] ?? 0;
   119→}
   120→
   121→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   122→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   123→}

   221→

   225→
   226→  const [viewMode, setViewMode] = useState<ViewType>("date");
   227→

   229→  const libraryDateRange = useMemo(() => {
   230→    const now = new Date();
   231→    const start = new Date(now);
   232→    start.setDate(start.getDate() - 30);
   233→    const end = new Date(now);
   234→    end.setDate(end.getDate() + 30);
   235→    return { start: start.getTime(), end: end.getTime() };
   236→  }, []);
   237→
   238→  const startOfDay = useMemo(() => new Date().setHours(0, 0, 0, 0), []);
   239→
   240→  // ── Export date range (1 year for comprehensive export) ──
   241→  const [dateFrom, setDateFrom] = useState(() => {
   242→    const d = new Date();
   243→    d.setMonth(d.getMonth() - 3);
   244→    return d.toISOString().split("T")[0];
   245→  });
   246→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   247→
   248→  // ── Convex Queries ──
   249→  // Library data with ±30 day window for monitoring
   250→  const evidenceData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   251→    view: viewMode,
   252→    startDate: libraryDateRange.start,
   253→    endDate: libraryDateRange.end,
   254→  });
   255→
   256→  // Export data with full year window for comprehensive exports
   257→  const exportData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   258→    view: "date" as const,
   259→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   260→    endDate: Date.now(),
   261→  }) as any | undefined;
   262→
   263→  const timelineData = useQuery(api.evidence.library.getEvidenceTimeline, {
   264→    date: startOfDay,
   265→  });
   266→
   267→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   268→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   269→
   270→  const isLoading = evidenceData === undefined || timelineData === undefined;
   271→  const exportLoading = exportData === undefined || clients === undefined;
   272→  const timedOut = useQueryTimeout(isLoading, 5000);
   273→  const showLoading = isLoading && !timedOut && !isDisconnected;
   274→
   275→  // ── Safe fallbacks ──
   276→  const safeEvidenceData = evidenceData ?? {
   277→    totalCount: 0,
   278→    disputeSuccessRate: 0,
   279→    contentQualityScore: 0,
   280→    gapPrediction: {
   281→      status: "no_gaps" as const,
   282→      message: "No Gaps Predicted",
   283→      description: "Start collecting evidence to build your protection.",
   284→      nextGapTime: null,
   285→      missingTypes: [],
   286→    },
   287→    evidenceItems: [],
   288→    healthScore: null,
   289→    disputeData: null,
   290→    contentData: null,
   291→  };
   292→
   293→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   294→
   295→  // ── Export State ──
   296→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   297→  const [selectedProject, setSelectedProject] = useState("all");
   298→  const [selectedClient, setSelectedClient] = useState("all");
   299→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   300→  const [isExporting, setIsExporting] = useState(false);
   301→  const [previewOpen, setPreviewOpen] = useState(false);
   302→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   303→
   304→  // ── Derived Export Data ──
   305→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   306→    if (!exportData) return [];
   307→    const items = exportData.evidenceItems ?? [];
   308→    const typeCounts: Record<string, number> = {};
   309→    const typeLabels: Record<string, string> = {
   310→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   311→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   312→    };
   313→    const typeIcons: Record<string, React.ReactNode> = {
   314→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,
   315→      url: <FileJson className="h-4 w-4" />, keyboard: <FileText className="h-4 w-4" />,
   316→      mouse: <FileText className="h-4 w-4" />, platform_status: <Shield className="h-4 w-4" />,
   317→    };
   318→    for (const item of items) {
   319→      const kind = item.type ?? "unknown";
   320→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   321→    }
   322→    return Object.entries(typeCounts).map(([id, count]) => ({
   323→      id,
   324→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   325→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   326→      count,
   327→    }));
   328→  }, [exportData]);
   329→
   330→  const projectOptions = useMemo(() => {
   331→    const all = [{ id: "all", label: "All Projects" }];
   332→    if (scopeDefinitions) {
   333→      for (const scope of scopeDefinitions) {
   334→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   335→      }
   336→    }
   337→    return all;
   338→  }, [scopeDefinitions]);
   339→
   340→  const clientOptions = useMemo(() => {
   341→    const all = [{ id: "all", label: "All Clients" }];
   342→    if (clients) {
   343→      for (const client of clients) {
   344→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   345→      }
   346→    }
   347→    return all;
   348→  }, [clients]);
   349→
   350→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;
   351→  const totalItems = exportData?.totalCount ?? safeEvidenceData.totalCount ?? 0;
   352→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;
   353→
   354→  // Auto-select all evidence types when they load
   355→  useEffect(() => {
   356→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   357→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   358→    }
   359→  }, [evidenceTypes]);
   360→
   361→  const totalEvidenceItems = useMemo(
   362→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   363→    [evidenceTypes, selectedEvidenceTypes]
   364→  );
   365→
   366→  const isFormatLocked = (format: ExportFormat): boolean => {
   367→    return !hasTierAccess(subscriptionTier, FORMAT_CONFIG[format].tierRequired);
   368→  };
   369→
   370→  const toggleEvidenceType = (id: string) => {
   371→    setSelectedEvidenceTypes((prev) =>
   372→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   373→    );
   374→  };
   375→
   376→  const handleExport = async () => {
   377→    if (selectedEvidenceTypes.length === 0) {
   378→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   379→      return;
   380→    }
   381→    if (isFormatLocked(selectedFormat)) {
   382→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   383→      return;
   384→    }
   385→    setIsExporting(true);
   386→    try {
   387→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   388→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   389→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   390→
   391→      exportEvidence(selectedFormat, {
   392→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   393→        dateRange: `${dateFrom} to ${dateTo}`,
   394→        project: selectedProjectLabel,
   395→        client: selectedClientLabel,
   396→        complianceScore,
   397→        totalItems: totalEvidenceItems,
   398→      });
   399→
   400→      const newExport: RecentExport = {
   401→        id: `exp_${Date.now()}`,
   402→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   403→        format: selectedFormat,
   404→        status: "completed",
   405→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   406→        date: new Date().toISOString().split("T")[0],
   407→        itemCount: totalEvidenceItems,
   408→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   409→        project: selectedProjectLabel,
   410→      };
   411→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   412→
   413→      toast.success("Export complete", {
   414→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   415→      });
   416→
   417→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   418→    } catch {
   419→      toast.error("Export failed", { description: "An error occurred while generating the export file." });
   420→    }
   421→    setIsExporting(false);
   422→  };
   423→
   424→  const handleFormatSelect = (format: ExportFormat) => {
   425→    if (isFormatLocked(format)) {
   426→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.` });
   427→      return;
   428→    }
   429→    setSelectedFormat(format);
   430→  };
   431→
   432→  const previewItems = useMemo(() => {
   433→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   434→  }, [evidenceTypes, selectedEvidenceTypes]);
   435→
   436→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   437→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   438→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   439→  const hasTierAccessFn = (requiredTier: string) => getTierLevel(subscriptionTier) >= getTierLevel(requiredTier);
   440→
   441→  // ── Loading / Timeout State ──
   442→  const [showTimeout, setShowTimeout] = useState(false);
   443→  useEffect(() => {
   444→    if (isLoading) {
   445→      const timer = setTimeout(() => setShowTimeout(true), 5000);
   446→      return () => clearTimeout(timer);
   447→    }
   448→    setShowTimeout(false);
   449→  }, [isLoading]);
   450→
   451→  return (
   452→    <div className="w-full min-h-screen bg-background">
   453→      <div className="p-8 space-y-8">
   454→        {/* ── Page Header ── */}
   455→        <div>
   456→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   457→            Evidence Library
   458→          </h1>
   459→          <p className="text-[16px] text-muted-foreground">
   460→            Monitor capture health, anticipate gaps early, and export airtight proof on demand.
   461→          </p>
   462→        </div>
   463→
   464→        {/* Demo Mode Banner */}
   465→        {!isAuthenticated && <DemoModeBanner />}
   466→
   467→        {/* ════════════════════════════════════════════════════════════════════════
   468→            SECTION 1: EVIDENCE MONITORING (was Evidence Library)
   469→            - Health dashboard, timeline, quality scorecard, item browser
   470→        ════════════════════════════════════════════════════════════════════════ */}
   471→        <section id="monitoring">
   472→          <div className="flex items-center gap-3 mb-4">
   473→            <div className="p-2 rounded-lg bg-primary/10">
   474→              <DatabaseIcon className="w-5 h-5 text-primary" />
   475→            </div>
   476→            <div>
   477→              <h2 className="text-xl font-semibold text-foreground">Evidence Monitoring</h2>
   478→              <p className="text-sm text-muted-foreground">Real-time health, quality analysis, and gap detection</p>
   479→            </div>
   480→          </div>
   481→
   482→          {showLoading ? (
   483→            <div className="text-center py-12">
   484→              <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary mx-auto"></div>
   485→              <p className="mt-4 text-muted-foreground">Loading evidence data...</p>
   486→              {showTimeout && (
   487→                <div className="mt-4 p-4 bg-yellow-50 dark:bg-yellow-900/20 border border-yellow-200 dark:border-yellow-800 rounded-lg">
   488→                  <p className="text-sm text-yellow-800 dark:text-yellow-200">
   489→                    Taking longer than expected. Check your connection or try refreshing the page.
   490→                  </p>
   491→                </div>
   492→              )}
   493→            </div>
   494→          ) : (
   495→            <div className="space-y-6">
   496→              <WorkContentAnalysis
   497→                qualityScore={safeEvidenceData.contentQualityScore}
   498→                hasAccess={hasTierAccessFn("pro")}
   499→                contentData={safeEvidenceData.contentData}
   500→              />
   501→
   502→              <EvidenceTimeline timeline={safeTimelineData.timeline} />
   503→
   504→              <EvidenceQualityScorecard
   505→                hasAccess={hasTierAccessFn("pro")}
   506→                healthScore={safeEvidenceData.healthScore}
   507→              />
   508→
   509→              <TeamValidation hasAccess={hasTierAccessFn("expert")} />
   510→
   511→              <EvidenceItemsList
   512→                evidenceItems={safeEvidenceData.evidenceItems}
   513→                viewMode={viewMode}
   514→                setViewMode={setViewMode}
   515→              />
   516→            </div>
   517→          )}
   518→        </section>
   519→
   520→        {/* ── Section Divider ── */}
   521→        <div className="relative py-4">
   522→          <Separator />
