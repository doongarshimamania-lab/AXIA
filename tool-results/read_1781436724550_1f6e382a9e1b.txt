   256→  const [statusFilter, setStatusFilter] = useState<string>("all");

   258→  const [editOpen, setEditOpen] = useState(false);

   262→

   271→  const [formStatus, setFormStatus] = useState("not_started");

   275→  const [isUpdating, setIsUpdating] = useState(false);

   280→  const { isDisconnected } = useConvexConnectionState();

   282→
   283→  const isLoading = !authLoading && goalsData === undefined && !timedOut && !isDisconnected;
   284→
   285→  // ─── Demo mode ──────────────────────────────────────────────────────────
   286→  const isDemoMode = !authLoading && !isAuthenticated;
   287→

   293→  const completedGoals = goals.filter((g: any) => g.status === "completed").length;
   294→  const completionRate =
   295→    goals.length > 0 ? Math.round((completedGoals / goals.length) * 100) : 0;
   296→  const longestStreak = goals.reduce((max: number, g: any) => Math.max(max, g.streak ?? 0), 0);
   297→

   299→  const filteredGoals =
   300→    statusFilter === "all"
   301→      ? goals

   306→    setFormTitle("");
   307→    setFormDescription("");
   308→    setFormType("custom");
   309→    setFormTarget("");
   310→    setFormCurrent("0");

   314→  };

   317→    if (goal.target === 0) return goal.status === "completed" ? 100 : 0;
   318→    return Math.min(Math.round((goal.current / goal.target) * 100), 100);
   319→  };

   322→  const handleCreate = async () => {
   323→    if (!formTitle.trim()) {
   324→      toast.error("Goal title is required");
   325→      return;
   326→    }
   327→    if (!formTarget) {
   328→      toast.error("Target value is required");
   329→      return;

   361→      setIsCreating(false);
   362→    }

   371→    setFormTarget(String(goal.target));

   373→    setFormUnit(goal.unit);
   374→    setFormDeadline(goal.deadline ? new Date(goal.deadline).toISOString().split("T")[0] : "");
   375→    setFormStatus(goal.status);
   376→    setEditOpen(true);
   377→  };
   378→

   380→    if (!editingGoal) return;
   381→    if (!formTitle.trim()) {
   382→      toast.error("Goal title is required");
   383→      return;
   384→    }
   385→
   386→    const target = Number(formTarget);
   387→    const deadline = formDeadline ? new Date(formDeadline).getTime() : undefined;
   388→
   389→    if (isDemoMode) {
   390→      toast.success(`Goal "${formTitle.trim()}" updated! (Demo mode)`);
   391→      resetForm();
   392→      setEditOpen(false);
   393→      setEditingGoal(null);
   394→      return;
   395→    }
   396→
   397→    setIsUpdating(true);
   398→    try {
   399→      await updateGoalMutation({
   400→        goalId: editingGoal._id,
   401→        title: formTitle.trim(),
   402→        description: formDescription.trim() || undefined,
   403→        type: formType,
   404→        target,
   405→        current: Number(formCurrent) || 0,
   406→        unit: formUnit,
   407→        deadline,
   408→        status: formStatus,
   409→      });
   410→      resetForm();
   411→      setEditOpen(false);
   412→      setEditingGoal(null);
   413→      toast.success(`Goal "${formTitle.trim()}" updated`);
   414→    } catch (err: any) {
   415→      toast.error(err?.message || "Failed to update goal");
   416→    } finally {
   417→      setIsUpdating(false);
   418→    }
   419→  };
   420→

   422→  const openDelete = (goal: any) => {
   423→    setDeletingGoal(goal);
   424→    setDeleteOpen(true);
   425→  };

   428→    if (!deletingGoal) return;
   429→
   430→    if (isDemoMode) {
   431→      toast.success(`Goal "${deletingGoal.title}" deleted! (Demo mode)`);
   432→      setDeleteOpen(false);
   433→      setDeletingGoal(null);
   434→      return;
   435→    }

   437→    setIsDeleting(true);
   438→    try {
   439→      await deleteGoalMutation({ goalId: deletingGoal._id });
   440→      setDeleteOpen(false);
   441→      setDeletingGoal(null);
   442→      toast.success(`Goal "${deletingGoal.title}" deleted`);
   443→    } catch (err: any) {
   444→      toast.error(err?.message || "Failed to delete goal");
   445→    } finally {
   446→      setIsDeleting(false);
   447→    }
   448→  };

   465→      toast.error(err?.message || "Failed to mark goal complete");
   466→    } finally {
   467→      setIsCompleting(null);
   468→    }

   478→    try {
   479→      await updateMilestoneMutation({

   486→    }
   487→  };

   490→  const FILTER_OPTIONS = [
   491→    { value: "all", label: "All" },
   492→    { value: "in_progress", label: "In Progress" },
   493→    { value: "not_started", label: "Not Started" },
   494→    { value: "completed", label: "Completed" },
   495→    { value: "abandoned", label: "Abandoned" },
   496→  ];
   497→
   498→  // ── Render ──
   499→  return (
   500→    <motion.div
   501→      initial={{ opacity: 0 }}
   502→      animate={{ opacity: 1 }}
   503→      className="w-full min-h-screen bg-background text-foreground"
   504→    >
   505→      <PageLayout spaced>
   506→        {/* ── Header ── */}
   507→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   508→          <div>
   509→            <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   510→              Goals
   511→            </h1>
   512→            <p className="text-[16px] text-muted-foreground">
   513→              Set and track professional goals to protect your income and grow your agency
   514→            </p>
   515→          </div>
   516→
   517→          <Dialog
   518→            open={createOpen}
   519→            onOpenChange={(open) => {

   521→              if (!open) resetForm();
   522→            }}
   523→          >
   524→            <DialogTrigger asChild>
   525→              <Button className="bg-primary hover:bg-primary/90 shrink-0">
   526→                <Plus className="mr-2 h-4 w-4" />
   527→                Create Goal
   528→              </Button>
   529→            </DialogTrigger>
   530→            <DialogContent className="sm:max-w-lg">
   531→              <DialogHeader>
   532→                <DialogTitle>Create New Goal</DialogTitle>
   533→                <DialogDescription>
   534→                  Define a measurable goal with a deadline to keep your professional growth on track.
   535→                </DialogDescription>
   536→              </DialogHeader>
   537→              <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   538→                <div className="space-y-2">
   539→                  <Label htmlFor="goal-title">Title</Label>
   540→                  <Input
   541→                    id="goal-title"
   542→                    placeholder="e.g. Reach $10K monthly revenue"
   543→                    value={formTitle}
   544→                    onChange={(e) => setFormTitle(e.target.value)}
   545→                  />
   546→                </div>
   547→                <div className="space-y-2">
   548→                  <Label htmlFor="goal-description">Description</Label>
   549→                  <Textarea
   550→                    id="goal-description"
