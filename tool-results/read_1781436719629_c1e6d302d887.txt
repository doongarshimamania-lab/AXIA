   268→function formatDate(timestamp: number | undefined): string {
   269→  if (!timestamp) return "—";
   270→  return new Intl.DateTimeFormat("en-US", {
   271→    month: "short",
   272→    day: "numeric",
   273→    year: "numeric",
   274→  }).format(new Date(timestamp));
   275→}

   277→function daysUntilClose(timestamp: number | undefined): number | null {
   278→  if (!timestamp) return null;
   279→  const diff = timestamp - Date.now();
   280→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   281→}
