   116→// --- Helpers ---
   117→function formatDate(ts: number): string {
   118→  return new Date(ts).toLocaleDateString("en-US", {
   119→    year: "numeric",
   120→    month: "short",
   121→    day: "numeric",
   122→  });
   123→}
   124→
   125→function timeAgo(ts: number | null): string {
   126→  if (!ts) return "Never";
   127→  const diff = Date.now() - ts;
   128→  const mins = Math.floor(diff / 60000);
   129→  if (mins < 1) return "Just now";
   130→  if (mins < 60) return `${mins}m ago`;
   131→  const hrs = Math.floor(mins / 60);
   132→  if (hrs < 24) return `${hrs}h ago`;
   133→  const days = Math.floor(hrs / 24);
   134→  return `${days}d ago`;
   135→}

   137→function generateMockKey(prefix: string = "tl_prod_"): string {
   138→  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
   139→  let result = prefix + "sk_";
   140→  for (let i = 0; i < 20; i++) {
   141→    result += chars.charAt(Math.floor(Math.random() * chars.length));
   142→  }
   143→  return result;
   144→}
   145→
   146→function maskKey(key: string): string {
   147→  const prefix = key.substring(0, key.indexOf("sk_") + 3);
   148→  const suffix = key.slice(-4);
   149→  return `${prefix}${"•".repeat(16)}${suffix}`;
   150→}
