    98→function formatTime(seconds: number) {
    99→  const h = Math.floor(seconds / 3600);
   100→  const m = Math.floor((seconds % 3600) / 60);
   101→  const s = seconds % 60;
   102→  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
   103→}
   104→
   105→function formatDuration(minutes: number | undefined) {
   106→  if (!minutes) return "0h 0m";
   107→  const hours = Math.floor(minutes / 60);
   108→  const mins = minutes % 60;
   109→  return `${hours}h ${mins}m`;
   110→}
   111→
   112→function formatDate(ts: number) {
   113→  return new Date(ts).toLocaleDateString("en-US", {
   114→    month: "short",
   115→    day: "numeric",
   116→    year: "numeric",
   117→  });
   118→}
