    45→// ─────────────────────────────────────────────
    46→// 1. CREATE
    47→// ─────────────────────────────────────────────
    48→export const create = mutation({
    49→  args: {
    50→    workspaceId: v.optional(v.id("workspaces")),
    51→    teamId: v.optional(v.id("teams")),
    52→    projectId: v.optional(v.id("projects")),
    53→    proposalId: v.optional(v.id("proposals")),
    54→    clientId: v.optional(v.id("clients")),
    55→    deliverables: v.array(deliverableValidator),
    56→    exclusions: v.optional(v.array(v.string())),
    57→    assumptions: v.optional(v.array(v.string())),
    58→    customFields: v.optional(v.any()),
    59→  },
    60→  handler: async (ctx, args) => {
    61→    const userId = await getAuthUserId(ctx);
    62→    if (!userId) throw new ConvexError("Not authenticated");
    63→
    64→    // If workspaceId provided, verify membership
    65→    if (args.workspaceId) {
    66→      await requireWorkspaceAccess(ctx, args.workspaceId, "member");
    67→    }
    68→
    69→    // Validate deliverables are not empty
    70→    if (args.deliverables.length === 0) {
    71→      throw new ConvexError("Scope must have at least one deliverable");
    72→    }
    73→
    74→    // Validate each deliverable
    75→    for (const d of args.deliverables) {
    76→      if (d.name.trim().length === 0) {
    77→        throw new ConvexError("Deliverable name cannot be empty");
    78→      }
    79→      if (d.revisionLimit < 0) {
    80→        throw new ConvexError("Revision limit cannot be negative");
    81→      }
    82→      if (d.revisionsUsed < 0) {
    83→        throw new ConvexError("Revisions used cannot be negative");
    84→      }
    85→      if (d.revisionsUsed > d.revisionLimit) {
    86→        throw new ConvexError("Revisions used cannot exceed revision limit");
    87→      }
    88→    }
    89→
    90→    // Validate project access if provided
    91→    if (args.projectId) {
    92→      const project = await ctx.db.get(args.projectId);
    93→      if (!project) throw new ConvexError("Project not found");
    94→      if (project.workspaceId) {
    95→        const access = await getRecordAccess(ctx, project, userId);
    96→        if (!access) throw new ConvexError("Project does not belong to this workspace");
    97→      } else if (project.userId !== userId) {
    98→        throw new ConvexError("Project does not belong to this user");
    99→      }
   100→    }
   101→
   102→    // Validate proposal access if provided
   103→    if (args.proposalId) {
   104→      const proposal = await ctx.db.get(args.proposalId);
   105→      if (!proposal) throw new ConvexError("Proposal not found");
   106→      if (proposal.workspaceId) {
   107→        const access = await getRecordAccess(ctx, proposal, userId);
   108→        if (!access) throw new ConvexError("Proposal does not belong to this workspace");
   109→      } else if (proposal.userId !== userId) {
   110→        throw new ConvexError("Proposal does not belong to this user");
   111→      }
   112→    }
   113→
   114→    // Validate client access if provided
   115→    if (args.clientId) {
   116→      const client = await ctx.db.get(args.clientId);
   117→      if (!client) throw new ConvexError("Client not found");
   118→      if (client.workspaceId) {
   119→        const access = await getRecordAccess(ctx, client, userId);
   120→        if (!access) throw new ConvexError("Client does not belong to this workspace");
   121→      } else if (client.userId !== userId) {
   122→        throw new ConvexError("Client does not belong to this user");
   123→      }
   124→    }
   125→
   126→    // Generate approval token
   127→    const clientApprovalToken = crypto.randomUUID();
   128→
   129→    const now = Date.now();
   130→    const { workspaceId, teamId, customFields, ...rest } = args;
   131→
   132→    const scopeId = await ctx.db.insert("scopeDefinitions", {
   133→      userId,
   134→      workspaceId: workspaceId ?? undefined,
   135→      createdBy: userId,
   136→      teamId: teamId ?? undefined,
   137→      customFields: customFields ?? undefined,
   138→      ...rest,
   139→      clientApprovedAt: undefined,
   140→      clientApprovalToken,
   141→      status: "draft",
   142→      createdAt: now,
   143→      updatedAt: now,
   144→    });
   145→
   146→    return scopeId;
   147→  },
   148→});

   227→      const client = await ctx.db.get(args.clientId);
   228→      if (!client) throw new ConvexError("Client not found");

   230→        const access = await getRecordAccess(ctx, client, userId);

   233→        throw new ConvexError("Client does not belong to this user");

   237→    // Build partial update object
   238→    const updates: Record<string, unknown> = { updatedAt: Date.now() };
   239→

   241→    if (args.proposalId !== undefined) updates.proposalId = args.proposalId;

   245→    if (args.assumptions !== undefined) updates.assumptions = args.assumptions;
   246→    if (args.teamId !== undefined) updates.teamId = args.teamId;
   247→

   249→
   250→    return args.scopeDefinitionId;
   251→  },

   294→// 4. CLIENT APPROVE (via approvalToken, no auth)
   295→// ─────────────────────────────────────────────
   296→export const clientApprove = mutation({
   297→  args: {
   298→    approvalToken: v.string(),
   299→  },
   300→  handler: async (ctx, args) => {
   301→    const scope = await ctx.db
   302→      .query("scopeDefinitions")
   303→      .withIndex("by_approval_token", (q) => q.eq("clientApprovalToken", args.approvalToken))
   304→      .first();
   305→
   306→    if (!scope) throw new ConvexError("Invalid or expired approval link");
   307→
   308→    // Only draft or active scopes can be approved
   309→    if (scope.status !== "draft" && scope.status !== "active") {
   310→      throw new ConvexError("This scope definition cannot be approved in its current state");
   311→    }
   312→
   313→    // Check if already approved
   314→    if (scope.clientApprovedAt) {
   315→      throw new ConvexError("This scope definition has already been approved");
   316→    }
   317→
   318→    const now = Date.now();
   319→
   320→    await ctx.db.patch(scope._id, {
   321→      clientApprovedAt: now,
   322→      updatedAt: now,
   323→    });
   324→
   325→    return { success: true, scopeDefinitionId: scope._id };
   326→  },
   327→});

   330→// 5. RECORD REVISION
   331→// ─────────────────────────────────────────────
   332→export const recordRevision = mutation({
   333→  args: {
   334→    scopeDefinitionId: v.id("scopeDefinitions"),
   335→    deliverableId: v.string(),
   336→  },
   337→  handler: async (ctx, args) => {
   338→    const userId = await getAuthUserId(ctx);
   339→    if (!userId) throw new ConvexError("Not authenticated");
   340→
   341→    const scope = await ctx.db.get(args.scopeDefinitionId);
   342→    if (!scope) throw new ConvexError("Scope definition not found");
   343→
   344→    // Check workspace access or direct ownership
   345→    if (scope.workspaceId) {
   346→      await requireRecordAccess(ctx, scope, "collaborate");
   347→    } else if (scope.userId !== userId) {
   348→      throw new ConvexError("Not authorized to record revisions for this scope");
   349→    }
   350→
   351→    // Only active scopes can have revisions recorded
   352→    if (scope.status !== "active") {
   353→      throw new ConvexError("Revisions can only be recorded for active scope definitions");
   354→    }
   355→
   356→    // Find the deliverable
   357→    const deliverableIndex = scope.deliverables.findIndex((d) => d.id === args.deliverableId);
   358→    if (deliverableIndex === -1) {
   359→      throw new ConvexError("Deliverable not found in this scope definition");
   360→    }
   361→
   362→    const deliverable = scope.deliverables[deliverableIndex];
   363→
   364→    // Check if already exceeded
   365→    if (deliverable.status === "revisions_exceeded") {
   366→      throw new ConvexError("Revision limit already exceeded for this deliverable");
   367→    }
   368→
   369→    // Increment revisionsUsed
   370→    const newRevisionsUsed = deliverable.revisionsUsed + 1;
   371→    const updatedDeliverables = [...scope.deliverables];
   372→    const now = Date.now();
   373→
   374→    let autoCreatedChangeOrder = false;
   375→
   376→    if (newRevisionsUsed >= deliverable.revisionLimit) {
   377→      // Set status to revisions_exceeded
   378→      updatedDeliverables[deliverableIndex] = {
   379→        ...deliverable,
   380→        revisionsUsed: newRevisionsUsed,
   381→        status: "revisions_exceeded",
   382→      };
   383→
   384→      // Auto-generate a change order
   385→      const changeOrderApprovalToken = crypto.randomUUID();
   386→
   387→      await ctx.db.insert("scopeChangeOrders", {
   388→        userId,
   389→        workspaceId: scope.workspaceId ?? undefined,
   390→        createdBy: userId,
   391→        teamId: scope.teamId ?? undefined,
   392→        scopeDefinitionId: args.scopeDefinitionId,
   393→        clientId: scope.clientId,
   394→        deliverableId: args.deliverableId,
   395→        requestedBy: "freelancer",
   396→        description: `Revision limit exceeded for "${deliverable.name}". ${newRevisionsUsed} of ${deliverable.revisionLimit} allowed revisions have been used. Additional revisions require a change order.`,
   397→        costImpact: undefined,
   398→        timelineImpactDays: undefined,
   399→        additionalRevisions: undefined,
   400→        status: "pending",
   401→        clientApprovedAt: undefined,
   402→        clientApprovalToken: changeOrderApprovalToken,
   403→        clientRejectedAt: undefined,
   404→        rejectionReason: undefined,
   405→        invoiceId: undefined,
   406→        createdAt: now,
   407→        updatedAt: now,
   408→      });
   409→
   410→      autoCreatedChangeOrder = true;
   411→    } else {
   412→      updatedDeliverables[deliverableIndex] = {
   413→        ...deliverable,
   414→        revisionsUsed: newRevisionsUsed,
   415→      };
   416→    }
   417→
   418→    await ctx.db.patch(args.scopeDefinitionId, {
   419→      deliverables: updatedDeliverables,
   420→      updatedAt: now,
   421→    });
   422→
   423→    return {
   424→      scopeDefinitionId: args.scopeDefinitionId,
   425→      deliverableId: args.deliverableId,
   426→      revisionsUsed: newRevisionsUsed,
   427→      revisionLimit: deliverable.revisionLimit,
   428→      status: updatedDeliverables[deliverableIndex].status,
   429→      autoCreatedChangeOrder,
   430→    };
   431→  },
   432→});

   583→  },
   584→  handler: async (ctx, args) => {
   585→    const changeOrder = await ctx.db
   586→      .query("scopeChangeOrders")
   587→      .withIndex("by_approval_token", (q) => q.eq("clientApprovalToken", args.approvalToken))
   588→      .first();
   589→
   590→    if (!changeOrder) throw new ConvexError("Invalid or expired change order approval link");
   591→
   592→    // Only pending change orders can be approved
   593→    if (changeOrder.status !== "pending") {
   594→      throw new ConvexError("This change order cannot be approved in its current state");
   595→    }
   596→
   597→    const now = Date.now();
   598→
   599→    await ctx.db.patch(changeOrder._id, {
   600→      status: "approved",
   601→      clientApprovedAt: now,
   602→      updatedAt: now,
   603→    });
   604→
   605→    // If additionalRevisions is specified, update the scope deliverable's revision limit
   606→    if (changeOrder.additionalRevisions && changeOrder.additionalRevisions > 0 && changeOrder.deliverableId) {
   607→      const scope = await ctx.db.get(changeOrder.scopeDefinitionId);
   608→      if (scope) {
   609→        const updatedDeliverables = scope.deliverables.map((d) => {
   610→          if (d.id === changeOrder.deliverableId) {
   611→            return {
   612→              ...d,
   613→              revisionLimit: d.revisionLimit + changeOrder.additionalRevisions!,
   614→              // If status was revisions_exceeded, reset to in_progress since we now have more revisions
   615→              status: d.status === "revisions_exceeded" ? "in_progress" as const : d.status,
   616→            };
   617→          }
   618→          return d;
   619→        });
   620→
   621→        await ctx.db.patch(changeOrder.scopeDefinitionId, {
   622→          deliverables: updatedDeliverables,
   623→          updatedAt: now,
   624→        });
   625→      }
   626→    }
   627→
   628→    return { success: true, changeOrderId: changeOrder._id };
   629→  },
   630→});

   692→// ─────────────────────────────────────────────
   693→// 11. LIST SCOPE DEFINITIONS (workspace-aware)
   694→// ─────────────────────────────────────────────
   695→export const list = query({
   696→  args: {
   697→    workspaceId: v.optional(v.id("workspaces")),
   698→    status: v.optional(scopeStatusValidator),
   699→  },
   700→  handler: async (ctx, args) => {
   701→    const userId = await getAuthUserId(ctx);
   702→    if (!userId) throw new ConvexError("Not authenticated");
   703→
   704→    // If workspaceId provided, filter by workspace with team-aware visibility
   705→    if (args.workspaceId) {
   706→      const visibility = await getUserVisibility(ctx, args.workspaceId);
   707→      if (!visibility) throw new ConvexError("Not a member of this workspace");
   708→
   709→      const allScopes = await ctx.db
   710→        .query("scopeDefinitions")
   711→        .withIndex("by_workspace", (q) => q.eq("workspaceId", args.workspaceId!))
   712→        .collect();
   713→
   714→      let visible = allScopes.filter((s) => isRecordVisible(s, visibility));
   715→
   716→      if (args.status) {
   717→        visible = visible.filter((s) => s.status === args.status);
   718→      }
   719→
   720→      return visible.sort((a, b) => b.updatedAt - a.updatedAt);
   721→    }
   722→
   723→    // Backward compat: no workspaceId
   724→    const allScopes = await ctx.db
   725→      .query("scopeDefinitions")
   726→      .withIndex("by_user", (q) => q.eq("userId", userId))
   727→      .collect();
   728→
   729→    if (args.status) {
   730→      return allScopes.filter((s) => s.status === args.status);
   731→    }
   732→
   733→    return allScopes.sort((a, b) => b.updatedAt - a.updatedAt);
   734→  },
   735→});

   796→// 14. LINK CHANGE ORDER TO INVOICE
   797→// ─────────────────────────────────────────────
   798→export const linkChangeOrderToInvoice = mutation({
   799→  args: {
   800→    changeOrderId: v.id("scopeChangeOrders"),
   801→    invoiceId: v.id("invoices"),
   802→  },
   803→  handler: async (ctx, args) => {
   804→    const userId = await getAuthUserId(ctx);
   805→    if (!userId) throw new ConvexError("Not authenticated");
   806→
   807→    const changeOrder = await ctx.db.get(args.changeOrderId);
   808→    if (!changeOrder) throw new ConvexError("Change order not found");
   809→
   810→    // Check workspace access or direct ownership
   811→    if (changeOrder.workspaceId) {
   812→      await requireRecordAccess(ctx, changeOrder, "collaborate");
   813→    } else if (changeOrder.userId !== userId) {
   814→      throw new ConvexError("Not authorized to update this change order");
   815→    }
   816→
   817→    // Change order must be approved before linking to invoice
   818→    if (changeOrder.status !== "approved") {
   819→      throw new ConvexError("Only approved change orders can be linked to invoices");
   820→    }
   821→
   822→    // Validate invoice access
   823→    const invoice = await ctx.db.get(args.invoiceId);
   824→    if (!invoice) throw new ConvexError("Invoice not found");
   825→
   826→    if (invoice.workspaceId) {
   827→      const access = await getRecordAccess(ctx, invoice, userId);
   828→      if (!access) throw new ConvexError("Invoice does not belong to this workspace");
   829→    } else if (invoice.userId !== userId) {
   830→      throw new ConvexError("Invoice does not belong to this user");
   831→    }
   832→
   833→    const now = Date.now();
   834→
   835→    await ctx.db.patch(args.changeOrderId, {
   836→      invoiceId: args.invoiceId,
   837→      updatedAt: now,
   838→    });
   839→
   840→    return args.changeOrderId;
   841→  },
   842→});
