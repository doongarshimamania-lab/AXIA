     8→export const getInvoices = query({
     9→  args: { workspaceId: v.optional(v.id("workspaces")), status: v.optional(v.string()) },
    10→  handler: async (ctx, { workspaceId, status }) => {
    11→    const userId = await getAuthUserId(ctx);
    12→    if (!userId) return [];
    13→
    14→    // If workspaceId provided, filter by workspace with membership check
    15→    if (workspaceId) {
    16→      const membership = await getWorkspaceMembership(ctx, workspaceId, userId);
    17→      if (!membership) return [];
    18→
    19→      const allInvoices = await ctx.db
    20→        .query("invoices")
    21→        .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
    22→        .order("desc")
    23→        .collect();
    24→
    25→      if (status) return allInvoices.filter((i: any) => i.status === status);
    26→      return allInvoices;
    27→    }
    28→
    29→    if (status) {
    30→      return await ctx.db
    31→        .query("invoices")
    32→        .withIndex("by_user_and_status", (q) => q.eq("userId", userId).eq("status", status as any))
    33→        .order("desc")
    34→        .collect();
    35→    }
    36→    return await ctx.db
    37→      .query("invoices")
    38→      .withIndex("by_user", (q) => q.eq("userId", userId))
    39→      .order("desc")
    40→      .collect();
    41→  },
    42→});

   103→export const getInvoiceStats = query({
   104→  args: { workspaceId: v.optional(v.id("workspaces")) },
   105→  handler: async (ctx, { workspaceId }) => {
   106→    const userId = await getAuthUserId(ctx);
   107→    if (!userId) return { total: 0, paid: 0, pending: 0, overdue: 0, totalRevenue: 0, withProof: 0 };
   108→
   109→    let invoices;
   110→    if (workspaceId) {
   111→      const membership = await getWorkspaceMembership(ctx, workspaceId, userId);
   112→      if (!membership) return { total: 0, paid: 0, pending: 0, overdue: 0, totalRevenue: 0, withProof: 0 };
   113→      invoices = await ctx.db
   114→        .query("invoices")
   115→        .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
   116→        .collect();
   117→    } else {
   118→      invoices = await ctx.db
   119→        .query("invoices")
   120→        .withIndex("by_user", (q) => q.eq("userId", userId))
   121→        .collect();
   122→    }
   123→
   124→    return {
   125→      total: invoices.length,
   126→      paid: invoices.filter(i => i.status === "paid").length,
   127→      pending: invoices.filter(i => i.status === "sent" || i.status === "viewed").length,
   128→      overdue: invoices.filter(i => i.status === "overdue").length,
   129→      draft: invoices.filter(i => i.status === "draft").length,
   130→      totalRevenue: invoices.filter(i => i.status === "paid").reduce((s, i) => s + i.total, 0),
   131→      totalOutstanding: invoices.filter(i => ["sent", "viewed", "overdue"].includes(i.status)).reduce((s, i) => s + i.total, 0),
   132→      withProof: invoices.filter(i => (i.proofCount ?? 0) > 0).length,
   133→    };
   134→  },
   135→});

   139→function generateInvoiceNumber(existing: any[]): string {
   140→  const nums = existing.map(inv => {
   141→    const match = inv.invoiceNumber?.match(/INV-(\d+)/);
   142→    return match ? parseInt(match[1]) : 0;
   143→  });
   144→  const max = nums.length > 0 ? Math.max(...nums) : 0;
   145→  return `INV-${String(max + 1).padStart(3, "0")}`;
   146→}

   148→function generateToken(): string {
   149→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
   150→  let result = "";
   151→  for (let i = 0; i < 32; i++) result += chars.charAt(Math.floor(Math.random() * chars.length));
   152→  return result;
   153→}

   156→  args: {

   320→        createdAt: Date.now(),
   321→      });
   322→    }

   595→export const getInvoiceTemplates = query({
   596→  args: { workspaceId: v.optional(v.id("workspaces")) },
   597→  handler: async (ctx, { workspaceId }) => {
   598→    const userId = await getAuthUserId(ctx);
   599→    const systemTemplates = await ctx.db
   600→      .query("invoiceTemplates")
   601→      .withIndex("by_system", (q) => q.eq("isSystem", true))
   602→      .collect();
   603→
   604→    let userTemplates: any[] = [];
   605→    if (userId) {
   606→      if (workspaceId) {
   607→        userTemplates = await ctx.db
   608→          .query("invoiceTemplates")
   609→          .withIndex("by_workspace", (q) => q.eq("workspaceId", workspaceId))
   610→          .collect();
   611→        // Also get user's own templates
   612→        const personalTemplates = await ctx.db
   613→          .query("invoiceTemplates")
   614→          .withIndex("by_user", (q) => q.eq("userId", userId))
   615→          .collect();
   616→        const ids = new Set(userTemplates.map((t: any) => t._id));
   617→        for (const t of personalTemplates) {
   618→          if (!ids.has(t._id)) userTemplates.push(t);
   619→        }
   620→      } else {
   621→        userTemplates = await ctx.db
   622→          .query("invoiceTemplates")
   623→          .withIndex("by_user", (q) => q.eq("userId", userId))
   624→          .collect();
   625→      }
   626→    }
   627→
   628→    return [...systemTemplates, ...userTemplates];
   629→  },
   630→});
