   140→   140→function statusColor(status: string): "default" | "secondary" | "destructive" | "outline" {
   141→   141→  if (["won", "signed", "paid", "completed"].includes(status)) return "default";
   142→   142→  if (["lost", "declined", "overdue", "expired", "rejected"].includes(status)) return "destructive";
   143→   143→  return "secondary";
   144→   144→}

   163→   163→// ─── Format currency ─────────────────────────────────────────────────────
   164→   164→function fmtCurrency(n: number): string {
   165→   165→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(n);
   166→   166→}

   168→   168→function fmtRelative(ts: number): string {
   169→   169→  const diff = Date.now() - ts;
   170→   170→  const mins = Math.floor(diff / 60000);
   171→   171→  if (mins < 1) return "just now";
   172→   172→  if (mins < 60) return `${mins}m ago`;
   173→   173→  const hrs = Math.floor(mins / 60);
   174→   174→  if (hrs < 24) return `${hrs}h ago`;
   175→   175→  const days = Math.floor(hrs / 24);
   176→   176→  if (days < 30) return `${days}d ago`;
   177→   177→  return new Date(ts).toLocaleDateString();
   178→   178→}
