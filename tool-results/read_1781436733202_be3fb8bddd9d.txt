    60→// Record evidence events in bulk
    61→export const recordEvents = mutation({
    62→  args: {
    63→    evidenceSessionId: v.id("evidenceSessions"),
    64→    events: v.array(v.object({
    65→      t: v.number(),
    66→      kind: v.union(
    67→        v.literal("mouse"),
    68→        v.literal("keyboard"),
    69→        v.literal("url"),
    70→        v.literal("screenshot_ref"),
    71→        v.literal("memo"),
    72→        v.literal("platform_status")
    73→      ),
    74→      data: v.any(),
    75→      url: v.optional(v.string()),
    76→    })),
    77→  },
    78→  handler: async (ctx, args) => {
    79→    const user = await getCurrentUser(ctx);
    80→    if (!user) throw new Error("Not authenticated");
    81→
    82→    // Validate evidence session exists and is active
    83→    const evidenceSession = await ctx.db.get(args.evidenceSessionId);
    84→    if (!evidenceSession || evidenceSession.status !== "active") {
    85→      throw new Error("Evidence session not found or not active");
    86→    }
    87→
    88→    // SECURITY: Verify the evidence session belongs to the authenticated user
    89→    if (evidenceSession.userId !== user._id) {
    90→      throw new Error("Not authorized to record events for this session");
    91→    }
    92→
    93→    // Chunk events to stay under Convex limits (max 500 per batch)
    94→    const chunks = [];
    95→    for (let i = 0; i < args.events.length; i += 500) {
    96→      chunks.push(args.events.slice(i, i + 500));
    97→    }
    98→
    99→    let insertedCount = 0;
   100→    for (const chunk of chunks) {
   101→      for (const event of chunk) {
   102→        await ctx.db.insert("evidenceEvents", {
   103→          workspaceId: evidenceSession.workspaceId ?? undefined,
   104→          evidenceSessionId: args.evidenceSessionId,
   105→          t: event.t,
   106→          kind: event.kind,
   107→          data: event.data,
   108→          url: event.url,
   109→        });
   110→        insertedCount++;
   111→      }
   112→    }
   113→
   114→    return { insertedCount };
   115→  },
   116→});

   253→// Generate universal dispute report
   254→export const generateUniversalReport = mutation({
   255→  args: {
   256→    sessionId: v.id("workSessions"),
   257→  },
   258→  handler: async (ctx, args) => {
   259→    const user = await getCurrentUser(ctx);

   262→    }
   263→
   264→    // Get work session (must belong to user)

   266→    if (!workSession || workSession.userId !== user._id) {
   267→      throw new Error("Work session not found or unauthorized");
   268→    }
   269→
   270→    // Enforce free tier limit (1/month) using generatedAt window like disputeReports.ts
   271→    if (user.subscriptionTier === "free") {
   272→      const now = new Date();
   273→      const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
   274→      const endOfMonth = new Date(
   275→        now.getFullYear(),
   276→        now.getMonth() + 1,
   277→        0,
   278→        23,
   279→        59,
   280→        59,
   281→        999
   282→      ).getTime();
   283→
   284→      const monthlyReports = await ctx.db
   285→        .query("disputeReports")
   286→        .withIndex("by_user", (q) => q.eq("userId", user._id))
   287→        .filter((q) =>
   288→          q.and(
   289→            q.gte(q.field("generatedAt"), startOfMonth),
   290→            q.lte(q.field("generatedAt"), endOfMonth)
   291→          )
   292→        )
   293→        .collect();
   294→
   295→      const sessionHours =
   296→        ((workSession.endTime ?? Date.now()) - workSession.startTime) /
   297→        (1000 * 60 * 60);
   298→      const sessionLoss = Math.max(
   299→        0,
   300→        Math.round(sessionHours * workSession.hourlyRate * 100) / 100
   301→      );
   302→
   303→      if (monthlyReports.length >= 1) {
   304→        const prevLoss = monthlyReports.reduce(
   305→          (sum, r) => sum + r.lostIncome,
   306→          0
   307→        );
   308→        const monthlyLoss =
   309→          Math.round((prevLoss + sessionLoss) * 100) / 100;
   310→        const monthlySavings =
   311→          Math.round(monthlyLoss * 0.83 * 100) / 100;
   312→
   313→        return {
   314→          limited: true,
   315→          monthlyLoss,
   316→          monthlySavings,
   317→        };
   318→      }
   319→    }

   321→    // Find evidence session for this work session

   323→      .query("evidenceSessions")
   324→      .withIndex("by_session", (q) => q.eq("sessionId", args.sessionId))
   325→      .first();
   326→
   327→    if (!evidenceSession) {
   328→      throw new Error("No evidence session found for this work session");
   329→    }
   330→
   331→    // Collect all evidence events ordered by time
   332→    const events = await ctx.db
   333→      .query("evidenceEvents")
   334→      .withIndex("by_session_and_time", (q) =>
   335→        q.eq("evidenceSessionId", evidenceSession._id)
   336→      )
   337→      .order("asc")
   338→      .collect();
   339→
   340→    const caseId = `TL-${Date.now()
   341→      .toString(36)
   342→      .toUpperCase()}-${Math.random().toString(36).slice(2, 6).toUpperCase()}`;
   343→

   345→    const universalCore = analyzeUniversalEvidence(

   347→      evidenceSession,
   348→      workSession

   351→      evidenceSession.platform,
   352→      events,
   353→      universalCore
   354→    );

   356→      events,
   357→      evidenceSession
   358→    );

   360→      caseId,
   361→      workSession,
   362→      evidenceSession,
   363→      universalCore,
   364→      platformAdaptation,
   365→      authenticity,
   366→      events,
   367→    });
   368→
   369→    // Compute safe values
   370→    const hours =
   371→      ((workSession.endTime ?? Date.now()) - workSession.startTime) /
   372→      (1000 * 60 * 60);
   373→    const lostIncome =
   374→      Math.max(0, Math.round(hours * workSession.hourlyRate * 100) / 100);
   375→
   376→    // Store minimal dispute report per schema
   377→    await ctx.db.insert("disputeReports", {
   378→      userId: user._id,
   379→      sessionId: args.sessionId,
   380→      caseId,
   381→      generatedAt: Date.now(),
   382→      rejectedHours: Math.max(0, Math.round(hours * 10) / 10),
   383→      lostIncome,
   384→      reportContent,
   385→      status: "generated",
   386→    });
   387→
   388→    return { caseId, reportContent, limited: false };
   389→  },
   390→});

   618→// Helper functions
   619→function analyzeUniversalEvidence(events: any[], evidenceSession: any, workSession: any) {
   620→  const mouseEvents = events.filter(e => e.kind === "mouse");
   621→  const keyboardEvents = events.filter(e => e.kind === "keyboard");
   622→  const urlEvents = events.filter(e => e.kind === "url");
   623→
   624→  const sessionDuration = (evidenceSession.endTime || Date.now()) - evidenceSession.startTime;
   625→  const activityDensity = (mouseEvents.length + keyboardEvents.length) / (sessionDuration / 60000); // per minute
   626→
   627→  return {
   628→    continuous_activity: {
   629→      mouseEvents: mouseEvents.length,
   630→      keyboardEvents: keyboardEvents.length,
   631→      activityDensity,
   632→      gapsAnalysis: analyzeActivityGaps(events),
   633→      status: activityDensity > 2 ? "PASS" : "FAIL",
   634→    },
   635→    work_context: {
   636→      sitesVisited: [...new Set(urlEvents.map(e => e.url).filter(Boolean))],
   637→      workSites: urlEvents.filter(e => isWorkSite(e.url || "")).length,
   638→      nonWorkSites: urlEvents.filter(e => !isWorkSite(e.url || "")).length,
   639→      relevanceScore: calculateWorkRelevance(urlEvents, workSession),
   640→      status: calculateWorkRelevance(urlEvents, workSession) > 0.7 ? "PASS" : "FAIL",
   641→    },
   642→    time_consistency: {
   643→      evidencePoints: events.length,
   644→      timeSpan: sessionDuration,

   806→function analyzeActivityGaps(events: any[]) {
   807→  const sortedEvents = events.filter(e => e.kind === "mouse" || e.kind === "keyboard").sort((a, b) => a.t - b.t);
   808→  let maxGap = 0;
   809→  let gapCount = 0;
   810→
   811→  for (let i = 1; i < sortedEvents.length; i++) {
   812→    const gap = sortedEvents[i].t - sortedEvents[i-1].t;
   813→    if (gap > 60000) { // > 1 minute
   814→      gapCount++;
   815→      maxGap = Math.max(maxGap, gap);
   816→    }
   817→  }
   818→
   819→  return `${gapCount} gaps > 1min (max: ${Math.round(maxGap/1000)}s)`;
   820→}

   837→function validateTimeConsistency(events: any[], workSession: any): boolean {
   838→  if (events.length === 0) return false;
   839→  const firstEvent = Math.min(...events.map(e => e.t));
   840→  const lastEvent = Math.max(...events.map(e => e.t));
   841→  
   842→  const sessionStart = workSession.startTime;
   843→  const sessionEnd = workSession.endTime || Date.now();
   844→  
   845→  return Math.abs(firstEvent - sessionStart) <= 2000 && Math.abs(lastEvent - sessionEnd) <= 2000;
   846→}
   847→
   848→function calculateTimestampConsistency(events: any[]): string {
   849→  const sortedEvents = events.sort((a, b) => a.t - b.t);
   850→  let inconsistencies = 0;
   851→  
   852→  for (let i = 1; i < sortedEvents.length; i++) {
   853→    if (sortedEvents[i].t < sortedEvents[i-1].t) {
   854→      inconsistencies++;
   855→    }
   856→  }
   857→  
   858→  return inconsistencies === 0 ? "Consistent" : `${inconsistencies} inconsistencies`;
   859→}
   860→
   861→function calculateMaxDrift(events: any[]): number {
   862→  if (events.length < 2) return 0;
   863→  const sortedEvents = events.sort((a, b) => a.t - b.t);
   864→  let maxDrift = 0;
   865→  
   866→  for (let i = 1; i < sortedEvents.length; i++) {
   867→    const drift = Math.abs(sortedEvents[i].t - sortedEvents[i-1].t);
   868→    maxDrift = Math.max(maxDrift, drift);
   869→  }
   870→  
   871→  return maxDrift;
   872→}
