   218→   218→export default function EvidenceLibrary() {
   219→   219→  const { tier: subscriptionTier } = useSubscriptionTier();
   220→   220→  const { isAuthenticated } = useConvexAuth();
   221→   221→  const { isDisconnected } = useConvexConnectionState();
   222→   222→
   223→   223→  // ── Workspace Context ──
   224→   224→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   225→   225→  const workspaceId = isConvexConnected ? (activeWorkspaceId as Id<"workspaces">) : undefined;
   226→   226→
   227→   227→  const [viewMode, setViewMode] = useState<ViewType>("date");
   228→   228→
   229→   229→  // ── Library date range (±30 days for monitoring) ──
   230→   230→  const libraryDateRange = useMemo(() => {
   231→   231→    const now = new Date();
   232→   232→    const start = new Date(now);
   233→   233→    start.setDate(start.getDate() - 30);
   234→   234→    const end = new Date(now);
   235→   235→    end.setDate(end.getDate() + 30);
   236→   236→    return { start: start.getTime(), end: end.getTime() };
   237→   237→  }, []);
   238→   238→
   239→   239→  const startOfDay = useMemo(() => new Date().setHours(0, 0, 0, 0), []);
   240→   240→
   241→   241→  // ── Export date range (1 year for comprehensive export) ──
   242→   242→  const [dateFrom, setDateFrom] = useState(() => {
   243→   243→    const d = new Date();
   244→   244→    d.setMonth(d.getMonth() - 3);
   245→   245→    return d.toISOString().split("T")[0];
   246→   246→  });
   247→   247→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   248→   248→
   249→   249→  // ── Convex Queries ──
   250→   250→
   251→   251→  // Build query args — only include workspaceId when it has a value so Convex
   252→   252→  // doesn't reject the call with an invalid optional ID.
   253→   253→  const libraryQueryArgs = useMemo(() => ({
   254→   254→    view: viewMode,
   255→   255→    startDate: libraryDateRange.start,
   256→   256→    endDate: libraryDateRange.end,
   257→   257→    ...(workspaceId ? { workspaceId } : {}),
   258→   258→  }), [viewMode, libraryDateRange.start, libraryDateRange.end, workspaceId]);
   259→   259→
   260→   260→  const exportQueryArgs = useMemo(() => ({
   261→   261→    view: "date" as const,
   262→   262→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   263→   263→    endDate: Date.now(),
   264→   264→    ...(workspaceId ? { workspaceId } : {}),
   265→   265→  }), [workspaceId]);
   266→   266→
   267→   267→  const timelineQueryArgs = useMemo(() => ({
   268→   268→    date: startOfDay,
   269→   269→    ...(workspaceId ? { workspaceId } : {}),
   270→   270→  }), [startOfDay, workspaceId]);
   271→   271→
   272→   272→  // Library data with ±30 day window for monitoring
   273→   273→  const evidenceData = useQuery(
   274→   274→    api.evidence.library.getEvidenceLibraryData,
   275→   275→    libraryQueryArgs
   276→   276→  );
   277→   277→
   278→   278→  // Export data with full year window for comprehensive exports
   279→   279→  const exportData = useQuery(
   280→   280→    api.evidence.library.getEvidenceLibraryData,
   281→   281→    exportQueryArgs
   282→   282→  ) as any | undefined;
   283→   283→
   284→   284→  const timelineData = useQuery(
   285→   285→    api.evidence.library.getEvidenceTimeline,
   286→   286→    timelineQueryArgs
   287→   287→  );
   288→   288→
   289→   289→  const clients = useQuery(api.clients.crud.getClients, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   290→   290→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, workspaceId ? { workspaceId } : "skip") as any[] | undefined;
   291→   291→
   292→   292→  const isLoading = evidenceData === undefined && timelineData === undefined;
   293→   293→  const exportLoading = exportData === undefined && clients === undefined;
   294→   294→  const timedOut = useQueryTimeout(isLoading, 5000);
   295→   295→  const showLoading = isLoading && !timedOut && !isDisconnected;
   296→   296→
   297→   297→  // ── Safe fallbacks ──
   298→   298→  const safeEvidenceData = evidenceData ?? {
   299→   299→    totalCount: 0,
   300→   300→    disputeSuccessRate: 0,
   301→   301→    contentQualityScore: 0,
   302→   302→    gapPrediction: {
   303→   303→      status: "no_gaps" as const,
   304→   304→      message: "No Gaps Predicted",
   305→   305→      description: "Start collecting evidence to build your protection.",
   306→   306→      nextGapTime: null,
   307→   307→      missingTypes: [],
   308→   308→    },
   309→   309→    evidenceItems: [],
   310→   310→    healthScore: null,
   311→   311→    disputeData: null,
   312→   312→    contentData: null,
   313→   313→  };
   314→   314→
   315→   315→  const safeTimelineData = timelineData ?? { protectedHours: 0, timeline: [] };
   316→   316→
   317→   317→  // ── Export State ──
   318→   318→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   319→   319→  const [selectedProject, setSelectedProject] = useState("all");
   320→   320→  const [selectedClient, setSelectedClient] = useState("all");
   321→   321→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   322→   322→  const [isExporting, setIsExporting] = useState(false);
   323→   323→  const [previewOpen, setPreviewOpen] = useState(false);
   324→   324→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   325→   325→
   326→   326→  // ── Derived Export Data ──
   327→   327→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   328→   328→    if (!exportData) return [];
   329→   329→    const items = exportData.evidenceItems ?? [];
   330→   330→    const typeCounts: Record<string, number> = {};
   331→   331→    const typeLabels: Record<string, string> = {
   332→   332→      screenshot_ref: "Screenshots", memo: "Work Memos", url: "URL Visits",
   333→   333→      keyboard: "Keyboard Activity", mouse: "Mouse Activity", platform_status: "Platform Status",
   334→   334→    };
   335→   335→    const typeIcons: Record<string, React.ReactNode> = {
   336→   336→      screenshot_ref: <Eye className="h-4 w-4" />, memo: <FileText className="h-4 w-4" />,
   337→   337→      url: <FileJson className="h-4 w-4" />, keyboard: <FileText className="h-4 w-4" />,
   338→   338→      mouse: <FileText className="h-4 w-4" />, platform_status: <Shield className="h-4 w-4" />,

   342→   342→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;

   344→   344→    return Object.entries(typeCounts).map(([id, count]) => ({

   350→   350→  }, [exportData]);
   351→   351→
   352→   352→  const projectOptions = useMemo(() => {
   353→   353→    const all = [{ id: "all", label: "All Projects" }];

   356→   356→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });

   358→   358→    }
   359→   359→    return all;
   360→   360→  }, [scopeDefinitions]);
   361→   361→
   362→   362→  const clientOptions = useMemo(() => {
   363→   363→    const all = [{ id: "all", label: "All Clients" }];

   367→   367→      }

   369→   369→    return all;
   370→   370→  }, [clients]);
   371→   371→
   372→   372→  const complianceScore = exportData?.healthScore?.score ?? safeEvidenceData.healthScore?.score ?? 0;
   373→   373→  const totalItems = exportData?.totalCount ?? safeEvidenceData.totalCount ?? 0;
   374→   374→  const disputeSuccessRate = exportData?.disputeSuccessRate ?? safeEvidenceData.disputeSuccessRate ?? 0;

   376→   376→  // Auto-select all evidence types when they load
   377→   377→  useEffect(() => {
   378→   378→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   379→   379→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   380→   380→    }
   381→   381→  }, [evidenceTypes]);
   382→   382→
   383→   383→  const totalEvidenceItems = useMemo(
   384→   384→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   385→   385→    [evidenceTypes, selectedEvidenceTypes]
   386→   386→  );
   387→   387→
   388→   388→  const isFormatLocked = (format: ExportFormat): boolean => {
   389→   389→    return !hasTierAccess(subscriptionTier, FORMAT_CONFIG[format].tierRequired);
   390→   390→  };
   391→   391→
   392→   392→  const toggleEvidenceType = (id: string) => {
   393→   393→    setSelectedEvidenceTypes((prev) =>
   394→   394→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   395→   395→    );
   396→   396→  };
   397→   397→
   398→   398→  const handleExport = async () => {
   399→   399→    if (selectedEvidenceTypes.length === 0) {
   400→   400→      toast.error("No evidence types selected", { description: "Select at least one evidence type to export." });
   401→   401→      return;
   402→   402→    }
   403→   403→    if (isFormatLocked(selectedFormat)) {
   404→   404→      toast.error("Upgrade required", { description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.` });
   405→   405→      return;
   406→   406→    }
   407→   407→    setIsExporting(true);
   408→   408→    try {
   409→   409→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   410→   410→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   411→   411→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   412→   412→
   413→   413→      exportEvidence(selectedFormat, {
   414→   414→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   415→   415→        dateRange: `${dateFrom} to ${dateTo}`,
   416→   416→        project: selectedProjectLabel,
   417→   417→        client: selectedClientLabel,
   418→   418→        complianceScore,
   419→   419→        totalItems: totalEvidenceItems,
   420→   420→      });
   421→   421→
   422→   422→      const newExport: RecentExport = {
   423→   423→        id: `exp_${Date.now()}`,
   424→   424→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   425→   425→        format: selectedFormat,
   426→   426→        status: "completed",
   427→   427→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   428→   428→        date: new Date().toISOString().split("T")[0],
   429→   429→        itemCount: totalEvidenceItems,
   430→   430→        complianceVerified: hasTierAccess(subscriptionTier, "pro"),
   431→   431→        project: selectedProjectLabel,
   432→   432→      };
   433→   433→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   434→   434→
   435→   435→      toast.success("Export complete", {
   436→   436→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   437→   437→      });
   438→   438→
   439→   439→      trackEvent(AnalyticsEvents.EVIDENCE_CAPTURED, { format: selectedFormat, itemCount: totalEvidenceItems });
   440→   440→    } catch {
