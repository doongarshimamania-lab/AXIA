    61→    61→function AnimatedNumber({ value, duration = 1200, prefix = "", suffix = "", decimals = 0 }: {
    62→    62→  value: number; duration?: number; prefix?: string; suffix?: string; decimals?: number;
    63→    63→}) {
    64→    64→  const [display, setDisplay] = useState(0);
    65→    65→  const prevValue = useRef(0);
    66→    66→  const rafRef = useRef<number>();
    67→    67→
    68→    68→  useEffect(() => {
    69→    69→    const start = prevValue.current;
    70→    70→    const end = value;
    71→    71→    const startTime = performance.now();
    72→    72→
    73→    73→    const animate = (now: number) => {
    74→    74→      const elapsed = now - startTime;
    75→    75→      const progress = Math.min(elapsed / duration, 1);
    76→    76→      // Ease out cubic
    77→    77→      const eased = 1 - Math.pow(1 - progress, 3);
    78→    78→      const current = start + (end - start) * eased;
    79→    79→      setDisplay(current);
    80→    80→      if (progress < 1) {
    81→    81→        rafRef.current = requestAnimationFrame(animate);
    82→    82→      } else {
    83→    83→        prevValue.current = end;
    84→    84→      }
    85→    85→    };
    86→    86→
    87→    87→    rafRef.current = requestAnimationFrame(animate);
    88→    88→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
    89→    89→  }, [value, duration]);
    90→    90→
    91→    91→  const formatted = decimals > 0
    92→    92→    ? display.toFixed(decimals)
    93→    93→    : Math.round(display).toLocaleString();
    94→    94→
    95→    95→  return <span>{prefix}{formatted}{suffix}</span>;
    96→    96→}

    99→    99→function Sparkline({ data, color = "#8B5CF6", height = 32, width = 80 }: {
   100→   100→  data: number[]; color?: string; height?: number; width?: number;
   101→   101→}) {
   102→   102→  if (!data || data.length < 2) return null;
   103→   103→
   104→   104→  const min = Math.min(...data);
   105→   105→  const max = Math.max(...data);
   106→   106→  const range = max - min || 1;
   107→   107→  const padding = 2;
   108→   108→
   109→   109→  const points = data.map((v, i) => {
   110→   110→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   111→   111→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   112→   112→    return `${x},${y}`;
   113→   113→  }).join(" ");
   114→   114→
   115→   115→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   116→   116→
   117→   117→  return (
   118→   118→    <svg width={width} height={height} className="overflow-visible">
   119→   119→      <defs>
   120→   120→        <linearGradient id={`spark-grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
   121→   121→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   122→   122→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   123→   123→        </linearGradient>
   124→   124→      </defs>
   125→   125→      <polygon points={areaPoints} fill={`url(#spark-grad-${color.replace("#", "")})`} />
   126→   126→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   127→   127→      {/* Last point dot */}
   128→   128→      {data.length > 0 && (
   129→   129→        <circle
   130→   130→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   131→   131→          cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)}
   132→   132→          r="2.5"
   133→   133→          fill={color}
   134→   134→          className="animate-pulse"
   135→   135→        />
   136→   136→      )}
   137→   137→    </svg>
   138→   138→  );
   139→   139→}

   142→   142→function ProgressRing({ value, size = 56, strokeWidth = 4, color = "#8B5CF6", label }: {
   143→   143→  value: number; size?: number; strokeWidth?: number; color?: string; label?: string;
   144→   144→}) {
   145→   145→  const radius = (size - strokeWidth) / 2;
   146→   146→  const circumference = 2 * Math.PI * radius;
   147→   147→  const offset = circumference - (value / 100) * circumference;
   148→   148→
   149→   149→  return (
   150→   150→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   151→   151→      <svg width={size} height={size} className="-rotate-90">
   152→   152→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   153→   153→        <motion.circle
   154→   154→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
   155→   155→          strokeLinecap="round"
   156→   156→          strokeDasharray={circumference}
   157→   157→          initial={{ strokeDashoffset: circumference }}
   158→   158→          animate={{ strokeDashoffset: offset }}
   159→   159→          transition={{ duration: 1.2, ease: "easeOut" }}
   160→   160→        />
   161→   161→      </svg>
   162→   162→      <span className="absolute text-[10px] font-bold" style={{ color }}>{value}%</span>
   163→   163→    </div>
   164→   164→  );
   165→   165→}
