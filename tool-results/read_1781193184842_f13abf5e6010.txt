   118→  const { activeWorkspaceId, isTeamMode, isOwner, canManageTeam, activeWorkspace } = useWorkspaceContext();
   119→
   120→  // ─── Convex Queries ────────────────────────────────────────────────────────
   121→  const hasRealWorkspaceId = activeWorkspaceId && !activeWorkspaceId.startsWith("ws_");
   122→
   123→  const convexMembers = useQuery(
   124→    hasRealWorkspaceId ? api.workspaces.members.getMembers : "skip",
   125→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   126→  ) as any[] | undefined;
   127→
   128→  const convexStats = useQuery(
   129→    hasRealWorkspaceId ? api.workspaces.crud.getWorkspaceStats : "skip",
   130→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   131→  ) as any | undefined;
   132→
   133→  const convexInvitations = useQuery(
   134→    hasRealWorkspaceId ? api.workspaces.invitations.getInvitations : "skip",
   135→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any, status: "pending" as const } : "skip"
   136→  ) as any[] | undefined;
   137→
   138→  // ─── Teams queries ─────────────────────────────────────────────────────────
   139→  const teamsApi = (api as any).teams?.crud;
   140→  const hasTeamsApi = !!teamsApi?.getTeams;
   141→
   142→  const convexTeams = useQuery(
   143→    hasTeamsApi && hasRealWorkspaceId ? teamsApi.getTeams : "skip",
   144→    hasRealWorkspaceId ? { workspaceId: activeWorkspaceId as any } : "skip"
   145→  ) as any[] | undefined;
   146→
   147→  // ─── Team Members per team ──────────────────────────────────────────────────
   148→  const hasTeamMembersApi = !!teamsApi?.getTeamMembers;
   149→
   150→  // ─── Mutations ─────────────────────────────────────────────────────────────
   151→  const inviteMemberMutation = useMutation(api.workspaces.invitations.createInvitation);
   152→  const removeMemberMutation = useMutation(api.workspaces.members.removeMember);
   153→  const updateRoleMutation = useMutation(api.workspaces.members.updateMemberRole);
   154→  const cancelInvitationMutation = useMutation(api.workspaces.invitations.cancelInvitation);
   155→
   156→  // Team mutations
   157→  const createTeamMutation = useMutation(hasTeamsApi ? teamsApi.createTeam : null);
   158→  const updateTeamMutation = useMutation(hasTeamsApi ? teamsApi.updateTeam : null);
   159→  const deleteTeamMutation = useMutation(hasTeamsApi ? teamsApi.deleteTeam : null);
   160→  const addTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.addTeamMember : null);
   161→  const removeTeamMemberMutation = useMutation(hasTeamsApi ? teamsApi.removeTeamMember : null);
   162→
   163→  // ─── State ─────────────────────────────────────────────────────────────────
   164→  const [activeTab, setActiveTab] = useState<string>("members");
   165→
   166→  // Invite dialog
   167→  const [showInviteDialog, setShowInviteDialog] = useState(false);
   168→  const [inviteEmail, setInviteEmail] = useState("");
   169→  const [inviteRole, setInviteRole] = useState<"manager" | "member">("member");
   170→  const [inviteMessage, setInviteMessage] = useState("");
   171→  const [isInviting, setIsInviting] = useState(false);
   172→
   173→  // Remove/change role
   174→  const [memberToRemove, setMemberToRemove] = useState<any>(null);
   175→  const [memberToChangeRole, setMemberToChangeRole] = useState<any>(null);
   176→  const [newRole, setNewRole] = useState<"manager" | "member">("member");
   177→
   178→  // Team CRUD state
   179→  const [showCreateTeamDialog, setShowCreateTeamDialog] = useState(false);
   180→  const [teamName, setTeamName] = useState("");
   181→  const [teamColor, setTeamColor] = useState(TEAM_COLORS[0]);
   182→  const [teamDescription, setTeamDescription] = useState("");
   183→  const [teamIsCrossTeam, setTeamIsCrossTeam] = useState(false);
   184→  const [isCreatingTeam, setIsCreatingTeam] = useState(false);
   185→
   186→  // Edit team
   187→  const [editingTeam, setEditingTeam] = useState<any>(null);
   188→  const [editTeamName, setEditTeamName] = useState("");
   189→  const [editTeamColor, setEditTeamColor] = useState("");
   190→  const [editTeamDescription, setEditTeamDescription] = useState("");
   191→  const [isUpdatingTeam, setIsUpdatingTeam] = useState(false);
   192→
   193→  // Delete team
   194→  const [teamToDelete, setTeamToDelete] = useState<any>(null);
   195→  const [isDeletingTeam, setIsDeletingTeam] = useState(false);
   196→
   197→  // Add member to team
   198→  const [teamToAddMember, setTeamToAddMember] = useState<any>(null);
   199→  const [addMemberUserId, setAddMemberUserId] = useState("");
   200→  const [addMemberRole, setAddMemberRole] = useState<"lead" | "member">("member");
   201→  const [isAddingTeamMember, setIsAddingTeamMember] = useState(false);
   202→
   203→  // Remove member from team
   204→  const [teamMemberToRemove, setTeamMemberToRemove] = useState<any>(null);
   205→  const [teamMemberRemoveTeamId, setTeamMemberRemoveTeamId] = useState<string>("");
   206→
   207→  // ─── Map Convex data to UI shape ───────────────────────────────────────────
   208→  const members = useMemo(() => {
   209→    if (convexMembers && convexMembers.length > 0) {
   210→      return convexMembers.map((m: any) => ({
   211→        _id: m._id,
   212→        userId: m.userId,
   213→        name: m.userName ?? m.userEmail ?? "Unknown",
   214→        displayName: m.userName ?? m.userEmail ?? "Unknown",
   215→        email: m.userEmail ?? "",
   216→        image: m.userImage ?? "",
   217→        role: m.role,
   218→        status: m.status,
   219→        joinedAt: m.joinedAt,
   220→        lastActiveAt: m.lastActiveAt ?? null,
   221→        projectsAssigned: 0,
   222→        hoursThisWeek: 0,
   223→      }));
   224→    }
   225→    return [];
   226→  }, [convexMembers]);
   227→
   228→  const teams = useMemo(() => {
   229→    if (convexTeams && convexTeams.length > 0) {
   230→      return convexTeams;
   231→    }
   232→    return [];
   233→  }, [convexTeams]);
   234→
   235→  const stats = useMemo(() => {
   236→    if (convexStats) {
   237→      return {
   238→        memberCount: convexStats.memberCount ?? 0,
   239→        clientCount: convexStats.clientCount ?? 0,
   240→        activeProjectCount: convexStats.activeProjectCount ?? 0,
   241→        pendingInvoiceCount: 0,
   242→        totalRevenue: 0,
   243→        totalHoursThisWeek: 0,
   244→        protectionScore: 0,
   245→      };
   246→    }
   247→    return null;
   248→  }, [convexStats]);
   249→
   250→  const invitations = useMemo(() => {
   251→    if (convexInvitations && convexInvitations.length > 0) {
   252→      return convexInvitations.map((inv: any) => ({
   253→        _id: inv._id,
   254→        name: inv.email,
   255→        displayName: inv.email,
   256→        email: inv.email,
   257→        role: inv.role,
   258→        invitedAt: inv.createdAt,
   259→        invitedBy: inv.inviterName,
   260→      }));
   261→    }
   262→    return [];
   263→  }, [convexInvitations]);
   264→
   265→  const isLoading = hasRealWorkspaceId && (convexMembers === undefined);
   266→  const isDemoMode = !hasRealWorkspaceId;
   267→
   268→  // ─── Handlers ──────────────────────────────────────────────────────────────
   269→  const handleInvite = async () => {
   270→    if (!inviteEmail.trim() || !activeWorkspaceId) return;
   271→    setIsInviting(true);
   272→    try {
   273→      if (hasRealWorkspaceId) {
   274→        await inviteMemberMutation({
   275→          workspaceId: activeWorkspaceId as any,
   276→          email: inviteEmail.trim(),
   277→          role: inviteRole,
   278→        });
   279→      }
   280→      toast.success(`Invitation sent to ${inviteEmail}`);
   281→      setInviteEmail("");
   282→      setInviteMessage("");
   283→      setShowInviteDialog(false);
   284→    } catch (e: any) {
   285→      toast.error(e.message || "Failed to send invitation");
   286→    } finally {
   287→      setIsInviting(false);
   288→    }
   289→  };
   290→
   291→  const handleRemoveMember = async () => {
   292→    if (!memberToRemove) return;
   293→    try {
   294→      await removeMemberMutation({ memberId: memberToRemove._id });
   295→      toast.success(`${memberToRemove.name} removed from workspace`);
   296→      setMemberToRemove(null);
   297→    } catch (e: any) {
   298→      toast.error(e.message || "Failed to remove member");
   299→    }
   300→  };
   301→
   302→  const handleChangeRole = async () => {
   303→    if (!memberToChangeRole) return;
   304→    try {
   305→      await updateRoleMutation({ memberId: memberToChangeRole._id, role: newRole });
   306→      toast.success(`Role updated to ${newRole}`);
   307→      setMemberToChangeRole(null);
   308→    } catch (e: any) {
   309→      toast.error(e.message || "Failed to update role");
   310→    }
   311→  };
   312→
   313→  const handleCancelInvitation = async (invitation: any) => {
   314→    try {
   315→      await cancelInvitationMutation({ invitationId: invitation._id });
   316→      toast.success(`Invitation to ${invitation.email || invitation.name} cancelled`);
   317→    } catch (e: any) {
   318→      toast.error(e.message || "Failed to cancel invitation");
   319→    }
   320→  };
   321→
   322→  // ─── Team CRUD Handlers ────────────────────────────────────────────────────
   323→  const handleCreateTeam = async () => {
   324→    if (!teamName.trim() || !activeWorkspaceId) return;
   325→    setIsCreatingTeam(true);
   326→    try {
   327→      if (createTeamMutation && hasRealWorkspaceId) {
   328→        await createTeamMutation({
   329→          workspaceId: activeWorkspaceId as any,
   330→          name: teamName.trim(),
   331→          color: teamColor,
   332→          description: teamDescription.trim() || undefined,
   333→          isCrossTeam: teamIsCrossTeam || undefined,
   334→        });
   335→      }
   336→      toast.success(`Team "${teamName}" created!`);
   337→      setShowCreateTeamDialog(false);
   338→      setTeamName("");
   339→      setTeamColor(TEAM_COLORS[0]);
   340→      setTeamDescription("");
   341→      setTeamIsCrossTeam(false);
   342→    } catch (e: any) {
   343→      toast.error(e.message || "Failed to create team");
   344→    } finally {
   345→      setIsCreatingTeam(false);
   346→    }
   347→  };
   348→
   349→  const handleUpdateTeam = async () => {
   350→    if (!editingTeam) return;
   351→    setIsUpdatingTeam(true);
   352→    try {
   353→      if (updateTeamMutation) {
   354→        await updateTeamMutation({
   355→          teamId: editingTeam._id,
   356→          name: editTeamName.trim() || undefined,
   357→          color: editTeamColor || undefined,
   358→          description: editTeamDescription.trim() || undefined,
   359→        });
   360→      }
   361→      toast.success("Team updated!");
   362→      setEditingTeam(null);
   363→    } catch (e: any) {
   364→      toast.error(e.message || "Failed to update team");
   365→    } finally {
   366→      setIsUpdatingTeam(false);
   367→    }
   368→  };
   369→
   370→  const handleDeleteTeam = async () => {
   371→    if (!teamToDelete) return;
   372→    setIsDeletingTeam(true);
   373→    try {
   374→      if (deleteTeamMutation) {
   375→        await deleteTeamMutation({ teamId: teamToDelete._id });
   376→      }
   377→      toast.success(`Team "${teamToDelete.name}" deleted`);
   378→      setTeamToDelete(null);
   379→    } catch (e: any) {
   380→      toast.error(e.message || "Failed to delete team");
   381→    } finally {
   382→      setIsDeletingTeam(false);
   383→    }
   384→  };
   385→
   386→  const handleAddTeamMember = async () => {
   387→    if (!teamToAddMember || !addMemberUserId) return;
   388→    setIsAddingTeamMember(true);
   389→    try {
   390→      if (addTeamMemberMutation) {
   391→        await addTeamMemberMutation({
   392→          teamId: teamToAddMember._id,
   393→          userId: addMemberUserId as any,
   394→          role: addMemberRole,
   395→        });
   396→      }
   397→      toast.success("Member added to team!");
   398→      setTeamToAddMember(null);
   399→      setAddMemberUserId("");
   400→      setAddMemberRole("member");
   401→    } catch (e: any) {
   402→      toast.error(e.message || "Failed to add member to team");
   403→    } finally {
   404→      setIsAddingTeamMember(false);
   405→    }
   406→  };
   407→
   408→  const handleRemoveTeamMember = async () => {
   409→    if (!teamMemberToRemove || !teamMemberRemoveTeamId) return;
   410→    try {
   411→      if (removeTeamMemberMutation) {
   412→        await removeTeamMemberMutation({
   413→          teamId: teamMemberRemoveTeamId as any,
   414→          userId: teamMemberToRemove.userId as any,
   415→        });
   416→      }
   417→      toast.success("Member removed from team");
   418→      setTeamMemberToRemove(null);
   419→      setTeamMemberRemoveTeamId("");
   420→    } catch (e: any) {
   421→      toast.error(e.message || "Failed to remove member from team");
   422→    }
   423→  };
   424→
   425→  // ─── Not in team mode: Show upgrade prompt ────────────────
   426→  if (!isTeamMode) {
   427→    return <SoloModePrompt />;
   428→  }
   429→
   430→  const activeMembers = members.filter((m: any) => m.status === "active");
   431→  const ownerCount = activeMembers.filter((m: any) => m.role === "owner").length;
   432→  const managerCount = activeMembers.filter((m: any) => m.role === "manager").length;
   433→  const memberCount = activeMembers.filter((m: any) => m.role === "member").length;
   434→
   435→  const roleIcon = (role: string) => {
   436→    switch (role) {
   437→      case "owner": return <Crown className="w-3.5 h-3.5 text-amber-500" />;
   438→      case "manager": return <Shield className="w-3.5 h-3.5 text-blue-500" />;
   439→      case "lead": return <Shield className="w-3.5 h-3.5 text-emerald-500" />;
   440→      case "member": return <Eye className="w-3.5 h-3.5 text-gray-500" />;
   441→      default: return null;
   442→    }
   443→  };
   444→
   445→  const roleBadge = (role: string) => {
   446→    const variants: Record<string, any> = {
   447→      owner: { className: "bg-amber-500/10 text-amber-600 border-amber-500/20", label: "Owner" },
   448→      manager: { className: "bg-blue-500/10 text-blue-600 border-blue-500/20", label: "Manager" },
   449→      lead: { className: "bg-emerald-500/10 text-emerald-600 border-emerald-500/20", label: "Lead" },
   450→      member: { className: "bg-gray-500/10 text-gray-600 border-gray-500/20", label: "Member" },
   451→    };
   452→    const v = variants[role] || variants.member;
   453→    return (
   454→      <Badge variant="outline" className={`gap-1 ${v.className}`}>
   455→        {roleIcon(role)}
   456→        {v.label}
   457→      </Badge>
   458→    );
   459→  };
   460→
   461→  const formatLastActive = (timestamp: number | null) => {
   462→    if (!timestamp) return "Never";
   463→    const diffMs = Date.now() - timestamp;
   464→    const diffMins = Math.floor(diffMs / 60000);
   465→    if (diffMins < 5) return "Just now";
   466→    if (diffMins < 60) return `${diffMins}m ago`;
   467→    const diffHours = Math.floor(diffMins / 60);
   468→    if (diffHours < 24) return `${diffHours}h ago`;
   469→    const diffDays = Math.floor(diffHours / 24);
   470→    return `${diffDays}d ago`;
   471→  };
   472→
   473→  const formatJoined = (timestamp: number | null) => {
   474→    if (!timestamp) return "";
   475→    const diffMs = Date.now() - timestamp;
   476→    const diffDays = Math.floor(diffMs / (24 * 60 * 60 * 1000));
   477→    if (diffDays < 1) return "Joined today";
   478→    if (diffDays < 30) return `Joined ${diffDays}d ago`;
   479→    const diffMonths = Math.floor(diffDays / 30);
   480→    return `Joined ${diffMonths}mo ago`;
   481→  };
   482→
   483→  return (
   484→    <div className="p-8">
   485→      <div className="max-w-6xl mx-auto space-y-6">
   486→        {/* Header */}
   487→        <div className="flex items-center justify-between">
   488→          <div>
   489→            <h1 className="text-3xl font-bold text-foreground mb-2 flex items-center gap-3">
   490→              <Building2 className="w-8 h-8 text-purple-500" />
   491→              {activeWorkspace?.name || "Team Workspace"}
   492→            </h1>
   493→            <p className="text-muted-foreground">
   494→              Manage your team members, teams, and invitations
   495→            </p>
   496→          </div>
   497→          {canManageTeam && (
   498→            <div className="flex gap-2">
   499→              <Button className="gap-2" onClick={() => setShowInviteDialog(true)}>
   500→                <UserPlus className="w-4 h-4" />
   501→                Invite Member
   502→              </Button>
   503→            </div>
   504→          )}
   505→        </div>
   506→
   507→        {/* Demo Mode Banner */}
   508→        {isDemoMode && <DemoModeBanner />}
   509→
   510→        {/* Stats Cards */}
   511→        {isLoading ? (
   512→          <StatsSkeleton />
   513→        ) : (
   514→          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
   515→            <Card className="p-5">
   516→              <div className="flex items-center gap-3">
   517→                <div className="p-2.5 bg-purple-500/10 rounded-lg"><Users className="w-5 h-5 text-purple-500" /></div>
   518→                <div>
   519→                  <p className="text-sm text-muted-foreground">Members</p>
   520→                  <p className="text-2xl font-bold">{stats?.memberCount ?? activeMembers.length}</p>
   521→                </div>
   522→              </div>
   523→            </Card>
   524→            <Card className="p-5">
   525→              <div className="flex items-center gap-3">
   526→                <div className="p-2.5 bg-emerald-500/10 rounded-lg"><Shield className="w-5 h-5 text-emerald-500" /></div>
   527→                <div>
   528→                  <p className="text-sm text-muted-foreground">Teams</p>
   529→                  <p className="text-2xl font-bold">{teams.length}</p>
   530→                </div>
   531→              </div>
   532→            </Card>
   533→            <Card className="p-5">
   534→              <div className="flex items-center gap-3">
