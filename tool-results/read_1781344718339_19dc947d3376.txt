    96→function getTierLevel(tier: string): number {
    97→  const levels: Record<string, number> = { free: 0, starter: 1, pro: 2, expert: 3, client: 0 };
    98→  return levels[tier] ?? 0;
    99→}
   100→
   101→function hasTierAccess(userTier: string, requiredTier: string): boolean {
   102→  return getTierLevel(userTier) >= getTierLevel(requiredTier);
   103→}

   241→// --- Main Component ---
   242→export default function EvidenceExport() {
   243→  const { tier } = useSubscriptionTier();
   244→  const { isAuthenticated } = useConvexAuth();
   245→
   246→  // ─── Convex Queries ────────────────────────────────────────────────────────
   247→  const evidenceData = useQuery(api.evidence.library.getEvidenceLibraryData, {
   248→    view: "date" as const,
   249→    startDate: Date.now() - 365 * 24 * 60 * 60 * 1000,
   250→    endDate: Date.now(),
   251→  }) as any | undefined;
   252→
   253→  const clients = useQuery(api.clients.crud.getClients, {}) as any[] | undefined;
   254→  const scopeDefinitions = useQuery(api.scope.crud.getScopeDefinitions, {}) as any[] | undefined;
   255→
   256→  const { isDisconnected } = useConvexConnectionState();
   257→  const isLoading = evidenceData === undefined || clients === undefined;
   258→  const timedOut = useQueryTimeout(isLoading, 3000);
   259→  const showLoading = isLoading && !timedOut && !isDisconnected;
   260→
   261→  // ─── Derived Data ──────────────────────────────────────────────────────────
   262→
   263→  // Build evidence type options from real data
   264→  const evidenceTypes: EvidenceTypeOption[] = useMemo(() => {
   265→    if (!evidenceData) return [];
   266→    const items = evidenceData.evidenceItems ?? [];
   267→    const typeCounts: Record<string, number> = {};
   268→    const typeLabels: Record<string, string> = {
   269→      screenshot_ref: "Screenshots",
   270→      memo: "Work Memos",
   271→      url: "URL Visits",
   272→      keyboard: "Keyboard Activity",
   273→      mouse: "Mouse Activity",
   274→      platform_status: "Platform Status",
   275→    };
   276→    const typeIcons: Record<string, React.ReactNode> = {
   277→      screenshot_ref: <Eye className="h-4 w-4" />,
   278→      memo: <FileText className="h-4 w-4" />,
   279→      url: <FileJson className="h-4 w-4" />,
   280→      keyboard: <FileText className="h-4 w-4" />,
   281→      mouse: <FileText className="h-4 w-4" />,
   282→      platform_status: <Shield className="h-4 w-4" />,
   283→    };
   284→
   285→    for (const item of items) {
   286→      const kind = item.type ?? "unknown";
   287→      typeCounts[kind] = (typeCounts[kind] ?? 0) + 1;
   288→    }
   289→
   290→    return Object.entries(typeCounts).map(([id, count]) => ({
   291→      id,
   292→      label: typeLabels[id] ?? id.charAt(0).toUpperCase() + id.slice(1).replace(/_/g, " "),
   293→      icon: typeIcons[id] ?? <FileText className="h-4 w-4" />,
   294→      count,
   295→    }));
   296→  }, [evidenceData]);
   297→
   298→  // Build project options from real scopes
   299→  const projectOptions = useMemo(() => {
   300→    const all = [{ id: "all", label: "All Projects" }];
   301→    if (scopeDefinitions) {
   302→      for (const scope of scopeDefinitions) {
   303→        all.push({ id: scope._id, label: scope.title ?? "Untitled Scope" });
   304→      }
   305→    }
   306→    return all;
   307→  }, [scopeDefinitions]);
   308→
   309→  // Build client options from real clients
   310→  const clientOptions = useMemo(() => {
   311→    const all = [{ id: "all", label: "All Clients" }];
   312→    if (clients) {
   313→      for (const client of clients) {
   314→        all.push({ id: client._id, label: client.clientName ?? "Unknown Client" });
   315→      }
   316→    }
   317→    return all;
   318→  }, [clients]);
   319→
   320→  // Compliance score from real data
   321→  const complianceScore = evidenceData?.healthScore?.score ?? 0;
   322→  const totalItems = evidenceData?.totalCount ?? 0;
   323→  const disputeSuccessRate = evidenceData?.disputeSuccessRate ?? 0;
   324→
   325→  // State
   326→  const [selectedFormat, setSelectedFormat] = useState<ExportFormat>("csv");
   327→  const [dateFrom, setDateFrom] = useState(() => {
   328→    const d = new Date();
   329→    d.setMonth(d.getMonth() - 3);
   330→    return d.toISOString().split("T")[0];
   331→  });
   332→  const [dateTo, setDateTo] = useState(() => new Date().toISOString().split("T")[0]);
   333→  const [selectedProject, setSelectedProject] = useState("all");
   334→  const [selectedClient, setSelectedClient] = useState("all");
   335→  const [selectedEvidenceTypes, setSelectedEvidenceTypes] = useState<string[]>([]);
   336→  const [isExporting, setIsExporting] = useState(false);
   337→  const [previewOpen, setPreviewOpen] = useState(false);
   338→  const [recentExports, setRecentExports] = useState<RecentExport[]>([]);
   339→
   340→  // Auto-select all evidence types when they load
   341→  useEffect(() => {
   342→    if (evidenceTypes.length > 0 && selectedEvidenceTypes.length === 0) {
   343→      setSelectedEvidenceTypes(evidenceTypes.map((t) => t.id));
   344→    }
   345→  }, [evidenceTypes]);
   346→
   347→  // Derived
   348→  const totalEvidenceItems = useMemo(
   349→    () => evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id)).reduce((sum, t) => sum + t.count, 0),
   350→    [evidenceTypes, selectedEvidenceTypes]
   351→  );
   352→
   353→  const isFormatLocked = (format: ExportFormat): boolean => {
   354→    const requiredTier = FORMAT_CONFIG[format].tierRequired;
   355→    return !hasTierAccess(tier, requiredTier);
   356→  };
   357→
   358→  const toggleEvidenceType = (id: string) => {
   359→    setSelectedEvidenceTypes((prev) =>
   360→      prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]
   361→    );
   362→  };
   363→
   364→  const handleExport = async () => {
   365→    if (selectedEvidenceTypes.length === 0) {
   366→      toast.error("No evidence types selected", {
   367→        description: "Select at least one evidence type to export.",
   368→      });
   369→      return;
   370→    }
   371→
   372→    if (isFormatLocked(selectedFormat)) {
   373→      toast.error("Upgrade required", {
   374→        description: `Export as ${FORMAT_CONFIG[selectedFormat].label} requires a ${FORMAT_CONFIG[selectedFormat].tierRequired} plan or higher.`,
   375→      });
   376→      return;
   377→    }
   378→
   379→    setIsExporting(true);
   380→
   381→    try {
   382→      const selectedTypes = evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   383→      const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   384→      const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   385→
   386→      exportEvidence(selectedFormat, {
   387→        evidenceTypes: selectedTypes.map((t) => ({ id: t.id, label: t.label, count: t.count })),
   388→        dateRange: `${dateFrom} to ${dateTo}`,
   389→        project: selectedProjectLabel,
   390→        client: selectedClientLabel,
   391→        complianceScore,
   392→        totalItems: totalEvidenceItems,
   393→      });
   394→
   395→      // Add to recent exports
   396→      const newExport: RecentExport = {
   397→        id: `exp_${Date.now()}`,
   398→        name: `${selectedProjectLabel} — ${FORMAT_CONFIG[selectedFormat].label} Export`,
   399→        format: selectedFormat,
   400→        status: "completed",
   401→        size: `~${(totalEvidenceItems * 0.12).toFixed(1)} MB`,
   402→        date: new Date().toISOString().split("T")[0],
   403→        itemCount: totalEvidenceItems,
   404→        complianceVerified: hasTierAccess(tier, "pro"),
   405→        project: selectedProjectLabel,
   406→      };
   407→      setRecentExports((prev) => [newExport, ...prev].slice(0, 10));
   408→
   409→      toast.success("Export complete", {
   410→        description: `Your ${FORMAT_CONFIG[selectedFormat].label} export with ${totalEvidenceItems} items has been downloaded.`,
   411→      });
   412→    } catch {
   413→      toast.error("Export failed", {
   414→        description: "An error occurred while generating the export file.",
   415→      });
   416→    }
   417→
   418→    setIsExporting(false);
   419→  };
   420→
   421→  const handleFormatSelect = (format: ExportFormat) => {
   422→    if (isFormatLocked(format)) {
   423→      toast.error("Upgrade required", {
   424→        description: `Export as ${FORMAT_CONFIG[format].label} requires a ${FORMAT_CONFIG[format].tierRequired} plan or higher.`,
   425→      });
   426→      return;
   427→    }
   428→    setSelectedFormat(format);
   429→  };
   430→
   431→  const previewItems = useMemo(() => {
   432→    return evidenceTypes.filter((t) => selectedEvidenceTypes.includes(t.id));
   433→  }, [evidenceTypes, selectedEvidenceTypes]);
   434→
   435→  const selectedProjectLabel = projectOptions.find((p) => p.id === selectedProject)?.label ?? "All Projects";
   436→  const selectedClientLabel = clientOptions.find((c) => c.id === selectedClient)?.label ?? "All Clients";
   437→
   438→  const hasData = totalItems > 0 || evidenceTypes.length > 0;
   439→
   440→  // ─── Render ───────────────────────────────────────────────────────────────
   441→
   442→  return (
   443→    <motion.div
   444→      initial={{ opacity: 0 }}
   445→      animate={{ opacity: 1 }}
   446→      className="w-full min-h-screen bg-background"
   447→    >
   448→      <PageLayout spaced>
   449→        {/* Header */}
   450→        <div>
   451→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">Evidence Export</h1>
   452→          <p className="text-[16px] text-muted-foreground">
   453→            Package and download your work evidence in court-ready formats with compliance verification.
   454→          </p>
   455→        </div>
   456→
   457→        {/* Demo Mode Banner */}
   458→        {!isAuthenticated && <DemoModeBanner />}
   459→
   460→        {/* Loading / Timeout State */}
   461→        {showLoading ? (
   462→          <>
   463→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   464→              {[0, 1, 2, 3].map((i) => <SummarySkeleton key={i} />)}
   465→            </div>
   466→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   467→              <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   468→              <div><Card><CardContent className="p-6"><Skeleton className="h-64 w-full" /></CardContent></Card></div>
   469→            </div>
   470→          </>
   471→        ) : !hasData ? (
   472→          <EmptyEvidenceState />
   473→        ) : (
   474→          <>
   475→            {/* Summary Dashboard */}
   476→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   477→              <SummaryCard
   478→                title="Total Items"
   479→                value={totalItems}
   480→                subtitle="Across all projects"
   481→                icon={<HardDrive className="h-5 w-5 text-primary-foreground" />}
   482→                accent="bg-primary"
   483→              />
   484→              <SummaryCard
   485→                title="Evidence Types"
   486→                value={evidenceTypes.length}
   487→                subtitle="Different categories"
   488→                icon={<Clock className="h-5 w-5 text-amber-700" />}
   489→                accent="bg-amber-100 dark:bg-amber-900/30"
   490→              />
   491→              <SummaryCard
   492→                title="Compliance Score"
   493→                value={`${complianceScore}%`}
   494→                subtitle={complianceScore >= 80 ? "Court-admissible ready" : "Needs improvement"}
   495→                icon={<ShieldCheck className="h-5 w-5 text-emerald-700" />}
   496→                accent="bg-emerald-100 dark:bg-emerald-900/30"
   497→              />
   498→              <SummaryCard
   499→                title="Dispute Success"
   500→                value={`${disputeSuccessRate}%`}
   501→                subtitle="Based on evidence quality"
   502→                icon={<Shield className="h-5 w-5 text-sky-700" />}
   503→                accent="bg-sky-100 dark:bg-sky-900/30"
   504→              />
   505→            </div>
   506→
   507→            {/* Export Configuration */}
   508→            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   509→              {/* Left Column: Format + Filters */}
   510→              <div className="lg:col-span-2 space-y-6">
   511→                {/* Format Selection */}
   512→                <Card>
   513→                  <CardHeader>
   514→                    <CardTitle className="flex items-center gap-2">
   515→                      <Download className="h-5 w-5 text-primary" />
   516→                      Export Format
   517→                    </CardTitle>
   518→                    <CardDescription>Choose the output format for your evidence package.</CardDescription>
   519→                  </CardHeader>
   520→                  <CardContent>
   521→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   522→                      {(Object.keys(FORMAT_CONFIG) as ExportFormat[]).map((formatKey) => (
   523→                        <FormatOption
   524→                          key={formatKey}
   525→                          formatKey={formatKey}
   526→                          selected={selectedFormat === formatKey}
   527→                          onSelect={() => handleFormatSelect(formatKey)}
   528→                          locked={isFormatLocked(formatKey)}
   529→                        />
   530→                      ))}
   531→                    </div>
   532→                    {isFormatLocked(selectedFormat) && (
   533→                      <div className="mt-4 flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-900/20 dark:border-amber-800 p-3">
   534→                        <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400 shrink-0" />
   535→                        <p className="text-sm text-amber-800 dark:text-amber-200">
   536→                          <span className="font-semibold">Upgrade to {FORMAT_CONFIG[selectedFormat].tierRequired}</span> to unlock{" "}
   537→                          {FORMAT_CONFIG[selectedFormat].label} exports with compliance verification.
   538→                        </p>
   539→                      </div>
   540→                    )}
   541→                  </CardContent>
   542→                </Card>
   543→
   544→                {/* Date Range & Filters */}
   545→                <Card>
