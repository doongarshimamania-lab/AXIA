   123→function formatCurrency(amount: number, currency: string = "USD"): string {
   124→  try {
   125→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   126→  } catch {
   127→    return `$${amount.toFixed(2)}`;
   128→  }
   129→}
   130→
   131→function formatDate(ts: number): string {
   132→  return new Date(ts).toLocaleDateString("en-US", {
   133→    month: "short",
   134→    day: "numeric",
   135→    year: "numeric",
   136→  });
   137→}
   138→
   139→function formatRelativeTime(ts: number): string {
   140→  const diff = Date.now() - ts;
   141→  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   142→  if (days === 0) return "Today";
   143→  if (days === 1) return "Yesterday";
   144→  if (days < 7) return `${days} days ago`;
   145→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   146→  return `${Math.floor(days / 30)}mo ago`;
   147→}

   306→  );
