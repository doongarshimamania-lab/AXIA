   144→// ─── Helpers ────────────────────────────────────────────────────────────────
   145→
   146→function generateId(): string {
   147→  return `li_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`;
   148→}
   149→
   150→function formatCurrency(amount: number, currency: string = "USD"): string {
   151→  try {
   152→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   153→  } catch {
   154→    return `$${amount.toFixed(2)}`;
   155→  }
   156→}
   157→
   158→function formatDate(ts: number): string {
   159→  return new Date(ts).toLocaleDateString("en-US", {
   160→    month: "short",
   161→    day: "numeric",
   162→    year: "numeric",
   163→  });
   164→}
   165→
   166→function dateToTimestamp(dateStr: string): number {
   167→  return new Date(dateStr + "T00:00:00").getTime();
   168→}
   169→
   170→function timestampToDate(ts: number): string {
   171→  const d = new Date(ts);
   172→  return d.toISOString().split("T")[0];
   173→}
