   146→// ─── Helpers ────────────────────────────────────────────────────────────────
   147→
   148→function generateId(): string {
   149→  return `li_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`;
   150→}
   151→
   152→function formatCurrency(amount: number, currency: string = "USD"): string {
   153→  try {
   154→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(amount);
   155→  } catch {
   156→    return `$${amount.toFixed(2)}`;
   157→  }
   158→}
   159→
   160→function formatDate(ts: number): string {
   161→  return new Date(ts).toLocaleDateString("en-US", {
   162→    month: "short",
   163→    day: "numeric",
   164→    year: "numeric",
   165→  });
   166→}
   167→
   168→function dateToTimestamp(dateStr: string): number {
   169→  return new Date(dateStr + "T00:00:00").getTime();
   170→}
   171→
   172→function timestampToDate(ts: number): string {
   173→  const d = new Date(ts);
   174→  return d.toISOString().split("T")[0];
   175→}
