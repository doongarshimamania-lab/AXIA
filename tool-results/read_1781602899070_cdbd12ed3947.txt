    42→// Authentication hook
    43→function useOwnerAuth() {
    44→  const [isAuthenticated, setIsAuthenticated] = useState(false);
    45→  const [password, setPassword] = useState("");
    46→  const [failedAttempts, setFailedAttempts] = useState(0);
    47→  const [showError, setShowError] = useState(false);
    48→  const [lastActivity, setLastActivity] = useState(Date.now());
    49→  const timeoutRef = useRef<number | null>(null);
    50→
    51→  const CORRECT_PASSWORD = "@@@@HHH$";
    52→  const SESSION_TIMEOUT = 10 * 60 * 1000; // 10 minutes
    53→
    54→  const logout = useCallback((message?: string) => {
    55→    setIsAuthenticated(false);
    56→    localStorage.removeItem("ownerSessionActive");
    57→    localStorage.removeItem("ownerLastActivity");
    58→    if (timeoutRef.current) {
    59→      window.clearTimeout(timeoutRef.current);
    60→    }
    61→    if (message) {
    62→      toast.error(message);
    63→    }
    64→  }, []);
    65→
    66→  const resetActivityTimer = useCallback(() => {
    67→    const now = Date.now();
    68→    setLastActivity(now);
    69→    localStorage.setItem("ownerLastActivity", now.toString());
    70→
    71→    if (timeoutRef.current) {
    72→      window.clearTimeout(timeoutRef.current);
    73→    }
    74→
    75→    timeoutRef.current = window.setTimeout(() => {
    76→      logout("Session expired");
    77→    }, SESSION_TIMEOUT);
    78→  }, [logout]);
    79→
    80→  const login = (inputPassword: string) => {
    81→    if (inputPassword === CORRECT_PASSWORD) {
    82→      setIsAuthenticated(true);
    83→      localStorage.setItem("ownerSessionActive", "true");
    84→      resetActivityTimer();
    85→      setFailedAttempts(0);
    86→      setShowError(false);
    87→    } else {
    88→      const newFailedAttempts = failedAttempts + 1;
    89→      setFailedAttempts(newFailedAttempts);
    90→      
    91→      if (newFailedAttempts >= 3) {
    92→        setShowError(true);
    93→        setTimeout(() => setShowError(false), 5000);
    94→      }
    95→    }
    96→    setPassword("");
    97→  };
    98→
    99→  useEffect(() => {
   100→    // Check existing session
   101→    const sessionActive = localStorage.getItem("ownerSessionActive") === "true";
   102→    const storedLastActivity = localStorage.getItem("ownerLastActivity");
   103→    
   104→    if (sessionActive && storedLastActivity) {
   105→      const timeSinceLastActivity = Date.now() - parseInt(storedLastActivity);
   106→      if (timeSinceLastActivity < SESSION_TIMEOUT) {
   107→        setIsAuthenticated(true);
   108→        resetActivityTimer();
   109→      } else {
   110→        logout("Session expired");
   111→      }
   112→    }
   113→
   114→    // Activity listeners
   115→    const activities = ['mousemove', 'keydown', 'click', 'touchstart'];
   116→    const handleActivity = () => {
   117→      if (isAuthenticated) {
   118→        resetActivityTimer();
   119→      }
   120→    };
   121→
   122→    activities.forEach(event => {
   123→      document.addEventListener(event, handleActivity);
   124→    });
   125→
   126→    return () => {
   127→      activities.forEach(event => {
   128→        document.removeEventListener(event, handleActivity);
   129→      });
   130→      if (timeoutRef.current) {
   131→        // Use browser-specific API to ensure the type is number
   132→        window.clearTimeout(timeoutRef.current);
   133→      }
   134→    };
   135→  }, [resetActivityTimer]);
   136→
   137→  return {
   138→    isAuthenticated,
   139→    password,
   140→    setPassword,
   141→    login,
   142→    logout,
   143→    showError,
   144→    failedAttempts
   145→  };
   146→}

   148→// Reusable ThemeToggle component (persists and applies global theme)
   149→function ThemeToggle() {
   150→  const [isDark, setIsDark] = useState<boolean>(() => {
   151→    try {
   152→      const saved = localStorage.getItem("theme");
   153→      if (saved) return saved === "dark";
   154→      return document.documentElement.classList.contains("dark");
   155→    } catch {
   156→      return false;
   157→    }
   158→  });
   159→
   160→  useEffect(() => {
   161→    if (isDark) {
   162→      document.documentElement.classList.add("dark");
   163→      localStorage.setItem("theme", "dark");
   164→    } else {
   165→      document.documentElement.classList.remove("dark");
   166→      localStorage.setItem("theme", "light");
   167→    }
   168→  }, [isDark]);
   169→
   170→  return (
   171→    <div className="flex items-center gap-2">
   172→      <span className="text-sm text-muted-foreground">Dark</span>
   173→      <Switch checked={isDark} onCheckedChange={setIsDark} />
   174→    </div>
   175→  );
   176→}

   183→  const percentage = hasData ? Math.round((SAFE_MRR / 500) * 100) : 0;
   184→  const usersNeeded = hasData ? Math.ceil((500 - SAFE_MRR) / 8) : Math.ceil(500 / 8);

   193→  const LABEL_R = R + 20;

   197→  const needleAngle = -90 + (SAFE_MRR / 500) * 180;

   200→    if (SAFE_MRR < 400) return "#DC2626";
   201→    if (SAFE_MRR < 475) return "#D97706";

   206→    if (usersNeeded > 10) return "#DC2626";
   207→    if (usersNeeded >= 6) return "#D97706";

   240→
   241→            {/* Tick marks and labels */}

   243→              const angle = (value / 500) * 180; // 0..180
   244→              const radian = (angle * Math.PI) / 180;
   245→              // Map so 0 -> left (-90°), 180 -> right (+90°)
   246→              const theta = Math.PI - radian; // matches prior logic
   247→              const x1 = CENTER_X + TICK_INNER * Math.cos(theta);
   248→              const y1 = CENTER_Y - TICK_INNER * Math.sin(theta);
   249→              const x2 = CENTER_X + TICK_OUTER * Math.cos(theta);
   250→              const y2 = CENTER_Y - TICK_OUTER * Math.sin(theta);
   251→              const labelX = CENTER_X + LABEL_R * Math.cos(theta);
   252→              const labelY = CENTER_Y - LABEL_R * Math.sin(theta);

   537→// Compliance Rule Tester
   538→function ComplianceRuleTester() {
   539→  const [testUrl, setTestUrl] = useState("");
   540→  const [status, setStatus] = useState<"idle" | "active" | "at_risk">("idle");
   541→  const [detectedDomain, setDetectedDomain] = useState("");
   542→  const [showFixRule, setShowFixRule] = useState(false);
   543→  const [workSites, setWorkSites] = useState<string[]>(["upwork.com", "github.com", "slack.com"]);
   544→
   545→  useEffect(() => {
   546→    if (testUrl) {
   547→      try {
   548→        const url = new URL(testUrl.startsWith('http') ? testUrl : `https://${testUrl}`);
   549→        const domain = url.hostname.replace('www.', '');
   550→        setDetectedDomain(domain);
   551→        
   552→        if (workSites.includes(domain)) {
   553→          setStatus("active");
   554→          setShowFixRule(false);
   555→        } else {
   556→          setStatus("at_risk");
   557→          setShowFixRule(true);
   558→        }
