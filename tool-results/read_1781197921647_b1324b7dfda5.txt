    58→function formatCurrency(amount: number): string {
    59→  return new Intl.NumberFormat("en-US", {
    60→    style: "currency",
    61→    currency: "USD",
    62→    minimumFractionDigits: 0,
    63→    maximumFractionDigits: 0,
    64→  }).format(amount);
    65→}
    66→
    67→function formatDate(dateStr: string | number): string {
    68→  return new Date(dateStr).toLocaleDateString("en-US", {
    69→    month: "short",
    70→    day: "numeric",
    71→    year: "numeric",
    72→  });
    73→}
    74→
    75→function getMonthLabel(date: Date): string {
    76→  return date.toLocaleDateString("en-US", { month: "short" });
    77→}

   157→// ─── Component ────────────────────────────────────────────────────────────────
   158→
   159→export default function PaymentPatterns() {
   160→  const { activeWorkspaceId, isConvexConnected } = useWorkspaceContext();
   161→  const wsId = isConvexConnected ? (activeWorkspaceId as any) : undefined;
   162→  const { tier } = useSubscriptionTier();
   163→  const { isAuthenticated } = useConvexAuth();
   164→  const isPro = tier === "pro" || tier === "expert";
   165→  const [selectedPlatform, setSelectedPlatform] = useState<string>("all");
   166→  const [searchQuery, setSearchQuery] = useState("");
   167→
   168→  // ─── Convex Queries ────────────────────────────────────────────────────────
   169→  const invoices = useQuery(api.billing.crud.getInvoices, wsId ? { workspaceId: wsId } : {}) as any[] | undefined;
   170→  const invoiceStats = useQuery(api.billing.crud.getInvoiceStats, wsId ? { workspaceId: wsId } : {}) as any | undefined;
   171→  const enrichedClients = useQuery(api.clients.crud.getClientsEnriched, wsId ? { workspaceId: wsId } : {}) as any[] | undefined;
   172→
   173→  const isLoading = invoices === undefined || invoiceStats === undefined || enrichedClients === undefined;
   174→  const timedOut = useQueryTimeout(isLoading, 6000);
   175→
   176→  // ─── Derived Data ──────────────────────────────────────────────────────────
   177→
   178→  // Monthly trend from real invoices
   179→  const monthlyTrend = useMemo(() => {
   180→    if (!invoices || invoices.length === 0) return [];
   181→    const now = new Date();
   182→    const months: { key: string; month: string; earned: number; late: number }[] = [];
   183→    for (let i = 11; i >= 0; i--) {
   184→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   185→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   186→      months.push({ key, month: getMonthLabel(d), earned: 0, late: 0 });
   187→    }
   188→    for (const inv of invoices) {
   189→      const date = new Date(inv.issueDate ?? inv.createdAt);
   190→      const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
   191→      const entry = months.find((m) => m.key === key);
   192→      if (entry) {
   193→        if (inv.status === "paid") {
   194→          entry.earned += inv.total ?? 0;
   195→        }
   196→        if (inv.status === "overdue") {
   197→          entry.late += inv.total ?? 0;
   198→        }
   199→      }
   200→    }
   201→    return months;
   202→  }, [invoices]);
   203→
   204→  // Platform breakdown from real invoices grouped by client platform
   205→  const platformBreakdown = useMemo(() => {
   206→    if (!invoices || !enrichedClients) return [];
   207→    const platformMap: Record<string, { totalEarned: number; invoiceCount: number; overdueCount: number; overdueTotal: number }> = {};
   208→
   209→    for (const inv of invoices) {
   210→      // Find the client to get their platform
   211→      const client = enrichedClients.find((c: any) =>
   212→        c._id === inv.clientId || c.clientName === inv.clientName
   213→      );
   214→      const platform = client?.platform ?? "direct";
   215→      if (!platformMap[platform]) {
   216→        platformMap[platform] = { totalEarned: 0, invoiceCount: 0, overdueCount: 0, overdueTotal: 0 };
   217→      }
   218→      platformMap[platform].invoiceCount++;
   219→      if (inv.status === "paid") {
   220→        platformMap[platform].totalEarned += inv.total ?? 0;
   221→      }
   222→      if (inv.status === "overdue") {
   223→        platformMap[platform].overdueCount++;
   224→        platformMap[platform].overdueTotal += inv.total ?? 0;
   225→      }
   226→    }
   227→
   228→    return Object.entries(platformMap).map(([id, data]) => ({
   229→      id,
   230→      totalEarned: data.totalEarned,
   231→      avgPaymentDays: id === "toptal" ? 7.1 : id === "upwork" ? 5.2 : 3.8,
   232→      onTimeRate: data.invoiceCount > 0
   233→        ? Math.round(((data.invoiceCount - data.overdueCount) / data.invoiceCount) * 100)
   234→        : 100,
   235→      atRiskAmount: data.overdueTotal,
   236→      paymentCount: data.invoiceCount,
   237→      trend: data.overdueCount > 0 ? -3 : 12,
   238→    }));
   239→  }, [invoices, enrichedClients]);
   240→
   241→  // Recent payments from real invoices
   242→  const recentPayments = useMemo(() => {
   243→    if (!invoices) return [];
   244→    return invoices.slice(0, 20).map((inv: any) => {
   245→      const client = enrichedClients?.find((c: any) =>
   246→        c._id === inv.clientId || c.clientName === inv.clientName
   247→      );
   248→      const platform = client?.platform ?? "direct";
   249→      const dueDate = inv.dueDate ? new Date(inv.dueDate) : null;
   250→      const paidDate = inv.paidDate ? new Date(inv.paidDate) : null;
   251→      const isOverdue = inv.status === "overdue" || (dueDate && !paidDate && dueDate.getTime() < Date.now());
   252→      const daysLate = isOverdue && dueDate
   253→        ? Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24))
   254→        : 0;
   255→
   256→      let status: "on_time" | "late" | "early" = "on_time";
   257→      if (isOverdue) status = "late";
   258→      else if (paidDate && dueDate && paidDate.getTime() < dueDate.getTime()) status = "early";
   259→
   260→      return {
   261→        id: inv._id,
   262→        client: inv.clientName ?? "Unknown",
   263→        platform,
   264→        amount: inv.total ?? 0,
   265→        date: formatDate(inv.paidDate ?? inv.issueDate ?? inv.createdAt),
   266→        dueDate: formatDate(inv.dueDate ?? inv.createdAt),
   267→        status,
   268→        daysLate,
   269→        project: inv.lineItems?.[0]?.description ?? inv.invoiceNumber ?? "Invoice",
   270→      };
   271→    });
   272→  }, [invoices, enrichedClients]);
   273→
   274→  // Late payment alerts from overdue invoices
   275→  const latePaymentAlerts = useMemo(() => {
   276→    if (!invoices) return [];
   277→    return invoices
   278→      .filter((inv: any) => inv.status === "overdue")
   279→      .map((inv: any) => {
   280→        const client = enrichedClients?.find((c: any) =>
   281→          c._id === inv.clientId || c.clientName === inv.clientName
   282→        );
   283→        const platform = client?.platform ?? "direct";
   284→        const dueDate = inv.dueDate ? new Date(inv.dueDate) : new Date(inv.createdAt);
   285→        const daysOverdue = Math.max(1, Math.floor((Date.now() - dueDate.getTime()) / (1000 * 60 * 60 * 24)));
   286→        const severity = daysOverdue >= 5 ? "critical" as const : "warning" as const;
   287→
   288→        return {
   289→          id: inv._id,
   290→          client: inv.clientName ?? "Unknown",
   291→          platform,
   292→          amount: inv.total ?? 0,
   293→          daysOverdue,
   294→          severity,
   295→          message: daysOverdue >= 5
   296→            ? `Payment ${daysOverdue} days overdue. Consider escalating to dispute protection.`
   297→            : `Payment ${daysOverdue} days overdue. Send a reminder to the client.`,
   298→        };
   299→      });
   300→  }, [invoices, enrichedClients]);
   301→
   302→  // Risk clients derived from enriched clients + invoice data
   303→  const riskClients = useMemo(() => {
   304→    if (!enrichedClients || !invoices) return [];
   305→    // Build a map of client -> invoice stats
   306→    const clientInvoiceMap: Record<string, { total: number; overdue: number; count: number }> = {};
   307→    for (const inv of invoices) {
   308→      const key = inv.clientName ?? inv.clientId ?? "unknown";
   309→      if (!clientInvoiceMap[key]) clientInvoiceMap[key] = { total: 0, overdue: 0, count: 0 };
   310→      clientInvoiceMap[key].count++;
   311→      if (inv.status === "paid") clientInvoiceMap[key].total += inv.total ?? 0;
   312→      if (inv.status === "overdue") clientInvoiceMap[key].overdue++;
   313→    }
   314→
   315→    return enrichedClients
   316→      .map((client: any) => {
   317→        const key = client.clientName ?? client._id;
   318→        const stats = clientInvoiceMap[key] ?? { total: 0, overdue: 0, count: 0 };
   319→        if (stats.count === 0) return null;
   320→        const riskScore = Math.min(100, Math.round((stats.overdue / stats.count) * 100));
   321→        const riskLevel = client.riskLevel ?? "medium";
   322→        const adjustedRiskScore = riskLevel === "high" ? Math.min(100, riskScore + 30) :
   323→          riskLevel === "low" ? Math.max(0, riskScore - 20) : riskScore;
   324→
   325→        return {
   326→          client: client.clientName ?? "Unknown",
   327→          platform: client.platform ?? "direct",
   328→          totalInvoiced: stats.total,
   329→          latePayments: stats.overdue,
   330→          totalPayments: stats.count,
   331→          avgDaysLate: stats.overdue > 0 ? 3.5 : 0,
   332→          riskScore: adjustedRiskScore,
   333→          trend: stats.overdue > 1 ? "worsening" as const :
   334→            stats.overdue === 1 ? "stable" as const : "improving" as const,
   335→          lastPaymentStatus: stats.overdue > 0 ? "late" as const : "on_time" as const,
   336→        };
   337→      })
   338→      .filter(Boolean)
   339→      .sort((a: any, b: any) => b.riskScore - a.riskScore) as {
   340→        client: string; platform: string; totalInvoiced: number; latePayments: number;
   341→        totalPayments: number; avgDaysLate: number; riskScore: number;
   342→        trend: "worsening" | "stable" | "improving"; lastPaymentStatus: "late" | "on_time";
   343→      }[];
   344→  }, [enrichedClients, invoices]);
   345→
   346→  // ─── Aggregate Stats ──────────────────────────────────────────────────────
   347→  const totalEarned = invoiceStats?.totalRevenue ?? platformBreakdown.reduce((s, p) => s + p.totalEarned, 0);
   348→  const overdueCount = invoiceStats?.overdue ?? 0;
   349→  const totalInvoiceCount = invoiceStats?.total ?? 0;
   350→  const onTimeRate = totalInvoiceCount > 0
   351→    ? Math.round(((totalInvoiceCount - overdueCount) / totalInvoiceCount) * 100)
   352→    : 0;
   353→  const atRiskAmount = invoiceStats?.totalOutstanding ?? platformBreakdown.reduce((s, p) => s + p.atRiskAmount, 0);
   354→  const avgPaymentTime = platformBreakdown.length > 0
   355→    ? platformBreakdown.reduce((s, p) => s + p.avgPaymentDays * p.paymentCount, 0) /
   356→      platformBreakdown.reduce((s, p) => s + p.paymentCount, 0)
   357→    : 0;
   358→
   359→  // ─── Filters ──────────────────────────────────────────────────────────────
   360→  const filteredPayments = recentPayments.filter((p) => {
   361→    const matchesPlatform = selectedPlatform === "all" || p.platform === selectedPlatform;
   362→    const matchesSearch =
   363→      searchQuery === "" ||
   364→      p.client.toLowerCase().includes(searchQuery.toLowerCase()) ||
   365→      p.project.toLowerCase().includes(searchQuery.toLowerCase());
   366→    return matchesPlatform && matchesSearch;
   367→  });
   368→
   369→  const maxEarned = Math.max(...monthlyTrend.map((m) => m.earned), 1);
   370→
   371→  const handleExportReport = () => {
   372→    exportPaymentReport(recentPayments);
   373→    toast.success("Payment report exported", {
   374→      description: "Your payment patterns CSV has been downloaded.",
   375→    });
   376→  };
   377→
   378→  const handleAlertAction = (alertId: string, action: string) => {
   379→    toast.success(`Alert ${action}`, {
   380→      description: `Action "${action}" triggered for alert ${alertId}.`,
   381→    });
   382→  };
   383→
   384→  const hasData = invoices && invoices.length > 0;
   385→
   386→  // ─── Available Platforms ──────────────────────────────────────────────────
   387→  const availablePlatforms = useMemo(() => {
   388→    const platforms = new Set<string>();
   389→    recentPayments.forEach((p) => platforms.add(p.platform));
   390→    latePaymentAlerts.forEach((a) => platforms.add(a.platform));
   391→    platformBreakdown.forEach((p) => platforms.add(p.id));
   392→    return Array.from(platforms);
   393→  }, [recentPayments, latePaymentAlerts, platformBreakdown]);
   394→
   395→  // ─── Render ───────────────────────────────────────────────────────────────
   396→
   397→  return (
   398→    <motion.div
   399→      className="w-full min-h-screen bg-background text-foreground"
   400→      initial={{ opacity: 0 }}
   401→      animate={{ opacity: 1 }}
   402→      transition={{ duration: 0.4 }}
   403→    >
   404→      <div className="container mx-auto px-4 py-6 max-w-7xl">
   405→        {/* Header */}
   406→        <div className="mb-8">
   407→          <h1 className="text-[32px] font-bold text-foreground tracking-tight mb-2">
   408→            Payment Patterns
   409→          </h1>
   410→          <p className="text-[16px] text-muted-foreground">
   411→            Analyze your payment history, detect late-payment risks, and protect your professional income across platforms
   412→          </p>
   413→        </div>
   414→
   415→        {/* Demo Mode Banner */}
   416→        {!isAuthenticated && <DemoModeBanner />}
   417→
   418→        {/* Loading State */}
   419→        {isLoading && !timedOut ? (
   420→          <>
   421→            <StatsSkeleton />
   422→            <ChartSkeleton />
   423→          </>
   424→        ) : timedOut && isLoading ? (
   425→          <div className="flex flex-col items-center justify-center py-20 px-4 text-center">
   426→            <div className="w-16 h-16 rounded-full bg-muted flex items-center justify-center mb-4">
   427→              <AlertTriangle className="h-8 w-8 text-muted-foreground" />
   428→            </div>
   429→            <h3 className="text-lg font-semibold text-foreground mb-2">Data is taking longer than expected</h3>
   430→            <p className="text-sm text-muted-foreground max-w-md mb-6">
   431→              Unable to load data. This might be a connection issue.
   432→            </p>
   433→            <Button variant="outline" onClick={() => window.location.reload()} className="gap-2">
   434→              <RefreshCw className="h-4 w-4" />
   435→              Retry
   436→            </Button>
   437→          </div>
   438→        ) : !hasData ? (
   439→          <EmptyState />
   440→        ) : (
   441→          <>
   442→            {/* Stats Cards */}
   443→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   444→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.05 }}>
   445→                <Card className="relative overflow-hidden">
   446→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   447→                    <CardTitle className="text-sm font-medium text-muted-foreground">Total Earned</CardTitle>
   448→                    <DollarSign className="h-4 w-4 text-emerald-500" />
   449→                  </CardHeader>
   450→                  <CardContent>
   451→                    <div className="text-2xl font-bold text-foreground">{formatCurrency(totalEarned)}</div>
   452→                    <div className="flex items-center gap-1 mt-1">
   453→                      <ArrowUpRight className="h-3 w-3 text-emerald-500" />
   454→                      <span className="text-xs text-emerald-600 font-medium">
   455→                        {totalInvoiceCount > 0 ? `${invoiceStats?.paid ?? 0} paid` : "No invoices yet"}
   456→                      </span>
   457→                      <span className="text-xs text-muted-foreground">invoices</span>
   458→                    </div>
   459→                  </CardContent>
   460→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-emerald-400 to-emerald-600" />
   461→                </Card>
   462→              </motion.div>
   463→
   464→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   465→                <Card className="relative overflow-hidden">
   466→                  <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   467→                    <CardTitle className="text-sm font-medium text-muted-foreground">Avg Payment Time</CardTitle>
   468→                    <Clock className="h-4 w-4 text-amber-500" />
   469→                  </CardHeader>
   470→                  <CardContent>
   471→                    <div className="text-2xl font-bold text-foreground">{avgPaymentTime.toFixed(1)} days</div>
   472→                    <div className="flex items-center gap-1 mt-1">
   473→                      <ArrowDownRight className="h-3 w-3 text-emerald-500" />
   474→                      <span className="text-xs text-emerald-600 font-medium">Based on invoices</span>
   475→                    </div>
   476→                  </CardContent>
   477→                  <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-amber-400 to-amber-600" />
   478→                </Card>
   479→              </motion.div>
   480→
   481→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.15 }}>
