   197→function formatCurrency(value: number): string {
   198→  return new Intl.NumberFormat("en-US", {
   199→    style: "currency",
   200→    currency: "USD",
   201→    minimumFractionDigits: 0,
   202→    maximumFractionDigits: 0,
   203→  }).format(value);
   204→}
   205→
   206→function formatDate(timestamp: number | undefined): string {
   207→  if (!timestamp) return "—";
   208→  return new Intl.DateTimeFormat("en-US", {
   209→    month: "short",
   210→    day: "numeric",
   211→    year: "numeric",
   212→  }).format(new Date(timestamp));
   213→}
   214→
   215→function daysUntilClose(timestamp: number | undefined): number | null {
   216→  if (!timestamp) return null;
   217→  const diff = timestamp - Date.now();
   218→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   219→}

   221→function getSourceInfo(source: string | undefined) {
   222→  return SOURCE_OPTIONS.find((s) => s.value === source) ?? SOURCE_OPTIONS[5];
