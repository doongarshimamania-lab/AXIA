   269→function formatDate(timestamp: number | undefined): string {
   270→  if (!timestamp) return "—";
   271→  return new Intl.DateTimeFormat("en-US", {
   272→    month: "short",
   273→    day: "numeric",
   274→    year: "numeric",
   275→  }).format(new Date(timestamp));
   276→}

   278→function daysUntilClose(timestamp: number | undefined): number | null {
   279→  if (!timestamp) return null;
   280→  const diff = timestamp - Date.now();
   281→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
   282→}
