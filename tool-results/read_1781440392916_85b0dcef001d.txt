    81→    81→
    82→    82→  const CORRECT_PASSWORD = "@@@@HHH$";
    83→    83→  const SESSION_TIMEOUT = 10 * 60 * 1000; // 10 minutes
    84→    84→
    85→    85→  const logout = useCallback((message?: string) => {
    86→    86→    setIsAuthenticated(false);
    87→    87→    localStorage.removeItem("ownerSessionActive");

    89→    89→    if (timeoutRef.current) {
    90→    90→      window.clearTimeout(timeoutRef.current);
    91→    91→    }

    93→    93→      toast.error(message);
    94→    94→    }
    95→    95→  }, []);
    96→    96→
    97→    97→  const resetActivityTimer = useCallback(() => {
    98→    98→    const now = Date.now();
    99→    99→    setLastActivity(now);
   100→   100→    localStorage.setItem("ownerLastActivity", now.toString());
   101→   101→
   102→   102→    if (timeoutRef.current) {
   103→   103→      window.clearTimeout(timeoutRef.current);
   104→   104→    }
   105→   105→
   106→   106→    timeoutRef.current = window.setTimeout(() => {
   107→   107→      logout("Session expired");

   109→   109→  }, [logout]);
   110→   110→
   111→   111→  const login = (inputPassword: string) => {
   112→   112→    if (inputPassword === CORRECT_PASSWORD) {
   113→   113→      setIsAuthenticated(true);
   114→   114→      localStorage.setItem("ownerSessionActive", "true");
   115→   115→      resetActivityTimer();
   116→   116→      setFailedAttempts(0);
   117→   117→      setShowError(false);

   135→   135→    if (sessionActive && storedLastActivity) {
   136→   136→      const timeSinceLastActivity = Date.now() - parseInt(storedLastActivity);
   137→   137→      if (timeSinceLastActivity < SESSION_TIMEOUT) {
   138→   138→        setIsAuthenticated(true);
   139→   139→        resetActivityTimer();
   140→   140→      } else {
   141→   141→        logout("Session expired");
   142→   142→      }

   149→   149→        resetActivityTimer();
   150→   150→      }
   151→   151→    };
   152→   152→
   153→   153→    activities.forEach(event => {
   154→   154→      document.addEventListener(event, handleActivity);
   155→   155→    });
   156→   156→
   157→   157→    return () => {
   158→   158→      activities.forEach(event => {
   159→   159→        document.removeEventListener(event, handleActivity);

   170→   170→    password,

   172→   172→    login,
   173→   173→    logout,
   174→   174→    showError,

   179→   179→// Reusable ThemeToggle component (persists and applies global theme)
   180→   180→function ThemeToggle() {
   181→   181→  const [isDark, setIsDark] = useState<boolean>(() => {
   182→   182→    try {
   183→   183→      const saved = localStorage.getItem("theme");
   184→   184→      if (saved) return saved === "dark";
   185→   185→      return document.documentElement.classList.contains("dark");
   186→   186→    } catch {
   187→   187→      return false;
   188→   188→    }
   189→   189→  });
   190→   190→
   191→   191→  useEffect(() => {
   192→   192→    if (isDark) {
   193→   193→      document.documentElement.classList.add("dark");
   194→   194→      localStorage.setItem("theme", "dark");
   195→   195→    } else {
   196→   196→      document.documentElement.classList.remove("dark");
   197→   197→      localStorage.setItem("theme", "light");
   198→   198→    }
   199→   199→  }, [isDark]);
   200→   200→
   201→   201→  return (
   202→   202→    <div className="flex items-center gap-2">
   203→   203→      <span className="text-sm text-muted-foreground">Dark</span>
   204→   204→      <Switch checked={isDark} onCheckedChange={setIsDark} />
   205→   205→    </div>
   206→   206→  );
   207→   207→}
